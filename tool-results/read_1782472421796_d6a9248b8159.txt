   189→   189→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)

   342→   342→      }
   343→   343→    } catch {
   344→   344→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   345→   345→    } finally {
   346→   346→      setLocalSubmitting(false)
   347→   347→    }

   359→   359→      const selectedEntryIds = selectedEntries.size > 0 ? Array.from(selectedEntries) : undefined
   360→   360→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   361→   361→        method: 'POST',

   364→   364→          targetCustomerId: transferForm.targetCustomerId,
   365→   365→          amount: parseFloat(transferForm.amount) || 0,
   366→   366→          description: transferForm.description,
   367→   367→          entryIds: selectedEntryIds,
   368→   368→        }),
   369→   369→      })
   370→   370→      if (res.ok) {
   371→   371→        const data = await res.json()
   372→   372→        const docNo = data?.source?.entry?.docNo || ''
   373→   373→        setEntrySuccess({ show: true, docNo, type: 'Transfer' })
   374→   374→        notify({ type: 'success', title: `Transfer ${docNo}`, message: data.message || `Transfer ${docNo} processed successfully` })
   375→   375→        closeDialogs()
   376→   376→        onRefresh?.()
   377→   377→        setTimeout(() => setEntrySuccess(null), 4000)
   378→   378→      } else {
   379→   379→        const e = await safeJson(res)
   380→   380→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   381→   381→      }
   382→   382→    } catch {
   383→   383→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   384→   384→    } finally {
   385→   385→      setLocalSubmitting(false)
   386→   386→    }
   387→   387→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   388→   388→
   389→   389→  // ─── Split payment helpers for checkout ───
