     5→// ─── Report Types ───
     6→type ReportType =
     7→  | 'summary' | 'outstanding' | 'payments' | 'transactions' | 'vip' | 'settlements' | 'refunds' | 'gst'
     8→  | 'activeFolio' | 'checkoutHistory' | 'creditBalance' | 'chargeSummary'
     9→  | 'discounts' | 'userCollections' | 'dailyCollections' | 'monthlyCollections' | 'audit'
    10→
    11→const VALID_REPORT_TYPES: ReportType[] = [
    12→  'summary', 'outstanding', 'payments', 'transactions', 'vip', 'settlements', 'refunds', 'gst',
    13→  'activeFolio', 'checkoutHistory', 'creditBalance', 'chargeSummary',
    14→  'discounts', 'userCollections', 'dailyCollections', 'monthlyCollections', 'audit',
    15→]
    16→
    17→// ─── Filter Interface ───
    18→interface ReportFilters {
    19→  fromDate: Date | null
    20→  toDate: Date | null
    21→  customer: string | null
    22→  folioNumber: string | null
    23→  user: string | null
    24→  status: string | null
    25→  paymentType: string | null
    26→  paymentMode: string | null
    27→  transactionType: string | null
    28→  dateFilter: Record<string, Date> | undefined
    29→}
    30→
    31→function parseFilters(searchParams: URLSearchParams): ReportFilters {
    32→  const from = searchParams.get('from')
    33→  const to = searchParams.get('to')
    34→  const fromDate = from ? new Date(from) : null
    35→  let toDate: Date | null = to ? new Date(to) : null
    36→  if (toDate) toDate.setHours(23, 59, 59, 999)
    37→
    38→  const dateFilter: Record<string, Date> | undefined =
    39→    fromDate && toDate ? { gte: fromDate, lte: toDate } :
    40→    fromDate ? { gte: fromDate } :
    41→    toDate ? { lte: toDate } :
    42→    undefined
    43→
    44→  // transactionType is alias for paymentType
    45→  const paymentType = searchParams.get('paymentType') || searchParams.get('transactionType') || null
    46→
    47→  return {
    48→    fromDate,
    49→    toDate,
    50→    customer: searchParams.get('customer') || null,
    51→    folioNumber: searchParams.get('folioNumber') || null,
    52→    user: searchParams.get('user') || null,
    53→    status: searchParams.get('status') || null,
    54→    paymentType,
    55→    paymentMode: searchParams.get('paymentMode') || null,
    56→    transactionType: paymentType,
    57→    dateFilter,
    58→  }
    59→}

    61→// Helper: apply common ledger entry filters
    62→function applyLedgerFilters(where: Record<string, unknown>, filters: ReportFilters, unitCode: string) {
    63→  where.unitCode = unitCode
    64→  if (filters.dateFilter) where.createdAt = filters.dateFilter
    65→  if (filters.customer) {
    66→    where.OR = [
    67→      { customer: { fullName: { contains: filters.customer } } },
    68→      { customer: { customerId: { contains: filters.customer } } },
    69→      { customer: { mobile: { contains: filters.customer } } },
    70→    ] as any[]
    71→  }
    72→  if (filters.paymentType && filters.paymentType !== 'All') where.type = filters.paymentType
    73→  if (filters.paymentMode && filters.paymentMode !== 'All') where.paymentMode = filters.paymentMode
    74→  if (filters.user) where.userName = { contains: filters.user }
    75→}

    77→// Helper: apply common customer filters
    78→function applyCustomerFilters(where: Record<string, unknown>, filters: ReportFilters, unitCode: string) {
    79→  where.unitCode = unitCode
    80→  if (filters.status && filters.status !== 'All') where.status = filters.status
    81→  if (filters.customer) {
    82→    where.OR = [
    83→      { fullName: { contains: filters.customer } },
    84→      { customerId: { contains: filters.customer } },
    85→      { mobile: { contains: filters.customer } },
    86→    ] as any[]
    87→  }
    88→}

    90→// Helper: get balance for a set of customer IDs
    91→async function getCustomerBalances(customerIds: string[], folioStatus?: string) {
    92→  const folioWhere: Record<string, unknown> = { customerId: { in: customerIds }, status: folioStatus || 'Active' }
    93→  const activeFolioIds = await db.folio.findMany({
    94→    where: folioWhere,
    95→    select: { id: true, customerId: true },
    96→  })
    97→  const folioIdSet = activeFolioIds.map(f => f.id)
    98→  if (folioIdSet.length === 0) return new Map<string, { balance: number; date: string | null }>()
    99→
   100→  // Use a two-step approach: get max createdAt per customerId, then fetch those rows
   101→  // This avoids the unreliable distinct+orderBy pattern in SQLite
   102→  const maxDates = await db.ledgerEntry.groupBy({
   103→    by: ['customerId'],
   104→    where: { folioId: { in: folioIdSet }, status: 'Active', customerId: { in: customerIds } },
   105→    _max: { createdAt: true },
   106→  })
   107→  const customerMaxDate = new Map(maxDates.map(m => [m.customerId, m._max.createdAt]))
   108→  if (customerMaxDate.size === 0) return new Map<string, { balance: number; date: string | null }>()
   109→
   110→  // Fetch the actual ledger entries that match the max dates
   111→  const latestEntries = await db.ledgerEntry.findMany({
   112→    where: {
   113→      folioId: { in: folioIdSet },
   114→      status: 'Active',
   115→      customerId: { in: customerIds },
   116→    },
   117→    select: { customerId: true, balance: true, createdAt: true },
   118→  })
   119→
   120→  // Pick only the entries that match the max date per customer
   121→  const result = new Map<string, { balance: number; date: string | null }>()
   122→  for (const entry of latestEntries) {
   123→    const maxDate = customerMaxDate.get(entry.customerId)
   124→    if (maxDate && entry.createdAt && entry.createdAt.getTime() === maxDate.getTime()) {
   125→      if (!result.has(entry.customerId)) {
   126→        result.set(entry.customerId, { balance: entry.balance, date: entry.createdAt?.toISOString() ?? null })
   127→      }
   128→    }
   129→  }
   130→  return result
   131→}

   133→// ─── GET Handler ───
   134→export async function GET(request: NextRequest) {
   135→  try {
   136→    const session = await getSessionFromRequest(request)
   137→    if (!session) {
   138→      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   139→    }
   140→
   141→    if (!sessionHasPermission(session, 'khata', 'access')) {
   142→      return NextResponse.json({ error: 'Access denied' }, { status: 403 })
   143→    }
   144→
   145→    const { searchParams } = new URL(request.url)
   146→    const type = searchParams.get('type') as ReportType | null
   147→
   148→    if (!type || !VALID_REPORT_TYPES.includes(type)) {
   149→      return NextResponse.json({ error: 'Invalid or missing report type', validTypes: VALID_REPORT_TYPES }, { status: 400 })
   150→    }
   151→
   152→    const filters = parseFilters(searchParams)
   153→
   154→    switch (type) {
   155→      case 'summary': return handleSummary(session.unitCode, filters)
   156→      case 'outstanding': return handleOutstanding(session.unitCode, filters)
   157→      case 'payments': return handlePayments(session.unitCode, filters)
   158→      case 'transactions': return handleTransactions(session.unitCode, filters)
   159→      case 'vip': return handleVip(session.unitCode, filters)
   160→      case 'settlements': return handleSettlements(session.unitCode, filters)
   161→      case 'refunds': return handleRefunds(session.unitCode, filters)
   162→      case 'gst': return handleGst(session.unitCode, filters)
   163→      case 'activeFolio': return handleActiveFolio(session.unitCode, filters)
   164→      case 'checkoutHistory': return handleCheckoutHistory(session.unitCode, filters)
   165→      case 'creditBalance': return handleCreditBalance(session.unitCode, filters)
   166→      case 'chargeSummary': return handleChargeSummary(session.unitCode, filters)
   167→      case 'discounts': return handleDiscounts(session.unitCode, filters)
   168→      case 'userCollections': return handleUserCollections(session.unitCode, filters)
   169→      case 'dailyCollections': return handleDailyCollections(session.unitCode, filters)
   170→      case 'monthlyCollections': return handleMonthlyCollections(session.unitCode, filters)
   171→      case 'audit': return handleAudit(session.unitCode, filters)
   172→      default:
   173→        return NextResponse.json({ error: 'Invalid report type' }, { status: 400 })
   174→    }
   175→  } catch (error) {
   176→    console.error('[FolioReports] Error:', error)
   177→    return NextResponse.json({ error: 'Failed to generate report' }, { status: 500 })
   178→  }
   179→}

   185→// ─── 1. Customer Summary Report ───
   186→async function handleSummary(unitCode: string, filters: ReportFilters) {
   187→  const where: Record<string, unknown> = {}
   188→  applyCustomerFilters(where, filters, unitCode)
   189→
   190→  const customers = await db.customer.findMany({
   191→    where,
   192→    select: {
   193→      id: true, customerId: true, fullName: true, mobile: true, email: true,
   194→      customerType: true, creditLimit: true, status: true, gstNumber: true,
   195→      openingBalance: true, createdAt: true,
   196→      tags: { select: { tagName: true } },
   197→      _count: { select: { folios: true, ledgerEntries: true, settlements: true, visits: true } },
   198→    },
   199→    orderBy: { createdAt: 'desc' },
   200→  })
   201→
   202→  const customerIds = customers.map(c => c.id)
   203→  const ledgerMap = await getCustomerBalances(customerIds)
   204→
   205→  const rows = customers.map(c => {
   206→    const ledger = ledgerMap.get(c.id)
   207→    const balance = ledger?.balance ?? 0
   208→    return {
   209→      customerId: c.customerId,
   210→      fullName: c.fullName,
   211→      mobile: c.mobile,
   212→      email: c.email,
   213→      customerType: c.customerType,
   214→      status: c.status,
   215→      balance,
   216→      creditLimit: c.creditLimit,
   217→      gstNumber: c.gstNumber,
   218→      totalVisits: c._count.visits,
   219→      totalTransactions: c._count.ledgerEntries,
   220→      totalSettlements: c._count.settlements,
   221→      tags: c.tags.map(t => t.tagName),
   222→      createdAt: c.createdAt.toISOString(),
   223→      lastTxnDate: ledger?.date ?? null,
   224→    }
   225→  })
   226→
   227→  const totalBalance = rows.reduce((s, r) => s + r.balance, 0)
   228→  const totalCreditLimit = rows.reduce((s, r) => s + r.creditLimit, 0)
   229→  const totalOutstanding = rows.filter(r => r.balance > 0).reduce((s, r) => s + r.balance, 0)
   230→  const outstandingCount = rows.filter(r => r.balance > 0).length
   231→
   232→  return NextResponse.json({
   233→    reportType: 'summary',
   234→    summary: {
   235→      totalCustomers: rows.length,
   236→      totalBalance,
   237→      totalCreditLimit,
   238→      totalOutstanding,
   239→      outstandingCount,
   240→      activeCount: rows.filter(r => r.status === 'Active').length,
   241→      settledCount: rows.filter(r => r.status === 'Settled').length,
   242→    },
   243→    data: rows,
   244→  })
   245→}

   247→// ─── 2. Outstanding Balance Report ───
   248→async function handleOutstanding(unitCode: string, filters: ReportFilters) {
   249→  const where: Record<string, unknown> = { status: { notIn: ['Archived', 'CheckedOut'] } }
   250→  applyCustomerFilters(where, filters, unitCode)
   251→
   252→  const customers = await db.customer.findMany({
   253→    where,
   254→    select: {
   255→      id: true, customerId: true, fullName: true, mobile: true, email: true,
   256→      customerType: true, creditLimit: true, status: true,
   257→      tags: { select: { tagName: true } },
   258→      _count: { select: { ledgerEntries: true, settlements: true, visits: true } },
   259→    },
   260→    orderBy: { createdAt: 'desc' },
   261→  })
   262→
   263→  const customerIds = customers.map(c => c.id)
   264→  const ledgerMap = await getCustomerBalances(customerIds)
   265→
   266→  const rows = customers
   267→    .map(c => {
   268→      const balance = ledgerMap.get(c.id)?.balance ?? 0
   269→      return {
   270→        customerId: c.customerId,
   271→        fullName: c.fullName,
   272→        mobile: c.mobile,
   273→        email: c.email,
   274→        customerType: c.customerType,
   275→        status: c.status,
   276→        balance,
   277→        creditLimit: c.creditLimit,
   278→        overdue: balance > c.creditLimit,
   279→        tags: c.tags.map(t => t.tagName),
   280→        lastTxnDate: ledgerMap.get(c.id)?.date ?? null,
   281→      }
   282→    })
   283→    .filter(r => r.balance > 0)
   284→    .sort((a, b) => b.balance - a.balance)
   285→
   286→  const totalOutstanding = rows.reduce((s, r) => s + r.balance, 0)
   287→  const avgOutstanding = rows.length > 0 ? totalOutstanding / rows.length : 0
   288→
   289→  return NextResponse.json({
   290→    reportType: 'outstanding',
   291→    summary: {
   292→      totalCustomersWithDue: rows.length,
   293→      totalOutstanding,
   294→      avgOutstanding: Math.round(avgOutstanding * 100) / 100,
   295→      highRiskCount: rows.filter(r => r.overdue).length,
   296→    },
   297→    data: rows,
   298→  })
   299→}

   301→// ─── 3. Payment Collection Report ───
   302→async function handlePayments(unitCode: string, filters: ReportFilters) {
   303→  const where: Record<string, unknown> = { status: 'Active', type: 'Payment' }
   304→  applyLedgerFilters(where, filters, unitCode)
   305→
   306→  const entries = await db.ledgerEntry.findMany({
   307→    where,
   308→    orderBy: { createdAt: 'desc' },
   309→    select: {
   310→      id: true, date: true, description: true, credit: true, balance: true,
   311→      paymentMode: true, userName: true, customerId: true, createdAt: true,
   312→      customer: { select: { fullName: true, customerId: true, mobile: true } },
   313→    },
   314→    take: 5000,
   315→  })
   316→
   317→  const rows = entries.map(e => ({
   318→    date: e.date,
   319→    customerName: e.customer.fullName,
   320→    customerId: e.customer.customerId,
   321→    mobile: e.customer.mobile,
   322→    description: e.description,
   323→    amount: e.credit,
   324→    paymentMode: e.paymentMode,
   325→    balanceAfter: e.balance,
   326→    recordedBy: e.userName,
   327→    createdAt: e.createdAt.toISOString(),
   328→  }))
   329→
   330→  const totalPayments = rows.reduce((s, r) => s + r.amount, 0)
   331→  const modeBreakdown = rows.reduce((acc: Record<string, number>, r) => {
   332→    const mode = r.paymentMode || 'Cash'
   333→    acc[mode] = (acc[mode] || 0) + r.amount
   334→    return acc
   335→  }, {})
   336→
   337→  return NextResponse.json({
   338→    reportType: 'payments',
   339→    summary: {
   340→      totalPayments,
   341→      paymentCount: rows.length,
   342→      avgPayment: rows.length > 0 ? Math.round((totalPayments / rows.length) * 100) / 100 : 0,
   343→      modeBreakdown,
   344→    },
   345→    data: rows,
   346→  })
   347→}

   349→// ─── 4. Transaction History Report ───
   350→async function handleTransactions(unitCode: string, filters: ReportFilters) {
   351→  const where: Record<string, unknown> = { status: 'Active' }
   352→  applyLedgerFilters(where, filters, unitCode)
   353→
   354→  const entries = await db.ledgerEntry.findMany({
   355→    where,
   356→    orderBy: { createdAt: 'desc' },
   357→    select: {
   358→      id: true, date: true, type: true, description: true, docNo: true, invoiceNo: true,
   359→      debit: true, credit: true, tax: true, discount: true, balance: true,
   360→      paymentMode: true, userName: true, createdAt: true,
   361→      customer: { select: { fullName: true, customerId: true } },
   362→    },
   363→    take: 5000,
   364→  })
   365→
   366→  const rows = entries.map(e => ({
   367→    date: e.date,
   368→    customerName: e.customer.fullName,
   369→    customerId: e.customer.customerId,
   370→    type: e.type,
   371→    description: e.description,
   372→    docNo: e.docNo,
   373→    invoiceNo: e.invoiceNo,
   374→    debit: e.debit,
   375→    credit: e.credit,
   376→    tax: e.tax,
   377→    discount: e.discount,
   378→    balance: e.balance,
   379→    paymentMode: e.paymentMode,
   380→    recordedBy: e.userName,
   381→    createdAt: e.createdAt.toISOString(),
   382→  }))
   383→
   384→  const totalDebits = rows.reduce((s, r) => s + r.debit, 0)
   385→  const totalCredits = rows.reduce((s, r) => s + r.credit, 0)
   386→  const totalTax = rows.reduce((s, r) => s + r.tax, 0)
   387→  const totalDiscount = rows.reduce((s, r) => s + r.discount, 0)
   388→
   389→  const typeBreakdown = rows.reduce((acc: Record<string, { count: number; debit: number; credit: number }>, r) => {
   390→    if (!acc[r.type]) acc[r.type] = { count: 0, debit: 0, credit: 0 }
   391→    acc[r.type].count++
   392→    acc[r.type].debit += r.debit
   393→    acc[r.type].credit += r.credit
   394→    return acc
   395→  }, {})
   396→
   397→  return NextResponse.json({
   398→    reportType: 'transactions',
   399→    summary: {
   400→      totalEntries: rows.length,
   401→      totalDebits,
   402→      totalCredits,
   403→      totalTax,
   404→      totalDiscount,
   405→      netFlow: totalCredits - totalDebits,
   406→      typeBreakdown,
   407→    },
   408→    data: rows,
   409→  })
   410→}

   412→// ─── 5. VIP Customer Report ───
   413→async function handleVip(unitCode: string, filters: ReportFilters) {
   414→  const where: Record<string, unknown> = {
   415→    OR: [
   416→      { customerType: 'VIP' },
   417→      { tags: { some: { tagName: 'VIP' } } },
   418→      { tags: { some: { tagName: 'Premium' } } },
   419→    ],
   420→  }
   421→  applyCustomerFilters(where, filters, unitCode)
   422→
   423→  const customers = await db.customer.findMany({
   424→    where,
   425→    select: {
   426→      id: true, customerId: true, fullName: true, mobile: true, email: true,
   427→      customerType: true, creditLimit: true, status: true, gstNumber: true,
   428→      panNumber: true, openingBalance: true, createdAt: true,
   429→      tags: { select: { tagName: true } },
   430→      _count: { select: { folios: true, ledgerEntries: true, settlements: true, visits: true } },
   431→    },
   432→    orderBy: { createdAt: 'desc' },
   433→  })
   434→
   435→  const customerIds = customers.map(c => c.id)
   436→  const ledgerMap = await getCustomerBalances(customerIds)
   437→
   438→  const lifetimeSpend = customerIds.length > 0
   439→    ? await db.ledgerEntry.groupBy({
   440→        by: ['customerId'],
   441→        where: { customerId: { in: customerIds }, status: 'Active', type: 'Charge' },
   442→        _sum: { debit: true },
   443→      })
   444→    : []
   445→  const spendMap = new Map(lifetimeSpend.map(l => [l.customerId, l._sum.debit ?? 0]))
   446→
   447→  const rows = customers.map(c => {
   448→    const balance = ledgerMap.get(c.id)?.balance ?? 0
   449→    const lifetimeSpendAmt = spendMap.get(c.id) ?? 0
   450→    return {
   451→      customerId: c.customerId,
   452→      fullName: c.fullName,
   453→      mobile: c.mobile,
   454→      email: c.email,
   455→      customerType: c.customerType,
   456→      status: c.status,
   457→      balance,
   458→      creditLimit: c.creditLimit,
   459→      gstNumber: c.gstNumber,
   460→      panNumber: c.panNumber,
   461→      lifetimeSpend: lifetimeSpendAmt,
   462→      totalVisits: c._count.visits,
   463→      totalTransactions: c._count.ledgerEntries,
   464→      totalSettlements: c._count.settlements,
   465→      tags: c.tags.map(t => t.tagName),
   466→      createdAt: c.createdAt.toISOString(),
   467→      lastTxnDate: ledgerMap.get(c.id)?.date ?? null,
   468→    }
   469→  })
   470→
   471→  const totalLifetimeSpend = rows.reduce((s, r) => s + r.lifetimeSpend, 0)
   472→  const totalOutstanding = rows.reduce((s, r) => s + r.balance, 0)
   473→
   474→  return NextResponse.json({
   475→    reportType: 'vip',
   476→    summary: {
   477→      totalVipCustomers: rows.length,
   478→      totalLifetimeSpend,
   479→      totalOutstanding,
   480→      avgLifetimeSpend: rows.length > 0 ? Math.round((totalLifetimeSpend / rows.length) * 100) / 100 : 0,
   481→      activeVipCount: rows.filter(r => r.status === 'Active').length,
   482→    },
   483→    data: rows,
   484→  })
   485→}

   646→async function handleActiveFolio(unitCode: string, filters: ReportFilters) {
   647→  const customerWhere: Record<string, unknown> = { status: 'Active' }
   648→  applyCustomerFilters(customerWhere, filters, unitCode)
   649→
   650→  const customers = await db.customer.findMany({

   652→    select: {
   653→      id: true, customerId: true, fullName: true, mobile: true, email: true,
   654→      customerType: true, creditLimit: true, status: true,
   655→      _count: { select: { ledgerEntries: true, folios: true, visits: true } },
   656→    },
   657→    orderBy: { createdAt: 'desc' },
   658→  })
   659→
   660→  const customerIds = customers.map(c => c.id)
   661→
   662→  // Get active folios with their details
   663→  const foliosWhere: Record<string, unknown> = { customerId: { in: customerIds }, status: 'Active' }
   664→  if (filters.folioNumber) foliosWhere.folioNumber = parseInt(filters.folioNumber, 10) || undefined
   665→  const folios = await db.folio.findMany({
   666→    where: foliosWhere,
   667→    select: {
   668→      id: true, folioId: true, folioNumber: true, customerId: true, visitNumber: true,
   669→      status: true, createdAt: true,
   670→      _count: { select: { ledgerEntries: true } },
   671→    },
   672→    orderBy: { createdAt: 'desc' },
   673→  })
   674→
   675→  const folioIds = folios.map(f => f.id)
   676→  // Use groupBy to reliably get the latest entry per folio (avoids distinct+orderBy issues in SQLite)
   677→  const folioBalanceMap = new Map<string, { balance: number; date: string | null }>()
   678→  if (folioIds.length > 0) {
   679→    const maxDates = await db.ledgerEntry.groupBy({
   680→      by: ['folioId'],
   681→      where: { folioId: { in: folioIds }, status: 'Active' },
   682→      _max: { createdAt: true },
   683→    })
   684→    const folioMaxDate = new Map(maxDates.map(m => [m.folioId, m._max.createdAt]))
   685→    const latestEntries = await db.ledgerEntry.findMany({
   686→      where: { folioId: { in: folioIds }, status: 'Active' },
   687→      select: { folioId: true, balance: true, createdAt: true },
   688→    })
   689→    for (const entry of latestEntries) {
   690→      const maxDate = folioMaxDate.get(entry.folioId)
   691→      if (maxDate && entry.createdAt && entry.createdAt.getTime() === maxDate.getTime()) {
   692→        if (!folioBalanceMap.has(entry.folioId)) {
   693→          folioBalanceMap.set(entry.folioId, { balance: entry.balance, date: entry.createdAt?.toISOString() ?? null })
   694→        }
   695→      }
   696→    }
   697→  }
   698→
   699→  const customerMap = new Map(customers.map(c => [c.id, c]))
   700→
   701→  const rows = folios.map(f => {
   702→    const cust = customerMap.get(f.customerId)
   703→    const bal = folioBalanceMap.get(f.id)
   704→    return {
   705→      customerId: cust?.customerId ?? '',
   706→      fullName: cust?.fullName ?? '',
   707→      mobile: cust?.mobile ?? '',
   708→      customerType: cust?.customerType ?? '',
   709→      folioId: f.folioId,
   710→      folioNumber: f.folioNumber,
   711→      visitNumber: f.visitNumber,
   712→      status: f.status,
   713→      balance: bal?.balance ?? 0,
   714→      entryCount: f._count.ledgerEntries,
   715→      lastTxnDate: bal?.date ?? null,
   716→      folioCreatedAt: f.createdAt.toISOString(),
   717→    }
   718→  })
   719→
   720→  const totalBalance = rows.reduce((s, r) => s + r.balance, 0)
   721→  const activeFolioCount = rows.length
   722→
   723→  return NextResponse.json({
   724→    reportType: 'activeFolio',
   725→    summary: {
   726→      activeFolioCount,
   727→      totalBalance,
   728→      avgBalance: activeFolioCount > 0 ? Math.round((totalBalance / activeFolioCount) * 100) / 100 : 0,
   729→      uniqueCustomers: new Set(rows.map(r => r.customerId)).size,

  1073→// ─── 16. Monthly Collection Report ───
  1074→async function handleMonthlyCollections(unitCode: string, filters: ReportFilters) {
  1075→  const where: Record<string, unknown> = { status: 'Active', type: 'Payment' }
  1076→  applyLedgerFilters(where, filters, unitCode)
  1077→
  1078→  const entries = await db.ledgerEntry.findMany({
  1079→    where,
  1080→    orderBy: { createdAt: 'desc' },
  1081→    select: {
  1082→      id: true, date: true, credit: true, paymentMode: true, createdAt: true,
  1083→    },
  1084→    take: 10000,
  1085→  })
  1086→
  1087→  // Group by month (YYYY-MM)
  1088→  const monthGroups = new Map<string, {
  1089→    month: string
  1090→    totalAmount: number
  1091→    count: number
  1092→    modeBreakdown: Record<string, number>
  1093→  }>()
  1094→
  1095→  for (const e of entries) {
  1096→    const d = new Date(e.date || e.createdAt.toISOString())
  1097→    const monthKey = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
  1098→    if (!monthGroups.has(monthKey)) {
  1099→      monthGroups.set(monthKey, { month: monthKey, totalAmount: 0, count: 0, modeBreakdown: {} })
  1100→    }
  1101→    const group = monthGroups.get(monthKey)!
  1102→    group.totalAmount += e.credit
  1103→    group.count++
  1104→    const mode = e.paymentMode || 'Cash'
  1105→    group.modeBreakdown[mode] = (group.modeBreakdown[mode] || 0) + e.credit
  1106→  }
  1107→
  1108→  const rows = Array.from(monthGroups.values()).sort((a, b) => b.month.localeCompare(a.month))
  1109→
  1110→  const grandTotal = rows.reduce((s, r) => s + r.totalAmount, 0)
  1111→  const bestMonth = rows.length > 0 ? rows.reduce((best, r) => r.totalAmount > best.totalAmount ? r : best, rows[0]) : null
  1112→
  1113→  return NextResponse.json({
  1114→    reportType: 'monthlyCollections',
  1115→    summary: {
  1116→      totalMonths: rows.length,
  1117→      grandTotal,
  1118→      avgMonthly: rows.length > 0 ? Math.round((grandTotal / rows.length) * 100) / 100 : 0,
  1119→      bestMonth: bestMonth ? { month: bestMonth.month, amount: bestMonth.totalAmount } : null,
  1120→      totalPayments: rows.reduce((s, r) => s + r.count, 0),
  1121→    },
  1122→    data: rows,
  1123→  })
  1124→}

  1126→// ─── 17. Audit Report ───
  1127→async function handleAudit(unitCode: string, filters: ReportFilters) {
  1128→  const where: Record<string, unknown> = { unitCode }
  1129→  if (filters.dateFilter) where.createdAt = filters.dateFilter
  1130→  if (filters.user) where.userName = { contains: filters.user }
  1131→  if (filters.customer) {
  1132→    // For audit, search in target or details or customerId
  1133→    where.OR = [
  1134→      { target: { contains: filters.customer } },
  1135→      { details: { contains: filters.customer } },
  1136→      { customerId: { contains: filters.customer } },
  1137→    ] as any[]
  1138→  }
  1139→  // Filter by action type (map paymentType to action where applicable)
  1140→  if (filters.paymentType && filters.paymentType !== 'All') {
  1141→    const actionMap: Record<string, string> = {
  1142→      Charge: 'Create', Payment: 'Create', Refund: 'Refund',
  1143→      Discount: 'Discount', Settlement: 'Settlement',
  1144→    }
  1145→    const action = actionMap[filters.paymentType]
  1146→    if (action) where.action = action
  1147→  }
  1148→
  1149→  const logs = await db.folioAuditLog.findMany({
  1150→    where,
  1151→    orderBy: { createdAt: 'desc' },
  1152→    select: {
  1153→      id: true, action: true, target: true, details: true,
  1154→      userName: true, ipAddress: true, device: true,
  1155→      customerId: true, folioId: true, createdAt: true,
  1156→    },
  1157→    take: 5000,
  1158→  })
  1159→
  1160→  const rows = logs.map(l => {
  1161→    let detailsParsed: Record<string, any> = {}
  1162→    try { detailsParsed = l.details ? JSON.parse(l.details) : {} } catch { /* ignore */ }
  1163→    return {
  1164→      date: l.createdAt.toISOString(),
  1165→      action: l.action,
  1166→      target: l.target,
  1167→      userName: l.userName,
  1168→      customerId: l.customerId,
  1169→      folioId: l.folioId,
  1170→      details: detailsParsed,
  1171→      ipAddress: l.ipAddress,
  1172→      device: l.device,
  1173→    }
  1174→  })
  1175→
  1176→  // Action breakdown
  1177→  const actionBreakdown = rows.reduce((acc: Record<string, number>, r) => {
  1178→    acc[r.action] = (acc[r.action] || 0) + 1
  1179→    return acc
  1180→  }, {})
  1181→
  1182→  return NextResponse.json({
  1183→    reportType: 'audit',
  1184→    summary: {
  1185→      totalLogs: rows.length,
  1186→      actionBreakdown,
  1187→      uniqueUsers: new Set(rows.map(r => r.userName).filter(Boolean)).size,
  1188→    },
  1189→    data: rows,
  1190→  })
  1191→}
