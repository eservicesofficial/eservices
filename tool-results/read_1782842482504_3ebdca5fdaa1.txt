     5→// ─── Report Types ───
     6→type ReportType =
     7→  | 'summary' | 'outstanding' | 'payments' | 'transactions' | 'vip' | 'settlements' | 'refunds' | 'gst'
     8→  | 'activeFolio' | 'checkoutHistory' | 'creditBalance' | 'chargeSummary'
     9→  | 'discounts' | 'userCollections' | 'dailyCollections' | 'monthlyCollections' | 'audit'
    10→
    11→const VALID_REPORT_TYPES: ReportType[] = [
    12→  'summary', 'outstanding', 'payments', 'transactions', 'vip', 'settlements', 'refunds', 'gst',
    13→  'activeFolio', 'checkoutHistory', 'creditBalance', 'chargeSummary',
    14→  'discounts', 'userCollections', 'dailyCollections', 'monthlyCollections', 'audit',
    15→]
    16→
    17→// ─── Filter Interface ───
    18→interface ReportFilters {
    19→  fromDate: Date | null
    20→  toDate: Date | null
    21→  customer: string | null
    22→  folioNumber: string | null
    23→  user: string | null
    24→  status: string | null
    25→  paymentType: string | null
    26→  paymentMode: string | null
    27→  transactionType: string | null
    28→  dateFilter: Record<string, Date> | undefined
    29→}
    30→
    31→function parseFilters(searchParams: URLSearchParams): ReportFilters {
    32→  const from = searchParams.get('from')
    33→  const to = searchParams.get('to')
    34→  const fromDate = from ? new Date(from) : null
    35→  let toDate: Date | null = to ? new Date(to) : null
    36→  if (toDate) toDate.setHours(23, 59, 59, 999)
    37→
    38→  const dateFilter: Record<string, Date> | undefined =
    39→    fromDate && toDate ? { gte: fromDate, lte: toDate } :
    40→    fromDate ? { gte: fromDate } :
    41→    toDate ? { lte: toDate } :
    42→    undefined
    43→
    44→  // transactionType is alias for paymentType
    45→  const paymentType = searchParams.get('paymentType') || searchParams.get('transactionType') || null
    46→
    47→  return {
    48→    fromDate,
    49→    toDate,
    50→    customer: searchParams.get('customer') || null,
    51→    folioNumber: searchParams.get('folioNumber') || null,
    52→    user: searchParams.get('user') || null,
    53→    status: searchParams.get('status') || null,
    54→    paymentType,
    55→    paymentMode: searchParams.get('paymentMode') || null,
    56→    transactionType: paymentType,
    57→    dateFilter,
    58→  }
    59→}

    61→// Helper: apply common ledger entry filters
    62→function applyLedgerFilters(where: Record<string, unknown>, filters: ReportFilters, unitCode: string) {
    63→  where.unitCode = unitCode
    64→  if (filters.dateFilter) where.createdAt = filters.dateFilter
    65→  if (filters.customer) {
    66→    where.OR = [
    67→      { customer: { fullName: { contains: filters.customer } } },
    68→      { customer: { customerId: { contains: filters.customer } } },
    69→      { customer: { mobile: { contains: filters.customer } } },
    70→    ] as any[]
    71→  }
    72→  if (filters.paymentType && filters.paymentType !== 'All') where.type = filters.paymentType
    73→  if (filters.paymentMode && filters.paymentMode !== 'All') where.paymentMode = filters.paymentMode
    74→  if (filters.user) where.userName = { contains: filters.user }
    75→}

    77→// Helper: apply common customer filters
    78→function applyCustomerFilters(where: Record<string, unknown>, filters: ReportFilters, unitCode: string) {
    79→  where.unitCode = unitCode
    80→  if (filters.status && filters.status !== 'All') where.status = filters.status
    81→  if (filters.customer) {
    82→    where.OR = [
    83→      { fullName: { contains: filters.customer } },
    84→      { customerId: { contains: filters.customer } },
    85→      { mobile: { contains: filters.customer } },
    86→    ] as any[]
    87→  }
    88→}

    90→// Helper: get balance for a set of customer IDs
    91→async function getCustomerBalances(customerIds: string[], folioStatus?: string) {
    92→  const folioWhere: Record<string, unknown> = { customerId: { in: customerIds }, status: folioStatus || 'Active' }
    93→  const activeFolioIds = await db.folio.findMany({
    94→    where: folioWhere,
    95→    select: { id: true, customerId: true },
    96→  })
    97→  const folioIdSet = activeFolioIds.map(f => f.id)
    98→  if (folioIdSet.length === 0) return new Map<string, { balance: number; date: string | null }>()
    99→
   100→  const lastLedgers = await db.ledgerEntry.findMany({
   101→    where: { folioId: { in: folioIdSet }, status: 'Active' },
   102→    distinct: ['customerId'],
   103→    orderBy: { createdAt: 'desc' },
   104→    select: { customerId: true, balance: true, createdAt: true },
   105→  })
   106→  return new Map(lastLedgers.map(l => [l.customerId, { balance: l.balance, date: l.createdAt?.toISOString() ?? null }]))
   107→}

   109→// ─── GET Handler ───
   110→export async function GET(request: NextRequest) {
   111→  try {
   112→    const session = await getSessionFromRequest(request)
   113→    if (!session) {
   114→      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   115→    }
   116→
   117→    if (!sessionHasPermission(session, 'khata', 'access')) {
   118→      return NextResponse.json({ error: 'Access denied' }, { status: 403 })
   119→    }
   120→
   121→    const { searchParams } = new URL(request.url)
   122→    const type = searchParams.get('type') as ReportType | null
   123→
   124→    if (!type || !VALID_REPORT_TYPES.includes(type)) {
   125→      return NextResponse.json({ error: 'Invalid or missing report type', validTypes: VALID_REPORT_TYPES }, { status: 400 })
   126→    }
   127→
   128→    const filters = parseFilters(searchParams)
   129→
   130→    switch (type) {
   131→      case 'summary': return handleSummary(session.unitCode, filters)
   132→      case 'outstanding': return handleOutstanding(session.unitCode, filters)
   133→      case 'payments': return handlePayments(session.unitCode, filters)
   134→      case 'transactions': return handleTransactions(session.unitCode, filters)
   135→      case 'vip': return handleVip(session.unitCode, filters)
   136→      case 'settlements': return handleSettlements(session.unitCode, filters)
   137→      case 'refunds': return handleRefunds(session.unitCode, filters)
   138→      case 'gst': return handleGst(session.unitCode, filters)
   139→      case 'activeFolio': return handleActiveFolio(session.unitCode, filters)
   140→      case 'checkoutHistory': return handleCheckoutHistory(session.unitCode, filters)
   141→      case 'creditBalance': return handleCreditBalance(session.unitCode, filters)
   142→      case 'chargeSummary': return handleChargeSummary(session.unitCode, filters)
   143→      case 'discounts': return handleDiscounts(session.unitCode, filters)
   144→      case 'userCollections': return handleUserCollections(session.unitCode, filters)
   145→      case 'dailyCollections': return handleDailyCollections(session.unitCode, filters)
   146→      case 'monthlyCollections': return handleMonthlyCollections(session.unitCode, filters)
   147→      case 'audit': return handleAudit(session.unitCode, filters)
   148→      default:
   149→        return NextResponse.json({ error: 'Invalid report type' }, { status: 400 })
   150→    }
   151→  } catch (error) {
   152→    console.error('[FolioReports] Error:', error)
   153→    return NextResponse.json({ error: 'Failed to generate report' }, { status: 500 })
   154→  }
   155→}

   161→// ─── 1. Customer Summary Report ───
   162→async function handleSummary(unitCode: string, filters: ReportFilters) {
   163→  const where: Record<string, unknown> = {}
   164→  applyCustomerFilters(where, filters, unitCode)
   165→
   166→  const customers = await db.customer.findMany({
   167→    where,
   168→    select: {
   169→      id: true, customerId: true, fullName: true, mobile: true, email: true,
   170→      customerType: true, creditLimit: true, status: true, gstNumber: true,
   171→      openingBalance: true, createdAt: true,
   172→      tags: { select: { tagName: true } },
   173→      _count: { select: { folios: true, ledgerEntries: true, settlements: true, visits: true } },
   174→    },
   175→    orderBy: { createdAt: 'desc' },
   176→  })
   177→
   178→  const customerIds = customers.map(c => c.id)
   179→  const ledgerMap = await getCustomerBalances(customerIds)
   180→
   181→  const rows = customers.map(c => {
   182→    const ledger = ledgerMap.get(c.id)
   183→    const balance = ledger?.balance ?? 0
   184→    return {
   185→      customerId: c.customerId,
   186→      fullName: c.fullName,
   187→      mobile: c.mobile,
   188→      email: c.email,
   189→      customerType: c.customerType,
   190→      status: c.status,
   191→      balance,
   192→      creditLimit: c.creditLimit,
   193→      gstNumber: c.gstNumber,
   194→      totalVisits: c._count.visits,
   195→      totalTransactions: c._count.ledgerEntries,
   196→      totalSettlements: c._count.settlements,
   197→      tags: c.tags.map(t => t.tagName),
   198→      createdAt: c.createdAt.toISOString(),
   199→      lastTxnDate: ledger?.date ?? null,
   200→    }
   201→  })
   202→
   203→  const totalBalance = rows.reduce((s, r) => s + r.balance, 0)
   204→  const totalCreditLimit = rows.reduce((s, r) => s + r.creditLimit, 0)
   205→  const totalOutstanding = rows.filter(r => r.balance > 0).reduce((s, r) => s + r.balance, 0)
   206→  const outstandingCount = rows.filter(r => r.balance > 0).length
   207→
   208→  return NextResponse.json({
   209→    reportType: 'summary',
   210→    summary: {
   211→      totalCustomers: rows.length,
   212→      totalBalance,
   213→      totalCreditLimit,
   214→      totalOutstanding,
   215→      outstandingCount,
   216→      activeCount: rows.filter(r => r.status === 'Active').length,
   217→      settledCount: rows.filter(r => r.status === 'Settled').length,
   218→    },
   219→    data: rows,
   220→  })
   221→}

   223→// ─── 2. Outstanding Balance Report ───
   224→async function handleOutstanding(unitCode: string, filters: ReportFilters) {
   225→  const where: Record<string, unknown> = { status: { notIn: ['Archived', 'CheckedOut'] } }
   226→  applyCustomerFilters(where, filters, unitCode)
   227→
   228→  const customers = await db.customer.findMany({
   229→    where,
   230→    select: {
   231→      id: true, customerId: true, fullName: true, mobile: true, email: true,
   232→      customerType: true, creditLimit: true, status: true,
   233→      tags: { select: { tagName: true } },
   234→      _count: { select: { ledgerEntries: true, settlements: true, visits: true } },
   235→    },
   236→    orderBy: { createdAt: 'desc' },
   237→  })
   238→
   239→  const customerIds = customers.map(c => c.id)
   240→  const ledgerMap = await getCustomerBalances(customerIds)
   241→
   242→  const rows = customers
   243→    .map(c => {
   244→      const balance = ledgerMap.get(c.id)?.balance ?? 0
   245→      return {
   246→        customerId: c.customerId,
   247→        fullName: c.fullName,
   248→        mobile: c.mobile,
   249→        email: c.email,
   250→        customerType: c.customerType,
   251→        status: c.status,
   252→        balance,
   253→        creditLimit: c.creditLimit,
   254→        overdue: balance > c.creditLimit,
   255→        tags: c.tags.map(t => t.tagName),
   256→        lastTxnDate: ledgerMap.get(c.id)?.date ?? null,
   257→      }
   258→    })
   259→    .filter(r => r.balance > 0)
   260→    .sort((a, b) => b.balance - a.balance)
   261→
   262→  const totalOutstanding = rows.reduce((s, r) => s + r.balance, 0)
   263→  const avgOutstanding = rows.length > 0 ? totalOutstanding / rows.length : 0
   264→
   265→  return NextResponse.json({
   266→    reportType: 'outstanding',
   267→    summary: {
   268→      totalCustomersWithDue: rows.length,
   269→      totalOutstanding,
   270→      avgOutstanding: Math.round(avgOutstanding * 100) / 100,
   271→      highRiskCount: rows.filter(r => r.overdue).length,
   272→    },
   273→    data: rows,
   274→  })
   275→}

   277→// ─── 3. Payment Collection Report ───
   278→async function handlePayments(unitCode: string, filters: ReportFilters) {
   279→  const where: Record<string, unknown> = { status: 'Active', type: 'Payment' }
   280→  applyLedgerFilters(where, filters, unitCode)
   281→
   282→  const entries = await db.ledgerEntry.findMany({
   283→    where,
   284→    orderBy: { createdAt: 'desc' },
   285→    select: {
   286→      id: true, date: true, description: true, credit: true, balance: true,
   287→      paymentMode: true, userName: true, customerId: true, createdAt: true,
   288→      customer: { select: { fullName: true, customerId: true, mobile: true } },
   289→    },
   290→    take: 5000,
   291→  })
   292→
   293→  const rows = entries.map(e => ({
   294→    date: e.date,
   295→    customerName: e.customer.fullName,
   296→    customerId: e.customer.customerId,
   297→    mobile: e.customer.mobile,
   298→    description: e.description,
   299→    amount: e.credit,
   300→    paymentMode: e.paymentMode,
   301→    balanceAfter: e.balance,
   302→    recordedBy: e.userName,
   303→    createdAt: e.createdAt.toISOString(),
   304→  }))
   305→
   306→  const totalPayments = rows.reduce((s, r) => s + r.amount, 0)
   307→  const modeBreakdown = rows.reduce((acc: Record<string, number>, r) => {
   308→    const mode = r.paymentMode || 'Cash'
   309→    acc[mode] = (acc[mode] || 0) + r.amount
   310→    return acc
   311→  }, {})
   312→
   313→  return NextResponse.json({
   314→    reportType: 'payments',
   315→    summary: {
   316→      totalPayments,
   317→      paymentCount: rows.length,
   318→      avgPayment: rows.length > 0 ? Math.round((totalPayments / rows.length) * 100) / 100 : 0,
   319→      modeBreakdown,
   320→    },
   321→    data: rows,
   322→  })
   323→}

   325→// ─── 4. Transaction History Report ───
   326→async function handleTransactions(unitCode: string, filters: ReportFilters) {
   327→  const where: Record<string, unknown> = { status: 'Active' }
   328→  applyLedgerFilters(where, filters, unitCode)
   329→
   330→  const entries = await db.ledgerEntry.findMany({
   331→    where,
   332→    orderBy: { createdAt: 'desc' },
   333→    select: {
   334→      id: true, date: true, type: true, description: true, docNo: true, invoiceNo: true,
   335→      debit: true, credit: true, tax: true, discount: true, balance: true,
   336→      paymentMode: true, userName: true, createdAt: true,
   337→      customer: { select: { fullName: true, customerId: true } },
   338→    },
   339→    take: 5000,
   340→  })
   341→
   342→  const rows = entries.map(e => ({
   343→    date: e.date,
   344→    customerName: e.customer.fullName,
   345→    customerId: e.customer.customerId,
   346→    type: e.type,
   347→    description: e.description,
   348→    docNo: e.docNo,
   349→    invoiceNo: e.invoiceNo,
   350→    debit: e.debit,
   351→    credit: e.credit,
   352→    tax: e.tax,
   353→    discount: e.discount,
   354→    balance: e.balance,
   355→    paymentMode: e.paymentMode,
   356→    recordedBy: e.userName,
   357→    createdAt: e.createdAt.toISOString(),
   358→  }))
   359→
   360→  const totalDebits = rows.reduce((s, r) => s + r.debit, 0)
   361→  const totalCredits = rows.reduce((s, r) => s + r.credit, 0)
   362→  const totalTax = rows.reduce((s, r) => s + r.tax, 0)
   363→  const totalDiscount = rows.reduce((s, r) => s + r.discount, 0)
   364→
   365→  const typeBreakdown = rows.reduce((acc: Record<string, { count: number; debit: number; credit: number }>, r) => {
   366→    if (!acc[r.type]) acc[r.type] = { count: 0, debit: 0, credit: 0 }
   367→    acc[r.type].count++
   368→    acc[r.type].debit += r.debit
   369→    acc[r.type].credit += r.credit
   370→    return acc
   371→  }, {})
   372→
   373→  return NextResponse.json({
   374→    reportType: 'transactions',
   375→    summary: {
   376→      totalEntries: rows.length,
   377→      totalDebits,
   378→      totalCredits,
   379→      totalTax,
   380→      totalDiscount,
   381→      netFlow: totalCredits - totalDebits,
   382→      typeBreakdown,
   383→    },
   384→    data: rows,
   385→  })
   386→}

   388→// ─── 5. VIP Customer Report ───
   389→async function handleVip(unitCode: string, filters: ReportFilters) {
   390→  const where: Record<string, unknown> = {
   391→    OR: [
   392→      { customerType: 'VIP' },
   393→      { tags: { some: { tagName: 'VIP' } } },
   394→      { tags: { some: { tagName: 'Premium' } } },
   395→    ],
   396→  }
   397→  applyCustomerFilters(where, filters, unitCode)
   398→
   399→  const customers = await db.customer.findMany({
   400→    where,
   401→    select: {
   402→      id: true, customerId: true, fullName: true, mobile: true, email: true,
   403→      customerType: true, creditLimit: true, status: true, gstNumber: true,
   404→      panNumber: true, openingBalance: true, createdAt: true,
   405→      tags: { select: { tagName: true } },
   406→      _count: { select: { folios: true, ledgerEntries: true, settlements: true, visits: true } },
   407→    },
   408→    orderBy: { createdAt: 'desc' },
   409→  })
   410→
   411→  const customerIds = customers.map(c => c.id)
   412→  const ledgerMap = await getCustomerBalances(customerIds)
   413→
   414→  const lifetimeSpend = customerIds.length > 0
   415→    ? await db.ledgerEntry.groupBy({
   416→        by: ['customerId'],
   417→        where: { customerId: { in: customerIds }, status: 'Active', type: 'Charge' },
   418→        _sum: { debit: true },
   419→      })
   420→    : []
   421→  const spendMap = new Map(lifetimeSpend.map(l => [l.customerId, l._sum.debit ?? 0]))
   422→
   423→  const rows = customers.map(c => {
   424→    const balance = ledgerMap.get(c.id)?.balance ?? 0
   425→    const lifetimeSpendAmt = spendMap.get(c.id) ?? 0
   426→    return {
   427→      customerId: c.customerId,
   428→      fullName: c.fullName,
   429→      mobile: c.mobile,
   430→      email: c.email,
   431→      customerType: c.customerType,
   432→      status: c.status,
   433→      balance,
   434→      creditLimit: c.creditLimit,
   435→      gstNumber: c.gstNumber,
   436→      panNumber: c.panNumber,
   437→      lifetimeSpend: lifetimeSpendAmt,
   438→      totalVisits: c._count.visits,
   439→      totalTransactions: c._count.ledgerEntries,
   440→      totalSettlements: c._count.settlements,
   441→      tags: c.tags.map(t => t.tagName),
   442→      createdAt: c.createdAt.toISOString(),
   443→      lastTxnDate: ledgerMap.get(c.id)?.date ?? null,
   444→    }
   445→  })
   446→
   447→  const totalLifetimeSpend = rows.reduce((s, r) => s + r.lifetimeSpend, 0)
   448→  const totalOutstanding = rows.reduce((s, r) => s + r.balance, 0)
   449→
   450→  return NextResponse.json({
   451→    reportType: 'vip',
   452→    summary: {
   453→      totalVipCustomers: rows.length,
   454→      totalLifetimeSpend,
   455→      totalOutstanding,
   456→      avgLifetimeSpend: rows.length > 0 ? Math.round((totalLifetimeSpend / rows.length) * 100) / 100 : 0,
   457→      activeVipCount: rows.filter(r => r.status === 'Active').length,
   458→    },
   459→    data: rows,
   460→  })
   461→}

  1062→      monthGroups.set(monthKey, { month: monthKey, totalAmount: 0, count: 0, modeBreakdown: {} })
  1063→    }
  1064→    const group = monthGroups.get(monthKey)!
  1065→    group.totalAmount += e.credit
  1066→    group.count++
  1067→    const mode = e.paymentMode || 'Cash'
  1068→    group.modeBreakdown[mode] = (group.modeBreakdown[mode] || 0) + e.credit
  1069→  }
  1070→
  1071→  const rows = Array.from(monthGroups.values()).sort((a, b) => b.month.localeCompare(a.month))
  1072→
  1073→  const grandTotal = rows.reduce((s, r) => s + r.totalAmount, 0)
  1074→  const bestMonth = rows.length > 0 ? rows.reduce((best, r) => r.totalAmount > best.totalAmount ? r : best, rows[0]) : null
  1075→
  1076→  return NextResponse.json({
  1077→    reportType: 'monthlyCollections',
  1078→    summary: {
  1079→      totalMonths: rows.length,
  1080→      grandTotal,
  1081→      avgMonthly: rows.length > 0 ? Math.round((grandTotal / rows.length) * 100) / 100 : 0,
  1082→      bestMonth: bestMonth ? { month: bestMonth.month, amount: bestMonth.totalAmount } : null,
  1083→      totalPayments: rows.reduce((s, r) => s + r.count, 0),
  1084→    },
  1085→    data: rows,
  1086→  })
  1087→}

  1089→// ─── 17. Audit Report ───
  1090→async function handleAudit(unitCode: string, filters: ReportFilters) {
  1091→  const where: Record<string, unknown> = { unitCode }
  1092→  if (filters.dateFilter) where.createdAt = filters.dateFilter
  1093→  if (filters.user) where.userName = { contains: filters.user }
  1094→  if (filters.customer) {
  1095→    // For audit, search in target or details or customerId
  1096→    where.OR = [
  1097→      { target: { contains: filters.customer } },
  1098→      { details: { contains: filters.customer } },
  1099→      { customerId: { contains: filters.customer } },
  1100→    ] as any[]
  1101→  }
  1102→  // Filter by action type (map paymentType to action where applicable)
  1103→  if (filters.paymentType && filters.paymentType !== 'All') {
  1104→    const actionMap: Record<string, string> = {
  1105→      Charge: 'Create', Payment: 'Create', Refund: 'Refund',
  1106→      Discount: 'Discount', Settlement: 'Settlement',
  1107→    }
  1108→    const action = actionMap[filters.paymentType]
  1109→    if (action) where.action = action
  1110→  }
  1111→
  1112→  const logs = await db.folioAuditLog.findMany({
  1113→    where,
  1114→    orderBy: { createdAt: 'desc' },
  1115→    select: {
  1116→      id: true, action: true, target: true, details: true,
  1117→      userName: true, ipAddress: true, device: true,
  1118→      customerId: true, folioId: true, createdAt: true,
  1119→    },
  1120→    take: 5000,
  1121→  })
  1122→
  1123→  const rows = logs.map(l => {
  1124→    let detailsParsed: Record<string, any> = {}
  1125→    try { detailsParsed = l.details ? JSON.parse(l.details) : {} } catch { /* ignore */ }
  1126→    return {
  1127→      date: l.createdAt.toISOString(),
  1128→      action: l.action,
  1129→      target: l.target,
  1130→      userName: l.userName,
  1131→      customerId: l.customerId,
  1132→      folioId: l.folioId,
  1133→      details: detailsParsed,
  1134→      ipAddress: l.ipAddress,
  1135→      device: l.device,
  1136→    }
  1137→  })
  1138→
  1139→  // Action breakdown
  1140→  const actionBreakdown = rows.reduce((acc: Record<string, number>, r) => {
  1141→    acc[r.action] = (acc[r.action] || 0) + 1
  1142→    return acc
  1143→  }, {})
  1144→
  1145→  return NextResponse.json({
  1146→    reportType: 'audit',
  1147→    summary: {
  1148→      totalLogs: rows.length,
  1149→      actionBreakdown,
  1150→      uniqueUsers: new Set(rows.map(r => r.userName).filter(Boolean)).size,
  1151→    },
  1152→    data: rows,
  1153→  })
  1154→}
