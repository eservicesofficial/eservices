    10→    10→// ─── Helper: Convert number to Indian Rupee words ───
    11→    11→function numberToIndianWords(num: number): string {
    12→    12→  if (num === 0) return 'ZERO'
    13→    13→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    14→    14→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    15→    15→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    16→    16→  let words = ''
    17→    17→  function convertBelowThousand(n: number): string {
    18→    18→    let w = ''
    19→    19→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    20→    20→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    21→    21→    else if (n > 0) { w += ones[n] }
    22→    22→    return w
    23→    23→  }
    24→    24→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    25→    25→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    26→    26→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    27→    27→  if (num > 0) { words += convertBelowThousand(num) }
    28→    28→  return words.trim().toUpperCase() + ' ONLY'
    29→    29→}

   189→   189→// ─── Helper: Compute the most-used payment mode from credit entries ───
   190→   190→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   191→   191→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   192→   192→  if (creditEntries.length === 0) return 'N/A'
   193→   193→  const modeCounts: Record<string, number> = {}
   194→   194→  for (const e of creditEntries) {
   195→   195→    const mode = e.paymentMode || 'Cash'
   196→   196→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   197→   197→  }
   198→   198→  let bestMode = 'Cash'
   199→   199→  let bestAmount = 0
   200→   200→  for (const [mode, amount] of Object.entries(modeCounts)) {
   201→   201→    if (amount > bestAmount) {
   202→   202→      bestAmount = amount
   203→   203→      bestMode = mode
   204→   204→    }
   205→   205→  }
   206→   206→  return bestMode
   207→   207→}

   209→   209→// ─── Helper: Build payment summary HTML ───
   210→   210→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   211→   211→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   212→   212→  if (creditEntries.length === 0) {
   213→   213→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   214→   214→  }
   215→   215→
   216→   216→  const modeTotals: Record<string, number> = {}
   217→   217→  for (const e of creditEntries) {
   218→   218→    const mode = e.paymentMode || 'Cash'
   219→   219→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   220→   220→  }
   221→   221→
   222→   222→  const modes = Object.entries(modeTotals)
   223→   223→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   224→   224→
   225→   225→  if (modes.length === 1) {
   226→   226→    const [mode, amount] = modes[0]
   227→   227→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   228→   228→  }
   229→   229→
   230→   230→  const rows = modes.map(([mode, amount]) => {
   231→   231→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   232→   232→  }).join('')
   233→   233→
   234→   234→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   235→   235→}

   237→   237→// ─── Helper: Get stable document number from entries (never regenerate) ───
   238→   238→// Searches ALL entries for the first non-empty docNo to avoid N/A
   239→   239→function getStableDocId(entries: { docNo?: string; invoiceNo?: string; id?: string; createdAt?: string }[]): string {
   240→   240→  if (entries.length === 0) return `DOC-${Date.now()}`
   241→   241→  for (const e of entries) {
   242→   242→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   243→   243→  }
   244→   244→  for (const e of entries) {
   245→   245→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   246→   246→  }
   247→   247→  // Fallback: use entry ID truncated or timestamp — never show N/A
   248→   248→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   249→   249→  return fallback
   250→   250→}
