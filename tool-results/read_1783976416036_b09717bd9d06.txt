    11→// ─── Helper: Convert number to Indian Rupee words ───
    12→function numberToIndianWords(num: number): string {
    13→  if (num === 0) return 'ZERO'
    14→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    15→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    16→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    17→  let words = ''
    18→  function convertBelowThousand(n: number): string {
    19→    let w = ''
    20→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    21→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    22→    else if (n > 0) { w += ones[n] }
    23→    return w
    24→  }
    25→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    26→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    27→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    28→  if (num > 0) { words += convertBelowThousand(num) }
    29→  return words.trim().toUpperCase() + ' ONLY'
    30→}

   210→// ─── Helper: Compute the most-used payment mode from credit entries ───
   211→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   212→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   213→  if (creditEntries.length === 0) return 'N/A'
   214→  const modeCounts: Record<string, number> = {}
   215→  for (const e of creditEntries) {
   216→    const mode = e.paymentMode || 'Cash'
   217→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   218→  }
   219→  let bestMode = 'Cash'
   220→  let bestAmount = 0
   221→  for (const [mode, amount] of Object.entries(modeCounts)) {
   222→    if (amount > bestAmount) {
   223→      bestAmount = amount
   224→      bestMode = mode
   225→    }
   226→  }
   227→  return bestMode
   228→}

   230→// ─── Helper: Build payment summary HTML ───
   231→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   232→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   233→  if (creditEntries.length === 0) {
   234→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   235→  }
   236→
   237→  const modeTotals: Record<string, number> = {}
   238→  for (const e of creditEntries) {
   239→    const mode = e.paymentMode || 'Cash'
   240→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   241→  }
   242→
   243→  const modes = Object.entries(modeTotals)
   244→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   245→
   246→  if (modes.length === 1) {
   247→    const [mode, amount] = modes[0]
   248→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   249→  }
   250→
   251→  const rows = modes.map(([mode, amount]) => {
   252→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   253→  }).join('')
   254→
   255→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   256→}

   258→// ─── Helper: Get stable document number from entries (never regenerate) ───
   259→// Searches ALL entries for the first non-empty docNo to avoid N/A
   260→function getStableDocId(entries: { docNo?: string; invoiceNo?: string; id?: string; createdAt?: string }[]): string {
   261→  if (entries.length === 0) return `DOC-${Date.now()}`
   262→  for (const e of entries) {
   263→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   264→  }
   265→  for (const e of entries) {
   266→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   267→  }
   268→  // Fallback: use entry ID truncated or timestamp — never show N/A
   269→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   270→  return fallback
   271→}

   383→// ═══════════════════════════════════════════════════════════════════
   384→// BUILD HOTEL-STYLE TAX INVOICE (Redesigned professional layout)
   385→// Supports: P4 specific folio, P8 multi-page, P10a GST inclusive, P11 amount in words
   386→// ═══════════════════════════════════════════════════════════════════
   387→export function buildHotelStyleInvoice(
   388→  settings: Record<string, string>,
   389→  detail: CustomerDetail,
   390→  entries: any[],
   391→  invoiceNo: string,
   392→  invoiceDate: string,
   393→  docTitle: string,
   394→  extraMeta?: { openingBalance?: number; closingBalance?: number; periodFrom?: string; periodTo?: string; specificFolioNumber?: string | number },
   395→  printedBy?: string
   396→): string {
   397→  const d = detail as any
   398→  const bizName = settings.shopName || 'Eservices'
   399→  const bizAddress = settings.shopAddress || ''
   400→  const bizPhone = settings.shopPhone || ''
   401→  const bizEmail = settings.shopEmail || ''
   402→  const gstNumber = settings.gstNumber || settings.gstin || ''
   403→  const panNumber = settings.panNumber || d.panNumber || ''
   404→  const gstState = settings.gstState || settings.gstStateCode || ''
   405→  const stateCode = gstState ? gstState.split('-')[0] || gstState : ''
   406→  const shopLogo = settings.shopLogo || settings.logo || settings.thermalLogo || ''
   407→
   408→  // P10a: GST settings
   409→  const gstEnabled = settings.enableGst === 'true'
   410→  const gstRate = parseFloat(settings.gstRate) || 0
   411→
   412→  // Customer info
   413→  const custName = d.fullName || ''
   414→  const custId = d.customerId || ''
   415→  const custMobile = d.mobile || ''
   416→  const custEmail = d.email || ''
   417→  const custAddr = d.address || ''
   418→  const custGst = d.gstNumber || ''
   419→  const custType = d.customerType || ''
   420→  const custPan = d.panNumber || ''
   421→  const custCompany = d.companyName || d.company || ''
   422→
   423→  // Calculate totals
   424→  const chargeEntries = entries.filter((e: any) => (e.debit || 0) > 0 && e.type !== 'Refund')
   425→  const paymentEntries = entries.filter((e: any) => (e.credit || 0) > 0)
   426→  const refundEntries = entries.filter((e: any) => e.type === 'Refund')
   427→  const totalDebits = chargeEntries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   428→  const totalCredits = paymentEntries.reduce((s: number, e: any) => s + (e.credit || 0), 0)
   429→  const totalRefunds = refundEntries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   430→
   431→  // P10a: GST Inclusive calculation
   432→  // If GST is enabled in settings, compute GST from charge amounts even if individual entries have tax=0
   433→  let totalTax = 0
   434→  let computedTaxFromSettings = false
   435→  if (gstEnabled && gstRate > 0) {
   436→    // Check if any charge has explicit tax > 0
   437→    const hasExplicitTax = chargeEntries.some((e: any) => (e.tax || 0) > 0)
   438→    if (hasExplicitTax) {
   439→      // Use the explicit tax values from entries
   440→      totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   441→    } else {
   442→      // P10a: No explicit GST on charges — compute from inclusive amounts using settings GST rate
   443→      // For inclusive GST: GST = Amount × Rate / (100 + Rate), Taxable = Amount - GST
   444→      totalTax = Math.round((totalDebits * gstRate / (100 + gstRate)) * 100) / 100
   445→      computedTaxFromSettings = true
   446→    }
   447→  } else {
   448→    // GST not enabled — use whatever tax is on entries (usually 0)
   449→    totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   450→  }
   451→
   452→  const totalDiscount = entries.reduce((s: number, e: any) => s + (e.discount || 0), 0)
   453→  const closingBal = roundBal(extraMeta?.closingBalance ?? (entries.length > 0 ? entries[entries.length - 1].balance : 0))
   454→  const openingBal = roundBal(extraMeta?.openingBalance ?? 0)
   455→
   456→  // Tax calculation (CGST/SGST split from total tax)
   457→  const cgstAmount = Math.round(totalTax / 2 * 100) / 100
   458→  const sgstAmount = Math.round((totalTax - cgstAmount) * 100) / 100
   459→  const netAmount = totalDebits - totalDiscount
   460→  const effectiveGstRate = netAmount > 0 ? ((totalTax / netAmount) * 100) : 0
   461→  const halfRate = effectiveGstRate > 0 ? +(effectiveGstRate / 2).toFixed(2) : 0
   462→  // Grand total = debits (tax-inclusive) - discount - credits + refunds
   463→  const grandTotal = roundBal(totalDebits - totalDiscount - totalCredits + totalRefunds)
   464→
   465→  // Payment mode info
   466→  const mostUsedMode = computeMostUsedPaymentMode(entries)
   467→  const lastPayment = paymentEntries.length > 0 ? paymentEntries[paymentEntries.length - 1] : null
   468→  const paymentDate = lastPayment ? fmtDate(lastPayment.date) : invoiceDate
   469→
   470→  // Arrival / departure from entries
   471→  const firstEntry = entries.length > 0 ? entries[0] : null
   472→  const lastEntry = entries.length > 0 ? entries[entries.length - 1] : null
   473→  const arrivalDate = firstEntry ? fmtDate(firstEntry.date) : '—'
   474→  const departureDate = lastEntry ? fmtDate(lastEntry.date) : '—'
   475→
   476→  // ── Build HTML ──
   477→  // P4: Use specificFolioNumber if provided (for proforma/single folio printing)
   478→  const specificFolioNo = extraMeta?.specificFolioNumber
   479→  let folioNo: string
   480→  if (specificFolioNo != null) {
   481→    folioNo = typeof specificFolioNo === 'number' ? fmtFolioNum(specificFolioNo) : String(specificFolioNo)
   482→  } else {
   483→    const rawFolioNo = d.folioNumber || (d.folios && d.folios[0] ? d.folios[0].folioNumber : null) || d.folioNo || null
   484→    folioNo = rawFolioNo != null ? (typeof rawFolioNo === 'number' ? fmtFolioNum(rawFolioNo) : String(rawFolioNo)) : 'Master Folio'
   485→  }
   486→
   487→  const visitNo = d.visitNumber || (d.activeFolio && d.activeFolio.visitNumber) || (d.folios && d.folios[0] && d.folios[0].visitNumber) || d.totalVisits || d.visitNo || '—'
   488→
   489→  // P8: Determine invoice number label based on docTitle
   490→  const invoiceLabel = docTitle.toUpperCase().includes('PROFORMA') ? 'Proforma No' : 'Invoice No'
   491→
   492→  let html = `<div class="inv-wrap">`
   493→
   494→  // 1) HEADER — white background, flex layout with logo
   495→  html += `<div class="inv-header">`

   498→  }
   499→  html += `<div class="header-info">`
   500→  html += `<div class="biz-name">${bizName}</div>`
   501→  if (settings.shopTagline || settings.shopSubtitle) html += `<div class="biz-subtitle">${settings.shopTagline || settings.shopSubtitle}</div>`
   502→  if (bizAddress) html += `<div class="biz-address">${bizAddress}</div>`
   503→  if (bizPhone || bizEmail) html += `<div class="biz-contact">${bizPhone}${bizPhone && bizEmail ? ' | ' : ''}${bizEmail}</div>`
   504→  if (gstNumber && gstEnabled) html += `<div class="biz-gstin">GSTIN: ${gstNumber}</div>`
   505→  html += `</div>`
   506→  html += `</div>`
   507→
   508→  // 2) TITLE BAR
   509→  html += `<div class="inv-title-section">`
   510→  html += `<div class="doc-title">${docTitle}</div>`
   511→  html += `<hr class="title-underline">`
   512→  html += `<div class="meta-row">`
   513→  html += `<span>${invoiceLabel}: <strong>${invoiceNo}</strong></span>`
   514→  html += `<span>Date: <strong>${invoiceDate}</strong></span>`
   515→  // P8: Use CSS counter for page numbers
   516→  html += `<span class="page-counter"></span>`
   517→  html += `</div>`
   518→  html += `</div>`
   519→
   520→  // 3) CUSTOMER INFO TABLE
   521→  html += `<div class="inv-cust-section"><table>`
   522→  html += `<tr>
   523→    <td class="lbl">Customer Id</td><td class="val">${custId || '—'}</td>
   524→    <td class="lbl">Type</td><td class="val">${custType || '—'}</td>
   525→  </tr>`
   526→  html += `<tr>
   527→    <td class="lbl">Folio No</td><td class="val">${folioNo}</td>
   528→    <td class="lbl">Visit No</td><td class="val">${visitNo}</td>

   533→  </tr>`
   534→  html += `<tr>
   535→    <td class="lbl">Mobile</td><td class="val">${custMobile || '—'}</td>
   536→    <td class="lbl">Email</td><td class="val">${custEmail || '—'}</td>

   539→  html += `<tr>
   540→    <td class="lbl">GST No</td><td class="val" style="font-family:'Courier New',monospace;font-size:9px">${custGst || '—'}</td>

   544→  html += `<tr>
   545→    <td class="lbl">Arrival</td><td class="val">${arrivalDate}</td>

   812→// ─── Print Voucher / Payment History ───
   813→// P5: Uses invoiceNo from payment entries (VOU-xxxx) as permanent voucher number
   814→export async function printVoucherHistory(
   815→  detail: CustomerDetail,
   816→  paymentEntries: LedgerEntry[],
   817→  hdrRef: RefObject<Record<string, string>>,
   818→  printedBy?: string
   819→) {
   820→  if (!detail) return
   821→  if (paymentEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No payment entries to print' }); return }
   822→  const shopSettings = await fetchSettingsForPrint(hdrRef)
   823→  const now = new Date()
   824→
   825→  const sorted = [...paymentEntries].sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
   826→  const totalAmount = sorted.reduce((s, e) => s + (e.credit || 0), 0)
   827→
   828→  // P5: Use first payment's invoiceNo (VOU-xxxx) as permanent voucher number
   829→  // Fallback to docNo if invoiceNo is empty (for entries created before this fix)
   830→  const voucherNo = sorted[0]?.invoiceNo || sorted[0]?.docNo || getStableDocId(sorted)
   831→
   832→  // Build entries in the format expected by buildHotelStyleInvoice
   833→  const voucherEntries = sorted.map(e => ({
   834→    ...e,
   835→    debit: 0,
   836→    credit: e.credit || 0,
   837→    balance: 0,
   838→    type: 'Payment',
   839→    description: e.description || `Payment via ${e.paymentMode || 'Cash'}`,
   840→  }))
   841→
   842→  // Compute running balance for display
   843→  let runBal = 0
   844→  for (const e of voucherEntries) {
   845→    runBal -= (e.credit || 0)
   846→    e.balance = Math.round(runBal * 100) / 100
   847→  }
   848→
   849→  const w = openPrintWindow(`Payment Voucher - ${detail.fullName}`)
   850→  if (!w) return
   851→
   852→  w.document.write(
   853→    `<!DOCTYPE html><html><head><title>Payment Voucher</title><style>${INVOICE_CSS}</style><style>${buildTemplateCSSFromSettings(shopSettings)}</style></head><body>` +
   854→    buildHotelStyleInvoice(shopSettings, detail, voucherEntries, voucherNo, fmtDate(now.toISOString()), 'PAYMENT VOUCHER', {
   855→      openingBalance: 0,
   856→      closingBalance: -totalAmount,
   857→    }, printedBy) +
   858→    `</body></html>`
   859→  )
   860→  w.document.close()
   861→  w.print()
   862→}
