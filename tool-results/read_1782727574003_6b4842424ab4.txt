   244→function formatDate(dateStr: string | null): string {
   245→  if (!dateStr) return '—';
   246→  try {
   247→    return new Date(dateStr).toLocaleDateString('en-IN', {
   248→      day: '2-digit', month: 'short', year: 'numeric',
   249→    });
   250→  } catch {
   251→    return '—';
   252→  }
   253→}
   254→
   255→function formatDateTime(dateStr: string | null): string {
   256→  if (!dateStr) return 'Never';
   257→  try {
   258→    return new Date(dateStr).toLocaleDateString('en-IN', {
   259→      day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   260→    });
   261→  } catch {
   262→    return '—';
   263→  }
   264→}

   266→function isPermGroup(obj: unknown): obj is Record<string, boolean> {
   267→  return typeof obj === 'object' && obj !== null && !Array.isArray(obj);
