    68→    68→    68→// ─── Helper: Compute the most-used payment mode from credit entries ───
    69→    69→    69→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
    70→    70→    70→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    71→    71→    71→  if (creditEntries.length === 0) return 'N/A'
    72→    72→    72→  const modeCounts: Record<string, number> = {}
    73→    73→    73→  for (const e of creditEntries) {
    74→    74→    74→    const mode = e.paymentMode || 'Cash'
    75→    75→    75→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
    76→    76→    76→  }
    77→    77→    77→  let bestMode = 'Cash'
    78→    78→    78→  let bestAmount = 0
    79→    79→    79→  for (const [mode, amount] of Object.entries(modeCounts)) {
    80→    80→    80→    if (amount > bestAmount) {
    81→    81→    81→      bestAmount = amount
    82→    82→    82→      bestMode = mode
    83→    83→    83→    }
    84→    84→    84→  }
    85→    85→    85→  return bestMode
    86→    86→    86→}

    89→    89→    89→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
    90→    90→    90→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    91→    91→    91→  if (creditEntries.length === 0) {
    92→    92→    92→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
    93→    93→    93→  }
    94→    94→    94→
    95→    95→    95→  const modeTotals: Record<string, number> = {}
    96→    96→    96→  for (const e of creditEntries) {
    97→    97→    97→    const mode = e.paymentMode || 'Cash'
    98→    98→    98→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
    99→    99→    99→  }
   100→   100→   100→
   101→   101→   101→  const modes = Object.entries(modeTotals)
   102→   102→   102→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   103→   103→   103→
   104→   104→   104→  if (modes.length === 1) {
   105→   105→   105→    const [mode, amount] = modes[0]
   106→   106→   106→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   107→   107→   107→  }
   108→   108→   108→
   109→   109→   109→  const rows = modes.map(([mode, amount]) => {
   110→   110→   110→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   111→   111→   111→  }).join('')
   112→   112→   112→
   113→   113→   113→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   114→   114→   114→}

   200→   200→   200→export async function printReceipt(
   201→   201→   201→  detail: CustomerDetail,
   202→   202→   202→  entries: LedgerEntry[],
   203→   203→   203→  hdrRef: RefObject<Record<string, string>>

   207→   207→   207→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }

   210→   210→   210→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   211→   211→   211→  const totalTax = activeEntries.reduce((s, e) => s + (e.tax || 0), 0)
   212→   212→   212→  const totalDiscounts = activeEntries.reduce((s, e) => s + (e.discount || 0), 0)

   220→   220→   220→  // Use stable doc ID from entries — never regenerate

   222→   222→   222→  const authorizedSignatory = (settings as Record<string, string>).authorizedSignatory || ''
   223→   223→   223→  const gstAddr = (detail as any).gstAddress || ''

   229→   229→   229→    `<!DOCTYPE html><html><head><title>Folio Invoice</title><style>${PRINT_CSS}</style></head><body>` +

   233→   233→   233→      <div style="text-align:center"><div class="inv-label">Date</div><div class="inv-value">${fmtDate(now.toISOString())}</div></div>
   234→   234→   234→      <div style="text-align:right"><div class="inv-label">Status</div><div class="inv-value">${detail.status === 'CheckedOut' ? 'Checked Out' : detail.status}</div></div>
   235→   235→   235→    </div>` +

   239→   239→   239→    `<hr class="divider">` +
   240→   240→   240→    `<table><tr><th>#</th><th>Date</th><th>TXN ID</th><th>Type</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
