   206→   245→  if (!dateStr) return '—';
   207→   246→  try {
   208→   247→    return new Date(dateStr).toLocaleDateString('en-IN', {
   209→   248→      day: '2-digit', month: 'short', year: 'numeric',
   210→   249→    });
   211→   250→  } catch {
   212→   251→    return '—';
   213→   252→  }
   214→   253→}

   219→   258→    return new Date(dateStr).toLocaleDateString('en-IN', {
   220→   259→      day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   221→   260→    });
   222→   261→  } catch {
   223→   262→    return '—';
   224→   263→  }
