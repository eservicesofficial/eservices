    11→    11→    11→function numberToIndianWords(num: number): string {
    12→    12→    12→  if (num === 0) return 'ZERO'
    13→    13→    13→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    14→    14→    14→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    15→    15→    15→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    16→    16→    16→  let words = ''
    17→    17→    17→  function convertBelowThousand(n: number): string {
    18→    18→    18→    let w = ''
    19→    19→    19→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    20→    20→    20→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    21→    21→    21→    else if (n > 0) { w += ones[n] }
    22→    22→    22→    return w
    23→    23→    23→  }
    24→    24→    24→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    25→    25→    25→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    26→    26→    26→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    27→    27→    27→  if (num > 0) { words += convertBelowThousand(num) }
    28→    28→    28→  return words.trim().toUpperCase() + ' ONLY'
    29→    29→    29→}

   193→   193→   193→// ─── Helper: Compute the most-used payment mode from credit entries ───
   194→   194→   194→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   195→   195→   195→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   196→   196→   196→  if (creditEntries.length === 0) return 'N/A'
   197→   197→   197→  const modeCounts: Record<string, number> = {}
   198→   198→   198→  for (const e of creditEntries) {
   199→   199→   199→    const mode = e.paymentMode || 'Cash'
   200→   200→   200→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   201→   201→   201→  }
   202→   202→   202→  let bestMode = 'Cash'
   203→   203→   203→  let bestAmount = 0
   204→   204→   204→  for (const [mode, amount] of Object.entries(modeCounts)) {
   205→   205→   205→    if (amount > bestAmount) {
   206→   206→   206→      bestAmount = amount
   207→   207→   207→      bestMode = mode
   208→   208→   208→    }
   209→   209→   209→  }
   210→   210→   210→  return bestMode
   211→   211→   211→}
