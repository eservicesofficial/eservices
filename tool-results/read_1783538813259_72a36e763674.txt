    11→    11→function numberToIndianWords(num: number): string {
    12→    12→  if (num === 0) return 'ZERO'
    13→    13→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    14→    14→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    15→    15→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    16→    16→  let words = ''
    17→    17→  function convertBelowThousand(n: number): string {
    18→    18→    let w = ''
    19→    19→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    20→    20→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    21→    21→    else if (n > 0) { w += ones[n] }
    22→    22→    return w
    23→    23→  }
    24→    24→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    25→    25→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    26→    26→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    27→    27→  if (num > 0) { words += convertBelowThousand(num) }
    28→    28→  return words.trim().toUpperCase() + ' ONLY'
    29→    29→}

   188→   188→// ─── Helper: Compute the most-used payment mode from credit entries ───
   189→   189→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   190→   190→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   191→   191→  if (creditEntries.length === 0) return 'N/A'
   192→   192→  const modeCounts: Record<string, number> = {}
   193→   193→  for (const e of creditEntries) {
   194→   194→    const mode = e.paymentMode || 'Cash'
   195→   195→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   196→   196→  }
   197→   197→  let bestMode = 'Cash'
   198→   198→  let bestAmount = 0
   199→   199→  for (const [mode, amount] of Object.entries(modeCounts)) {
   200→   200→    if (amount > bestAmount) {
   201→   201→      bestAmount = amount
   202→   202→      bestMode = mode
   203→   203→    }
   204→   204→  }
   205→   205→  return bestMode
   206→   206→}

   208→   208→// ─── Helper: Build payment summary HTML ───
   209→   209→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   210→   210→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   211→   211→  if (creditEntries.length === 0) {
   212→   212→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   213→   213→  }
   214→   214→
   215→   215→  const modeTotals: Record<string, number> = {}
   216→   216→  for (const e of creditEntries) {
   217→   217→    const mode = e.paymentMode || 'Cash'
   218→   218→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   219→   219→  }
   220→   220→
   221→   221→  const modes = Object.entries(modeTotals)
   222→   222→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   223→   223→
   224→   224→  if (modes.length === 1) {
   225→   225→    const [mode, amount] = modes[0]
   226→   226→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   227→   227→  }
   228→   228→
   229→   229→  const rows = modes.map(([mode, amount]) => {
   230→   230→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   231→   231→  }).join('')
   232→   232→
   233→   233→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   234→   234→}

   236→   236→// ─── Helper: Get stable document number from entries (never regenerate) ───
   237→   237→// Searches ALL entries for the first non-empty docNo to avoid N/A
   238→   238→function getStableDocId(entries: { docNo?: string; invoiceNo?: string; id?: string; createdAt?: string }[]): string {
   239→   239→  if (entries.length === 0) return `DOC-${Date.now()}`
   240→   240→  for (const e of entries) {
   241→   241→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   242→   242→  }
   243→   243→  for (const e of entries) {
   244→   244→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   245→   245→  }
   246→   246→  // Fallback: use entry ID truncated or timestamp — never show N/A
   247→   247→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   248→   248→  return fallback
   249→   249→}
