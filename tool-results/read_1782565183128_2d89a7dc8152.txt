   236→function ChangePasswordSection() {
   237→  const { currentUser, token } = useAuth();
   238→  const [currentPassword, setCurrentPassword] = useState('');
   239→  const [newPassword, setNewPassword] = useState('');
   240→  const [confirmPassword, setConfirmPassword] = useState('');
   241→  const [showCurrent, setShowCurrent] = useState(false);
   242→  const [showNew, setShowNew] = useState(false);
   243→  const [showConfirm, setShowConfirm] = useState(false);
   244→  const [loading, setLoading] = useState(false);
   245→
   246→  const handleChangePassword = async () => {
   247→    if (!currentPassword || !newPassword || !confirmPassword) {
   248→      notify({ type: 'error', title: 'Error', message: 'All fields are required' });
   249→      return;
   250→    }
   251→    if (newPassword.length < 4) {
   252→      notify({ type: 'error', title: 'Error', message: 'New password must be at least 4 characters' });
   253→      return;
   254→    }
   255→    if (newPassword !== confirmPassword) {
   256→      notify({ type: 'error', title: 'Error', message: 'New passwords do not match' });
   257→      return;
   258→    }
   259→    setLoading(true);
   260→    try {
   261→      const res = await fetch(`/api/users/${currentUser?.userId}`, {
   262→        method: 'PUT',
   263→        headers: {
   264→          'Content-Type': 'application/json',
   265→          'Authorization': `Bearer ${token}`,
   266→        },
   267→        body: JSON.stringify({
   268→          currentPassword,
   269→          password: newPassword,
   270→          _passwordOnly: true,
   271→        }),
   272→      });
   273→      if (res.ok) {
   274→        notify({ type: 'success', title: 'Password Updated', message: 'Password changed successfully' });
   275→        setCurrentPassword('');
   276→        setNewPassword('');
   277→        setConfirmPassword('');
   278→      } else {
   279→        const data = await res.json().catch(() => null);
   280→        notify({ type: 'error', title: 'Error', message: data?.error || 'Failed to change password' });
   281→      }
   282→    } catch {
   283→      notify({ type: 'error', title: 'Error', message: 'Network error' });
   284→    } finally {
   285→      setLoading(false);
   286→    }
   287→  };
   288→
   289→  return (
   290→    <Card className="overflow-hidden border-0 shadow-sm">
   291→      <CardHeader className="pb-2">
   292→        <div className="flex items-center gap-3">
   293→          <div className="flex size-9 items-center justify-center rounded-lg bg-amber-100 text-amber-600 shrink-0">
   294→            <Lock className="size-4" />
   295→          </div>
   296→          <div className="min-w-0">
   297→            <CardTitle className="text-base">Change Password</CardTitle>
   298→            <CardDescription className="mt-0.5">Update your account password</CardDescription>
   299→          </div>
   300→        </div>
   301→      </CardHeader>
   302→      <div className="px-6 pb-6 space-y-4">
   303→        <div className="grid gap-2 border-l-2 border-l-amber-400 pl-3">
   304→          <Label className="text-sm font-medium">Current Password</Label>
   305→          <div className="relative">
   306→            <Input
   307→              type={showCurrent ? 'text' : 'password'}
   308→              placeholder="Enter current password"
   309→              value={currentPassword}
   310→              onChange={(e) => setCurrentPassword(e.target.value)}
   311→              className="pr-10"
   312→            />
   313→            <button type="button" onClick={() => setShowCurrent(!showCurrent)} className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground">
   314→              {showCurrent ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   315→            </button>
   316→          </div>
   317→        </div>
   318→        <div className="grid gap-2 border-l-2 border-l-amber-400 pl-3">
   319→          <Label className="text-sm font-medium">New Password</Label>
   320→          <div className="relative">
   321→            <Input
   322→              type={showNew ? 'text' : 'password'}
   323→              placeholder="Enter new password (min 4 chars)"
   324→              value={newPassword}
   325→              onChange={(e) => setNewPassword(e.target.value)}
   326→              className="pr-10"
   327→            />
   328→            <button type="button" onClick={() => setShowNew(!showNew)} className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground">
   329→              {showNew ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   330→            </button>
   331→          </div>
   332→        </div>
   333→        <div className="grid gap-2 border-l-2 border-l-amber-400 pl-3">
   334→          <Label className="text-sm font-medium">Confirm New Password</Label>
   335→          <div className="relative">
   336→            <Input
   337→              type={showConfirm ? 'text' : 'password'}
   338→              placeholder="Confirm new password"
   339→              value={confirmPassword}
   340→              onChange={(e) => setConfirmPassword(e.target.value)}
   341→              className="pr-10"
   342→            />
   343→            <button type="button" onClick={() => setShowConfirm(!showConfirm)} className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground">
   344→              {showConfirm ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   345→            </button>
   346→          </div>
   347→        </div>
   348→        <div className="flex justify-end">
   349→          <Button onClick={handleChangePassword} disabled={loading} className="gap-1.5 bg-amber-600 hover:bg-amber-700 text-white">
   350→            {loading ? <Loader2 className="size-4 animate-spin" /> : <><Lock className="size-4" />Update Password</>}
   351→          </Button>
   352→        </div>
   353→      </div>
   354→    </Card>
   355→  );
   356→}
