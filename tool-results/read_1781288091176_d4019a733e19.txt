   168→   168→  // ─── Internal State ───

   178→   178→
   179→   179→  // ─── History state ───
   180→   180→  const [historyLoading, setHistoryLoading] = React.useState(false)
   181→   181→  const [checkedOutCustomers, setCheckedOutCustomers] = React.useState<CustomerHistory[]>([])
   182→   182→  const [expandedFolioCustomer, setExpandedFolioCustomer] = React.useState<string | null>(null)
   183→   183→  const [expandedFolioEntries, setExpandedFolioEntries] = React.useState<Record<string, any[]>>({})
   184→   184→
   185→   185→  // Fetch checked-out customers for history
   186→   186→  React.useEffect(() => {
   187→   187→    if (historyOpen) {
   188→   188→      fetchCheckedOutHistory()
   189→   189→    }
   190→   190→  }, [historyOpen])
   191→   191→
   192→   192→  const fetchCheckedOutHistory = React.useCallback(async () => {
   193→   193→    try {
   194→   194→      setHistoryLoading(true)
   195→   195→      const res = await fetch('/api/folio/customers?status=CheckedOut&limit=100', { headers: apiHeaders })
   196→   196→      if (res.ok) {
   197→   197→        const data = await res.json()
   198→   198→        const list = Array.isArray(data) ? data : (data.customers || [])
   199→   199→        // Fetch history for each customer
   200→   200→        const histories = await Promise.all(
   201→   201→          list.slice(0, 20).map(async (c: any) => {
   202→   202→            try {
   203→   203→              const hRes = await fetch(`/api/folio/customers/${c.id}/history`, { headers: apiHeaders })
   204→   204→              if (hRes.ok) return await hRes.json() as CustomerHistory
   205→   205→            } catch { /* ignore */ }

   207→   207→          })

   213→   213→

   217→   217→      setExpandedFolioEntries(prev => { const n = { ...prev }; delete n[key]; return n })

   219→   219→      const customer = checkedOutCustomers.find(c => c.customer.id === customerId)
   220→   220→      const folio = customer?.historicalFolios.find(f => f.folioId === folioId)
   221→   221→      if (folio) {
   222→   222→        setExpandedFolioEntries(prev => ({ ...prev, [key]: (folio.entries || folio.ledgerEntries || []) }))
   223→   223→      }
   224→   224→    }
   225→   225→  }
   226→   226→
   227→   227→  // ─── Track checkout transition ───

   231→   231→      setPostCheckoutOpen(true)
   232→   232→    }

   236→   236→  // ─── Show TXN number when lastCreatedDocNo changes ───

   242→   242→    }
   243→   243→  }, [lastCreatedDocNo, lastCreatedType])
   244→   244→
   245→   245→  // ─── Fetch invoice number ───

   247→   247→    if (!detail || !apiHeaders) return

   249→   249→      setInvNumLoading(true)
   250→   250→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   251→   251→        method: 'POST',
   252→   252→        headers: apiHeaders,
   253→   253→      })
   254→   254→      if (res.ok) {
   255→   255→        const data = await res.json()
   256→   256→        setInvoiceNumber(data.invoiceNumber)
   257→   257→      }
   258→   258→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   259→   259→  }, [detail?.id, apiHeaders])
   260→   260→
   261→   261→  // ─── Fetch invoice number when invoice dialog opens ───

   263→   263→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber) {
   264→   264→      fetchInvoiceNumber()
   265→   265→    }
   266→   266→  }, [invoiceOpen, detail?.id, apiHeaders])
   267→   267→
   268→   268→  React.useEffect(() => {
   269→   269→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   270→   270→      fetchInvoiceData()
   271→   271→    }
   272→   272→  }, [invoiceOpen])
   273→   273→
   274→   274→  // ─── Entry creation with TXN tracking ───
   275→   275→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   276→   276→    if (!detail || !selectedFolioId || !apiHeaders) {
   277→   277→      return null
   278→   278→    }
   279→   279→    try {
   280→   280→      setLocalSubmitting(true)
   281→   281→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {

   283→   283→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },

   303→   303→    } finally {

   333→   333→        onRefresh?.()
   334→   334→        setTimeout(() => setEntrySuccess(null), 4000)
   335→   335→      } else {
   336→   336→        const e = await res.json()
   337→   337→        notify({ type: 'error', title: 'Error', message: e.error || 'Failed' })
   338→   338→      }
   339→   339→    } catch {
   340→   340→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   341→   341→    } finally {
   342→   342→      setLocalSubmitting(false)
   343→   343→    }

   358→   358→          targetCustomerId: transferForm.targetCustomerId,
   359→   359→          amount: parseFloat(transferForm.amount),
   360→   360→          description: transferForm.description,
   361→   361→        }),
   362→   362→      })

   364→   364→        const data = await res.json()
   365→   365→        const docNo = data?.source?.entry?.docNo || ''

   368→   368→        closeDialogs()
   369→   369→        onRefresh?.()
   370→   370→        setTimeout(() => setEntrySuccess(null), 4000)
   371→   371→      } else {
   372→   372→        const e = await res.json()
   373→   373→        notify({ type: 'error', title: 'Error', message: e.error || 'Failed' })
   374→   374→      }
   375→   375→    } catch {
   376→   376→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   377→   377→    } finally {
   378→   378→      setLocalSubmitting(false)
   379→   379→    }
   380→   380→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   381→   381→
   382→   382→  // ─── Split payment helpers for checkout ───
   383→   383→  const addSplitItem = () => {
   384→   384→    setLocalSplitItems(prev => [...prev, { method: 'Cash', amount: '', reference: '' }])
   385→   385→  }
   386→   386→  const removeSplitItem = (idx: number) => {
   387→   387→    setLocalSplitItems(prev => prev.filter((_, i) => i !== idx))
   388→   388→  }
   389→   389→  const updateSplitItem = (idx: number, field: string, value: string) => {
   390→   390→    setLocalSplitItems(prev => prev.map((item, i) => i === idx ? { ...item, [field]: value } : item))
   391→   391→  }
   392→   392→  const totalSplitAmount = localSplitItems.reduce((sum, item) => sum + (parseFloat(item.amount) || 0), 0)
   393→   393→  const splitRemaining = checkoutSettlementData ? Math.max(0, checkoutSettlementData.totalOutstanding - totalSplitAmount) : 0
   394→   394→
   395→   395→  // ─── Computed GST data from invoiceData ───
   396→   396→  const gstData = React.useMemo(() => {
   397→   397→    if (!invoiceData?.entries) return null
