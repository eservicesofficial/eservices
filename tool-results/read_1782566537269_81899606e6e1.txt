    68→// ─── Helper: Compute the most-used payment mode from credit entries ───
    69→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
    70→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    71→  if (creditEntries.length === 0) return 'N/A'
    72→  const modeCounts: Record<string, number> = {}
    73→  for (const e of creditEntries) {
    74→    const mode = e.paymentMode || 'Cash'
    75→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
    76→  }
    77→  let bestMode = 'Cash'
    78→  let bestAmount = 0
    79→  for (const [mode, amount] of Object.entries(modeCounts)) {
    80→    if (amount > bestAmount) {
    81→      bestAmount = amount
    82→      bestMode = mode
    83→    }
    84→  }
    85→  return bestMode
    86→}

    88→// ─── Helper: Build payment summary HTML ───
    89→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
    90→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    91→  if (creditEntries.length === 0) {
    92→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
    93→  }
    94→
    95→  const modeTotals: Record<string, number> = {}
    96→  for (const e of creditEntries) {
    97→    const mode = e.paymentMode || 'Cash'
    98→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
    99→  }
   100→
   101→  const modes = Object.entries(modeTotals)
   102→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   103→
   104→  if (modes.length === 1) {
   105→    const [mode, amount] = modes[0]
   106→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   107→  }
   108→
   109→  const rows = modes.map(([mode, amount]) => {
   110→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   111→  }).join('')
   112→
   113→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   114→}

   131→  documentTitle: string,
   132→  txnId: string
   133→): string {
   134→  const bizName = settings.shopName || 'Eservices'
   135→  const bizAddress = settings.shopAddress || ''
   136→  const bizPhone = settings.shopPhone || ''
   137→  const bizEmail = settings.shopEmail || ''
   138→  const gstNumber = settings.gstNumber || settings.gstin || ''
   139→  const shopLogo = settings.shopLogo || ''
   140→

   191→// ─── Print Receipt ───
   192→export async function printReceipt(
   193→  detail: CustomerDetail,
   194→  entries: LedgerEntry[],
   195→  hdrRef: RefObject<Record<string, string>>
   196→) {
   197→  if (!detail) return
   198→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   199→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   200→  const settings = await fetchSettingsForPrint(hdrRef)
   201→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   202→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   203→  const totalTax = activeEntries.reduce((s, e) => s + (e.tax || 0), 0)
   204→  const totalDiscounts = activeEntries.reduce((s, e) => s + (e.discount || 0), 0)
   205→  const finalBal = activeEntries[activeEntries.length - 1].balance
   206→  const now = new Date()
   207→  const rows = buildEntryRows(activeEntries)
   208→
   209→  const txnId = await fetchTxnId(detail.id)
   210→
   211→  const w = openPrintWindow(`Receipt - ${detail.fullName}`)
   212→  if (!w) return
   213→
   214→  w.document.write(
   215→    `<!DOCTYPE html><html><head><title>Receipt</title><style>${PRINT_CSS}</style></head><body>` +
   216→    buildUnifiedHeader(settings, 'PAYMENT RECEIPT', txnId) +
   217→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +
   218→    `<div class="datetime">${detail.status === 'CheckedOut' ? 'Status: Checked Out' : `Status: ${detail.status}`}</div>` +
   219→    `<hr class="divider">` +
   220→    `<table><tr><th>#</th><th>Date</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   221→    `<tr class="totals"><td colspan="3" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(finalBal)}</td></tr></table>` +
   222→    buildPaymentSummary(activeEntries) +
   223→    `<div class="tax-summary"><div class="tax-item"><span class="tax-label">Total Tax: </span><span class="tax-val">${fmtINR(totalTax)}</span></div><div class="tax-item"><span class="tax-label">Discounts: </span><span class="tax-val">${fmtINR(totalDiscounts)}</span></div><div class="tax-item"><span class="tax-label">Entries: </span><span class="tax-val">${activeEntries.length}</span></div></div>` +
   224→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Final Balance</div><div class="s-value" style="color:${finalBal >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(finalBal)}</div></div></div>` +
   225→    buildFooter() +
   226→    `</body></html>`
   227→  )
   228→  w.document.close()
   229→  w.print()
   230→}

   393→async function fetchShopSettings(hdrRef?: RefObject<Record<string, string>>): Promise<ShopSettings> {
   394→  try {

   398→      const db = (await res.json()) as Record<string, string>

   411→function openThermalPrintWindow(thermalHtml: string, title: string): Window | null {

   425→// ─── Thermal Print Receipt ───
   426→export async function thermalPrintReceipt(
   427→  detail: CustomerDetail,
   428→  entries: LedgerEntry[],
   429→  hdrRef: RefObject<Record<string, string>>
   430→) {
   431→  if (!detail) return
   432→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   433→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   434→
   435→  const [tConfig, shop] = await Promise.all([
   436→    fetchThermalSettings(),
   437→    fetchShopSettings(hdrRef),
   438→  ])
   439→
   440→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   441→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   442→  const finalBal = activeEntries[activeEntries.length - 1].balance
   443→
   444→  const items: ThermalReceiptData['items'] = activeEntries.map(e => ({
   445→    name: stripInternalDescriptions(e.description || e.type || '—'),
   446→    qty: 1,
   447→    rate: e.debit > 0 ? e.debit : e.credit,
   448→    total: e.debit > 0 ? e.debit : e.credit,
   449→  }))
   450→
   451→  const actualPaymentMethod = computeMostUsedPaymentMode(activeEntries)
   452→
   453→  const receiptData: ThermalReceiptData = {
   454→    shopName: shop.shopName,
   455→    shopAddress: shop.shopAddress,
   456→    shopPhone: shop.shopPhone,
   457→    shopGst: shop.shopGst,
   458→    shopGstState: shop.shopGstState,
   459→    showGst: false,
   460→    invoiceNumber: detail.customerId,
   461→    date: fmtDate(new Date().toISOString()),
   462→    orderType: 'PAYMENT RECEIPT',
   463→    customerName: detail.fullName,
   464→    customerPhone: detail.mobile,
   465→    customerAddress: '',
   466→    items,
   467→    subtotal: totalDebits,
   468→    cgst: 0,
   469→    sgst: 0,
   470→    gstRate: 0,
   471→    deliveryCharge: 0,
   472→    discountAmount: 0,
   473→    grandTotal: totalDebits - totalCredits,
   474→    paidAmount: totalCredits,
   475→    balanceAmount: finalBal,
   476→    paymentMethod: actualPaymentMethod,
   477→    billStatus: finalBal <= 0 ? 'Paid' : 'Balance',
   478→  }
   479→
   480→  const thermalHtml = buildRichThermalHtml(receiptData, tConfig, `Receipt - ${detail.fullName}`)
   481→  openThermalPrintWindow(thermalHtml, `Receipt - ${detail.fullName}`)
   482→}
