   169→  detail, allFolios, selectedEntries, submitting, settlingMode,

   189→  handleRemove, handleCheckout, handleDelete, handleAddDoc, handleAddNote,
   190→  handleToggleTag, handleDeleteDoc, handleDeleteNote,
   191→  fetchInvoiceData, fetchFolioStatement, printInvoice, printCompleteInvoice, printSingleFolio, printVoucherHistory,
   192→  closeDialogs, searchTransferTargets,
   193→  deleteFolioOpen, folioToDelete,
   194→  setDeleteFolioOpen, handleDeleteFolio,
   195→  // Enhanced props
   196→  apiHeaders, onRefresh, canDelete = false, isAdmin = false, onPrintReceipt, onThermalPrintReceipt,
   197→  onThermalPrintInvoice, onThermalPrintVoucher,
   198→  lastCreatedDocNo, lastCreatedType,
   199→  // History
   200→  historyOpen, setHistoryOpen, onPrintHistoryFolio,
   201→}: DialogsProps) {
   202→
   203→  // ─── Internal State ───
   204→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   205→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   206→
   207→  // Clear invoice number when customer changes to prevent stale numbers
   208→  React.useEffect(() => {
   209→    setInvoiceNumber('')
   210→  }, [detail?.id])
   211→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   212→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   213→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   214→  // ─── History state ───
   215→  const [historyLoading, setHistoryLoading] = React.useState(false)
   216→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   217→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   218→  const [selectedVoucherIds, setSelectedVoucherIds] = React.useState<Set<string>>(new Set())
   219→  const [editingInvoiceId, setEditingInvoiceId] = React.useState<string | null>(null)
   220→  const [editInvoiceValue, setEditInvoiceValue] = React.useState('')
   221→  const [savingInvoiceNum, setSavingInvoiceNum] = React.useState(false)
   222→
   223→  // Fetch history for the CURRENT customer (all records, no date filter)
   224→  const fetchCustomerHistory = React.useCallback(async () => {
   225→    if (!detail || !apiHeaders) return
   226→    try {
   227→      setHistoryLoading(true)
   228→      const res = await fetch(`/api/folio/customers/${detail.id}/history`, { headers: apiHeaders })
   229→      if (res.ok) {
   230→        const data = await res.json() as CustomerHistory
   231→        setCustomerHistory(data)
   232→      }
   233→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to fetch checkout history' }) } finally { setHistoryLoading(false) }
   234→  }, [detail?.id, apiHeaders])
   235→
   236→  React.useEffect(() => {
   237→    if (historyOpen && detail && apiHeaders && fetchCustomerHistory) {
   238→      fetchCustomerHistory()
   239→    } else if (!historyOpen) {
   240→      setCustomerHistory(null)
   241→      setExpandedFolioId(null)
   242→    }
   243→  }, [historyOpen, detail?.id, fetchCustomerHistory, apiHeaders])
   244→
   245→  // ─── Print helpers for checkout records ───
   246→  const getTplCSS = (s: Record<string, string>) => {
   247→    const tid = s.invoiceTemplate || 'modern'
   248→    let cust: InvoiceCustomizations = {}
   249→    try { if (s.invoiceCustomizations) cust = JSON.parse(s.invoiceCustomizations) } catch { /* */ }
   250→    return getTemplateCSS(tid, cust)
   251→  }
   252→
   253→  // Build frozen customer detail from checkout snapshot (P2: never use live data for reprint)
   254→  const buildFrozenDetail = (co: CheckoutRecord): CustomerDetail => {
   255→    const base = detail || {} as CustomerDetail
   256→    return {
   257→      ...base,
   258→      fullName: co.customerName || base.fullName,
   259→      mobile: co.customerMobile || base.mobile,
   260→      email: co.customerEmail || base.email,
   261→      address: co.customerAddress || base.address,
   262→      customerType: co.customerType || base.customerType,
   263→      customerId: co.customerIdRef || base.customerId,
   264→      gstNumber: co.customerGstNumber || base.gstNumber || '',
   265→      gstAddress: co.customerGstAddress || base.gstAddress || '',
   266→      panNumber: co.customerPanNumber || base.panNumber || '',
   267→      companyName: co.customerCompanyName || base.companyName || '',
   268→      totalVisits: co.visitNumber || base.totalVisits,
   269→    } as CustomerDetail
   270→  }
   271→
   272→  const printCheckoutFolioBill = async (co: CheckoutRecord) => {
   273→    if (!co.transactions.length) return
   274→    let shopSettings: Record<string, string> = {}
   275→    try {
   276→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   277→    } catch { /* use defaults */ }
   278→    const tplCss = getTplCSS(shopSettings)
   279→    const printDetail = buildFrozenDetail(co)
   280→    const invNo = co.customInvoiceNumber || co.invoiceNumber || co.checkoutNumber
   281→    const w = window.open('', '_blank')
   282→    if (!w) return
   283→    w.document.write(`<!DOCTYPE html><html><head><title>Folio Bill - ${invNo}</title><style>${INVOICE_CSS}</style><style>${tplCss}</style></head><body>${buildHotelStyleInvoice(shopSettings, printDetail, co.transactions, invNo, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'FOLIO BILL', undefined, co.checkoutUser)}</body></html>`)
   284→    w.document.close(); w.print()
   285→  }
   286→
   287→  const printCheckoutInvoice = async (co: CheckoutRecord) => {
   288→    if (!co.transactions.length) return
   289→    let shopSettings: Record<string, string> = {}
   290→    try {
   291→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   292→    } catch { /* use defaults */ }
   293→    const tplCss = getTplCSS(shopSettings)
   294→    const printDetail = buildFrozenDetail(co)
   295→    const invNo = co.customInvoiceNumber || co.invoiceNumber || co.checkoutNumber
   296→    const w = window.open('', '_blank')
   297→    if (!w) return
   298→    w.document.write(`<!DOCTYPE html><html><head><title>Tax Invoice - ${invNo}</title><style>${INVOICE_CSS}</style><style>${tplCss}</style></head><body>${buildHotelStyleInvoice(shopSettings, printDetail, co.transactions, invNo, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'Tax Invoice', { openingBalance: co.openingBalance, closingBalance: co.closingBalance }, co.checkoutUser)}</body></html>`)
   299→    w.document.close(); w.print()
   300→  }
   301→
   302→  const saveCustomInvoiceNumber = async (co: CheckoutRecord) => {
   303→    if (!detail || !apiHeaders) return
   304→    setSavingInvoiceNum(true)
   305→    try {
   306→      const res = await fetch(`/api/folio/customers/${detail.id}/history`, {
   307→        method: 'PATCH',
   308→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   309→        body: JSON.stringify({ checkoutRecordId: co.id, customInvoiceNumber: editInvoiceValue }),
   310→      })
   311→      if (res.ok) {
   312→        notify({ type: 'success', title: 'Saved', message: 'Invoice number updated' })
   313→        setEditingInvoiceId(null)
   314→        fetchCustomerHistory()
   315→      } else {
   316→        notify({ type: 'error', title: 'Error', message: 'Failed to save' })
   317→      }
   318→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to save' }) }
   319→    finally { setSavingInvoiceNum(false) }
   320→  }
   321→
   322→  // ─── Track checkout transition ───
   323→  const wasCheckedOut = React.useRef(isCheckedOut)
   324→  React.useEffect(() => {
   325→    if (isCheckedOut && !wasCheckedOut.current) {
   326→      setPostCheckoutOpen(true)
   327→    }
   328→    wasCheckedOut.current = isCheckedOut
   329→  }, [isCheckedOut])
   330→
   331→  // ─── Show TXN number when lastCreatedDocNo changes ───
   332→  React.useEffect(() => {
   333→    if (lastCreatedDocNo && lastCreatedType) {
   334→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   335→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   336→      return () => clearTimeout(timer)
   337→    }
   338→  }, [lastCreatedDocNo, lastCreatedType])
   339→
   340→  // ─── Items for charge selector (Req 5) ───
   341→  const [items, setItems] = React.useState<{ id: string; name: string; price: number }[]>([])
   342→  const [itemsLoading, setItemsLoading] = React.useState(false)
   343→  React.useEffect(() => {
   344→    if (chargeOpen && apiHeaders) {
   345→      setItemsLoading(true)
   346→      fetch('/api/items', { headers: apiHeaders }).then(r => r.ok ? r.json() : []).then((list: any[]) => {
   347→        setItems((list || []).map((it: any) => ({ id: it.id, name: it.name, price: it.price || 0 })))
   348→      }).catch(() => {}).finally(() => setItemsLoading(false))
   349→    }
   350→  }, [chargeOpen, apiHeaders])
   351→
   352→  // Handle item selection
   353→  const handleItemSelect = React.useCallback((itemId: string) => {
   354→    const item = items.find(it => it.id === itemId)
   355→    if (item) {
   356→      setChargeForm({
   357→        ...chargeForm,
   358→        selectedItem: itemId,
   359→        selectedItemPrice: String(item.price),
   360→        description: item.name,
   361→        amount: item.price > 0 ? String(item.price) : chargeForm.amount,
   362→      })
   363→    }
   364→  }, [items, chargeForm])
   365→
   366→  // ─── Fetch invoice number ───
   367→  const fetchInvoiceNumber = React.useCallback(async () => {
   368→    if (!detail || !apiHeaders) return
   369→    try {
   370→      setInvNumLoading(true)
   371→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   372→        method: 'POST',
   373→        headers: apiHeaders,
   374→      })
   375→      if (res.ok) {
   376→        const data = await res.json()
   377→        setInvoiceNumber(data.invoiceNumber)
   378→      }
   379→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   380→  }, [detail?.id, apiHeaders])
   381→
   382→  // ─── Fetch invoice number when CHECKOUT dialog opens only ───
   383→  // Invoice dialog uses getStableDocId from existing entries (no number consumed)
   384→  React.useEffect(() => {
   385→    if (checkoutOpen && detail && apiHeaders && !invoiceNumber && fetchInvoiceNumber) {
   386→      fetchInvoiceNumber()
   387→    }
   388→  }, [checkoutOpen, detail?.id, apiHeaders, invoiceNumber, fetchInvoiceNumber])
   389→
   390→  React.useEffect(() => {
   391→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   392→      fetchInvoiceData()
   393→    }
   394→  }, [invoiceOpen, invoiceData, detail?.id, selectedFolioId])
   395→
   396→  // ─── Entry creation with TXN tracking ───
   397→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   398→    if (!detail || !selectedFolioId || !apiHeaders) {
   399→      return null
   400→    }
   401→    try {
   402→      setLocalSubmitting(true)
   403→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   404→        method: 'POST',
   405→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   406→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   407→      })
   408→      if (res.ok) {
   409→        const entry = await res.json()
   410→        const docNo = entry?.docNo || ''
   411→        setEntrySuccess({ show: true, docNo, type })
   412→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   413→        closeDialogs()
   414→        onRefresh?.()
   415→        setTimeout(() => setEntrySuccess(null), 4000)
   416→        return entry
   417→      } else {
   418→        const e = await safeJson(res)
   419→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   420→        return null
   421→      }
   422→    } catch {
