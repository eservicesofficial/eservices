    10→// ─── Helper: Convert number to Indian Rupee words ───
    11→function numberToIndianWords(num: number): string {
    12→  if (num === 0) return 'ZERO'
    13→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    14→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    15→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    16→  let words = ''
    17→  function convertBelowThousand(n: number): string {
    18→    let w = ''
    19→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    20→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    21→    else if (n > 0) { w += ones[n] }
    22→    return w
    23→  }
    24→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    25→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    26→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    27→  if (num > 0) { words += convertBelowThousand(num) }
    28→  return words.trim().toUpperCase() + ' ONLY'
    29→}

   199→// ─── Helper: Compute the most-used payment mode from credit entries ───
   200→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   201→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   202→  if (creditEntries.length === 0) return 'N/A'
   203→  const modeCounts: Record<string, number> = {}
   204→  for (const e of creditEntries) {
   205→    const mode = e.paymentMode || 'Cash'
   206→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   207→  }
   208→  let bestMode = 'Cash'
   209→  let bestAmount = 0
   210→  for (const [mode, amount] of Object.entries(modeCounts)) {
   211→    if (amount > bestAmount) {
   212→      bestAmount = amount
   213→      bestMode = mode
   214→    }
   215→  }
   216→  return bestMode
   217→}

   219→// ─── Helper: Build payment summary HTML ───
   220→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   221→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   222→  if (creditEntries.length === 0) {
   223→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   224→  }
   225→
   226→  const modeTotals: Record<string, number> = {}
   227→  for (const e of creditEntries) {
   228→    const mode = e.paymentMode || 'Cash'
   229→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   230→  }
   231→
   232→  const modes = Object.entries(modeTotals)
   233→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   234→
   235→  if (modes.length === 1) {
   236→    const [mode, amount] = modes[0]
   237→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   238→  }
   239→
   240→  const rows = modes.map(([mode, amount]) => {
   241→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   242→  }).join('')
   243→
   244→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   245→}

   247→// ─── Helper: Get stable document number from entries (never regenerate) ───
   248→// Searches ALL entries for the first non-empty docNo to avoid N/A
   249→function getStableDocId(entries: { docNo?: string; invoiceNo?: string; id?: string; createdAt?: string }[]): string {
   250→  if (entries.length === 0) return `DOC-${Date.now()}`
   251→  for (const e of entries) {
   252→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   253→  }
   254→  for (const e of entries) {
   255→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   256→  }
   257→  // Fallback: use entry ID truncated or timestamp — never show N/A
   258→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   259→  return fallback
   260→}

   756→// ─── Print Voucher / Payment History ───
   757→export async function printVoucherHistory(
   758→  detail: CustomerDetail,
   759→  paymentEntries: LedgerEntry[],
   760→  hdrRef: RefObject<Record<string, string>>
   761→) {
   762→  if (!detail) return
   763→  if (paymentEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No payment entries to print' }); return }
   764→  const settings = await fetchSettingsForPrint(hdrRef)
   765→  const now = new Date()
   766→
   767→  const sorted = [...paymentEntries].sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
   768→  const totalAmount = sorted.reduce((s, e) => s + (e.credit || 0), 0)
   769→
   770→  const docId = getStableDocId(sorted)
   771→
   772→  const rows = sorted.map((e, i) => {
   773→    const desc = stripInternalDescriptions(e.description || '—')
   774→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td>${desc}</td><td style="text-align:right;font-weight:700;color:#16a34a">${fmtINR(e.credit || 0)}</td><td style="text-align:center">${e.paymentMode || '—'}</td></tr>`
   775→  }).join('')
   776→
   777→  const w = openPrintWindow(`Voucher History - ${detail.fullName}`)
   778→  if (!w) return
   779→
   780→  w.document.write(
   781→    `<!DOCTYPE html><html><head><title>Payment Voucher</title><style>${PRINT_CSS}</style></head><body>` +
   782→    buildUnifiedHeader(settings, 'PAYMENT VOUCHER', docId) +
   783→    buildCustomerInfoBlock(detail) +
   784→    `<div class="datetime">Printed: ${fmtDateTime(now.toISOString())}</div>` +
   785→    `<table><tr><th>#</th><th>Date</th><th>Description</th><th style="text-align:right">Amount (₹)</th><th>Mode</th></tr>${rows}` +
   786→    `<tr class="totals"><td colspan="3" style="text-align:right;font-size:11px;color:#555">TOTAL</td><td style="color:#16a34a;font-weight:700">${fmtINR(totalAmount)}</td><td>${sorted.length} entries</td></tr></table>` +
   787→    buildPaymentSummary(sorted) +
   788→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Total Payments</div><div class="s-value" style="color:#16a34a">${fmtINR(totalAmount)}</div></div><div class="summary-item"><div class="s-label">Entries</div><div class="s-value">${sorted.length}</div></div></div>` +
   789→    buildFooter() +
   790→    `</body></html>`
   791→  )
   792→  w.document.close()
   793→  w.print()
   794→}

   798→interface ShopSettings {
   799→  shopName: string
   800→  shopAddress: string
   801→  shopPhone: string
   802→  shopGst: string
   803→  shopGstState: string
   804→}
   805→
   806→async function fetchShopSettings(hdrRef?: RefObject<Record<string, string>>): Promise<ShopSettings> {
   807→  try {
   808→    const headers = hdrRef?.current || undefined
   809→    const res = await fetch('/api/settings', headers ? { headers } : undefined)
   810→    if (res.ok) {
   811→      const db = (await res.json()) as Record<string, string>
   812→      return {
   813→        shopName: db.shopName || 'Eservices',
   814→        shopAddress: db.shopAddress || '',
   815→        shopPhone: db.shopPhone || '',
   816→        shopGst: db.gstNumber || db.gstin || '',
   817→        shopGstState: db.gstState || '',
   818→      }
   819→    }
   820→  } catch { /* ignore */ }
   821→  return { shopName: 'Eservices', shopAddress: '', shopPhone: '', shopGst: '', shopGstState: '' }
   822→}
   823→
   824→function openThermalPrintWindow(thermalHtml: string, title: string): Window | null {
   825→  const w = window.open('', '_blank', 'width=350,height=700')
   826→  if (!w) {
   827→    notify({ type: 'error', title: 'Error', message: 'Please allow pop-ups to print thermal receipts' })
   828→    return null
   829→  }
   830→  w.document.write(thermalHtml)
   831→  w.document.close()
   832→  setTimeout(() => {
   833→    try { if (!w.closed) w.print() } catch { /* window closed */ }
   834→  }, 600)
   835→  return w
   836→}

   838→// ─── Thermal Print Receipt ───
   839→export async function thermalPrintReceipt(
   840→  detail: CustomerDetail,
   841→  entries: LedgerEntry[],
   842→  hdrRef: RefObject<Record<string, string>>
   843→) {
   844→  if (!detail) return
   845→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   846→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   847→
   848→  const [tConfig, shop] = await Promise.all([
   849→    fetchThermalSettings(),
   850→    fetchShopSettings(hdrRef),
   851→  ])
   852→
   853→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   854→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   855→  const finalBal = activeEntries[activeEntries.length - 1].balance
   856→
   857→  const items: ThermalReceiptData['items'] = activeEntries.map(e => ({
   858→    name: stripInternalDescriptions(e.description || e.type || '—'),
   859→    qty: 1,
   860→    rate: e.debit > 0 ? e.debit : e.credit,
   861→    total: e.debit > 0 ? e.debit : e.credit,
   862→  }))
   863→
   864→  const actualPaymentMethod = computeMostUsedPaymentMode(activeEntries)
   865→
   866→  const receiptData: ThermalReceiptData = {
   867→    shopName: shop.shopName,
   868→    shopAddress: shop.shopAddress,
   869→    shopPhone: shop.shopPhone,
   870→    shopGst: shop.shopGst,
   871→    shopGstState: shop.shopGstState,
   872→    showGst: false,
   873→    invoiceNumber: detail.customerId,
   874→    date: fmtDate(new Date().toISOString()),
   875→    orderType: 'PAYMENT RECEIPT',
   876→    customerName: detail.fullName,
   877→    customerPhone: detail.mobile,
   878→    customerAddress: (detail as any).address || '',
   879→    items,
   880→    subtotal: totalDebits,
   881→    cgst: 0,
   882→    sgst: 0,
   883→    gstRate: 0,
   884→    deliveryCharge: 0,
   885→    discountAmount: 0,
   886→    grandTotal: totalDebits - totalCredits,
   887→    paidAmount: totalCredits,
   888→    balanceAmount: finalBal,
   889→    paymentMethod: actualPaymentMethod,
   890→    billStatus: finalBal <= 0 ? 'Paid' : 'Balance',
   891→  }
   892→
   893→  const thermalHtml = buildRichThermalHtml(receiptData, tConfig, `Receipt - ${detail.fullName}`)
   894→  openThermalPrintWindow(thermalHtml, `Receipt - ${detail.fullName}`)
   895→}
