   168→export function CustomerFolioDialogs({
   169→  detail, allFolios, selectedEntries, submitting, settlingMode,
   170→  selectedFolioId, bal, selectedEntryObj, isActive, isCheckedOut,
   171→  invoiceData, invoiceLoading, folioStatementData, customerTags,
   172→  transferResults, paymentEntries,
   173→  addOpen, editOpen, chargeOpen, paymentOpen, refundOpen, transferOpen,
   174→  discountOpen, docOpen, noteOpen, tagsOpen, checkoutOpen, deleteOpen,
   175→  splitAmountOpen, shiftOpen, removeOpen, voucherOpen, invoiceOpen,
   176→  detailsSheetOpen,
   177→  form, chargeForm, paymentForm, refundForm, transferForm, discountForm,
   178→  docForm, noteForm, splitAmountForm, invoiceFrom, invoiceTo,
   179→  shiftTargetFolioId, duplicate,
   180→  setAddOpen, setEditOpen, setChargeOpen, setPaymentOpen, setRefundOpen,
   181→  setTransferOpen, setDiscountOpen, setDocOpen, setNoteOpen, setTagsOpen,
   182→  setCheckoutOpen, setDeleteOpen, setSplitAmountOpen, setShiftOpen,
   183→  setRemoveOpen, setVoucherOpen, setInvoiceOpen, setDetailsSheetOpen,
   184→  setForm, setChargeForm, setPaymentForm, setRefundForm, setTransferForm,
   185→  setDiscountForm, setDocForm, setNoteForm, setSplitAmountForm,
   186→  setInvoiceFrom, setInvoiceTo, setShiftTargetFolioId, setDuplicate,
   187→  handleCreate, handleUpdate, handleCharge, handlePayment, handleSettle,
   188→  handleRefund, handleDiscount, handleTransfer, handleShift, handleSplitAmount,
   189→  handleRemove, handleCheckout, handleDelete, handleAddDoc, handleAddNote,
   190→  handleToggleTag, handleDeleteDoc, handleDeleteNote,
   191→  fetchInvoiceData, fetchFolioStatement, printInvoice, printCompleteInvoice, printSingleFolio, printVoucherHistory,
   192→  closeDialogs, searchTransferTargets,
   193→  deleteFolioOpen, folioToDelete,
   194→  setDeleteFolioOpen, handleDeleteFolio,
   195→  // Enhanced props
   196→  apiHeaders, onRefresh, canDelete = false, isAdmin = false, onPrintReceipt, onThermalPrintReceipt,
   197→  onThermalPrintInvoice, onThermalPrintVoucher,
   198→  lastCreatedDocNo, lastCreatedType,
   199→  // History
   200→  historyOpen, setHistoryOpen, onPrintHistoryFolio,
   201→}: DialogsProps) {
   202→
   203→  // ─── Internal State ───
   204→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   205→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   206→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   207→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   208→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   209→  // ─── History state ───
   210→  const [historyLoading, setHistoryLoading] = React.useState(false)
   211→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   212→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   213→  const [selectedVoucherIds, setSelectedVoucherIds] = React.useState<Set<string>>(new Set())
   214→  const [historyDate, setHistoryDate] = React.useState(() => {
   215→    const today = new Date()
   216→    return today.toISOString().split('T')[0] // "2025-01-15"
   217→  })
   218→
   219→  // Fetch history for the CURRENT customer
   220→  const fetchCustomerHistory = React.useCallback(async (date?: string) => {
   221→    if (!detail || !apiHeaders) return
   222→    try {
   223→      setHistoryLoading(true)
   224→      const q = date ? `?date=${date}` : ''
   225→      const res = await fetch(`/api/folio/customers/${detail.id}/history${q}`, { headers: apiHeaders })
   226→      if (res.ok) {
   227→        const data = await res.json() as CustomerHistory
   228→        setCustomerHistory(data)
   229→      }
   230→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to fetch checkout history' }) } finally { setHistoryLoading(false) }
   231→  }, [detail?.id, apiHeaders])
   232→
   233→  React.useEffect(() => {
   234→    if (historyOpen && detail && apiHeaders && fetchCustomerHistory) {
   235→      fetchCustomerHistory(historyDate)
   236→    } else if (!historyOpen) {
   237→      setCustomerHistory(null)
   238→      setExpandedFolioId(null)
   239→    }
   240→  }, [historyOpen, detail?.id, fetchCustomerHistory, apiHeaders, historyDate])
   241→
   242→  // ─── Print helpers for checkout records ───
   243→  const getTplCSS = (s: Record<string, string>) => {
   244→    const tid = s.invoiceTemplate || 'modern'
   245→    let cust: InvoiceCustomizations = {}
   246→    try { if (s.invoiceCustomizations) cust = JSON.parse(s.invoiceCustomizations) } catch { /* */ }
   247→    return getTemplateCSS(tid, cust)
   248→  }
   249→
   250→  const printCheckoutFolioBill = async (co: CheckoutRecord) => {
   251→    if (!co.transactions.length || !detail) return
   252→    let shopSettings: Record<string, string> = {}
   253→    try {
   254→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   255→    } catch { /* use defaults */ }
   256→    const tplCss = getTplCSS(shopSettings)
   257→    // Use current customer data (detail prop) for up-to-date info
   258→    const printDetail = { ...detail, totalVisits: co.visitNumber || detail.totalVisits } as unknown as CustomerDetail
   259→    const w = window.open('', '_blank')
   260→    if (!w) return
   261→    w.document.write(`<!DOCTYPE html><html><head><title>Tax Invoice - ${co.checkoutNumber}</title><style>${INVOICE_CSS}</style><style>${tplCss}</style></head><body>${buildHotelStyleInvoice(shopSettings, printDetail, co.transactions, co.invoiceNumber, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'Tax Invoice', undefined, co.checkoutUser)}</body></html>`)
   262→    w.document.close(); w.print()
   263→  }
   264→
   265→  const printCheckoutInvoice = async (co: CheckoutRecord) => {
   266→    if (!co.transactions.length || !detail) return
   267→    let shopSettings: Record<string, string> = {}
   268→    try {
   269→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   270→    } catch { /* use defaults */ }
   271→    const tplCss = getTplCSS(shopSettings)
   272→    // Use current customer data (detail prop) for up-to-date info
   273→    const printDetail = { ...detail, totalVisits: co.visitNumber || detail.totalVisits } as unknown as CustomerDetail
   274→    const w = window.open('', '_blank')
   275→    if (!w) return
   276→    w.document.write(`<!DOCTYPE html><html><head><title>Tax Invoice - ${co.invoiceNumber}</title><style>${INVOICE_CSS}</style><style>${tplCss}</style></head><body>${buildHotelStyleInvoice(shopSettings, printDetail, co.transactions, co.invoiceNumber, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'Tax Invoice', { openingBalance: co.openingBalance, closingBalance: co.closingBalance }, co.checkoutUser)}</body></html>`)
   277→    w.document.close(); w.print()
   278→  }
   279→
   280→  // ─── Track checkout transition ───
   281→  const wasCheckedOut = React.useRef(isCheckedOut)
   282→  React.useEffect(() => {
   283→    if (isCheckedOut && !wasCheckedOut.current) {
   284→      setPostCheckoutOpen(true)
   285→    }
   286→    wasCheckedOut.current = isCheckedOut
   287→  }, [isCheckedOut])
   288→
   289→  // ─── Show TXN number when lastCreatedDocNo changes ───
   290→  React.useEffect(() => {
   291→    if (lastCreatedDocNo && lastCreatedType) {
   292→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   293→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   294→      return () => clearTimeout(timer)
   295→    }
   296→  }, [lastCreatedDocNo, lastCreatedType])
   297→
   298→  // ─── Items for charge selector (Req 5) ───
   299→  const [items, setItems] = React.useState<{ id: string; name: string; price: number }[]>([])
   300→  const [itemsLoading, setItemsLoading] = React.useState(false)
   301→  React.useEffect(() => {
   302→    if (chargeOpen && apiHeaders) {
   303→      setItemsLoading(true)
   304→      fetch('/api/items', { headers: apiHeaders }).then(r => r.ok ? r.json() : []).then((list: any[]) => {
   305→        setItems((list || []).map((it: any) => ({ id: it.id, name: it.name, price: it.price || 0 })))
   306→      }).catch(() => {}).finally(() => setItemsLoading(false))
   307→    }
   308→  }, [chargeOpen, apiHeaders])
   309→
   310→  // Handle item selection
   311→  const handleItemSelect = React.useCallback((itemId: string) => {
   312→    const item = items.find(it => it.id === itemId)
   313→    if (item) {
   314→      setChargeForm({
   315→        ...chargeForm,
   316→        selectedItem: itemId,
   317→        selectedItemPrice: String(item.price),
   318→        description: item.name,
   319→        amount: item.price > 0 ? String(item.price) : chargeForm.amount,
   320→      })
   321→    }
   322→  }, [items, chargeForm])
   323→
   324→  // ─── Fetch invoice number ───
   325→  const fetchInvoiceNumber = React.useCallback(async () => {
   326→    if (!detail || !apiHeaders) return
   327→    try {
   328→      setInvNumLoading(true)
   329→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   330→        method: 'POST',
   331→        headers: apiHeaders,
   332→      })
   333→      if (res.ok) {
   334→        const data = await res.json()
   335→        setInvoiceNumber(data.invoiceNumber)
   336→      }
   337→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   338→  }, [detail?.id, apiHeaders])
   339→
   340→  // ─── Fetch invoice number when invoice dialog opens ───
   341→  React.useEffect(() => {
   342→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber && fetchInvoiceNumber) {
   343→      fetchInvoiceNumber()
   344→    }
   345→  }, [invoiceOpen, detail?.id, apiHeaders, invoiceNumber, fetchInvoiceNumber])
   346→
   347→  React.useEffect(() => {
   348→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   349→      fetchInvoiceData()
   350→    }
   351→  }, [invoiceOpen, invoiceData, detail?.id, selectedFolioId])
   352→
   353→  // ─── Entry creation with TXN tracking ───
   354→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   355→    if (!detail || !selectedFolioId || !apiHeaders) {
   356→      return null
   357→    }
   358→    try {
   359→      setLocalSubmitting(true)
   360→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   361→        method: 'POST',
   362→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   363→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   364→      })
   365→      if (res.ok) {
   366→        const entry = await res.json()
   367→        const docNo = entry?.docNo || ''
   368→        setEntrySuccess({ show: true, docNo, type })
   369→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   370→        closeDialogs()
   371→        onRefresh?.()
   372→        setTimeout(() => setEntrySuccess(null), 4000)
   373→        return entry
   374→      } else {
   375→        const e = await safeJson(res)
   376→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   377→        return null
   378→      }
   379→    } catch {
   380→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   381→      return null
   382→    } finally {
   383→      setLocalSubmitting(false)
   384→    }
   385→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   386→
   387→  // ─── Settlement with TXN tracking ───
   388→  const settleWithTxn = React.useCallback(async () => {
   389→    if (!detail || !paymentForm.amount || !apiHeaders) {
   390→      handleSettle()
   391→      return
   392→    }
   393→    try {
   394→      setLocalSubmitting(true)
   395→      // When master folio is selected, use autoAllocate to distribute across all folios
   396→      const isMaster = selectedFolioId === 'master'
   397→      const payload = isMaster
   398→        ? { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   399→        : { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   400→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   401→        method: 'POST',
   402→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   403→        body: JSON.stringify(payload),
   404→      })
   405→      if (res.ok) {
   406→        const data = await res.json()
   407→        // Auto-allocate may return settledFolios array instead of single ledgerEntry
   408→        const docNo = data?.ledgerEntry?.docNo || (data?.settledFolios?.[0] ? 'Multi-Folio' : '')
   409→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })
   410→        notify({ type: 'success', title: `Settlement ${docNo}`, message: data.message || `Settlement ${docNo} processed successfully` })
   411→        closeDialogs()
   412→        onRefresh?.()
   413→        setTimeout(() => setEntrySuccess(null), 4000)
   414→      } else {
   415→        const e = await safeJson(res)
   416→        notify({ type: 'error', title: 'Settlement Failed', message: (e.error as string) || 'Failed to process settlement' })
   417→      }
   418→    } catch {
