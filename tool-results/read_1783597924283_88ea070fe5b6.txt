    12→    12→    12→function numberToIndianWords(num: number): string {
    13→    13→    13→  if (num === 0) return 'ZERO'
    14→    14→    14→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    15→    15→    15→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    16→    16→    16→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    17→    17→    17→  let words = ''
    18→    18→    18→  function convertBelowThousand(n: number): string {
    19→    19→    19→    let w = ''
    20→    20→    20→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    21→    21→    21→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    22→    22→    22→    else if (n > 0) { w += ones[n] }
    23→    23→    23→    return w
    24→    24→    24→  }
    25→    25→    25→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    26→    26→    26→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    27→    27→    27→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    28→    28→    28→  if (num > 0) { words += convertBelowThousand(num) }
    29→    29→    29→  return words.trim().toUpperCase() + ' ONLY'
    30→    30→    30→}

   201→   201→   201→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   202→   202→   202→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   203→   203→   203→  if (creditEntries.length === 0) return 'N/A'
   204→   204→   204→  const modeCounts: Record<string, number> = {}
   205→   205→   205→  for (const e of creditEntries) {
   206→   206→   206→    const mode = e.paymentMode || 'Cash'
   207→   207→   207→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   208→   208→   208→  }
   209→   209→   209→  let bestMode = 'Cash'
   210→   210→   210→  let bestAmount = 0
   211→   211→   211→  for (const [mode, amount] of Object.entries(modeCounts)) {
   212→   212→   212→    if (amount > bestAmount) {
   213→   213→   213→      bestAmount = amount
   214→   214→   214→      bestMode = mode
   215→   215→   215→    }
   216→   216→   216→  }
   217→   217→   217→  return bestMode
   218→   218→   218→}
