   271→function exportToCSV<T extends Record<string, unknown>>(
   272→  data: T[],
   273→  filename: string,
   274→  columns: CsvColumn[]
   275→) {
   276→  if (data.length === 0) return;
   277→
   278→  const escapeCSV = (value: unknown): string => {
   279→    const str = String(value ?? '');
   280→    if (str.includes('"') || str.includes(',') || str.includes('\n')) {
   281→      return `"${str.replace(/"/g, '""')}"`;
   282→    }
   283→    return str;
   284→  };
   285→
   286→  const header = columns.map((c) => escapeCSV(c.label)).join(',');
   287→  const rows = data.map((row) =>
   288→    columns.map((c) => escapeCSV(row[c.key])).join(',')
   289→  );
   290→  const csv = [header, ...rows].join('\n');
   291→
   292→  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' });
   293→  const url = URL.createObjectURL(blob);
   294→  const link = document.createElement('a');
   295→  link.href = url;
   296→  link.download = filename;
   297→  document.body.appendChild(link);
   298→  link.click();
   299→  document.body.removeChild(link);
   300→  URL.revokeObjectURL(url);
   301→}

   330→    case 'completed':

   333→    case 'pending':
   334→    case 'balance':
   335→    case 'partial':
   336→      return 'secondary';

   338→    case 'cancelled':

   341→      return 'outline';
