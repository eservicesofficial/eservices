   186→   186→// ─── CSV Export with proper headers ───
   187→   187→function downloadCSV(data: Record<string, any>[], columns: ColDef[], filename: string) {
   188→   188→  if (!data.length) return
   189→   189→  const headers = columns.map(c => c.label)
   190→   190→  const csvRows = data.map(row =>
   191→   191→    columns.map(col => {
   192→   192→      const val = row[col.key]
   193→   193→      if (Array.isArray(val)) return `"${val.join(', ').replace(/"/g, '""')}"`
   194→   194→      if (col.format === 'currency' || col.format === 'amount') return Number(val) || 0
   195→   195→      return `"${String(val ?? '').replace(/"/g, '""')}"`
   196→   196→    }).join(',')
   197→   197→  )
   198→   198→  const csv = [headers.join(','), ...csvRows].join('\n')
   199→   199→  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' })
   200→   200→  const url = URL.createObjectURL(blob)
   201→   201→  const a = document.createElement('a')
   202→   202→  a.href = url
   203→   203→  a.download = `${filename}.csv`
   204→   204→  document.body.appendChild(a)
   205→   205→  a.click()
   206→   206→  document.body.removeChild(a)
   207→   207→  URL.revokeObjectURL(url)
   208→   208→}

   210→   210→// ─── Format cell value ───
   211→   211→function formatCellValue(value: unknown, col: ColDef): React.ReactNode {
   212→   212→  if (value === null || value === undefined || value === '') return <span className="text-muted-foreground">—</span>
   213→   213→  const strVal = String(value)
   214→   214→
   215→   215→  switch (col.format) {
   216→   216→    case 'currency':
   217→   217→      return <span className="font-medium tabular-nums">{fmtINR(Number(value))}</span>
   218→   218→    case 'date':
   219→   219→      return <span className="tabular-nums">{fmtDate(strVal)}</span>
   220→   220→    case 'datetime':
   221→   221→      return <span className="tabular-nums text-xs">{fmtDateTime(strVal)}</span>
   222→   222→    case 'amount':
   223→   223→      return <span className="font-medium tabular-nums">{fmtCur(Number(value))}</span>
   224→   224→    case 'tags': {
   225→   225→      const tags = Array.isArray(value) ? value : [value]
   226→   226→      return (
   227→   227→        <div className="flex flex-wrap gap-1">
   228→   228→          {tags.map((tag: string, i: number) => (
   229→   229→            <Badge key={i} variant="outline" className={`text-[10px] px-1.5 py-0 ${TAG_BADGE_MAP[tag] || ''}`}>
   230→   230→              {tag}
   231→   231→            </Badge>
   232→   232→          ))}
   233→   233→        </div>
   234→   234→      )
   235→   235→    }
   236→   236→    case 'badge':
   237→   237→      return (
   238→   238→        <Badge variant="outline" className={`text-[10px] px-1.5 py-0 ${TYPE_BADGE_MAP[strVal] || 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400'}`}>
   239→   239→          {strVal}
   240→   240→        </Badge>
   241→   241→      )
   242→   242→    default:
   243→   243→      return <span className="text-xs">{strVal}</span>
   244→   244→  }
   245→   245→}
