     5→     5→// ─── Report Types ───
     6→     6→type ReportType =
     7→     7→  | 'summary' | 'outstanding' | 'payments' | 'transactions' | 'vip' | 'settlements' | 'refunds' | 'gst'
     8→     8→  | 'activeFolio' | 'checkoutHistory' | 'creditBalance' | 'chargeSummary'
     9→     9→  | 'discounts' | 'userCollections' | 'dailyCollections' | 'monthlyCollections' | 'audit'
    10→    10→
    11→    11→const VALID_REPORT_TYPES: ReportType[] = [
    12→    12→  'summary', 'outstanding', 'payments', 'transactions', 'vip', 'settlements', 'refunds', 'gst',
    13→    13→  'activeFolio', 'checkoutHistory', 'creditBalance', 'chargeSummary',
    14→    14→  'discounts', 'userCollections', 'dailyCollections', 'monthlyCollections', 'audit',
    15→    15→]

    17→    17→// ─── Filter Interface ───
    18→    18→interface ReportFilters {
    19→    19→  fromDate: Date | null
    20→    20→  toDate: Date | null
    21→    21→  customer: string | null
    22→    22→  folioNumber: string | null
    23→    23→  user: string | null
    24→    24→  status: string | null
    25→    25→  paymentType: string | null
    26→    26→  paymentMode: string | null
    27→    27→  transactionType: string | null
    28→    28→  dateFilter: Record<string, Date> | undefined
    29→    29→}
    30→    30→
    31→    31→function parseFilters(searchParams: URLSearchParams): ReportFilters {
    32→    32→  const from = searchParams.get('from')
    33→    33→  const to = searchParams.get('to')
    34→    34→  const fromDate = from ? new Date(from) : null
    35→    35→  let toDate: Date | null = to ? new Date(to) : null
    36→    36→  if (toDate) toDate.setHours(23, 59, 59, 999)
    37→    37→
    38→    38→  const dateFilter: Record<string, Date> | undefined =
    39→    39→    fromDate && toDate ? { gte: fromDate, lte: toDate } :
    40→    40→    fromDate ? { gte: fromDate } :
    41→    41→    toDate ? { lte: toDate } :
    42→    42→    undefined
    43→    43→
    44→    44→  // transactionType is alias for paymentType
    45→    45→  const paymentType = searchParams.get('paymentType') || searchParams.get('transactionType') || null
    46→    46→
    47→    47→  return {
    48→    48→    fromDate,
    49→    49→    toDate,
    50→    50→    customer: searchParams.get('customer') || null,
    51→    51→    folioNumber: searchParams.get('folioNumber') || null,
    52→    52→    user: searchParams.get('user') || null,
    53→    53→    status: searchParams.get('status') || null,
    54→    54→    paymentType,
    55→    55→    paymentMode: searchParams.get('paymentMode') || null,
    56→    56→    transactionType: paymentType,
    57→    57→    dateFilter,
    58→    58→  }
    59→    59→}

    61→    61→// Helper: apply common ledger entry filters
    62→    62→function applyLedgerFilters(where: Record<string, unknown>, filters: ReportFilters, unitCode: string) {
    63→    63→  where.unitCode = unitCode
    64→    64→  if (filters.dateFilter) where.createdAt = filters.dateFilter
    65→    65→  if (filters.customer) {
    66→    66→    where.OR = [
    67→    67→      { customer: { fullName: { contains: filters.customer } } },
    68→    68→      { customer: { customerId: { contains: filters.customer } } },
    69→    69→      { customer: { mobile: { contains: filters.customer } } },
    70→    70→    ] as any[]
    71→    71→  }
    72→    72→  if (filters.paymentType && filters.paymentType !== 'All') where.type = filters.paymentType
    73→    73→  if (filters.paymentMode && filters.paymentMode !== 'All') where.paymentMode = filters.paymentMode
    74→    74→  if (filters.user) where.userName = { contains: filters.user }
    75→    75→}

    77→    77→// Helper: apply common customer filters
    78→    78→function applyCustomerFilters(where: Record<string, unknown>, filters: ReportFilters, unitCode: string) {
    79→    79→  where.unitCode = unitCode
    80→    80→  if (filters.status && filters.status !== 'All') where.status = filters.status
    81→    81→  if (filters.customer) {
    82→    82→    where.OR = [
    83→    83→      { fullName: { contains: filters.customer } },
    84→    84→      { customerId: { contains: filters.customer } },
    85→    85→      { mobile: { contains: filters.customer } },
    86→    86→    ] as any[]
    87→    87→  }
    88→    88→}

    90→    90→// Helper: get balance for a set of customer IDs
    91→    91→async function getCustomerBalances(customerIds: string[], folioStatus?: string) {
    92→    92→  const folioWhere: Record<string, unknown> = { customerId: { in: customerIds }, status: folioStatus || 'Active' }
    93→    93→  const activeFolioIds = await db.folio.findMany({
    94→    94→    where: folioWhere,
    95→    95→    select: { id: true, customerId: true },
    96→    96→  })
    97→    97→  const folioIdSet = activeFolioIds.map(f => f.id)
    98→    98→  if (folioIdSet.length === 0) return new Map<string, { balance: number; date: string | null }>()
    99→    99→
   100→   100→  const lastLedgers = await db.ledgerEntry.findMany({
   101→   101→    where: { folioId: { in: folioIdSet }, status: 'Active' },
   102→   102→    distinct: ['customerId'],
   103→   103→    orderBy: { createdAt: 'desc' },
   104→   104→    select: { customerId: true, balance: true, createdAt: true },
   105→   105→  })
   106→   106→  return new Map(lastLedgers.map(l => [l.customerId, { balance: l.balance, date: l.createdAt?.toISOString() ?? null }]))
   107→   107→}

   109→   109→// ─── GET Handler ───
   110→   110→export async function GET(request: NextRequest) {
   111→   111→  try {
   112→   112→    const session = await getSessionFromRequest(request)
   113→   113→    if (!session) {
   114→   114→      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   115→   115→    }
   116→   116→
   117→   117→    if (!sessionHasPermission(session, 'khata', 'access')) {
   118→   118→      return NextResponse.json({ error: 'Access denied' }, { status: 403 })
   119→   119→    }
   120→   120→
   121→   121→    const { searchParams } = new URL(request.url)
   122→   122→    const type = searchParams.get('type') as ReportType | null
   123→   123→
   124→   124→    if (!type || !VALID_REPORT_TYPES.includes(type)) {
   125→   125→      return NextResponse.json({ error: 'Invalid or missing report type', validTypes: VALID_REPORT_TYPES }, { status: 400 })
   126→   126→    }
   127→   127→
   128→   128→    const filters = parseFilters(searchParams)
   129→   129→
   130→   130→    switch (type) {
   131→   131→      case 'summary': return handleSummary(session.unitCode, filters)
   132→   132→      case 'outstanding': return handleOutstanding(session.unitCode, filters)
   133→   133→      case 'payments': return handlePayments(session.unitCode, filters)
   134→   134→      case 'transactions': return handleTransactions(session.unitCode, filters)
   135→   135→      case 'vip': return handleVip(session.unitCode, filters)
   136→   136→      case 'settlements': return handleSettlements(session.unitCode, filters)
   137→   137→      case 'refunds': return handleRefunds(session.unitCode, filters)
   138→   138→      case 'gst': return handleGst(session.unitCode, filters)
   139→   139→      case 'activeFolio': return handleActiveFolio(session.unitCode, filters)
   140→   140→      case 'checkoutHistory': return handleCheckoutHistory(session.unitCode, filters)
   141→   141→      case 'creditBalance': return handleCreditBalance(session.unitCode, filters)
   142→   142→      case 'chargeSummary': return handleChargeSummary(session.unitCode, filters)
   143→   143→      case 'discounts': return handleDiscounts(session.unitCode, filters)
   144→   144→      case 'userCollections': return handleUserCollections(session.unitCode, filters)
   145→   145→      case 'dailyCollections': return handleDailyCollections(session.unitCode, filters)
   146→   146→      case 'monthlyCollections': return handleMonthlyCollections(session.unitCode, filters)
   147→   147→      case 'audit': return handleAudit(session.unitCode, filters)
   148→   148→      default:
   149→   149→        return NextResponse.json({ error: 'Invalid report type' }, { status: 400 })
   150→   150→    }
   151→   151→  } catch (error) {
   152→   152→    console.error('[FolioReports] Error:', error)
   153→   153→    return NextResponse.json({ error: 'Failed to generate report' }, { status: 500 })
   154→   154→  }
   155→   155→}

   161→   161→// ─── 1. Customer Summary Report ───
   162→   162→async function handleSummary(unitCode: string, filters: ReportFilters) {
   163→   163→  const where: Record<string, unknown> = {}
   164→   164→  applyCustomerFilters(where, filters, unitCode)
   165→   165→
   166→   166→  const customers = await db.customer.findMany({
   167→   167→    where,
   168→   168→    select: {
   169→   169→      id: true, customerId: true, fullName: true, mobile: true, email: true,
   170→   170→      customerType: true, creditLimit: true, status: true, gstNumber: true,
   171→   171→      openingBalance: true, createdAt: true,
   172→   172→      tags: { select: { tagName: true } },
   173→   173→      _count: { select: { folios: true, ledgerEntries: true, settlements: true, visits: true } },
   174→   174→    },
   175→   175→    orderBy: { createdAt: 'desc' },
   176→   176→  })
   177→   177→
   178→   178→  const customerIds = customers.map(c => c.id)
   179→   179→  const ledgerMap = await getCustomerBalances(customerIds)
   180→   180→
   181→   181→  const rows = customers.map(c => {
   182→   182→    const ledger = ledgerMap.get(c.id)
   183→   183→    const balance = ledger?.balance ?? 0
   184→   184→    return {
   185→   185→      customerId: c.customerId,
   186→   186→      fullName: c.fullName,
   187→   187→      mobile: c.mobile,
   188→   188→      email: c.email,
   189→   189→      customerType: c.customerType,
   190→   190→      status: c.status,
   191→   191→      balance,
   192→   192→      creditLimit: c.creditLimit,
   193→   193→      gstNumber: c.gstNumber,
   194→   194→      totalVisits: c._count.visits,
   195→   195→      totalTransactions: c._count.ledgerEntries,
   196→   196→      totalSettlements: c._count.settlements,
   197→   197→      tags: c.tags.map(t => t.tagName),
   198→   198→      createdAt: c.createdAt.toISOString(),
   199→   199→      lastTxnDate: ledger?.date ?? null,
   200→   200→    }
   201→   201→  })
   202→   202→
   203→   203→  const totalBalance = rows.reduce((s, r) => s + r.balance, 0)
   204→   204→  const totalCreditLimit = rows.reduce((s, r) => s + r.creditLimit, 0)
   205→   205→  const totalOutstanding = rows.filter(r => r.balance > 0).reduce((s, r) => s + r.balance, 0)
   206→   206→  const outstandingCount = rows.filter(r => r.balance > 0).length
   207→   207→
   208→   208→  return NextResponse.json({
   209→   209→    reportType: 'summary',
   210→   210→    summary: {
   211→   211→      totalCustomers: rows.length,
   212→   212→      totalBalance,
   213→   213→      totalCreditLimit,
   214→   214→      totalOutstanding,
   215→   215→      outstandingCount,
   216→   216→      activeCount: rows.filter(r => r.status === 'Active').length,
   217→   217→      settledCount: rows.filter(r => r.status === 'Settled').length,
   218→   218→    },
   219→   219→    data: rows,
   220→   220→  })
   221→   221→}

   223→   223→// ─── 2. Outstanding Balance Report ───
   224→   224→async function handleOutstanding(unitCode: string, filters: ReportFilters) {
   225→   225→  const where: Record<string, unknown> = { status: { notIn: ['Archived', 'CheckedOut'] } }
   226→   226→  applyCustomerFilters(where, filters, unitCode)
   227→   227→
   228→   228→  const customers = await db.customer.findMany({
   229→   229→    where,
   230→   230→    select: {
   231→   231→      id: true, customerId: true, fullName: true, mobile: true, email: true,
   232→   232→      customerType: true, creditLimit: true, status: true,
   233→   233→      tags: { select: { tagName: true } },
   234→   234→      _count: { select: { ledgerEntries: true, settlements: true, visits: true } },
   235→   235→    },
   236→   236→    orderBy: { createdAt: 'desc' },
   237→   237→  })
   238→   238→
   239→   239→  const customerIds = customers.map(c => c.id)
   240→   240→  const ledgerMap = await getCustomerBalances(customerIds)
   241→   241→
   242→   242→  const rows = customers
   243→   243→    .map(c => {
   244→   244→      const balance = ledgerMap.get(c.id)?.balance ?? 0
   245→   245→      return {
   246→   246→        customerId: c.customerId,
   247→   247→        fullName: c.fullName,
   248→   248→        mobile: c.mobile,
   249→   249→        email: c.email,
   250→   250→        customerType: c.customerType,
   251→   251→        status: c.status,
   252→   252→        balance,
   253→   253→        creditLimit: c.creditLimit,
   254→   254→        overdue: balance > c.creditLimit,
   255→   255→        tags: c.tags.map(t => t.tagName),
   256→   256→        lastTxnDate: ledgerMap.get(c.id)?.date ?? null,
   257→   257→      }
   258→   258→    })
   259→   259→    .filter(r => r.balance > 0)
   260→   260→    .sort((a, b) => b.balance - a.balance)
   261→   261→
   262→   262→  const totalOutstanding = rows.reduce((s, r) => s + r.balance, 0)
   263→   263→  const avgOutstanding = rows.length > 0 ? totalOutstanding / rows.length : 0
   264→   264→
   265→   265→  return NextResponse.json({
   266→   266→    reportType: 'outstanding',
   267→   267→    summary: {
   268→   268→      totalCustomersWithDue: rows.length,
   269→   269→      totalOutstanding,
   270→   270→      avgOutstanding: Math.round(avgOutstanding * 100) / 100,
   271→   271→      highRiskCount: rows.filter(r => r.overdue).length,
   272→   272→    },
   273→   273→    data: rows,
   274→   274→  })
   275→   275→}

   277→   277→// ─── 3. Payment Collection Report ───
   278→   278→async function handlePayments(unitCode: string, filters: ReportFilters) {
   279→   279→  const where: Record<string, unknown> = { status: 'Active', type: 'Payment' }
   280→   280→  applyLedgerFilters(where, filters, unitCode)
   281→   281→
   282→   282→  const entries = await db.ledgerEntry.findMany({
   283→   283→    where,
   284→   284→    orderBy: { createdAt: 'desc' },
   285→   285→    select: {
   286→   286→      id: true, date: true, description: true, credit: true, balance: true,
   287→   287→      paymentMode: true, userName: true, customerId: true, createdAt: true,
   288→   288→      customer: { select: { fullName: true, customerId: true, mobile: true } },
   289→   289→    },
   290→   290→    take: 5000,
   291→   291→  })
   292→   292→
   293→   293→  const rows = entries.map(e => ({
   294→   294→    date: e.date,
   295→   295→    customerName: e.customer.fullName,
   296→   296→    customerId: e.customer.customerId,
   297→   297→    mobile: e.customer.mobile,
   298→   298→    description: e.description,
   299→   299→    amount: e.credit,
   300→   300→    paymentMode: e.paymentMode,
   301→   301→    balanceAfter: e.balance,
   302→   302→    recordedBy: e.userName,
   303→   303→    createdAt: e.createdAt.toISOString(),
   304→   304→  }))
   305→   305→
   306→   306→  const totalPayments = rows.reduce((s, r) => s + r.amount, 0)
   307→   307→  const modeBreakdown = rows.reduce((acc: Record<string, number>, r) => {
   308→   308→    const mode = r.paymentMode || 'Cash'
   309→   309→    acc[mode] = (acc[mode] || 0) + r.amount
   310→   310→    return acc
   311→   311→  }, {})
   312→   312→
   313→   313→  return NextResponse.json({
   314→   314→    reportType: 'payments',
   315→   315→    summary: {
   316→   316→      totalPayments,
   317→   317→      paymentCount: rows.length,
   318→   318→      avgPayment: rows.length > 0 ? Math.round((totalPayments / rows.length) * 100) / 100 : 0,
   319→   319→      modeBreakdown,
   320→   320→    },
   321→   321→    data: rows,
   322→   322→  })
   323→   323→}

   325→   325→// ─── 4. Transaction History Report ───
   326→   326→async function handleTransactions(unitCode: string, filters: ReportFilters) {
   327→   327→  const where: Record<string, unknown> = { status: 'Active' }
   328→   328→  applyLedgerFilters(where, filters, unitCode)
   329→   329→
   330→   330→  const entries = await db.ledgerEntry.findMany({
   331→   331→    where,
   332→   332→    orderBy: { createdAt: 'desc' },
   333→   333→    select: {
   334→   334→      id: true, date: true, type: true, description: true, docNo: true, invoiceNo: true,
   335→   335→      debit: true, credit: true, tax: true, discount: true, balance: true,
   336→   336→      paymentMode: true, userName: true, createdAt: true,
   337→   337→      customer: { select: { fullName: true, customerId: true } },
   338→   338→    },
   339→   339→    take: 5000,
   340→   340→  })
   341→   341→
   342→   342→  const rows = entries.map(e => ({
   343→   343→    date: e.date,
   344→   344→    customerName: e.customer.fullName,
   345→   345→    customerId: e.customer.customerId,
   346→   346→    type: e.type,
   347→   347→    description: e.description,
   348→   348→    docNo: e.docNo,
   349→   349→    invoiceNo: e.invoiceNo,
   350→   350→    debit: e.debit,
   351→   351→    credit: e.credit,
   352→   352→    tax: e.tax,
   353→   353→    discount: e.discount,
   354→   354→    balance: e.balance,
   355→   355→    paymentMode: e.paymentMode,
   356→   356→    recordedBy: e.userName,
   357→   357→    createdAt: e.createdAt.toISOString(),
   358→   358→  }))
   359→   359→
   360→   360→  const totalDebits = rows.reduce((s, r) => s + r.debit, 0)
   361→   361→  const totalCredits = rows.reduce((s, r) => s + r.credit, 0)
   362→   362→  const totalTax = rows.reduce((s, r) => s + r.tax, 0)
   363→   363→  const totalDiscount = rows.reduce((s, r) => s + r.discount, 0)
   364→   364→
   365→   365→  const typeBreakdown = rows.reduce((acc: Record<string, { count: number; debit: number; credit: number }>, r) => {
   366→   366→    if (!acc[r.type]) acc[r.type] = { count: 0, debit: 0, credit: 0 }
   367→   367→    acc[r.type].count++
   368→   368→    acc[r.type].debit += r.debit
   369→   369→    acc[r.type].credit += r.credit
   370→   370→    return acc
   371→   371→  }, {})
   372→   372→
   373→   373→  return NextResponse.json({
   374→   374→    reportType: 'transactions',
   375→   375→    summary: {
   376→   376→      totalEntries: rows.length,
   377→   377→      totalDebits,
   378→   378→      totalCredits,
   379→   379→      totalTax,
   380→   380→      totalDiscount,
   381→   381→      netFlow: totalCredits - totalDebits,
   382→   382→      typeBreakdown,
   383→   383→    },
   384→   384→    data: rows,
   385→   385→  })
   386→   386→}

   388→   388→// ─── 5. VIP Customer Report ───
   389→   389→async function handleVip(unitCode: string, filters: ReportFilters) {
   390→   390→  const where: Record<string, unknown> = {
   391→   391→    OR: [
   392→   392→      { customerType: 'VIP' },
   393→   393→      { tags: { some: { tagName: 'VIP' } } },
   394→   394→      { tags: { some: { tagName: 'Premium' } } },
   395→   395→    ],
   396→   396→  }
   397→   397→  applyCustomerFilters(where, filters, unitCode)
   398→   398→
   399→   399→  const customers = await db.customer.findMany({
   400→   400→    where,
   401→   401→    select: {
   402→   402→      id: true, customerId: true, fullName: true, mobile: true, email: true,
   403→   403→      customerType: true, creditLimit: true, status: true, gstNumber: true,
   404→   404→      panNumber: true, openingBalance: true, createdAt: true,
   405→   405→      tags: { select: { tagName: true } },
   406→   406→      _count: { select: { folios: true, ledgerEntries: true, settlements: true, visits: true } },
   407→   407→    },
   408→   408→    orderBy: { createdAt: 'desc' },
   409→   409→  })
   410→   410→
   411→   411→  const customerIds = customers.map(c => c.id)
   412→   412→  const ledgerMap = await getCustomerBalances(customerIds)
   413→   413→
   414→   414→  const lifetimeSpend = customerIds.length > 0
   415→   415→    ? await db.ledgerEntry.groupBy({
   416→   416→        by: ['customerId'],
   417→   417→        where: { customerId: { in: customerIds }, status: 'Active', type: 'Charge' },
   418→   418→        _sum: { debit: true },
   419→   419→      })
   420→   420→    : []
   421→   421→  const spendMap = new Map(lifetimeSpend.map(l => [l.customerId, l._sum.debit ?? 0]))
   422→   422→
   423→   423→  const rows = customers.map(c => {
   424→   424→    const balance = ledgerMap.get(c.id)?.balance ?? 0
   425→   425→    const lifetimeSpendAmt = spendMap.get(c.id) ?? 0
   426→   426→    return {
   427→   427→      customerId: c.customerId,
   428→   428→      fullName: c.fullName,
   429→   429→      mobile: c.mobile,
   430→   430→      email: c.email,
   431→   431→      customerType: c.customerType,
   432→   432→      status: c.status,
   433→   433→      balance,
   434→   434→      creditLimit: c.creditLimit,
   435→   435→      gstNumber: c.gstNumber,
   436→   436→      panNumber: c.panNumber,
   437→   437→      lifetimeSpend: lifetimeSpendAmt,
   438→   438→      totalVisits: c._count.visits,
   439→   439→      totalTransactions: c._count.ledgerEntries,
   440→   440→      totalSettlements: c._count.settlements,
   441→   441→      tags: c.tags.map(t => t.tagName),
   442→   442→      createdAt: c.createdAt.toISOString(),
