   187→   187→  // ─── History state ───
   188→   188→  const [historyLoading, setHistoryLoading] = React.useState(false)
   189→   189→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   190→   190→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)

   192→   192→  const [selectedVoucherIds, setSelectedVoucherIds] = React.useState<Set<string>>(new Set())
   193→   193→

   196→   196→    if (historyOpen && detail && apiHeaders) {
   197→   197→      fetchCustomerHistory()

   199→   199→      setCustomerHistory(null)
   200→   200→      setExpandedFolioId(null)

   212→   212→        setCustomerHistory(data)
   213→   213→      }
   214→   214→    } catch { /* ignore */ } finally { setHistoryLoading(false) }

   221→   221→      const folio = customerHistory?.historicalFolios.find(f => f.folioId === folioId)
   222→   222→      if (folio) {
   223→   223→        setExpandedFolioId(folioId)
   224→   224→        setExpandedFolioEntries(prev => ({
   225→   225→          ...prev,
   226→   226→          [folioId]: (folio.entries || []),
   227→   227→        }))
   228→   228→      }
   229→   229→    }
   230→   230→  }
   231→   231→
   232→   232→  // ─── Track checkout transition ───
   233→   233→  const wasCheckedOut = React.useRef(isCheckedOut)
   234→   234→  React.useEffect(() => {
   235→   235→    if (isCheckedOut && !wasCheckedOut.current) {
   236→   236→      setPostCheckoutOpen(true)
   237→   237→    }
   238→   238→    wasCheckedOut.current = isCheckedOut
   239→   239→  }, [isCheckedOut])
   240→   240→
   241→   241→  // ─── Show TXN number when lastCreatedDocNo changes ───
   242→   242→  React.useEffect(() => {
   243→   243→    if (lastCreatedDocNo && lastCreatedType) {
   244→   244→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   245→   245→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   246→   246→      return () => clearTimeout(timer)
   247→   247→    }
   248→   248→  }, [lastCreatedDocNo, lastCreatedType])
   249→   249→
   250→   250→  // ─── Items for charge selector (Req 5) ───
   251→   251→  const [items, setItems] = React.useState<{ id: string; name: string; price: number }[]>([])
   252→   252→  const [itemsLoading, setItemsLoading] = React.useState(false)
   253→   253→  React.useEffect(() => {
   254→   254→    if (chargeOpen && apiHeaders) {
   255→   255→      setItemsLoading(true)
   256→   256→      fetch('/api/items', { headers: apiHeaders }).then(r => r.ok ? r.json() : []).then((list: any[]) => {
   257→   257→        setItems((list || []).map((it: any) => ({ id: it.id, name: it.name, price: it.price || 0 })))
   258→   258→      }).catch(() => {}).finally(() => setItemsLoading(false))
   259→   259→    }
   260→   260→  }, [chargeOpen, apiHeaders])
   261→   261→

   272→   272→      })
   273→   273→    }
   274→   274→  }, [items, chargeForm])
   275→   275→

   278→   278→    if (!detail || !apiHeaders) return

   309→   309→    }

   314→   314→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   315→   315→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   316→   316→      })
   317→   317→      if (res.ok) {
   318→   318→        const entry = await res.json()
   319→   319→        const docNo = entry?.docNo || ''
   320→   320→        setEntrySuccess({ show: true, docNo, type })

   322→   322→        closeDialogs()
   323→   323→        onRefresh?.()
   324→   324→        setTimeout(() => setEntrySuccess(null), 4000)
   325→   325→        return entry
   326→   326→      } else {
   327→   327→        const e = await safeJson(res)
   328→   328→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   329→   329→        return null
   330→   330→      }
   331→   331→    } catch {
   332→   332→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   333→   333→      return null
   334→   334→    } finally {
   335→   335→      setLocalSubmitting(false)
   336→   336→    }
   337→   337→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   338→   338→
   339→   339→  // ─── Settlement with TXN tracking ───
   340→   340→  const settleWithTxn = React.useCallback(async () => {
   341→   341→    if (!detail || !paymentForm.amount || !apiHeaders) {
   342→   342→      handleSettle()
   343→   343→      return
   344→   344→    }

   346→   346→      setLocalSubmitting(true)

   351→   351→        : { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   352→   352→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   353→   353→        method: 'POST',
   354→   354→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   355→   355→        body: JSON.stringify(payload),
   356→   356→      })
   357→   357→      if (res.ok) {
   358→   358→        const data = await res.json()
   359→   359→        // Auto-allocate may return settledFolios array instead of single ledgerEntry
   360→   360→        const docNo = data?.ledgerEntry?.docNo || (data?.settledFolios?.[0] ? 'Multi-Folio' : '')
   361→   361→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })
   362→   362→        notify({ type: 'success', title: `Settlement ${docNo}`, message: data.message || `Settlement ${docNo} processed successfully` })
   363→   363→        closeDialogs()
   364→   364→        onRefresh?.()
   365→   365→        setTimeout(() => setEntrySuccess(null), 4000)
   366→   366→      } else {
   367→   367→        const e = await safeJson(res)
   368→   368→        notify({ type: 'error', title: 'Settlement Failed', message: (e.error as string) || 'Failed to process settlement' })
   369→   369→      }
   370→   370→    } catch {
   371→   371→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   372→   372→    } finally {
   373→   373→      setLocalSubmitting(false)
   374→   374→    }
   375→   375→  }, [detail?.id, selectedFolioId, apiHeaders, paymentForm, handleSettle, closeDialogs, onRefresh])
   376→   376→
   377→   377→  // ─── Transfer with TXN tracking ───
   378→   378→  const transferWithTxn = React.useCallback(async () => {
   379→   379→    if (!detail || !transferForm.targetCustomerId || !apiHeaders) {
   380→   380→      handleTransfer()
   381→   381→      return
   382→   382→    }
   383→   383→    try {
   384→   384→      setLocalSubmitting(true)
   385→   385→      // If entries are selected, transfer only those; otherwise pure balance transfer
   386→   386→      const selectedEntryIds = selectedEntries.size > 0 ? Array.from(selectedEntries) : undefined
   387→   387→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   388→   388→        method: 'POST',
   389→   389→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   390→   390→        body: JSON.stringify({
   391→   391→          targetCustomerId: transferForm.targetCustomerId,
   392→   392→          amount: parseFloat(transferForm.amount) || 0,
   393→   393→          description: transferForm.description,
   394→   394→          entryIds: selectedEntryIds,
