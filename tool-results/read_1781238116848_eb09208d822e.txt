   337→      const res = await fetch(`/api/folio/customers/${detail.id}/split`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify({ entryId, targetFolioId: splitAmountForm.targetFolioId, splitAmount: parseFloat(splitAmountForm.amount), description: 'Split Amount' }) })
   338→      if (res.ok) { notify({ type: 'success', title: 'Split Complete' }); setSplitAmountOpen(false); setSplitAmountForm({ ...DEFAULT_SPLIT_AMOUNT_FORM }); setSelectedEntries(new Set()); fetchFolioEntries(detail.id, selectedFolioId!); fetchDetail(detail.id, true) }
   339→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   340→    } catch { /* ignore */ } finally { setSubmitting(false) }
   341→  }
   342→
   343→  const handleCheckout = async () => {
   344→    if (!detail) return
   345→    try { setSubmitting(true)
   346→      const res = await fetch(`/api/folio/customers/${detail.id}/checkout`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' } })
   347→      if (res.ok) { notify({ type: 'success', title: 'Checkout Complete' }); setCheckoutOpen(false); refreshAll() }
   348→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   349→    } catch { /* ignore */ } finally { setSubmitting(false) }
   350→  }
   351→
   352→  // ── Handle Checkout Settlement (check balances first) ──
   353→  const handleCheckoutSettlement = async () => {
   354→    if (!detail) return
   355→    try { setSubmitting(true)
   356→      const res = await fetch(`/api/folio/customers/${detail.id}/settlement-overview`, { headers: hdrRef.current })
   357→      if (res.ok) { const data = await res.json(); setCheckoutSettlementData(data); setCheckoutSettlementOpen(true) }
   358→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   359→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to check settlement status' }) } finally { setSubmitting(false) }
   360→  }
   361→
   362→  // ── Auto-settle all outstanding folios ──
   363→  const handleAutoSettleAll = async () => {
   364→    if (!detail || !checkoutSettlementData) return
   365→    try { setSubmitting(true)
   366→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   367→        method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' },
   368→        body: JSON.stringify({ amount: checkoutSettlementData.totalOutstanding, mode: 'Cash', reference: '', description: 'Auto-settlement (all folios)', autoAllocate: true }),
   369→      })
   370→      if (res.ok) { notify({ type: 'success', title: 'All Folios Settled', message: `Settled across all folios` }); setCheckoutSettlementOpen(false); refreshAll(); fetchCheckoutSettlementData() }
   371→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   372→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to settle' }) } finally { setSubmitting(false) }
   373→  }
   374→
   375→  const fetchCheckoutSettlementData = async () => {
   376→    if (!detail) return
   377→    try {
   378→      const res = await fetch(`/api/folio/customers/${detail.id}/settlement-overview`, { headers: hdrRef.current })
   379→      if (res.ok) setCheckoutSettlementData(await res.json())
   380→    } catch { /* ignore */ }
   381→  }
   382→
   383→  // ── Multi-folio settlement (settle specific folio) ──
   384→  const handleMultiFolioSettlement = async (payload: SettlementPayload) => {
   385→    if (!detail) return
   386→    try { setSubmitting(true)
   387→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   388→        method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(payload),
   389→      })
   390→      if (res.ok) { notify({ type: 'success', title: 'Settlement Complete' }); fetchCheckoutSettlementData(); if (selectedFolioId && selectedFolioId !== 'master') fetchFolioEntries(detail.id, selectedFolioId); fetchMasterFolioEntries(detail.id); refreshAll() }
   391→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   392→    } catch { /* ignore */ } finally { setSubmitting(false) }
   393→  }
   394→
   395→  const handleDeleteFolio = async () => {
   396→    if (!detail || !folioToDelete) return
   397→    try { setSubmitting(true)
   398→      const res = await fetch(`/api/folio/customers/${detail.id}/folios/${folioToDelete}`, { method: 'DELETE', headers: hdrRef.current })
   399→      if (res.ok) { notify({ type: 'success', title: 'Folio Deleted' }); setDeleteFolioOpen(false); setFolioToDelete(null); fetchDetail(detail.id) }
   400→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   401→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to delete folio' }) } finally { setSubmitting(false) }
   402→  }
   403→
   404→  const handleReopen = async () => {
   405→    if (!detail) return
   406→    try { setSubmitting(true)
   407→      const res = await fetch(`/api/folio/customers/${detail.id}/reopen`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' } })
   408→      if (res.ok) { notify({ type: 'success', title: 'Reopened' }); refreshAll() }
   409→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   410→    } catch { /* ignore */ } finally { setSubmitting(false) }
   411→  }
   412→
   413→  const handleDelete = async () => {
   414→    if (!detail) return
   415→    try { setSubmitting(true)
   416→      const res = await fetch(`/api/folio/customers/${detail.id}`, { method: 'DELETE', headers: hdrRef.current })
   417→      if (res.ok) { notify({ type: 'success', title: 'Archived' }); setDeleteOpen(false); goBack(); fetchCustomers(); onRefreshData() }
   418→      else notify({ type: 'error', title: 'Error', message: 'Failed' })
   419→    } catch { /* ignore */ } finally { setSubmitting(false) }
   420→  }
   421→
   422→  const handleAddDoc = async () => { if (!detail || !docForm.name) { notify({ type: 'error', title: 'Error', message: 'Name required' }); return } try { setSubmitting(true); const res = await fetch(`/api/folio/customers/${detail.id}/documents`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(docForm) }); if (res.ok) { notify({ type: 'success', title: 'Document Added' }); setDocForm({ ...DEFAULT_DOC_FORM }); await fetchDetail(detail.id); } } catch { /* ignore */ } finally { setSubmitting(false) } }
   423→  const handleDeleteDoc = async (docId: string) => { if (!detail) return; try { await fetch(`/api/folio/customers/${detail.id}/documents/${docId}`, { method: 'DELETE', headers: hdrRef.current }); notify({ type: 'success', title: 'Deleted' }); await fetchDetail(detail.id) } catch { /* ignore */ } }
   424→  const handleAddNote = async () => { if (!detail || !noteForm.content) { notify({ type: 'error', title: 'Error', message: 'Content required' }); return } try { setSubmitting(true); const res = await fetch(`/api/folio/customers/${detail.id}/notes`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(noteForm) }); if (res.ok) { notify({ type: 'success', title: 'Note Added' }); setNoteForm({ ...DEFAULT_NOTE_FORM }); await fetchDetail(detail.id); } } catch { /* ignore */ } finally { setSubmitting(false) } }
   425→  const handleDeleteNote = async (noteId: string) => { if (!detail) return; try { await fetch(`/api/folio/customers/${detail.id}/notes/${noteId}`, { method: 'DELETE', headers: hdrRef.current }); notify({ type: 'success', title: 'Deleted' }); await fetchDetail(detail.id) } catch { /* ignore */ } }
   426→  const handleToggleTag = async (tagName: string) => { if (!detail) return; const has = customerTags.includes(tagName); try { if (has) { await fetch(`/api/folio/customers/${detail.id}/tags?tagName=${encodeURIComponent(tagName)}`, { method: 'DELETE', headers: hdrRef.current }); setCustomerTags(prev => prev.filter(t => t !== tagName)) } else { await fetch(`/api/folio/customers/${detail.id}/tags`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify({ tagName }) }); setCustomerTags(prev => [...prev, tagName]) } fetchDetail(detail.id, true); fetchCustomers(true) } catch { /* ignore */ } }
   427→
   428→  // ── Invoice / statement helpers ──
   429→  const fetchInvoiceData = async () => {
   430→    if (!detail) return
   431→    try { setInvoiceLoading(true); const params = new URLSearchParams(); if (invoiceFrom) params.set('from', invoiceFrom); if (invoiceTo) params.set('to', invoiceTo)
   432→      const res = await fetch(`/api/folio/customers/${detail.id}/statement?${params}`, { headers: hdrRef.current }); if (res.ok) setInvoiceData(await res.json()); else notify({ type: 'error', title: 'Error', message: 'Failed' })
   433→    } catch { /* ignore */ } finally { setInvoiceLoading(false) }
   434→  }
   435→  const fetchFolioStatement = async (folioId: string) => {
   436→    if (!detail) return
   437→    try { setFolioStatementData(null)
   438→      const res = await fetch(`/api/folio/customers/${detail.id}/statement?folioId=${folioId}`, { headers: hdrRef.current }); if (res.ok) setFolioStatementData(await res.json())
   439→    } catch { /* ignore */ }
   440→  }
   441→
   442→  // ── Print wrappers (async, using ref) ──
   443→  const bal = detail?.balance ?? 0
   444→  const isActive = detail?.status === 'Active'
   445→  const isCheckedOut = detail?.status === 'CheckedOut'
   446→  // Include all payment/settlement/refund entries from ALL folios for voucher history
   447→  const paymentEntries = masterFolioEntries.filter(e => e.type === 'Payment' || e.type === 'Settlement' || e.type === 'Refund')
   448→  const selectedEntryObj = selectedEntries.size === 1 ? folioEntries.find(e => e.id === Array.from(selectedEntries)[0]) : null
   449→
   450→  const doPrintReceipt = async () => {
   451→    // After checkout, folioEntries may be empty — use masterFolioEntries as fallback
   452→    const entriesToPrint = folioEntries.length > 0 ? folioEntries : masterFolioEntries
   453→    await _printReceipt(detail!, entriesToPrint, hdrRef as any)
   454→  }
   455→  const doPrintInvoice = () => { _printInvoice(detail!, invoiceData, invoiceFrom, invoiceTo) }
   456→  const doPrintSingleFolio = () => { _printSingleFolio(detail!, folioStatementData) }
   457→  const doPrintVoucherHistory = async () => { await _printVoucherHistory(detail!, paymentEntries, hdrRef as any) }
   458→
   459→  // ── RENDER ──
   460→  return (
   461→    <FolioErrorBoundary>
   462→    <TooltipProvider delayDuration={200}>
   463→    <>
   464→    {!canAccess && <ReadOnlyBanner module="Customer Folio" noAccess />}
   465→
   466→    <div className={isFullscreen ? 'fixed inset-0 z-[60] flex flex-col' : ''}>
   467→      <div className={`flex flex-col ${isFullscreen ? 'h-full' : 'min-h-0'} border border-slate-200 dark:border-slate-700 ${isFullscreen ? 'border-0 rounded-none' : 'rounded-lg'} overflow-hidden bg-white dark:bg-slate-900 shadow-sm`}>
   468→
   469→        {/* ── HEADER BAR ── */}
   470→        <div className="bg-white dark:bg-slate-900 border-b border-slate-200 dark:border-slate-700 px-4 py-2.5 flex items-center justify-between shrink-0">
   471→          <div className="flex items-center gap-3">
   472→            <div className="w-9 h-9 rounded-xl bg-gradient-to-br from-emerald-600 to-emerald-700 flex items-center justify-center shadow-sm shadow-emerald-600/20">
   473→              <span className="text-white text-sm font-bold">CF</span>
   474→            </div>
   475→            <div>
   476→              <h2 className="text-sm font-bold text-slate-800 dark:text-slate-100">Customer Folio</h2>
   477→              <p className="text-[10px] text-slate-400 dark:text-slate-500">Guest profiles &amp; billing management</p>
   478→            </div>
   479→          </div>
   480→          <div className="flex items-center gap-2">
   481→            <button onClick={() => setIsFullscreen(!isFullscreen)} className="text-[10px] text-slate-500 dark:text-slate-400 hover:text-slate-700 dark:hover:text-slate-200 px-2 py-1 rounded border border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-800 transition-colors">
   482→              {isFullscreen ? 'Exit Full' : 'Fullscreen'}
   483→            </button>
   484→          </div>
   485→        </div>
   486→
   487→        {/* ── MAIN BODY ── */}
   488→        <div className="flex flex-col flex-1 min-h-0">
   489→          {/* CUSTOMER PROFILE GRID VIEW (when no customer selected) */}
   490→          {!selectedId && (
   491→            <CustomerFolioProfileGrid
   492→              customers={customers} loading={loading} search={search} setSearch={setSearch}
   493→              filter={filter} setFilter={setFilter} sort={sort} setSort={setSort}
   494→              canCreate={canCreate} onAddCustomer={() => { resetForm(); setAddOpen(true) }}
   495→              onSelectCustomer={openCustomer}
   496→            />
   497→          )}
   498→
   499→          {/* CUSTOMER DETAIL VIEW (when customer selected) — full width */}
   500→          {selectedId && detail && detailLoading && !folioEntries.length ? (
   501→            <div className="flex-1 flex flex-col items-center justify-center">
   502→              <Loader2 className="size-8 animate-spin text-emerald-500 mb-3" />
   503→              <p className="text-sm text-muted-foreground">Loading customer details...</p>
   504→            </div>
   505→          ) : selectedId && detail ? (
   506→            <CustomerFolioManagement
   507→              detail={detail} allFolios={allFolios} folioEntries={folioEntries}
   508→              selectedEntries={selectedEntries} selectedFolioId={selectedFolioId}
   509→              folioEntriesLoading={folioEntriesLoading} submitting={submitting}
   510→              isAdmin={isAdmin} isActive={isActive} isCheckedOut={isCheckedOut}
   511→              folioStatementData={folioStatementData}
   512→              masterFolioEntries={masterFolioEntries}
   513→              onCheckoutSettlement={handleCheckoutSettlement}
   514→              goBack={goBack} setSelectedFolioId={setSelectedFolioId}
   515→              setSelectedEntries={setSelectedEntries} fetchFolioEntries={fetchFolioEntries}

   529→              onShift={() => setShiftOpen(true)}
   530→              onSplit={() => { if (selectedEntries.size === 1) { setSplitAmountForm({ ...DEFAULT_SPLIT_AMOUNT_FORM }); setSplitAmountOpen(true) } else notify({ type: 'info', title: 'Info', message: 'Select 1 entry to split' }) }}
   531→              onRemove={() => setRemoveOpen(true)}
   532→              onEdit={() => { populateEditForm(); setEditOpen(true) }}
   533→              onDetails={() => setDetailsSheetOpen(true)}
   534→              onTags={() => setTagsOpen(true)}
   535→              onNotes={() => setNoteOpen(true)}
   536→              onDocs={() => setDocOpen(true)}
   537→              onDelete={() => setDeleteOpen(true)}
   538→              onDeleteFolio={(folioId: string) => { setFolioToDelete(folioId); setDeleteFolioOpen(true) }}
   539→              isFullscreen={isFullscreen}
   540→            />
   541→          ) : null}

   546→      {/* All Dialogs */}
   547→        <CustomerFolioDialogs

   549→          submitting={submitting} settlingMode={settlingMode} selectedFolioId={selectedFolioId}
   550→          bal={bal} selectedEntryObj={selectedEntryObj} isActive={isActive}
   551→          isCheckedOut={isCheckedOut} invoiceData={invoiceData} invoiceLoading={invoiceLoading}
   552→          folioStatementData={folioStatementData} customerTags={customerTags}
   553→          transferResults={transferResults} paymentEntries={paymentEntries}
   554→          addOpen={addOpen} editOpen={editOpen} chargeOpen={chargeOpen}
   555→          paymentOpen={paymentOpen} refundOpen={refundOpen} transferOpen={transferOpen}
   556→          discountOpen={discountOpen} docOpen={docOpen} noteOpen={noteOpen}
   557→          tagsOpen={tagsOpen} checkoutOpen={checkoutOpen} deleteOpen={deleteOpen}

   560→          form={form} chargeForm={chargeForm} paymentForm={paymentForm}
   561→          refundForm={refundForm} transferForm={transferForm} discountForm={discountForm}
   562→          docForm={docForm} noteForm={noteForm} splitAmountForm={splitAmountForm}
   563→          invoiceFrom={invoiceFrom} invoiceTo={invoiceTo} shiftTargetFolioId={shiftTargetFolioId}
   564→          duplicate={duplicate}
   565→          setAddOpen={setAddOpen} setEditOpen={setEditOpen} setChargeOpen={setChargeOpen}
   566→          setPaymentOpen={setPaymentOpen} setRefundOpen={setRefundOpen} setTransferOpen={setTransferOpen}
   567→          setDiscountOpen={setDiscountOpen} setDocOpen={setDocOpen} setNoteOpen={setNoteOpen}
   568→          setTagsOpen={setTagsOpen} setCheckoutOpen={setCheckoutOpen} setDeleteOpen={setDeleteOpen}
