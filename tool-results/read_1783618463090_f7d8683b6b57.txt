    38→// ─── Section-level Error Boundary ───
    39→// Catches rendering errors in ANY section so they don't crash the entire page
    40→class SectionErrorBoundary extends React.Component<{ children: React.ReactNode; sectionName: string }, { hasError: boolean; error: Error | null }> {
    41→  state = { hasError: false, error: null as Error | null }
    42→  static getDerivedStateFromError(error: Error) {
    43→    return { hasError: true, error }
    44→  }
    45→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    46→    console.error(`[SectionErrorBoundary:${this.props.sectionName}]`, error?.message, error?.stack, info?.componentStack)
    47→    // Auto-retry ChunkLoadError once after a short delay (module system may recover)
    48→    const isChunkError = error?.name === 'ChunkLoadError' || (error?.message && error.message.includes('Loading chunk'))
    49→    if (isChunkError) {
    50→      setTimeout(() => {
    51→        this.setState({ hasError: false, error: null })
    52→      }, 800)
    53→    }
    54→  }
    55→  handleRetry = () => {
    56→    const isChunkError = this.state.error?.name === 'ChunkLoadError' || (this.state.error?.message && this.state.error.message.includes('Loading chunk'))
    57→    if (isChunkError) {
    58→      // For chunk errors, a full page reload is the most reliable fix
    59→      window.location.reload()
    60→    } else {
    61→      this.setState({ hasError: false, error: null })
    62→    }
    63→  }
    64→  render() {
    65→    if (this.state.hasError) {
    66→      const errMsg = this.state.error?.message || 'Unknown error'
    67→      const errName = this.state.error?.name || 'Error'
    68→      const isChunkError = errName === 'ChunkLoadError' || errMsg.includes('Loading chunk')
    69→      const stackLines = (this.state.error?.stack || '').split('\n').filter((l: string) => l.includes('src/')).slice(0, 2).map((l: string) => l.trim())
    70→      return (
    71→        <div className="flex flex-col items-center justify-center py-20 px-6">
    72→          <div className="flex size-14 items-center justify-center rounded-2xl bg-red-100 dark:bg-red-900/30 mb-4">
    73→            <AlertCircle className="size-7 text-red-500" />
    74→          </div>
    75→          <h3 className="text-base font-bold text-slate-800 dark:text-slate-200 mb-1">Section Error: {this.props.sectionName}</h3>
    76→          <p className="text-xs text-red-500 dark:text-red-400 mb-1 font-mono">{errName}: {errMsg}</p>
    77→          {isChunkError && (
    78→            <p className="text-xs text-slate-500 dark:text-slate-400 mb-2 mt-1">This usually fixes itself after a page reload.</p>
    79→          )}
    80→          {stackLines.length > 0 && !isChunkError && (
    81→            <pre className="text-[10px] text-slate-400 dark:text-slate-500 max-w-lg text-center overflow-x-auto mb-4 whitespace-pre-wrap mt-1">{stackLines.join('\n')}</pre>
    82→          )}
    83→          <Button variant="outline" size="sm" onClick={this.handleRetry} className="gap-1.5 mt-2">
    84→            <RefreshCcw className="size-3" /> {isChunkError ? 'Reload Page' : 'Try Again'}
    85→          </Button>
    86→        </div>
    87→      )
    88→    }
    89→    return this.props.children
    90→  }
    91→}
