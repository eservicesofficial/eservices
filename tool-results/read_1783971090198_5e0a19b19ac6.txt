    12→    12→function numberToIndianWords(num: number): string {
    13→    13→  if (num === 0) return 'ZERO'
    14→    14→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    15→    15→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    16→    16→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    17→    17→  let words = ''
    18→    18→  function convertBelowThousand(n: number): string {
    19→    19→    let w = ''
    20→    20→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    21→    21→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    22→    22→    else if (n > 0) { w += ones[n] }
    23→    23→    return w
    24→    24→  }
    25→    25→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    26→    26→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    27→    27→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    28→    28→  if (num > 0) { words += convertBelowThousand(num) }
    29→    29→  return words.trim().toUpperCase() + ' ONLY'
    30→    30→}

   210→   210→// ─── Helper: Compute the most-used payment mode from credit entries ───
   211→   211→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   212→   212→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   213→   213→  if (creditEntries.length === 0) return 'N/A'
   214→   214→  const modeCounts: Record<string, number> = {}
   215→   215→  for (const e of creditEntries) {
   216→   216→    const mode = e.paymentMode || 'Cash'
   217→   217→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   218→   218→  }
   219→   219→  let bestMode = 'Cash'
   220→   220→  let bestAmount = 0
   221→   221→  for (const [mode, amount] of Object.entries(modeCounts)) {
   222→   222→    if (amount > bestAmount) {
   223→   223→      bestAmount = amount
   224→   224→      bestMode = mode
   225→   225→    }
   226→   226→  }
   227→   227→  return bestMode
   228→   228→}

   230→   230→// ─── Helper: Build payment summary HTML ───
   231→   231→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   232→   232→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   233→   233→  if (creditEntries.length === 0) {
   234→   234→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   235→   235→  }
   236→   236→
   237→   237→  const modeTotals: Record<string, number> = {}
   238→   238→  for (const e of creditEntries) {
   239→   239→    const mode = e.paymentMode || 'Cash'
   240→   240→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   241→   241→  }
   242→   242→
   243→   243→  const modes = Object.entries(modeTotals)
   244→   244→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   245→   245→
   246→   246→  if (modes.length === 1) {
   247→   247→    const [mode, amount] = modes[0]
   248→   248→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   249→   249→  }

   399→   399→export function buildHotelStyleInvoice(

   402→   402→  entries: any[],

   405→   405→  docTitle: string,

   408→   408→): string {
   409→   409→  const d = detail as any
   410→   410→  const bizName = settings.shopName || 'Eservices'
   411→   411→  const bizAddress = settings.shopAddress || ''
   412→   412→  const bizPhone = settings.shopPhone || ''
   413→   413→  const bizEmail = settings.shopEmail || ''
   414→   414→  const gstNumber = settings.gstNumber || settings.gstin || ''

   416→   416→  const gstState = settings.gstState || settings.gstStateCode || ''
   417→   417→  const stateCode = gstState ? gstState.split('-')[0] || gstState : ''

   421→   421→  const gstEnabled = settings.enableGst === 'true'
   422→   422→  const gstRate = parseFloat(settings.gstRate) || 0

   424→   424→  // Customer info
   425→   425→  const custName = d.fullName || ''
   426→   426→  const custId = d.customerId || ''
   427→   427→  const custMobile = d.mobile || ''
   428→   428→  const custEmail = d.email || ''
   429→   429→  const custAddr = d.address || ''
   430→   430→  const custGst = d.gstNumber || ''

   432→   432→  const custPan = d.panNumber || ''
   433→   433→  const custCompany = d.companyName || d.company || ''
   434→   434→
   435→   435→  // Calculate totals
   436→   436→  const chargeEntries = entries.filter((e: any) => (e.debit || 0) > 0 && e.type !== 'Refund')
   437→   437→  const paymentEntries = entries.filter((e: any) => (e.credit || 0) > 0)
   438→   438→  const refundEntries = entries.filter((e: any) => e.type === 'Refund')
   439→   439→  const totalDebits = chargeEntries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   440→   440→  const totalCredits = paymentEntries.reduce((s: number, e: any) => s + (e.credit || 0), 0)
   441→   441→  const totalRefunds = refundEntries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   442→   442→
   443→   443→  // P10a: GST Inclusive calculation
   444→   444→  // If GST is enabled in settings, compute GST from charge amounts even if individual entries have tax=0
   445→   445→  let totalTax = 0
   446→   446→  let computedTaxFromSettings = false
   447→   447→  if (gstEnabled && gstRate > 0) {
   448→   448→    // Check if any charge has explicit tax > 0
   449→   449→    const hasExplicitTax = chargeEntries.some((e: any) => (e.tax || 0) > 0)
   450→   450→    if (hasExplicitTax) {
   451→   451→      // Use the explicit tax values from entries
   452→   452→      totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   453→   453→    } else {
   454→   454→      // P10a: No explicit GST on charges — compute from inclusive amounts using settings GST rate
   455→   455→      // For inclusive GST: GST = Amount × Rate / (100 + Rate), Taxable = Amount - GST
   456→   456→      totalTax = Math.round((totalDebits * gstRate / (100 + gstRate)) * 100) / 100
   457→   457→      computedTaxFromSettings = true
   458→   458→    }
   459→   459→  } else {
   460→   460→    // GST not enabled — use whatever tax is on entries (usually 0)

   472→   472→  const effectiveGstRate = netAmount > 0 ? ((totalTax / netAmount) * 100) : 0

   826→   826→export async function printVoucherHistory(
   827→   827→  detail: CustomerDetail,
   828→   828→  paymentEntries: LedgerEntry[],
   829→   829→  hdrRef: RefObject<Record<string, string>>,
   830→   830→  printedBy?: string
   831→   831→) {

   835→   835→  const now = new Date()
   836→   836→
   837→   837→  const sorted = [...paymentEntries].sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
   838→   838→  const totalAmount = sorted.reduce((s, e) => s + (e.credit || 0), 0)
   839→   839→
   840→   840→  // P5: Use first payment's invoiceNo (VOU-xxxx) as permanent voucher number
   841→   841→  // Fallback to docNo if invoiceNo is empty (for entries created before this fix)
   842→   842→  const voucherNo = sorted[0]?.invoiceNo || sorted[0]?.docNo || getStableDocId(sorted)
   843→   843→

   845→   845→  const voucherEntries = sorted.map(e => ({
   846→   846→    ...e,
   847→   847→    debit: 0,
   848→   848→    credit: e.credit || 0,
   849→   849→    balance: 0,
   850→   850→    type: 'Payment',

   852→   852→  }))
   853→   853→
   854→   854→  // Compute running balance for display
   855→   855→  let runBal = 0
   856→   856→  for (const e of voucherEntries) {
   857→   857→    runBal -= (e.credit || 0)
   858→   858→    e.balance = Math.round(runBal * 100) / 100
   859→   859→  }
   860→   860→
   861→   861→  const w = openPrintWindow(`Payment Voucher - ${detail.fullName}`)
   862→   862→  if (!w) return
   863→   863→

   865→   865→    `<!DOCTYPE html><html><head><title>Payment Voucher</title><style>${INVOICE_CSS}</style><style>${buildTemplateCSSFromSettings(shopSettings)}</style></head><body>` +
   866→   866→    buildHotelStyleInvoice(shopSettings, detail, voucherEntries, voucherNo, fmtDate(now.toISOString()), 'PAYMENT VOUCHER', {
   867→   867→      openingBalance: 0,
   868→   868→      closingBalance: -totalAmount,
   869→   869→    }, printedBy) +
   870→   870→    `</body></html>`
   871→   871→  )
   872→   872→  w.document.close()
   873→   873→  w.print()
