   138→export function CustomerFolioDialogs({
   139→  detail, allFolios, selectedEntries, submitting, settlingMode,
   140→  selectedFolioId, bal, selectedEntryObj, isActive, isCheckedOut,
   141→  invoiceData, invoiceLoading, folioStatementData, customerTags,

   143→  addOpen, editOpen, chargeOpen, paymentOpen, refundOpen, transferOpen,

   147→  form, chargeForm, paymentForm, refundForm, transferForm, discountForm,

   155→  setCheckoutOpen, setDeleteOpen, setSplitAmountOpen, setShiftOpen,
   156→  setRemoveOpen, setVoucherOpen, setInvoiceOpen, setDetailsSheetOpen,
   157→  setForm, setChargeForm, setPaymentForm, setRefundForm, setTransferForm,
   158→  setDiscountForm, setDocForm, setNoteForm, setSplitAmountForm,

   162→  handleRemove, handleCheckout, handleDelete, handleAddDoc, handleAddNote,

   165→  closeDialogs, searchTransferTargets,
   166→  deleteFolioOpen, folioToDelete,
   167→  setDeleteFolioOpen, handleDeleteFolio,
   168→  // Enhanced props
   169→  apiHeaders, onRefresh, onPrintReceipt, onThermalPrintReceipt,
   170→  onThermalPrintInvoice, onThermalPrintVoucher,
   171→  lastCreatedDocNo, lastCreatedType,
   172→  // History
   173→  historyOpen, setHistoryOpen, onPrintHistoryFolio,
   174→}: DialogsProps) {
   175→
   176→  // ─── Internal State ───
   177→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   178→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   179→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   180→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   181→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   182→  const [splitMode, setSplitMode] = React.useState(false)
   183→  const [localSplitItems, setLocalSplitItems] = React.useState<{ method: string; amount: string; reference: string }[]>([
   184→    { method: 'Cash', amount: '', reference: '' },
   185→  ])
   186→
   187→  // ─── History state ───
   188→  const [historyLoading, setHistoryLoading] = React.useState(false)
   189→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   190→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   191→  const [expandedFolioEntries, setExpandedFolioEntries] = React.useState<Record<string, any[]>>({})
   192→  const [selectedVoucherIds, setSelectedVoucherIds] = React.useState<Set<string>>(new Set())
   193→
   194→  // Fetch history for the CURRENT customer (not all checked-out customers)
   195→  React.useEffect(() => {
   196→    if (historyOpen && detail && apiHeaders) {
   197→      fetchCustomerHistory()
   198→    } else if (!historyOpen) {
   199→      setCustomerHistory(null)
   200→      setExpandedFolioId(null)
   201→      setExpandedFolioEntries({})
   202→    }
   203→  }, [historyOpen, detail?.id])
   204→
   205→  const fetchCustomerHistory = React.useCallback(async () => {
   206→    if (!detail || !apiHeaders) return
   207→    try {
   208→      setHistoryLoading(true)
   209→      const res = await fetch(`/api/folio/customers/${detail.id}/history`, { headers: apiHeaders })
   210→      if (res.ok) {
   211→        const data = await res.json() as CustomerHistory
   212→        setCustomerHistory(data)
   213→      }
   214→    } catch { /* ignore */ } finally { setHistoryLoading(false) }
   215→  }, [detail?.id, apiHeaders])
   216→
   217→  const toggleFolioEntries = (folioId: string) => {
   218→    if (expandedFolioId === folioId) {
   219→      setExpandedFolioId(null)
   220→    } else {
   221→      const folio = customerHistory?.historicalFolios.find(f => f.folioId === folioId)
   222→      if (folio) {
   223→        setExpandedFolioId(folioId)
   224→        setExpandedFolioEntries(prev => ({
   225→          ...prev,
   226→          [folioId]: (folio.entries || []),
   227→        }))
   228→      }
   229→    }
   230→  }
   231→
   232→  // ─── Track checkout transition ───
   233→  const wasCheckedOut = React.useRef(isCheckedOut)
   234→  React.useEffect(() => {
   235→    if (isCheckedOut && !wasCheckedOut.current) {
   236→      setPostCheckoutOpen(true)
   237→    }
   238→    wasCheckedOut.current = isCheckedOut
   239→  }, [isCheckedOut])
   240→
   241→  // ─── Show TXN number when lastCreatedDocNo changes ───
   242→  React.useEffect(() => {
   243→    if (lastCreatedDocNo && lastCreatedType) {
   244→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   245→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   246→      return () => clearTimeout(timer)
   247→    }
   248→  }, [lastCreatedDocNo, lastCreatedType])
   249→
   250→  // ─── Items for charge selector (Req 5) ───
   251→  const [items, setItems] = React.useState<{ id: string; name: string; price: number }[]>([])
   252→  const [itemsLoading, setItemsLoading] = React.useState(false)
   253→  React.useEffect(() => {
   254→    if (chargeOpen && apiHeaders) {
   255→      setItemsLoading(true)
   256→      fetch('/api/items', { headers: apiHeaders }).then(r => r.ok ? r.json() : []).then((list: any[]) => {
   257→        setItems((list || []).map((it: any) => ({ id: it.id, name: it.name, price: it.price || 0 })))
   258→      }).catch(() => {}).finally(() => setItemsLoading(false))
   259→    }
   260→  }, [chargeOpen, apiHeaders])
   261→
   262→  // Handle item selection
   263→  const handleItemSelect = React.useCallback((itemId: string) => {
   264→    const item = items.find(it => it.id === itemId)
   265→    if (item) {
   266→      setChargeForm({
   267→        ...chargeForm,
   268→        selectedItem: itemId,
   269→        selectedItemPrice: String(item.price),
   270→        description: item.name,
   271→        amount: item.price > 0 ? String(item.price) : chargeForm.amount,
   272→      })
   273→    }
   274→  }, [items, chargeForm])
   275→
   276→  // ─── Fetch invoice number ───
   277→  const fetchInvoiceNumber = React.useCallback(async () => {
   278→    if (!detail || !apiHeaders) return
   279→    try {
   280→      setInvNumLoading(true)
   281→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   282→        method: 'POST',
   283→        headers: apiHeaders,
   284→      })
   285→      if (res.ok) {
   286→        const data = await res.json()
   287→        setInvoiceNumber(data.invoiceNumber)
   288→      }
   289→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   290→  }, [detail?.id, apiHeaders])
   291→
   292→  // ─── Fetch invoice number when invoice dialog opens ───
   293→  React.useEffect(() => {
   294→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber) {
   295→      fetchInvoiceNumber()
   296→    }
   297→  }, [invoiceOpen, detail?.id, apiHeaders])
   298→
   299→  React.useEffect(() => {
   300→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   301→      fetchInvoiceData()
   302→    }
   303→  }, [invoiceOpen])
   304→
   305→  // ─── Entry creation with TXN tracking ───
   306→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   307→    if (!detail || !selectedFolioId || !apiHeaders) {
   308→      return null
   309→    }
   310→    try {
   311→      setLocalSubmitting(true)
   312→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   313→        method: 'POST',
   314→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   315→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   316→      })
   317→      if (res.ok) {
   318→        const entry = await res.json()
   319→        const docNo = entry?.docNo || ''
   320→        setEntrySuccess({ show: true, docNo, type })
   321→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   322→        closeDialogs()
   323→        onRefresh?.()
   324→        setTimeout(() => setEntrySuccess(null), 4000)
   325→        return entry
   326→      } else {
   327→        const e = await safeJson(res)
   328→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   329→        return null
   330→      }
   331→    } catch {
   332→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   333→      return null
   334→    } finally {
   335→      setLocalSubmitting(false)
   336→    }
   337→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   338→
   339→  // ─── Settlement with TXN tracking ───
   340→  const settleWithTxn = React.useCallback(async () => {
   341→    if (!detail || !paymentForm.amount || !apiHeaders) {
   342→      handleSettle()
   343→      return
   344→    }
   345→    try {
   346→      setLocalSubmitting(true)
   347→      // When master folio is selected, use autoAllocate to distribute across all folios
   348→      const isMaster = selectedFolioId === 'master'
   349→      const payload = isMaster
   350→        ? { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   351→        : { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   352→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   353→        method: 'POST',
   354→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   355→        body: JSON.stringify(payload),
   356→      })
   357→      if (res.ok) {
   358→        const data = await res.json()
   359→        // Auto-allocate may return settledFolios array instead of single ledgerEntry
   360→        const docNo = data?.ledgerEntry?.docNo || (data?.settledFolios?.[0] ? 'Multi-Folio' : '')
   361→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })
   362→        notify({ type: 'success', title: `Settlement ${docNo}`, message: data.message || `Settlement ${docNo} processed successfully` })
   363→        closeDialogs()
   364→        onRefresh?.()
   365→        setTimeout(() => setEntrySuccess(null), 4000)
   366→      } else {
   367→        const e = await safeJson(res)
   368→        notify({ type: 'error', title: 'Settlement Failed', message: (e.error as string) || 'Failed to process settlement' })
   369→      }
   370→    } catch {
   371→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   372→    } finally {
   373→      setLocalSubmitting(false)
   374→    }
   375→  }, [detail?.id, selectedFolioId, apiHeaders, paymentForm, handleSettle, closeDialogs, onRefresh])
   376→
   377→  // ─── Transfer with TXN tracking ───
   378→  const transferWithTxn = React.useCallback(async () => {
   379→    if (!detail || !transferForm.targetCustomerId || !apiHeaders) {
   380→      handleTransfer()
   381→      return
   382→    }
   383→    try {
   384→      setLocalSubmitting(true)
   385→      // If entries are selected, transfer only those; otherwise pure balance transfer
   386→      const selectedEntryIds = selectedEntries.size > 0 ? Array.from(selectedEntries) : undefined
   387→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   388→        method: 'POST',
   389→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   390→        body: JSON.stringify({
   391→          targetCustomerId: transferForm.targetCustomerId,
   392→          amount: parseFloat(transferForm.amount) || 0,
   393→          description: transferForm.description,
   394→          entryIds: selectedEntryIds,
   395→        }),
   396→      })
   397→      if (res.ok) {
   398→        const data = await res.json()
   399→        const docNo = data?.source?.entry?.docNo || ''
   400→        setEntrySuccess({ show: true, docNo, type: 'Transfer' })
   401→        notify({ type: 'success', title: `Transfer ${docNo}`, message: data.message || `Transfer ${docNo} processed successfully` })
   402→        closeDialogs()
   403→        onRefresh?.()
   404→        setTimeout(() => setEntrySuccess(null), 4000)
   405→      } else {
   406→        const e = await safeJson(res)
   407→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   408→      }
   409→    } catch {
   410→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   411→    } finally {
   412→      setLocalSubmitting(false)
   413→    }
   414→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   415→
   416→  // ─── Split payment helpers for checkout ───
   417→  const addSplitItem = () => {
   418→    setLocalSplitItems(prev => [...prev, { method: 'Cash', amount: '', reference: '' }])
   419→  }
   420→  const removeSplitItem = (idx: number) => {
   421→    setLocalSplitItems(prev => prev.filter((_, i) => i !== idx))
   422→  }
   423→  const updateSplitItem = (idx: number, field: string, value: string) => {
   424→    setLocalSplitItems(prev => prev.map((item, i) => i === idx ? { ...item, [field]: value } : item))
   425→  }
   426→  const totalSplitAmount = localSplitItems.reduce((sum, item) => sum + (parseFloat(item.amount) || 0), 0)
   427→  const effectiveOutstanding = checkoutSettlementData ? Math.max(0, checkoutSettlementData.totalOutstanding) : 0
   428→  const splitRemaining = checkoutSettlementData ? Math.max(0, effectiveOutstanding - totalSplitAmount) : 0
   429→
   430→  // ─── Computed GST data from invoiceData ───
   431→  const gstData = React.useMemo(() => {
   432→    if (!invoiceData?.entries) return null
   433→    const entries = invoiceData.entries as any[]
   434→    const taxableEntries = entries.filter((e: any) => e.debit > 0)
   435→    const taxableAmount = taxableEntries.reduce((sum: number, e: any) => sum + (e.debit || 0) - (e.tax || 0) - (e.discount || 0), 0)
   436→    const totalTax = entries.reduce((sum: number, e: any) => sum + (e.tax || 0), 0)
   437→    const totalDiscount = entries.reduce((sum: number, e: any) => sum + (e.discount || 0), 0)
   438→    const totalCharges = entries.reduce((sum: number, e: any) => sum + (e.debit || 0), 0)
   439→    const cgst = totalTax / 2
   440→    const sgst = totalTax / 2
   441→    const totalPayments = entries.reduce((sum: number, e: any) => sum + (e.credit || 0), 0)
   442→    const advancePayment = bal < 0 ? Math.abs(bal) : 0
   443→    return { taxableAmount, totalTax, totalDiscount, totalCharges, cgst, sgst, totalPayments, advancePayment, entryCount: taxableEntries.length }
   444→  }, [invoiceData, bal])
   445→
   446→  // ─── Active submitting state (local or prop) ───
   447→  const isSubmitting = localSubmitting || submitting
   448→
   449→  // ─── Entry success banner component ───
   450→  const EntrySuccessBanner = () => {
   451→    if (!entrySuccess?.show) return null
   452→    return (
