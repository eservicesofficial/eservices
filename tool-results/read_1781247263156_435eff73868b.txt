   136→   185→function downloadCSV(data: Record<string, any>[], filename: string) {
   137→   186→  if (!data.length) return
   138→   187→  const headers = Object.keys(data[0])
   139→   188→  const csv = [
   140→   189→    headers.join(','),
   141→   190→    ...data.map(row => headers.map(h => `"${String(row[h] ?? '').replace(/"/g, '""')}"`).join(','))
   142→   191→  ].join('\n')
   143→   192→  const blob = new Blob([csv], { type: 'text/csv' })
   144→   193→  const url = URL.createObjectURL(blob)
   145→   194→  const a = document.createElement('a')
   146→   195→  a.href = url; a.download = `${filename}.csv`; a.click()
   147→   196→  URL.revokeObjectURL(url)
   148→   197→}
