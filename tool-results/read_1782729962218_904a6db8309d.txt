   350→   350→      }
   351→   351→    } catch {
   352→   352→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   353→   353→    } finally {
   354→   354→      setLocalSubmitting(false)
   355→   355→    }

   363→   363→    }

   367→   367→      const selectedEntryIds = selectedEntries.size > 0 ? Array.from(selectedEntries) : undefined
   368→   368→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   369→   369→        method: 'POST',
   370→   370→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   371→   371→        body: JSON.stringify({
   372→   372→          targetCustomerId: transferForm.targetCustomerId,
   373→   373→          amount: parseFloat(transferForm.amount) || 0,
   374→   374→          description: transferForm.description,
   375→   375→          entryIds: selectedEntryIds,
   376→   376→        }),
   377→   377→      })
   378→   378→      if (res.ok) {
   379→   379→        const data = await res.json()
   380→   380→        const docNo = data?.source?.entry?.docNo || ''
   381→   381→        setEntrySuccess({ show: true, docNo, type: 'Transfer' })
   382→   382→        notify({ type: 'success', title: `Transfer ${docNo}`, message: data.message || `Transfer ${docNo} processed successfully` })
   383→   383→        closeDialogs()
   384→   384→        onRefresh?.()
   385→   385→        setTimeout(() => setEntrySuccess(null), 4000)
   386→   386→      } else {
   387→   387→        const e = await safeJson(res)
   388→   388→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   389→   389→      }
   390→   390→    } catch {
   391→   391→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   392→   392→    } finally {
   393→   393→      setLocalSubmitting(false)
   394→   394→    }
   395→   395→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   396→   396→
