   289→      return null
   290→    }
   291→    try {
   292→      setLocalSubmitting(true)
   293→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   294→        method: 'POST',
   295→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },

   297→      })

   389→      }
   390→    } catch {
   391→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   392→    } finally {
   393→      setLocalSubmitting(false)
   394→    }
   395→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])

   398→  const gstData = React.useMemo(() => {

   401→    const taxableEntries = entries.filter((e: any) => e.debit > 0)
   402→    const taxableAmount = taxableEntries.reduce((sum: number, e: any) => sum + (e.debit || 0) - (e.tax || 0) - (e.discount || 0), 0)
   403→    const totalTax = entries.reduce((sum: number, e: any) => sum + (e.tax || 0), 0)
   404→    const totalDiscount = entries.reduce((sum: number, e: any) => sum + (e.discount || 0), 0)
   405→    const totalCharges = entries.reduce((sum: number, e: any) => sum + (e.debit || 0), 0)
   406→    const cgst = totalTax / 2
   407→    const sgst = totalTax / 2
   408→    const totalPayments = entries.reduce((sum: number, e: any) => sum + (e.credit || 0), 0)
   409→    const advancePayment = bal < 0 ? Math.abs(bal) : 0
   410→    return { taxableAmount, totalTax, totalDiscount, totalCharges, cgst, sgst, totalPayments, advancePayment, entryCount: taxableEntries.length }
   411→  }, [invoiceData, bal])

   425→          <div className="min-w-0">
   426→            <div className="text-xs font-bold">{entrySuccess.type} Created</div>
   427→            <div className="text-[11px] text-emerald-100 flex items-center gap-1 mt-0.5">
   428→              <Hash className="size-3" />
   429→              <span className="font-mono font-bold">{entrySuccess.docNo}</span>
   430→            </div>
   431→          </div>
   432→        </div>
   433→      </div>
   434→    )
   435→  }
   436→
   437→  // ─── Tax calculation preview ───
   438→  const TaxPreview = ({ chargeForm }: { chargeForm: ChargeForm }) => {
   439→    if (!chargeForm.taxEnabled || !chargeForm.amount || parseFloat(chargeForm.amount) <= 0) return null
   440→    const taxPct = parseFloat(chargeForm.tax) || 0
   441→    const amt = parseFloat(chargeForm.amount) || 0
   442→    if (taxPct <= 0) return null
   443→    if (chargeForm.taxType === 'exclude') {
   444→      const taxAmt = amt * (taxPct / 100)
   445→      const total = amt + taxAmt
   446→      return <div className="pl-6 bg-emerald-50 dark:bg-emerald-900/20 rounded-md p-2 text-[11px] space-y-0.5"><div className="text-muted-foreground">Base: <span className="font-semibold text-foreground">₹{amt.toLocaleString('en-IN')}</span></div><div className="text-muted-foreground">Tax ({taxPct}%): <span className="font-semibold text-emerald-600">+₹{taxAmt.toFixed(2)}</span></div><div className="font-bold">Total Charge: ₹{total.toFixed(2)}</div></div>
   447→    }
