   129→
   130→export function CustomerFolioDialogs({
   131→  detail, allFolios, selectedEntries, submitting, settlingMode,
   132→  selectedFolioId, bal, selectedEntryObj, isActive, isCheckedOut,
   133→  invoiceData, invoiceLoading, folioStatementData, customerTags,
   134→  transferResults, paymentEntries,
   135→  addOpen, editOpen, chargeOpen, paymentOpen, refundOpen, transferOpen,
   136→  discountOpen, docOpen, noteOpen, tagsOpen, checkoutOpen, deleteOpen,
   137→  splitAmountOpen, shiftOpen, removeOpen, voucherOpen, invoiceOpen,
   138→  detailsSheetOpen,
   139→  form, chargeForm, paymentForm, refundForm, transferForm, discountForm,
   140→  docForm, noteForm, splitAmountForm, invoiceFrom, invoiceTo,
   141→  shiftTargetFolioId, duplicate,
   142→  checkoutSettlementData, checkoutSettlementOpen, splitPayments,
   143→  setCheckoutSettlementOpen, setSplitPayments,
   144→  handleMultiFolioSettlement, handleAutoSettleAll,
   145→  setAddOpen, setEditOpen, setChargeOpen, setPaymentOpen, setRefundOpen,
   146→  setTransferOpen, setDiscountOpen, setDocOpen, setNoteOpen, setTagsOpen,
   147→  setCheckoutOpen, setDeleteOpen, setSplitAmountOpen, setShiftOpen,
   148→  setRemoveOpen, setVoucherOpen, setInvoiceOpen, setDetailsSheetOpen,
   149→  setForm, setChargeForm, setPaymentForm, setRefundForm, setTransferForm,
   150→  setDiscountForm, setDocForm, setNoteForm, setSplitAmountForm,
   151→  setInvoiceFrom, setInvoiceTo, setShiftTargetFolioId, setDuplicate,
   152→  handleCreate, handleUpdate, handleCharge, handlePayment, handleSettle,
   153→  handleRefund, handleDiscount, handleTransfer, handleShift, handleSplitAmount,
   154→  handleRemove, handleCheckout, handleDelete, handleAddDoc, handleAddNote,
   155→  handleToggleTag, handleDeleteDoc, handleDeleteNote,
   156→  fetchInvoiceData, fetchFolioStatement, printInvoice, printSingleFolio, printVoucherHistory,
   157→  closeDialogs, searchTransferTargets,
   158→  deleteFolioOpen, folioToDelete,
   159→  setDeleteFolioOpen, handleDeleteFolio,
   160→  // Enhanced props
   161→  apiHeaders, onRefresh, onPrintReceipt, onThermalPrintReceipt,
   162→  onThermalPrintInvoice, onThermalPrintVoucher,
   163→  lastCreatedDocNo, lastCreatedType,
   164→  // History
   165→  historyOpen, setHistoryOpen, onPrintHistoryFolio,
   166→}: DialogsProps) {
   167→
   168→  // ─── Internal State ───
   169→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   170→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   171→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   172→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   173→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   174→  const [splitMode, setSplitMode] = React.useState(false)
   175→  const [localSplitItems, setLocalSplitItems] = React.useState<{ method: string; amount: string; reference: string }[]>([
   176→    { method: 'Cash', amount: '', reference: '' },
   177→  ])
   178→
   179→  // ─── History state ───
   180→  const [historyLoading, setHistoryLoading] = React.useState(false)
   181→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   182→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   183→  const [expandedFolioEntries, setExpandedFolioEntries] = React.useState<Record<string, any[]>>({})
   184→
   185→  // Fetch history for the CURRENT customer (not all checked-out customers)
   186→  React.useEffect(() => {
   187→    if (historyOpen && detail && apiHeaders) {
   188→      fetchCustomerHistory()
   189→    } else if (!historyOpen) {
   190→      setCustomerHistory(null)
   191→      setExpandedFolioId(null)
   192→      setExpandedFolioEntries({})
   193→    }
   194→  }, [historyOpen, detail?.id])
   195→
   196→  const fetchCustomerHistory = React.useCallback(async () => {
   197→    if (!detail || !apiHeaders) return
   198→    try {
   199→      setHistoryLoading(true)
   200→      const res = await fetch(`/api/folio/customers/${detail.id}/history`, { headers: apiHeaders })
   201→      if (res.ok) {
   202→        const data = await res.json() as CustomerHistory
   203→        setCustomerHistory(data)
   204→      }
   205→    } catch { /* ignore */ } finally { setHistoryLoading(false) }
   206→  }, [detail?.id, apiHeaders])
   207→
   208→  const toggleFolioEntries = (folioId: string) => {
   209→    if (expandedFolioId === folioId) {
   210→      setExpandedFolioId(null)
   211→    } else {
   212→      const folio = customerHistory?.historicalFolios.find(f => f.folioId === folioId)
   213→      if (folio) {
   214→        setExpandedFolioId(folioId)
   215→        setExpandedFolioEntries(prev => ({
   216→          ...prev,
   217→          [folioId]: (folio.entries || folio.ledgerEntries || []),
   218→        }))
   219→      }
   220→    }
   221→  }
   222→
   223→  // ─── Track checkout transition ───
   224→  const wasCheckedOut = React.useRef(isCheckedOut)
   225→  React.useEffect(() => {
   226→    if (isCheckedOut && !wasCheckedOut.current) {
   227→      setPostCheckoutOpen(true)
   228→    }
   229→    wasCheckedOut.current = isCheckedOut
   230→  }, [isCheckedOut])
   231→
   232→  // ─── Show TXN number when lastCreatedDocNo changes ───
   233→  React.useEffect(() => {
   234→    if (lastCreatedDocNo && lastCreatedType) {
   235→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   236→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   237→      return () => clearTimeout(timer)
   238→    }
   239→  }, [lastCreatedDocNo, lastCreatedType])
   240→
   241→  // ─── Fetch invoice number ───
   242→  const fetchInvoiceNumber = React.useCallback(async () => {
   243→    if (!detail || !apiHeaders) return
   244→    try {
   245→      setInvNumLoading(true)
   246→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   247→        method: 'POST',
   248→        headers: apiHeaders,
   249→      })
   250→      if (res.ok) {
   251→        const data = await res.json()
   252→        setInvoiceNumber(data.invoiceNumber)
   253→      }
   254→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   255→  }, [detail?.id, apiHeaders])
   256→
   257→  // ─── Fetch invoice number when invoice dialog opens ───
   258→  React.useEffect(() => {
   259→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber) {
   260→      fetchInvoiceNumber()
   261→    }
   262→  }, [invoiceOpen, detail?.id, apiHeaders])
   263→
   264→  React.useEffect(() => {
   265→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   266→      fetchInvoiceData()
   267→    }
   268→  }, [invoiceOpen])
   269→
   270→  // ─── Entry creation with TXN tracking ───
   271→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   272→    if (!detail || !selectedFolioId || !apiHeaders) {
   273→      return null
   274→    }
   275→    try {
   276→      setLocalSubmitting(true)
   277→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   278→        method: 'POST',
   279→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   280→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   281→      })
   282→      if (res.ok) {
   283→        const entry = await res.json()
   284→        const docNo = entry?.docNo || ''
   285→        setEntrySuccess({ show: true, docNo, type })
   286→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   287→        closeDialogs()
   288→        onRefresh?.()
   289→        setTimeout(() => setEntrySuccess(null), 4000)
   290→        return entry
   291→      } else {
   292→        const e = await res.json()
   293→        notify({ type: 'error', title: 'Error', message: e.error || 'Failed' })
   294→        return null
   295→      }
   296→    } catch {
   297→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   298→      return null
   299→    } finally {
   300→      setLocalSubmitting(false)
   301→    }
   302→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   303→
   304→  // ─── Settlement with TXN tracking ───
   305→  const settleWithTxn = React.useCallback(async () => {
   306→    if (!detail || !paymentForm.amount || !apiHeaders) {
   307→      handleSettle()
   308→      return
   309→    }
   310→    try {
   311→      setLocalSubmitting(true)
   312→      // When master folio is selected, use autoAllocate to distribute across all folios
   313→      const isMaster = selectedFolioId === 'master'
   314→      const payload = isMaster
   315→        ? { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   316→        : { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   317→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   318→        method: 'POST',
   319→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   320→        body: JSON.stringify(payload),
   321→      })
   322→      if (res.ok) {
   323→        const data = await res.json()
   324→        // Auto-allocate may return settledFolios array instead of single ledgerEntry
   325→        const docNo = data?.ledgerEntry?.docNo || (data?.settledFolios?.[0] ? 'Multi-Folio' : '')
   326→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })
   327→        notify({ type: 'success', title: `Settlement ${docNo}`, message: data.message || `Settlement ${docNo} processed successfully` })
   328→        closeDialogs()
   329→        onRefresh?.()
   330→        setTimeout(() => setEntrySuccess(null), 4000)
   331→      } else {
   332→        const e = await res.json()
   333→        notify({ type: 'error', title: 'Settlement Failed', message: e.error || 'Failed to process settlement' })
   334→      }
   335→    } catch {
   336→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   337→    } finally {
   338→      setLocalSubmitting(false)
   339→    }
   340→  }, [detail?.id, selectedFolioId, apiHeaders, paymentForm, handleSettle, closeDialogs, onRefresh])
   341→
   342→  // ─── Transfer with TXN tracking ───
   343→  const transferWithTxn = React.useCallback(async () => {
   344→    if (!detail || !transferForm.targetCustomerId || !apiHeaders) {
   345→      handleTransfer()
   346→      return
   347→    }
   348→    try {
   349→      setLocalSubmitting(true)
   350→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   351→        method: 'POST',
   352→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   353→        body: JSON.stringify({
   354→          targetCustomerId: transferForm.targetCustomerId,
   355→          amount: parseFloat(transferForm.amount) || 0,
   356→          description: transferForm.description,
   357→          transferEntries: true,
   358→        }),
   359→      })
   360→      if (res.ok) {
   361→        const data = await res.json()
   362→        const docNo = data?.source?.entry?.docNo || ''
   363→        setEntrySuccess({ show: true, docNo, type: 'Transfer' })
   364→        notify({ type: 'success', title: `Transfer ${docNo}`, message: data.message || `Transfer ${docNo} processed successfully` })
   365→        closeDialogs()
   366→        onRefresh?.()
   367→        setTimeout(() => setEntrySuccess(null), 4000)
   368→      } else {
   369→        const e = await res.json()
   370→        notify({ type: 'error', title: 'Error', message: e.error || 'Failed' })
   371→      }
   372→    } catch {
   373→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   374→    } finally {
   375→      setLocalSubmitting(false)
   376→    }
   377→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   378→
   379→  // ─── Split payment helpers for checkout ───
   380→  const addSplitItem = () => {
   381→    setLocalSplitItems(prev => [...prev, { method: 'Cash', amount: '', reference: '' }])
   382→  }
   383→  const removeSplitItem = (idx: number) => {
   384→    setLocalSplitItems(prev => prev.filter((_, i) => i !== idx))
   385→  }
   386→  const updateSplitItem = (idx: number, field: string, value: string) => {
   387→    setLocalSplitItems(prev => prev.map((item, i) => i === idx ? { ...item, [field]: value } : item))
   388→  }
   389→  const totalSplitAmount = localSplitItems.reduce((sum, item) => sum + (parseFloat(item.amount) || 0), 0)
   390→  const splitRemaining = checkoutSettlementData ? Math.max(0, checkoutSettlementData.totalOutstanding - totalSplitAmount) : 0
   391→
   392→  // ─── Computed GST data from invoiceData ───
   393→  const gstData = React.useMemo(() => {
   394→    if (!invoiceData?.entries) return null
   395→    const entries = invoiceData.entries as any[]
   396→    const taxableEntries = entries.filter((e: any) => e.debit > 0)
   397→    const taxableAmount = taxableEntries.reduce((sum: number, e: any) => sum + (e.debit || 0) - (e.tax || 0) - (e.discount || 0), 0)
   398→    const totalTax = entries.reduce((sum: number, e: any) => sum + (e.tax || 0), 0)
   399→    const totalDiscount = entries.reduce((sum: number, e: any) => sum + (e.discount || 0), 0)
   400→    const totalCharges = entries.reduce((sum: number, e: any) => sum + (e.debit || 0), 0)
   401→    const cgst = totalTax / 2
   402→    const sgst = totalTax / 2
   403→    const totalPayments = entries.reduce((sum: number, e: any) => sum + (e.credit || 0), 0)
   404→    const advancePayment = bal < 0 ? Math.abs(bal) : 0
   405→    return { taxableAmount, totalTax, totalDiscount, totalCharges, cgst, sgst, totalPayments, advancePayment, entryCount: taxableEntries.length }
   406→  }, [invoiceData, bal])
   407→
   408→  // ─── Active submitting state (local or prop) ───
   409→  const isSubmitting = localSubmitting || submitting
   410→
   411→  // ─── Entry success banner component ───
   412→  const EntrySuccessBanner = () => {
   413→    if (!entrySuccess?.show) return null
   414→    return (
   415→      <div className="fixed bottom-4 right-4 z-[100] animate-in slide-in-from-bottom-2 fade-in duration-300">
   416→        <div className="bg-emerald-600 text-white rounded-xl shadow-lg px-4 py-3 flex items-center gap-3 min-w-[260px]">
   417→          <div className="w-8 h-8 rounded-full bg-emerald-500 flex items-center justify-center shrink-0">
   418→            <CircleCheck className="size-4 text-white" />
   419→          </div>
   420→          <div className="min-w-0">
   421→            <div className="text-xs font-bold">{entrySuccess.type} Created</div>
   422→            <div className="text-[11px] text-emerald-100 flex items-center gap-1 mt-0.5">
   423→              <Hash className="size-3" />
   424→              <span className="font-mono font-bold">{entrySuccess.docNo}</span>
   425→            </div>
   426→          </div>
   427→        </div>
   428→      </div>
   429→    )
   430→  }
   431→
   432→  // ─── Per-entry-type success wrapper ───
   433→  const SuccessOverlay = ({ docNo, type, label }: { docNo: string; type: string; label: string }) => (
   434→    <div className="flex flex-col items-center justify-center py-6 gap-3">
   435→      <div className="w-14 h-14 rounded-2xl bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   436→        <PartyPopper className="size-7 text-emerald-600 dark:text-emerald-400" />
   437→      </div>
   438→      <div className="text-center">
   439→        <div className="text-sm font-bold text-slate-800 dark:text-slate-100">{label}</div>
   440→        <div className="mt-2 inline-flex items-center gap-2 bg-emerald-50 dark:bg-emerald-950/40 border border-emerald-200 dark:border-emerald-800 rounded-lg px-4 py-2.5">
   441→          <Hash className="size-4 text-emerald-600 dark:text-emerald-400" />
   442→          <span className="font-mono text-base font-bold text-emerald-700 dark:text-emerald-300">{docNo}</span>
   443→        </div>
   444→        <div className="text-[11px] text-slate-400 mt-2">Transaction recorded successfully</div>
   445→      </div>
   446→    </div>
   447→  )
   448→
   449→  return (
   450→    <>
   451→    {/* Entry Success Banner (floating) */}
   452→    <EntrySuccessBanner />
