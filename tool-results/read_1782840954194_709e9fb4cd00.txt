   277→function formatDate(dateStr: string | null): string {
   278→  if (!dateStr) return '—';
   279→  try {
   280→    return new Date(dateStr).toLocaleDateString('en-IN', {
   281→      day: '2-digit', month: 'short', year: 'numeric',
   282→    });
   283→  } catch {
   284→    return '—';
   285→  }
   286→}
   287→
   288→function formatDateTime(dateStr: string | null): string {
   289→  if (!dateStr) return 'Never';
   290→  try {
   291→    return new Date(dateStr).toLocaleDateString('en-IN', {
   292→      day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit',
   293→    });
   294→  } catch {
   295→    return '—';
   296→  }
   297→}

   299→function isPermGroup(obj: unknown): obj is Record<string, boolean> {
   300→  return typeof obj === 'object' && obj !== null && !Array.isArray(obj);
   301→}
