   136→export function CustomerFolioDialogs({
   137→  detail, allFolios, selectedEntries, submitting, settlingMode,
   138→  selectedFolioId, bal, selectedEntryObj, isActive, isCheckedOut,
   139→  invoiceData, invoiceLoading, folioStatementData, customerTags,
   140→  transferResults, paymentEntries,
   141→  addOpen, editOpen, chargeOpen, paymentOpen, refundOpen, transferOpen,
   142→  discountOpen, docOpen, noteOpen, tagsOpen, checkoutOpen, deleteOpen,
   143→  splitAmountOpen, shiftOpen, removeOpen, voucherOpen, invoiceOpen,
   144→  detailsSheetOpen,
   145→  form, chargeForm, paymentForm, refundForm, transferForm, discountForm,
   146→  docForm, noteForm, splitAmountForm, invoiceFrom, invoiceTo,
   147→  shiftTargetFolioId, duplicate,
   148→  checkoutSettlementData, checkoutSettlementOpen, splitPayments,
   149→  setCheckoutSettlementOpen, setSplitPayments,
   150→  handleMultiFolioSettlement, handleAutoSettleAll,

   152→  setTransferOpen, setDiscountOpen, setDocOpen, setNoteOpen, setTagsOpen,
   153→  setCheckoutOpen, setDeleteOpen, setSplitAmountOpen, setShiftOpen,
   154→  setRemoveOpen, setVoucherOpen, setInvoiceOpen, setDetailsSheetOpen,
   155→  setForm, setChargeForm, setPaymentForm, setRefundForm, setTransferForm,
   156→  setDiscountForm, setDocForm, setNoteForm, setSplitAmountForm,
   157→  setInvoiceFrom, setInvoiceTo, setShiftTargetFolioId, setDuplicate,
   158→  handleCreate, handleUpdate, handleCharge, handlePayment, handleSettle,
   159→  handleRefund, handleDiscount, handleTransfer, handleShift, handleSplitAmount,
   160→  handleRemove, handleCheckout, handleDelete, handleAddDoc, handleAddNote,
   161→  handleToggleTag, handleDeleteDoc, handleDeleteNote,
   162→  fetchInvoiceData, fetchFolioStatement, printInvoice, printCompleteInvoice, printSingleFolio, printVoucherHistory,
   163→  closeDialogs, searchTransferTargets,
   164→  deleteFolioOpen, folioToDelete,
   165→  setDeleteFolioOpen, handleDeleteFolio,
   166→  // Enhanced props
   167→  apiHeaders, onRefresh, onPrintReceipt, onThermalPrintReceipt,
   168→  onThermalPrintInvoice, onThermalPrintVoucher,
   169→  lastCreatedDocNo, lastCreatedType,
   170→  // History
   171→  historyOpen, setHistoryOpen, onPrintHistoryFolio,
   172→}: DialogsProps) {
   173→
   174→  // ─── Internal State ───
   175→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   176→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   177→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   178→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   179→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   180→  const [splitMode, setSplitMode] = React.useState(false)
   181→  const [localSplitItems, setLocalSplitItems] = React.useState<{ method: string; amount: string; reference: string }[]>([
   182→    { method: 'Cash', amount: '', reference: '' },
   183→  ])
   184→
   185→  // ─── History state ───
   186→  const [historyLoading, setHistoryLoading] = React.useState(false)
   187→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   188→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   189→  const [expandedFolioEntries, setExpandedFolioEntries] = React.useState<Record<string, any[]>>({})
   190→
   191→  // Fetch history for the CURRENT customer (not all checked-out customers)
   192→  React.useEffect(() => {
   193→    if (historyOpen && detail && apiHeaders) {
   194→      fetchCustomerHistory()
   195→    } else if (!historyOpen) {
   196→      setCustomerHistory(null)
   197→      setExpandedFolioId(null)
   198→      setExpandedFolioEntries({})
   199→    }
   200→  }, [historyOpen, detail?.id])
   201→
   202→  const fetchCustomerHistory = React.useCallback(async () => {
   203→    if (!detail || !apiHeaders) return
   204→    try {
   205→      setHistoryLoading(true)
   206→      const res = await fetch(`/api/folio/customers/${detail.id}/history`, { headers: apiHeaders })
   207→      if (res.ok) {
   208→        const data = await res.json() as CustomerHistory
   209→        setCustomerHistory(data)
   210→      }
   211→    } catch { /* ignore */ } finally { setHistoryLoading(false) }
   212→  }, [detail?.id, apiHeaders])
   213→
   214→  const toggleFolioEntries = (folioId: string) => {
   215→    if (expandedFolioId === folioId) {
   216→      setExpandedFolioId(null)
   217→    } else {
   218→      const folio = customerHistory?.historicalFolios.find(f => f.folioId === folioId)
   219→      if (folio) {
   220→        setExpandedFolioId(folioId)
   221→        setExpandedFolioEntries(prev => ({
   222→          ...prev,
   223→          [folioId]: (folio.entries || folio.ledgerEntries || []),
   224→        }))
   225→      }
   226→    }
   227→  }
   228→
   229→  // ─── Track checkout transition ───
   230→  const wasCheckedOut = React.useRef(isCheckedOut)
   231→  React.useEffect(() => {
   232→    if (isCheckedOut && !wasCheckedOut.current) {
   233→      setPostCheckoutOpen(true)
   234→    }
   235→    wasCheckedOut.current = isCheckedOut
   236→  }, [isCheckedOut])
   237→
   238→  // ─── Show TXN number when lastCreatedDocNo changes ───
   239→  React.useEffect(() => {
   240→    if (lastCreatedDocNo && lastCreatedType) {
   241→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   242→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   243→      return () => clearTimeout(timer)
   244→    }
   245→  }, [lastCreatedDocNo, lastCreatedType])
   246→
   247→  // ─── Fetch invoice number ───
   248→  const fetchInvoiceNumber = React.useCallback(async () => {
   249→    if (!detail || !apiHeaders) return
   250→    try {
   251→      setInvNumLoading(true)
   252→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   253→        method: 'POST',
   254→        headers: apiHeaders,
   255→      })
   256→      if (res.ok) {
   257→        const data = await res.json()
   258→        setInvoiceNumber(data.invoiceNumber)
   259→      }
   260→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   261→  }, [detail?.id, apiHeaders])
   262→
   263→  // ─── Fetch invoice number when invoice dialog opens ───
   264→  React.useEffect(() => {
   265→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber) {
   266→      fetchInvoiceNumber()
   267→    }
   268→  }, [invoiceOpen, detail?.id, apiHeaders])
   269→
   270→  React.useEffect(() => {
   271→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   272→      fetchInvoiceData()
   273→    }
   274→  }, [invoiceOpen])
   275→
   276→  // ─── Entry creation with TXN tracking ───
   277→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   278→    if (!detail || !selectedFolioId || !apiHeaders) {
   279→      return null
   280→    }
   281→    try {
   282→      setLocalSubmitting(true)
   283→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   284→        method: 'POST',
   285→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   286→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   287→      })
   288→      if (res.ok) {
   289→        const entry = await res.json()
   290→        const docNo = entry?.docNo || ''
   291→        setEntrySuccess({ show: true, docNo, type })
   292→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   293→        closeDialogs()
   294→        onRefresh?.()
   295→        setTimeout(() => setEntrySuccess(null), 4000)
   296→        return entry
   297→      } else {
   298→        const e = await safeJson(res)
   299→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   300→        return null
   301→      }
   302→    } catch {
   303→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   304→      return null
   305→    } finally {
   306→      setLocalSubmitting(false)
   307→    }
   308→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   309→
   310→  // ─── Settlement with TXN tracking ───
   311→  const settleWithTxn = React.useCallback(async () => {
   312→    if (!detail || !paymentForm.amount || !apiHeaders) {
   313→      handleSettle()
   314→      return
   315→    }
   316→    try {
   317→      setLocalSubmitting(true)
   318→      // When master folio is selected, use autoAllocate to distribute across all folios
   319→      const isMaster = selectedFolioId === 'master'
   320→      const payload = isMaster
   321→        ? { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   322→        : { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   323→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   324→        method: 'POST',
   325→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   326→        body: JSON.stringify(payload),
   327→      })
   328→      if (res.ok) {
   329→        const data = await res.json()
   330→        // Auto-allocate may return settledFolios array instead of single ledgerEntry
   331→        const docNo = data?.ledgerEntry?.docNo || (data?.settledFolios?.[0] ? 'Multi-Folio' : '')
   332→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })
   333→        notify({ type: 'success', title: `Settlement ${docNo}`, message: data.message || `Settlement ${docNo} processed successfully` })
   334→        closeDialogs()
   335→        onRefresh?.()
   336→        setTimeout(() => setEntrySuccess(null), 4000)
   337→      } else {
   338→        const e = await safeJson(res)
   339→        notify({ type: 'error', title: 'Settlement Failed', message: (e.error as string) || 'Failed to process settlement' })
   340→      }
   341→    } catch {
   342→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   343→    } finally {
   344→      setLocalSubmitting(false)
   345→    }
   346→  }, [detail?.id, selectedFolioId, apiHeaders, paymentForm, handleSettle, closeDialogs, onRefresh])
   347→
   348→  // ─── Transfer with TXN tracking ───
   349→  const transferWithTxn = React.useCallback(async () => {
   350→    if (!detail || !transferForm.targetCustomerId || !apiHeaders) {
   351→      handleTransfer()
   352→      return
   353→    }
   354→    try {
   355→      setLocalSubmitting(true)
   356→      // If entries are selected, transfer only those; otherwise pure balance transfer
   357→      const selectedEntryIds = selectedEntries.size > 0 ? Array.from(selectedEntries) : undefined
   358→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   359→        method: 'POST',
   360→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   361→        body: JSON.stringify({
   362→          targetCustomerId: transferForm.targetCustomerId,
   363→          amount: parseFloat(transferForm.amount) || 0,
   364→          description: transferForm.description,
   365→          entryIds: selectedEntryIds,
   366→        }),
   367→      })
   368→      if (res.ok) {
   369→        const data = await res.json()
   370→        const docNo = data?.source?.entry?.docNo || ''
   371→        setEntrySuccess({ show: true, docNo, type: 'Transfer' })
   372→        notify({ type: 'success', title: `Transfer ${docNo}`, message: data.message || `Transfer ${docNo} processed successfully` })
   373→        closeDialogs()
   374→        onRefresh?.()
   375→        setTimeout(() => setEntrySuccess(null), 4000)
   376→      } else {
   377→        const e = await safeJson(res)
   378→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   379→      }
   380→    } catch {
   381→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   382→    } finally {
   383→      setLocalSubmitting(false)
   384→    }
   385→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   386→
   387→  // ─── Split payment helpers for checkout ───
   388→  const addSplitItem = () => {
   389→    setLocalSplitItems(prev => [...prev, { method: 'Cash', amount: '', reference: '' }])
   390→  }
   391→  const removeSplitItem = (idx: number) => {
   392→    setLocalSplitItems(prev => prev.filter((_, i) => i !== idx))
   393→  }
   394→  const updateSplitItem = (idx: number, field: string, value: string) => {
   395→    setLocalSplitItems(prev => prev.map((item, i) => i === idx ? { ...item, [field]: value } : item))
   396→  }
   397→  const totalSplitAmount = localSplitItems.reduce((sum, item) => sum + (parseFloat(item.amount) || 0), 0)
   398→  const splitRemaining = checkoutSettlementData ? Math.max(0, checkoutSettlementData.totalOutstanding - totalSplitAmount) : 0
   399→
   400→  // ─── Computed GST data from invoiceData ───
   401→  const gstData = React.useMemo(() => {
   402→    if (!invoiceData?.entries) return null
   403→    const entries = invoiceData.entries as any[]
   404→    const taxableEntries = entries.filter((e: any) => e.debit > 0)
   405→    const taxableAmount = taxableEntries.reduce((sum: number, e: any) => sum + (e.debit || 0) - (e.tax || 0) - (e.discount || 0), 0)
   406→    const totalTax = entries.reduce((sum: number, e: any) => sum + (e.tax || 0), 0)
   407→    const totalDiscount = entries.reduce((sum: number, e: any) => sum + (e.discount || 0), 0)
   408→    const totalCharges = entries.reduce((sum: number, e: any) => sum + (e.debit || 0), 0)
   409→    const cgst = totalTax / 2
   410→    const sgst = totalTax / 2
   411→    const totalPayments = entries.reduce((sum: number, e: any) => sum + (e.credit || 0), 0)
   412→    const advancePayment = bal < 0 ? Math.abs(bal) : 0
   413→    return { taxableAmount, totalTax, totalDiscount, totalCharges, cgst, sgst, totalPayments, advancePayment, entryCount: taxableEntries.length }
   414→  }, [invoiceData, bal])
   415→
   416→  // ─── Active submitting state (local or prop) ───
   417→  const isSubmitting = localSubmitting || submitting
   418→
   419→  // ─── Entry success banner component ───
   420→  const EntrySuccessBanner = () => {
   421→    if (!entrySuccess?.show) return null
   422→    return (
   423→      <div className="fixed bottom-4 right-4 z-[100] animate-in slide-in-from-bottom-2 fade-in duration-300">
   424→        <div className="bg-emerald-600 text-white rounded-xl shadow-lg px-4 py-3 flex items-center gap-3 min-w-[260px]">
   425→          <div className="w-8 h-8 rounded-full bg-emerald-500 flex items-center justify-center shrink-0">
   426→            <CircleCheck className="size-4 text-white" />
   427→          </div>
   428→          <div className="min-w-0">
   429→            <div className="text-xs font-bold">{entrySuccess.type} Created</div>
   430→            <div className="text-[11px] text-emerald-100 flex items-center gap-1 mt-0.5">
   431→              <Hash className="size-3" />
   432→              <span className="font-mono font-bold">{entrySuccess.docNo}</span>
   433→            </div>
   434→          </div>
   435→        </div>
   436→      </div>
   437→    )
   438→  }
   439→
   440→  // ─── Per-entry-type success wrapper ───
   441→  const SuccessOverlay = ({ docNo, type, label }: { docNo: string; type: string; label: string }) => (
   442→    <div className="flex flex-col items-center justify-center py-6 gap-3">
   443→      <div className="w-14 h-14 rounded-2xl bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   444→        <PartyPopper className="size-7 text-emerald-600 dark:text-emerald-400" />
   445→      </div>
   446→      <div className="text-center">
