   223→   223→        setExpandedFolioId(folioId)

   225→   225→          ...prev,
   226→   226→          [folioId]: (folio.entries || []),

   229→   229→    }
   230→   230→  }

   232→   232→  // ─── Track checkout transition ───

   234→   234→  React.useEffect(() => {
   235→   235→    if (isCheckedOut && !wasCheckedOut.current) {
   236→   236→      setPostCheckoutOpen(true)
   237→   237→    }

   239→   239→  }, [isCheckedOut])
   240→   240→
   241→   241→  // ─── Show TXN number when lastCreatedDocNo changes ───

   244→   244→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })

   247→   247→    }
   248→   248→  }, [lastCreatedDocNo, lastCreatedType])
   249→   249→
   250→   250→  // ─── Items for charge selector (Req 5) ───

   252→   252→  const [itemsLoading, setItemsLoading] = React.useState(false)

   316→   316→      })

   328→   328→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   329→   329→        return null
   330→   330→      }
   331→   331→    } catch {
   332→   332→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   333→   333→      return null
   334→   334→    } finally {
   335→   335→      setLocalSubmitting(false)
   336→   336→    }
   337→   337→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   338→   338→
   339→   339→  // ─── Settlement with TXN tracking ───

   354→   354→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   355→   355→        body: JSON.stringify(payload),
   356→   356→      })
   357→   357→      if (res.ok) {
   358→   358→        const data = await res.json()
   359→   359→        // Auto-allocate may return settledFolios array instead of single ledgerEntry
   360→   360→        const docNo = data?.ledgerEntry?.docNo || (data?.settledFolios?.[0] ? 'Multi-Folio' : '')
   361→   361→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })

   363→   363→        closeDialogs()
   364→   364→        onRefresh?.()
   365→   365→        setTimeout(() => setEntrySuccess(null), 4000)
   366→   366→      } else {
   367→   367→        const e = await safeJson(res)
   368→   368→        notify({ type: 'error', title: 'Settlement Failed', message: (e.error as string) || 'Failed to process settlement' })
   369→   369→      }
   370→   370→    } catch {
   371→   371→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   372→   372→    } finally {
   373→   373→      setLocalSubmitting(false)
   374→   374→    }
   375→   375→  }, [detail?.id, selectedFolioId, apiHeaders, paymentForm, handleSettle, closeDialogs, onRefresh])
   376→   376→
   377→   377→  // ─── Transfer with TXN tracking ───
   378→   378→  const transferWithTxn = React.useCallback(async () => {
   379→   379→    if (!detail || !transferForm.targetCustomerId || !apiHeaders) {
   380→   380→      handleTransfer()
   381→   381→      return
   382→   382→    }
   383→   383→    try {
   384→   384→      setLocalSubmitting(true)
   385→   385→      // If entries are selected, transfer only those; otherwise pure balance transfer
   386→   386→      const selectedEntryIds = selectedEntries.size > 0 ? Array.from(selectedEntries) : undefined
   387→   387→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   388→   388→        method: 'POST',
   389→   389→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   390→   390→        body: JSON.stringify({
   391→   391→          targetCustomerId: transferForm.targetCustomerId,
   392→   392→          amount: parseFloat(transferForm.amount) || 0,
   393→   393→          description: transferForm.description,
   394→   394→          entryIds: selectedEntryIds,
