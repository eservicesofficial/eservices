   151→      // VIP and Due are tag/balance filters — don't set status for them

   153→        params.set('status', filter)

   155→        params.set('status', 'CheckedOut')
   156→      if (debounced) params.set('search', debounced)
   157→      if (sort) params.set('sort', sort === 'name' ? 'name-asc' : sort)
   158→      if (filter === 'VIP') params.set('tag', 'VIP')

   165→      }
   166→    } catch { /* ignore */ } finally { setLoading(false) }
   167→  }, [debounced, filter, sort])

   208→        const ledgerEntries: LedgerEntry[] = activeFolio?.ledgerEntries || []
   209→        const balance = lifetimeStats.currentBalance ?? (ledgerEntries.length > 0 ? ledgerEntries[ledgerEntries.length - 1].balance : 0)
   210→        setDetail({
   211→          ...customer,
   212→          openingBalance: customer.openingBalance ?? 0,
   213→          creditLimit: customer.creditLimit ?? 0,
   214→          balance: balance ?? 0,
   215→          totalVisits: d.visits?.length ?? lifetimeStats.totalVisits ?? 0,
   216→          lastTxnDate: lifetimeStats.lastTransactionDate || '',
   217→          tags: tagAssignments, folios: d.folios || [], ledgerEntries,
   218→          documents: d.documents || [], notes: d.notes || [], visits: d.visits || [],
   219→          settlements: d.settlements || [], auditLogs: [],
   220→          lifetimeRevenue: lifetimeStats.totalCharges ?? 0,
   221→          lifetimePayments: lifetimeStats.totalPayments ?? 0,
   222→          lifetimeRefunds: lifetimeStats.totalRefunds ?? 0,
   223→          totalBills: 0, totalTransactions: ledgerEntries.length,

   226→        })
   227→        setCustomerTags(tagAssignments.map(t => t.tagName))
   228→        setAllFolios(d.folios || [])
   229→        if (activeFolio) { setSelectedFolioId(activeFolio.id); setFolioEntries(ledgerEntries) }
   230→        else if (d.folios?.length > 0) { setSelectedFolioId(d.folios[0].id); fetchFolioEntries(id, d.folios[0].id) }
   231→        else { setSelectedFolioId('master'); fetchMasterFolioEntries(id) }
   232→        setSelectedEntries(new Set())
   233→      }
   234→    } catch { /* ignore */ } finally { setDetailLoading(false) }
   235→  }, [fetchFolioEntries, fetchMasterFolioEntries])

   262→  // ── Form helpers ──

   303→  const handleCreate = async () => {
   304→    if (!form.fullName.trim() || !form.mobile.trim()) { notify({ type: 'error', title: 'Error', message: 'Name and Mobile required' }); return }

   306→      const res = await fetch('/api/folio/customers', { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(form) })
   307→      if (res.ok) { notify({ type: 'success', title: 'Customer Created' }); setAddOpen(false); resetForm(); fetchCustomers(); onRefreshData() }
   308→      else { const e = await safeJson(res); if (e.duplicate) { setDuplicate(e.duplicate); notify({ type: 'warning', title: 'Duplicate' }) } else notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' }) }
   309→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed' }) } finally { setSubmitting(false) }
   310→  }
   311→
   312→  const handleUpdate = async () => {
   313→    if (!detail || !form.fullName.trim()) return

   315→      const res = await fetch(`/api/folio/customers/${detail.id}`, { method: 'PUT', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(form) })
   316→      if (res.ok) { notify({ type: 'success', title: 'Updated' }); setEditOpen(false); fetchDetail(detail.id); fetchCustomers(true) }
   317→      else { const e = await safeJson(res); notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' }) }
   318→    } catch { /* ignore */ } finally { setSubmitting(false) }
   319→  }
   320→
   321→  const addLedgerEntry = async (type: string, data: any) => {
   322→    if (!detail || !selectedFolioId) return
   323→    try { setSubmitting(true)
   324→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify({ type, folioId: selectedFolioId, ...data }) })

   327→    } catch { /* ignore */ } finally { setSubmitting(false) }
   328→  }
   329→
   330→  const handleCharge = () => { if (!chargeForm.amount || !chargeForm.description) { notify({ type: 'error', title: 'Error', message: 'Amount and description required' }); return }; addLedgerEntry('Charge', { description: chargeForm.description, amount: parseFloat(chargeForm.amount), tax: parseFloat(chargeForm.tax) || 0, invoiceNo: chargeForm.invoiceNo }) }

   332→  const handleRefund = () => { if (!refundForm.amount || !refundForm.description) { notify({ type: 'error', title: 'Error', message: 'Amount and reason required' }); return }; addLedgerEntry('Refund', { description: `${refundForm.reason}: ${refundForm.description}`, amount: parseFloat(refundForm.amount), invoiceNo: refundForm.invoiceNo }) }
   333→  const handleDiscount = () => { if (!discountForm.amount) { notify({ type: 'error', title: 'Error', message: 'Amount required' }); return }; addLedgerEntry('Discount', { description: discountForm.description || 'Discount applied', amount: parseFloat(discountForm.amount), discountType: discountForm.discountType }) }
   334→
   335→  const handleSettle = async () => {
   336→    if (!detail || !paymentForm.amount) return
   337→    try { setSubmitting(true)
   338→      const settleAmount = parseFloat(paymentForm.amount)
   339→      // When master folio is selected, use autoAllocate to distribute across all folios
   340→      const isMaster = selectedFolioId === 'master'
   341→      const payload = isMaster
   342→        ? { amount: settleAmount, mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   343→        : { amount: settleAmount, mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   344→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(payload) })
   345→      if (res.ok) {
   346→        notify({ type: 'success', title: 'Settlement Complete' })
   347→        closeDialogs()
   348→        if (!isMaster && selectedFolioId) fetchFolioEntries(detail.id, selectedFolioId)
   349→        fetchMasterFolioEntries(detail.id)
   350→        fetchDetail(detail.id, true)
   351→        fetchCustomers(true)
   352→        onRefreshData()
   353→      }
