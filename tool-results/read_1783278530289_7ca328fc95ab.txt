   210→        const ledgerEntries: LedgerEntry[] = activeFolio?.ledgerEntries || []
   211→        const balance = lifetimeStats.currentBalance ?? (ledgerEntries.length > 0 ? ledgerEntries[ledgerEntries.length - 1].balance : 0)
   212→        setDetail({
   213→          ...customer,
   214→          openingBalance: customer.openingBalance ?? 0,
   215→          creditLimit: customer.creditLimit ?? 0,
   216→          balance: balance ?? 0,
   217→          totalVisits: d.visits?.length ?? lifetimeStats.totalVisits ?? 0,
   218→          lastTxnDate: lifetimeStats.lastTransactionDate || '',
   219→          tags: tagAssignments, folios: d.folios || [], ledgerEntries,
   220→          documents: d.documents || [], notes: d.notes || [], visits: d.visits || [],
   221→          settlements: d.settlements || [], auditLogs: [],
   222→          lifetimeRevenue: lifetimeStats.totalCharges ?? 0,
   223→          lifetimePayments: lifetimeStats.totalPayments ?? 0,
   224→          lifetimeRefunds: lifetimeStats.totalRefunds ?? 0,
   225→          totalBills: 0, totalTransactions: ledgerEntries.length,

   228→        })

   230→        setAllFolios(d.folios || [])
   231→        if (activeFolio) { setSelectedFolioId(activeFolio.id); setFolioEntries(ledgerEntries) }
   232→        else if (d.folios?.length > 0) { setSelectedFolioId(d.folios[0].id); fetchFolioEntries(id, d.folios[0].id) }
   233→        else { setSelectedFolioId('master'); fetchMasterFolioEntries(id) }
   234→        setSelectedEntries(new Set())
   235→      }
   236→    } catch { /* ignore */ } finally { setDetailLoading(false) }

   306→    if (!form.fullName.trim() || !form.mobile.trim()) { notify({ type: 'error', title: 'Error', message: 'Name and Mobile required' }); return }

   308→      const res = await fetch('/api/folio/customers', { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(form) })
   309→      if (res.ok) { notify({ type: 'success', title: 'Customer Created' }); setAddOpen(false); resetForm(); fetchCustomers(); onRefreshData() }
   310→      else { const e = await safeJson(res); if (e.duplicate) { setDuplicate(e.duplicate as Customer); notify({ type: 'warning', title: 'Duplicate', message: 'A customer with similar details exists' }) } else notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' }) }
   311→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed' }) } finally { setSubmitting(false) }

   319→      else { const e = await safeJson(res); notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' }) }

   334→  const handleRefund = () => { if (!refundForm.amount || !refundForm.description) { notify({ type: 'error', title: 'Error', message: 'Amount and reason required' }); return }; addLedgerEntry('Refund', { description: `${refundForm.reason}: ${refundForm.description}`, amount: parseFloat(refundForm.amount), invoiceNo: refundForm.invoiceNo }) }
   335→  const handleDiscount = () => { if (!discountForm.amount) { notify({ type: 'error', title: 'Error', message: 'Amount required' }); return }; addLedgerEntry('Discount', { description: discountForm.description || 'Discount applied', amount: parseFloat(discountForm.amount), discountType: discountForm.discountType }) }

   337→  const handleSettle = async () => {

   340→      const settleAmount = parseFloat(paymentForm.amount)
   341→      // When master folio is selected, use autoAllocate to distribute across all folios
   342→      const isMaster = selectedFolioId === 'master'
   343→      const payload = isMaster
   344→        ? { amount: settleAmount, mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   345→        : { amount: settleAmount, mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   346→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(payload) })
   347→      if (res.ok) {
   348→        notify({ type: 'success', title: 'Settlement Complete' })
   349→        closeDialogs()
   350→        if (!isMaster && selectedFolioId) fetchFolioEntries(detail.id, selectedFolioId)
   351→        fetchMasterFolioEntries(detail.id)
   352→        fetchDetail(detail.id, true)
   353→        fetchCustomers(true)
