   156→export function CustomerFolioDialogs({
   157→  detail, allFolios, selectedEntries, submitting, settlingMode,
   158→  selectedFolioId, bal, selectedEntryObj, isActive, isCheckedOut,
   159→  invoiceData, invoiceLoading, folioStatementData, customerTags,
   160→  transferResults, paymentEntries,
   161→  addOpen, editOpen, chargeOpen, paymentOpen, refundOpen, transferOpen,
   162→  discountOpen, docOpen, noteOpen, tagsOpen, checkoutOpen, deleteOpen,
   163→  splitAmountOpen, shiftOpen, removeOpen, voucherOpen, invoiceOpen,
   164→  detailsSheetOpen,
   165→  form, chargeForm, paymentForm, refundForm, transferForm, discountForm,
   166→  docForm, noteForm, splitAmountForm, invoiceFrom, invoiceTo,
   167→  shiftTargetFolioId, duplicate,
   168→  setAddOpen, setEditOpen, setChargeOpen, setPaymentOpen, setRefundOpen,
   169→  setTransferOpen, setDiscountOpen, setDocOpen, setNoteOpen, setTagsOpen,
   170→  setCheckoutOpen, setDeleteOpen, setSplitAmountOpen, setShiftOpen,
   171→  setRemoveOpen, setVoucherOpen, setInvoiceOpen, setDetailsSheetOpen,
   172→  setForm, setChargeForm, setPaymentForm, setRefundForm, setTransferForm,
   173→  setDiscountForm, setDocForm, setNoteForm, setSplitAmountForm,
   174→  setInvoiceFrom, setInvoiceTo, setShiftTargetFolioId, setDuplicate,
   175→  handleCreate, handleUpdate, handleCharge, handlePayment, handleSettle,
   176→  handleRefund, handleDiscount, handleTransfer, handleShift, handleSplitAmount,
   177→  handleRemove, handleCheckout, handleDelete, handleAddDoc, handleAddNote,
   178→  handleToggleTag, handleDeleteDoc, handleDeleteNote,
   179→  fetchInvoiceData, fetchFolioStatement, printInvoice, printCompleteInvoice, printSingleFolio, printVoucherHistory,
   180→  closeDialogs, searchTransferTargets,
   181→  deleteFolioOpen, folioToDelete,
   182→  setDeleteFolioOpen, handleDeleteFolio,
   183→  // Enhanced props
   184→  apiHeaders, onRefresh, onPrintReceipt, onThermalPrintReceipt,
   185→  onThermalPrintInvoice, onThermalPrintVoucher,
   186→  lastCreatedDocNo, lastCreatedType,
   187→  // History
   188→  historyOpen, setHistoryOpen, onPrintHistoryFolio,
   189→}: DialogsProps) {
   190→
   191→  // ─── Internal State ───
   192→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   193→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   194→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   195→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   196→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   197→  // ─── History state ───
   198→  const [historyLoading, setHistoryLoading] = React.useState(false)
   199→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   200→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   201→  const [expandedFolioEntries, setExpandedFolioEntries] = React.useState<Record<string, any[]>>({})
   202→  const [selectedVoucherIds, setSelectedVoucherIds] = React.useState<Set<string>>(new Set())
   203→
   204→  // Fetch history for the CURRENT customer (not all checked-out customers)
   205→  React.useEffect(() => {
   206→    if (historyOpen && detail && apiHeaders) {
   207→      fetchCustomerHistory()
   208→    } else if (!historyOpen) {
   209→      setCustomerHistory(null)
   210→      setExpandedFolioId(null)
   211→      setExpandedFolioEntries({})
   212→    }
   213→  }, [historyOpen, detail?.id])
   214→
   215→  const fetchCustomerHistory = React.useCallback(async () => {
   216→    if (!detail || !apiHeaders) return
   217→    try {
   218→      setHistoryLoading(true)
   219→      const res = await fetch(`/api/folio/customers/${detail.id}/history`, { headers: apiHeaders })
   220→      if (res.ok) {
   221→        const data = await res.json() as CustomerHistory
   222→        setCustomerHistory(data)
   223→      }
   224→    } catch { /* ignore */ } finally { setHistoryLoading(false) }
   225→  }, [detail?.id, apiHeaders])
   226→
   227→  const toggleFolioEntries = (folioId: string) => {
   228→    if (expandedFolioId === folioId) {
   229→      setExpandedFolioId(null)
   230→    } else {
   231→      const folio = customerHistory?.historicalFolios.find(f => f.folioId === folioId)
   232→      if (folio) {
   233→        setExpandedFolioId(folioId)
   234→        setExpandedFolioEntries(prev => ({
   235→          ...prev,
   236→          [folioId]: (folio.entries || []),
   237→        }))
   238→      }
   239→    }
   240→  }
   241→
   242→  // ─── Track checkout transition ───
   243→  const wasCheckedOut = React.useRef(isCheckedOut)
   244→  React.useEffect(() => {
   245→    if (isCheckedOut && !wasCheckedOut.current) {
   246→      setPostCheckoutOpen(true)
   247→    }
   248→    wasCheckedOut.current = isCheckedOut
   249→  }, [isCheckedOut])
   250→
   251→  // ─── Show TXN number when lastCreatedDocNo changes ───
   252→  React.useEffect(() => {
   253→    if (lastCreatedDocNo && lastCreatedType) {
   254→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   255→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   256→      return () => clearTimeout(timer)
   257→    }
   258→  }, [lastCreatedDocNo, lastCreatedType])
   259→
   260→  // ─── Items for charge selector (Req 5) ───
   261→  const [items, setItems] = React.useState<{ id: string; name: string; price: number }[]>([])
   262→  const [itemsLoading, setItemsLoading] = React.useState(false)
   263→  React.useEffect(() => {
   264→    if (chargeOpen && apiHeaders) {
   265→      setItemsLoading(true)
   266→      fetch('/api/items', { headers: apiHeaders }).then(r => r.ok ? r.json() : []).then((list: any[]) => {
   267→        setItems((list || []).map((it: any) => ({ id: it.id, name: it.name, price: it.price || 0 })))
   268→      }).catch(() => {}).finally(() => setItemsLoading(false))
   269→    }
   270→  }, [chargeOpen, apiHeaders])
   271→
   272→  // Handle item selection
   273→  const handleItemSelect = React.useCallback((itemId: string) => {
   274→    const item = items.find(it => it.id === itemId)
   275→    if (item) {
   276→      setChargeForm({
   277→        ...chargeForm,
   278→        selectedItem: itemId,
   279→        selectedItemPrice: String(item.price),
   280→        description: item.name,
   281→        amount: item.price > 0 ? String(item.price) : chargeForm.amount,
   282→      })
   283→    }
   284→  }, [items, chargeForm])
   285→
   286→  // ─── Fetch invoice number ───
   287→  const fetchInvoiceNumber = React.useCallback(async () => {
   288→    if (!detail || !apiHeaders) return
   289→    try {
   290→      setInvNumLoading(true)
   291→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   292→        method: 'POST',
   293→        headers: apiHeaders,
   294→      })
   295→      if (res.ok) {
   296→        const data = await res.json()
   297→        setInvoiceNumber(data.invoiceNumber)
   298→      }
   299→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   300→  }, [detail?.id, apiHeaders])
   301→
   302→  // ─── Fetch invoice number when invoice dialog opens ───
   303→  React.useEffect(() => {
   304→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber) {
   305→      fetchInvoiceNumber()
   306→    }
   307→  }, [invoiceOpen, detail?.id, apiHeaders])
   308→
   309→  React.useEffect(() => {
   310→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   311→      fetchInvoiceData()
   312→    }
   313→  }, [invoiceOpen])
   314→
   315→  // ─── Entry creation with TXN tracking ───
   316→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   317→    if (!detail || !selectedFolioId || !apiHeaders) {
   318→      return null
   319→    }
   320→    try {
   321→      setLocalSubmitting(true)
   322→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   323→        method: 'POST',
   324→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   325→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   326→      })
   327→      if (res.ok) {
   328→        const entry = await res.json()
   329→        const docNo = entry?.docNo || ''
   330→        setEntrySuccess({ show: true, docNo, type })
   331→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   332→        closeDialogs()
   333→        onRefresh?.()
   334→        setTimeout(() => setEntrySuccess(null), 4000)
   335→        return entry
   336→      } else {
   337→        const e = await safeJson(res)
   338→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   339→        return null
   340→      }
   341→    } catch {
   342→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   343→      return null
   344→    } finally {
   345→      setLocalSubmitting(false)
   346→    }
   347→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   348→
   349→  // ─── Settlement with TXN tracking ───
   350→  const settleWithTxn = React.useCallback(async () => {
   351→    if (!detail || !paymentForm.amount || !apiHeaders) {
   352→      handleSettle()
   353→      return
   354→    }
   355→    try {
   356→      setLocalSubmitting(true)
   357→      // When master folio is selected, use autoAllocate to distribute across all folios
   358→      const isMaster = selectedFolioId === 'master'
   359→      const payload = isMaster
   360→        ? { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   361→        : { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   362→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   363→        method: 'POST',
   364→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   365→        body: JSON.stringify(payload),
   366→      })
   367→      if (res.ok) {
   368→        const data = await res.json()
   369→        // Auto-allocate may return settledFolios array instead of single ledgerEntry
   370→        const docNo = data?.ledgerEntry?.docNo || (data?.settledFolios?.[0] ? 'Multi-Folio' : '')
   371→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })
   372→        notify({ type: 'success', title: `Settlement ${docNo}`, message: data.message || `Settlement ${docNo} processed successfully` })
   373→        closeDialogs()
   374→        onRefresh?.()
   375→        setTimeout(() => setEntrySuccess(null), 4000)
   376→      } else {
   377→        const e = await safeJson(res)
   378→        notify({ type: 'error', title: 'Settlement Failed', message: (e.error as string) || 'Failed to process settlement' })
   379→      }
   380→    } catch {
   381→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   382→    } finally {
   383→      setLocalSubmitting(false)
   384→    }
   385→  }, [detail?.id, selectedFolioId, apiHeaders, paymentForm, handleSettle, closeDialogs, onRefresh])
   386→
   387→  // ─── Transfer with TXN tracking ───
   388→  const transferWithTxn = React.useCallback(async () => {
   389→    if (!detail || !transferForm.targetCustomerId || !apiHeaders) {
   390→      handleTransfer()
   391→      return
   392→    }
   393→    try {
   394→      setLocalSubmitting(true)
   395→      // If entries are selected, transfer only those; otherwise pure balance transfer
   396→      const selectedEntryIds = selectedEntries.size > 0 ? Array.from(selectedEntries) : undefined
   397→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   398→        method: 'POST',
   399→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   400→        body: JSON.stringify({
   401→          targetCustomerId: transferForm.targetCustomerId,
   402→          amount: parseFloat(transferForm.amount) || 0,
   403→          description: transferForm.description,
   404→          entryIds: selectedEntryIds,
   405→        }),
   406→      })
   407→      if (res.ok) {
   408→        const data = await res.json()
   409→        const docNo = data?.source?.entry?.docNo || ''
   410→        setEntrySuccess({ show: true, docNo, type: 'Transfer' })
   411→        notify({ type: 'success', title: `Transfer ${docNo}`, message: data.message || `Transfer ${docNo} processed successfully` })
   412→        closeDialogs()
   413→        onRefresh?.()
   414→        setTimeout(() => setEntrySuccess(null), 4000)
   415→      } else {
   416→        const e = await safeJson(res)
   417→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   418→      }
   419→    } catch {
   420→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   421→    } finally {
   422→      setLocalSubmitting(false)
   423→    }
   424→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   425→
   426→  // ─── Computed GST data from invoiceData ───
   427→  const gstData = React.useMemo(() => {
   428→    if (!invoiceData?.entries) return null
   429→    const entries = invoiceData.entries as any[]
   430→    const taxableEntries = entries.filter((e: any) => e.debit > 0)
   431→    const taxableAmount = taxableEntries.reduce((sum: number, e: any) => sum + (e.debit || 0) - (e.tax || 0) - (e.discount || 0), 0)
   432→    const totalTax = entries.reduce((sum: number, e: any) => sum + (e.tax || 0), 0)
   433→    const totalDiscount = entries.reduce((sum: number, e: any) => sum + (e.discount || 0), 0)
   434→    const totalCharges = entries.reduce((sum: number, e: any) => sum + (e.debit || 0), 0)
   435→    const cgst = totalTax / 2
   436→    const sgst = totalTax / 2
   437→    const totalPayments = entries.reduce((sum: number, e: any) => sum + (e.credit || 0), 0)
   438→    const advancePayment = bal < 0 ? Math.abs(bal) : 0
