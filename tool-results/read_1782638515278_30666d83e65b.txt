    74→// ─── Helper: Compute the most-used payment mode from credit entries ───
    75→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
    76→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    77→  if (creditEntries.length === 0) return 'N/A'
    78→  const modeCounts: Record<string, number> = {}
    79→  for (const e of creditEntries) {
    80→    const mode = e.paymentMode || 'Cash'
    81→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
    82→  }
    83→  let bestMode = 'Cash'
    84→  let bestAmount = 0
    85→  for (const [mode, amount] of Object.entries(modeCounts)) {
    86→    if (amount > bestAmount) {
    87→      bestAmount = amount
    88→      bestMode = mode
    89→    }
    90→  }
    91→  return bestMode
    92→}

    94→// ─── Helper: Build payment summary HTML ───
    95→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
    96→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    97→  if (creditEntries.length === 0) {
    98→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
    99→  }
   100→
   101→  const modeTotals: Record<string, number> = {}
   102→  for (const e of creditEntries) {
   103→    const mode = e.paymentMode || 'Cash'
   104→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   105→  }
   106→
   107→  const modes = Object.entries(modeTotals)
   108→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   109→
   110→  if (modes.length === 1) {
   111→    const [mode, amount] = modes[0]
   112→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   113→  }
   114→
   115→  const rows = modes.map(([mode, amount]) => {
   116→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   117→  }).join('')
   118→
   119→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   120→}

   122→// ─── Helper: Get stable document number from entries (never regenerate) ───
   123→// Searches ALL entries for the first non-empty docNo to avoid N/A
   124→function getStableDocId(entries: { docNo?: string; invoiceNo?: string }[]): string {
   125→  if (entries.length === 0) return 'N/A'
   126→  for (const e of entries) {
   127→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   128→  }
   129→  for (const e of entries) {
   130→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   131→  }
   132→  return entries[0].docNo || entries[0].invoiceNo || 'N/A'
   133→}

   242→// ─── Print Folio Invoice (formerly "Receipt") ───
   243→// Uses stable doc IDs from existing entries — NEVER regenerates numbers on reprint
   244→export async function printReceipt(
   245→  detail: CustomerDetail,
   246→  entries: LedgerEntry[],
   247→  hdrRef: RefObject<Record<string, string>>
   248→) {
   249→  if (!detail) return
   250→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   251→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   252→  const settings = await fetchSettingsForPrint(hdrRef)
   253→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   254→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   255→  const totalTax = activeEntries.reduce((s, e) => s + (e.tax || 0), 0)
   256→  const totalDiscounts = activeEntries.reduce((s, e) => s + (e.discount || 0), 0)
   257→  const finalBal = activeEntries[activeEntries.length - 1].balance
   258→  const now = new Date()
   259→  const rows = activeEntries.map((e, i) => {
   260→    const desc = stripInternalDescriptions(e.description || e.type || '—')
   261→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td style="font-size:9px;color:#666;font-family:monospace">${e.docNo || ''}</td><td>${e.type || ''}</td><td>${desc}</td><td style="text-align:right;color:#dc2626;font-weight:600">${e.debit > 0 ? fmtINR(e.debit) : '—'}</td><td style="text-align:right;color:#16a34a;font-weight:600">${e.credit > 0 ? fmtINR(e.credit) : '—'}</td><td style="font-weight:700;color:${e.balance > 0 ? '#dc2626' : e.balance < 0 ? '#16a34a' : '#666'}">${fmtINR(e.balance)}</td></tr>`
   262→  }).join('')
   263→
   264→  // Use stable doc ID from entries — never regenerate
   265→  const docId = getStableDocId(activeEntries)
   266→  const authorizedSignatory = (settings as Record<string, string>).authorizedSignatory || ''
   267→
   268→  const w = openPrintWindow(`Folio Invoice - ${detail.fullName}`)
   269→  if (!w) return
   270→
   271→  w.document.write(
   272→    `<!DOCTYPE html><html><head><title>Folio Invoice</title><style>${PRINT_CSS}</style></head><body>` +
   273→    buildUnifiedHeader(settings, 'FOLIO INVOICE', docId) +
   274→    `<div class="invoice-meta">
   275→      <div><div class="inv-label">Doc No.</div><div class="inv-value">${docId}</div></div>
   276→      <div style="text-align:center"><div class="inv-label">Date</div><div class="inv-value">${fmtDate(now.toISOString())}</div></div>
   277→      <div style="text-align:right"><div class="inv-label">Status</div><div class="inv-value">${detail.status === 'CheckedOut' ? 'Checked Out' : detail.status}</div></div>
   278→    </div>` +
   279→    buildCustomerInfoBlock(detail) +
   280→    `<hr class="divider">` +
   281→    `<table><tr><th>#</th><th>Date</th><th>TXN ID</th><th>Type</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   282→    `<tr class="totals"><td colspan="5" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(finalBal)}</td></tr></table>` +
   283→    buildPaymentSummary(activeEntries) +
   284→    `<div class="tax-summary"><div class="tax-item"><span class="tax-label">Total Tax: </span><span class="tax-val">${fmtINR(totalTax)}</span></div><div class="tax-item"><span class="tax-label">Discount: </span><span class="tax-val">${fmtINR(totalDiscounts)}</span></div><div class="tax-item"><span class="tax-label">Net Charges: </span><span class="tax-val">${fmtINR(totalDebits - totalCredits)}</span></div></div>` +
   285→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Total Charges</div><div class="s-value" style="color:#dc2626">${fmtINR(totalDebits)}</div></div><div class="summary-item"><div class="s-label">Total Payments</div><div class="s-value" style="color:#16a34a">${fmtINR(totalCredits)}</div></div><div class="summary-item"><div class="s-label">Final Balance</div><div class="s-value" style="color:${finalBal > 0 ? '#dc2626' : finalBal < 0 ? '#16a34a' : '#666'}">${fmtINR(finalBal)}</div></div></div>` +
   286→    `<div class="signature-section">
   287→      <div class="signature-box"><div class="signature-line"><div class="sig-label">Customer Signature</div><div class="sig-sub">Authorized Representative</div></div></div>
   288→      <div class="signature-box"><div class="signature-line"><div class="sig-label">${authorizedSignatory || 'Authorized Signatory'}</div><div class="sig-sub">${authorizedSignatory ? 'Authorized Signatory' : 'For ' + (settings.shopName || 'Eservices')}</div></div></div>
   289→    </div>` +
   290→    buildFooter() +
   291→    `</body></html>`
   292→  )
   293→  w.document.close()
   294→  w.print()
   295→}

   297→// ─── Print Invoice (Professional) ───
   298→export async function printInvoice(
   299→  detail: CustomerDetail,
   300→  invoiceData: { entries?: any[]; closingBalance?: number; openingBalance?: number },
   301→  invoiceFrom: string,
   302→  invoiceTo: string
   303→) {
   304→  if (!detail || !invoiceData) return
   305→  const entries = invoiceData.entries || []
   306→  if (entries.length === 0) return
   307→  const totalDebits = entries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   308→  const totalCredits = entries.reduce((s: number, e: any) => s + (e.credit || 0), 0)
   309→  const totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   310→  const totalDiscount = entries.reduce((s: number, e: any) => s + (e.discount || 0), 0)
   311→  const closingBal = invoiceData.closingBalance ?? 0
   312→  const openingBal = invoiceData.openingBalance ?? 0
   313→  const now = new Date()
   314→
   315→  // Fetch shop settings for professional header
   316→  let shopSettings: Record<string, string> = {}
   317→  try {
   318→    const res = await fetch('/api/settings')
   319→    if (res.ok) shopSettings = await res.json()
   320→  } catch { /* ignore */ }
   321→
   322→  // Use stable doc ID from entries — never regenerate on reprint
   323→  const docId = getStableDocId(entries)
   324→  const authorizedSignatory = shopSettings.authorizedSignatory || ''
