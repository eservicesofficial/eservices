   237→function ChangePasswordSection() {
   238→  const { currentUser, token } = useAuth();
   239→  const [currentPassword, setCurrentPassword] = useState('');
   240→  const [newPassword, setNewPassword] = useState('');
   241→  const [confirmPassword, setConfirmPassword] = useState('');
   242→  const [showCurrent, setShowCurrent] = useState(false);
   243→  const [showNew, setShowNew] = useState(false);
   244→  const [showConfirm, setShowConfirm] = useState(false);
   245→  const [loading, setLoading] = useState(false);
   246→
   247→  const handleChangePassword = async () => {
   248→    if (!currentPassword || !newPassword || !confirmPassword) {
   249→      notify({ type: 'error', title: 'Error', message: 'All fields are required' });
   250→      return;
   251→    }
   252→    if (newPassword.length < 4) {
   253→      notify({ type: 'error', title: 'Error', message: 'New password must be at least 4 characters' });
   254→      return;
   255→    }
   256→    if (newPassword !== confirmPassword) {
   257→      notify({ type: 'error', title: 'Error', message: 'New passwords do not match' });
   258→      return;
   259→    }
   260→    setLoading(true);
   261→    try {
   262→      const res = await fetch(`/api/users/${currentUser?.userId}`, {
   263→        method: 'PUT',
   264→        headers: {
   265→          'Content-Type': 'application/json',
   266→          'Authorization': `Bearer ${token}`,
   267→        },
   268→        body: JSON.stringify({
   269→          currentPassword,
   270→          password: newPassword,
   271→          _passwordOnly: true,
   272→        }),
   273→      });
   274→      if (res.ok) {
   275→        notify({ type: 'success', title: 'Password Updated', message: 'Password changed successfully' });
   276→        setCurrentPassword('');
   277→        setNewPassword('');
   278→        setConfirmPassword('');
   279→      } else {
   280→        const data = await res.json().catch(() => null);
   281→        notify({ type: 'error', title: 'Error', message: data?.error || 'Failed to change password' });
   282→      }
   283→    } catch {
   284→      notify({ type: 'error', title: 'Error', message: 'Network error' });
   285→    } finally {
   286→      setLoading(false);
   287→    }
   288→  };
   289→
   290→  return (
   291→    <Card className="overflow-hidden border-0 shadow-sm">
   292→      <CardHeader className="pb-2">
   293→        <div className="flex items-center gap-3">
   294→          <div className="flex size-9 items-center justify-center rounded-lg bg-amber-100 text-amber-600 shrink-0">
   295→            <Lock className="size-4" />
   296→          </div>
   297→          <div className="min-w-0">
   298→            <CardTitle className="text-base">Change Password</CardTitle>
   299→            <CardDescription className="mt-0.5">Update your account password</CardDescription>
   300→          </div>
   301→        </div>
   302→      </CardHeader>
   303→      <div className="px-6 pb-6 space-y-4">
   304→        <div className="grid gap-2 border-l-2 border-l-amber-400 pl-3">
   305→          <Label className="text-sm font-medium">Current Password</Label>
   306→          <div className="relative">
   307→            <Input
   308→              type={showCurrent ? 'text' : 'password'}
   309→              placeholder="Enter current password"
   310→              value={currentPassword}
   311→              onChange={(e) => setCurrentPassword(e.target.value)}
   312→              className="pr-10"
   313→            />
   314→            <button type="button" onClick={() => setShowCurrent(!showCurrent)} className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground">
   315→              {showCurrent ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   316→            </button>
   317→          </div>
   318→        </div>
   319→        <div className="grid gap-2 border-l-2 border-l-amber-400 pl-3">
   320→          <Label className="text-sm font-medium">New Password</Label>
   321→          <div className="relative">
   322→            <Input
   323→              type={showNew ? 'text' : 'password'}
   324→              placeholder="Enter new password (min 4 chars)"
   325→              value={newPassword}
   326→              onChange={(e) => setNewPassword(e.target.value)}
   327→              className="pr-10"
   328→            />
   329→            <button type="button" onClick={() => setShowNew(!showNew)} className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground">
   330→              {showNew ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   331→            </button>
   332→          </div>
   333→        </div>
   334→        <div className="grid gap-2 border-l-2 border-l-amber-400 pl-3">
   335→          <Label className="text-sm font-medium">Confirm New Password</Label>
   336→          <div className="relative">
   337→            <Input
   338→              type={showConfirm ? 'text' : 'password'}
   339→              placeholder="Confirm new password"
   340→              value={confirmPassword}
   341→              onChange={(e) => setConfirmPassword(e.target.value)}
   342→              className="pr-10"
   343→            />
   344→            <button type="button" onClick={() => setShowConfirm(!showConfirm)} className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground">
   345→              {showConfirm ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   346→            </button>
   347→          </div>
   348→        </div>
   349→        <div className="flex justify-end">
   350→          <Button onClick={handleChangePassword} disabled={loading} className="gap-1.5 bg-amber-600 hover:bg-amber-700 text-white">
   351→            {loading ? <Loader2 className="size-4 animate-spin" /> : <><Lock className="size-4" />Update Password</>}
   352→          </Button>
   353→        </div>
   354→      </div>
   355→    </Card>
   356→  );
   357→}
