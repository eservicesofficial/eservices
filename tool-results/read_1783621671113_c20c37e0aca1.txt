   168→export function CustomerFolioDialogs({
   169→  detail, allFolios, selectedEntries, submitting, settlingMode,
   170→  selectedFolioId, bal, selectedEntryObj, isActive, isCheckedOut,
   171→  invoiceData, invoiceLoading, folioStatementData, customerTags,
   172→  transferResults, paymentEntries,
   173→  addOpen, editOpen, chargeOpen, paymentOpen, refundOpen, transferOpen,
   174→  discountOpen, docOpen, noteOpen, tagsOpen, checkoutOpen, deleteOpen,
   175→  splitAmountOpen, shiftOpen, removeOpen, voucherOpen, invoiceOpen,
   176→  detailsSheetOpen,
   177→  form, chargeForm, paymentForm, refundForm, transferForm, discountForm,
   178→  docForm, noteForm, splitAmountForm, invoiceFrom, invoiceTo,
   179→  shiftTargetFolioId, duplicate,
   180→  setAddOpen, setEditOpen, setChargeOpen, setPaymentOpen, setRefundOpen,
   181→  setTransferOpen, setDiscountOpen, setDocOpen, setNoteOpen, setTagsOpen,
   182→  setCheckoutOpen, setDeleteOpen, setSplitAmountOpen, setShiftOpen,
   183→  setRemoveOpen, setVoucherOpen, setInvoiceOpen, setDetailsSheetOpen,
   184→  setForm, setChargeForm, setPaymentForm, setRefundForm, setTransferForm,
   185→  setDiscountForm, setDocForm, setNoteForm, setSplitAmountForm,
   186→  setInvoiceFrom, setInvoiceTo, setShiftTargetFolioId, setDuplicate,
   187→  handleCreate, handleUpdate, handleCharge, handlePayment, handleSettle,
   188→  handleRefund, handleDiscount, handleTransfer, handleShift, handleSplitAmount,
   189→  handleRemove, handleCheckout, handleDelete, handleAddDoc, handleAddNote,
   190→  handleToggleTag, handleDeleteDoc, handleDeleteNote,
   191→  fetchInvoiceData, fetchFolioStatement, printInvoice, printCompleteInvoice, printSingleFolio, printVoucherHistory,
   192→  closeDialogs, searchTransferTargets,
   193→  deleteFolioOpen, folioToDelete,
   194→  setDeleteFolioOpen, handleDeleteFolio,
   195→  // Enhanced props
   196→  apiHeaders, onRefresh, canDelete = false, isAdmin = false, onPrintReceipt, onThermalPrintReceipt,
   197→  onThermalPrintInvoice, onThermalPrintVoucher,
   198→  lastCreatedDocNo, lastCreatedType,
   199→  // History
   200→  historyOpen, setHistoryOpen, onPrintHistoryFolio,
   201→}: DialogsProps) {
   202→
   203→  // ─── Internal State ───
   204→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   205→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   206→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   207→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   208→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   209→  // ─── History state ───
   210→  const [historyLoading, setHistoryLoading] = React.useState(false)
   211→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   212→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   213→  const [selectedVoucherIds, setSelectedVoucherIds] = React.useState<Set<string>>(new Set())
   214→  const [historyDate, setHistoryDate] = React.useState(() => {
   215→    const today = new Date()
   216→    return today.toISOString().split('T')[0] // "2025-01-15"
   217→  })
   218→
   219→  // Fetch history for the CURRENT customer
   220→  const fetchCustomerHistory = React.useCallback(async (date?: string) => {
   221→    if (!detail || !apiHeaders) return
   222→    try {
   223→      setHistoryLoading(true)
   224→      const q = date ? `?date=${date}` : ''
   225→      const res = await fetch(`/api/folio/customers/${detail.id}/history${q}`, { headers: apiHeaders })
   226→      if (res.ok) {
   227→        const data = await res.json() as CustomerHistory
   228→        setCustomerHistory(data)
   229→      }
   230→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to fetch checkout history' }) } finally { setHistoryLoading(false) }
   231→  }, [detail?.id, apiHeaders])
   232→
   233→  React.useEffect(() => {
   234→    if (historyOpen && detail && apiHeaders && fetchCustomerHistory) {
   235→      fetchCustomerHistory(historyDate)
   236→    } else if (!historyOpen) {
   237→      setCustomerHistory(null)
   238→      setExpandedFolioId(null)
   239→    }
   240→  }, [historyOpen, detail?.id, fetchCustomerHistory, apiHeaders, historyDate])
   241→
   242→  // ─── Print helpers for checkout records ───
   243→  const getTplCSS = (s: Record<string, string>) => {
   244→    const tid = s.invoiceTemplate || 'modern'
   245→    let cust: InvoiceCustomizations = {}
   246→    try { if (s.invoiceCustomizations) cust = JSON.parse(s.invoiceCustomizations) } catch { /* */ }
   247→    return getTemplateCSS(tid, cust)
   248→  }
   249→
   250→  // Build frozen customer detail from checkout snapshot (P2: never use live data for reprint)
   251→  const buildFrozenDetail = (co: CheckoutRecord): CustomerDetail => {
   252→    const base = detail || {} as CustomerDetail
   253→    return {
   254→      ...base,
   255→      fullName: co.customerName || base.fullName,
   256→      mobile: co.customerMobile || base.mobile,
   257→      email: co.customerEmail || base.email,
   258→      address: co.customerAddress || base.address,
   259→      customerType: co.customerType || base.customerType,
   260→      customerId: co.customerIdRef || base.customerId,
   261→      gstNumber: co.customerGstNumber || base.gstNumber || '',
   262→      gstAddress: co.customerGstAddress || base.gstAddress || '',
   263→      panNumber: co.customerPanNumber || base.panNumber || '',
   264→      companyName: co.customerCompanyName || base.companyName || '',
   265→      totalVisits: co.visitNumber || base.totalVisits,
   266→    } as CustomerDetail
   267→  }
   268→
   269→  const printCheckoutFolioBill = async (co: CheckoutRecord) => {
   270→    if (!co.transactions.length) return
   271→    let shopSettings: Record<string, string> = {}
   272→    try {
   273→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   274→    } catch { /* use defaults */ }
   275→    const tplCss = getTplCSS(shopSettings)
   276→    const printDetail = buildFrozenDetail(co)
   277→    const w = window.open('', '_blank')
   278→    if (!w) return
   279→    w.document.write(`<!DOCTYPE html><html><head><title>Folio Bill - ${co.checkoutNumber}</title><style>${INVOICE_CSS}</style><style>${tplCss}</style></head><body>${buildHotelStyleInvoice(shopSettings, printDetail, co.transactions, co.invoiceNumber, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'FOLIO BILL', undefined, co.checkoutUser)}</body></html>`)
   280→    w.document.close(); w.print()
   281→  }
   282→
   283→  const printCheckoutInvoice = async (co: CheckoutRecord) => {
   284→    if (!co.transactions.length) return
   285→    let shopSettings: Record<string, string> = {}
   286→    try {
   287→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   288→    } catch { /* use defaults */ }
   289→    const tplCss = getTplCSS(shopSettings)
   290→    const printDetail = buildFrozenDetail(co)
   291→    const w = window.open('', '_blank')
   292→    if (!w) return
   293→    w.document.write(`<!DOCTYPE html><html><head><title>Tax Invoice - ${co.invoiceNumber}</title><style>${INVOICE_CSS}</style><style>${tplCss}</style></head><body>${buildHotelStyleInvoice(shopSettings, printDetail, co.transactions, co.invoiceNumber, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'Tax Invoice', { openingBalance: co.openingBalance, closingBalance: co.closingBalance }, co.checkoutUser)}</body></html>`)
   294→    w.document.close(); w.print()
   295→  }
   296→
   297→  // ─── Track checkout transition ───
   298→  const wasCheckedOut = React.useRef(isCheckedOut)
   299→  React.useEffect(() => {
   300→    if (isCheckedOut && !wasCheckedOut.current) {
   301→      setPostCheckoutOpen(true)
   302→    }
   303→    wasCheckedOut.current = isCheckedOut
   304→  }, [isCheckedOut])
   305→
   306→  // ─── Show TXN number when lastCreatedDocNo changes ───
   307→  React.useEffect(() => {
   308→    if (lastCreatedDocNo && lastCreatedType) {
   309→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   310→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   311→      return () => clearTimeout(timer)
   312→    }
   313→  }, [lastCreatedDocNo, lastCreatedType])
   314→
   315→  // ─── Items for charge selector (Req 5) ───
   316→  const [items, setItems] = React.useState<{ id: string; name: string; price: number }[]>([])
   317→  const [itemsLoading, setItemsLoading] = React.useState(false)
   318→  React.useEffect(() => {
   319→    if (chargeOpen && apiHeaders) {
   320→      setItemsLoading(true)
   321→      fetch('/api/items', { headers: apiHeaders }).then(r => r.ok ? r.json() : []).then((list: any[]) => {
   322→        setItems((list || []).map((it: any) => ({ id: it.id, name: it.name, price: it.price || 0 })))
   323→      }).catch(() => {}).finally(() => setItemsLoading(false))
   324→    }
   325→  }, [chargeOpen, apiHeaders])
   326→
   327→  // Handle item selection
   328→  const handleItemSelect = React.useCallback((itemId: string) => {
   329→    const item = items.find(it => it.id === itemId)
   330→    if (item) {
   331→      setChargeForm({
   332→        ...chargeForm,
   333→        selectedItem: itemId,
   334→        selectedItemPrice: String(item.price),
   335→        description: item.name,
   336→        amount: item.price > 0 ? String(item.price) : chargeForm.amount,
   337→      })
   338→    }
   339→  }, [items, chargeForm])
   340→
   341→  // ─── Fetch invoice number ───
   342→  const fetchInvoiceNumber = React.useCallback(async () => {
   343→    if (!detail || !apiHeaders) return
   344→    try {
   345→      setInvNumLoading(true)
   346→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   347→        method: 'POST',
   348→        headers: apiHeaders,
   349→      })
   350→      if (res.ok) {
   351→        const data = await res.json()
   352→        setInvoiceNumber(data.invoiceNumber)
   353→      }
   354→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   355→  }, [detail?.id, apiHeaders])
   356→
   357→  // ─── Fetch invoice number when invoice dialog opens ───
   358→  React.useEffect(() => {
   359→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber && fetchInvoiceNumber) {
   360→      fetchInvoiceNumber()
   361→    }
   362→  }, [invoiceOpen, detail?.id, apiHeaders, invoiceNumber, fetchInvoiceNumber])
   363→
   364→  React.useEffect(() => {
   365→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   366→      fetchInvoiceData()
   367→    }
   368→  }, [invoiceOpen, invoiceData, detail?.id, selectedFolioId])
   369→
   370→  // ─── Entry creation with TXN tracking ───
   371→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   372→    if (!detail || !selectedFolioId || !apiHeaders) {
   373→      return null
   374→    }
   375→    try {
   376→      setLocalSubmitting(true)
   377→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   378→        method: 'POST',
   379→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   380→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   381→      })
   382→      if (res.ok) {
   383→        const entry = await res.json()
   384→        const docNo = entry?.docNo || ''
   385→        setEntrySuccess({ show: true, docNo, type })
   386→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   387→        closeDialogs()
   388→        onRefresh?.()
   389→        setTimeout(() => setEntrySuccess(null), 4000)
   390→        return entry
   391→      } else {
   392→        const e = await safeJson(res)
   393→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   394→        return null
   395→      }
   396→    } catch {
   397→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   398→      return null
   399→    } finally {
   400→      setLocalSubmitting(false)
   401→    }
   402→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   403→
   404→  // ─── Settlement with TXN tracking ───
   405→  const settleWithTxn = React.useCallback(async () => {
   406→    if (!detail || !paymentForm.amount || !apiHeaders) {
   407→      handleSettle()
   408→      return
   409→    }
   410→    try {
   411→      setLocalSubmitting(true)
   412→      // When master folio is selected, use autoAllocate to distribute across all folios
   413→      const isMaster = selectedFolioId === 'master'
   414→      const payload = isMaster
   415→        ? { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   416→        : { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   417→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   418→        method: 'POST',
   419→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   420→        body: JSON.stringify(payload),
   421→      })
   422→      if (res.ok) {
