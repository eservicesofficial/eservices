   205→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   206→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   207→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   208→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   209→  // ─── History state ───
   210→  const [historyLoading, setHistoryLoading] = React.useState(false)
   211→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   212→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   213→  const [selectedVoucherIds, setSelectedVoucherIds] = React.useState<Set<string>>(new Set())
   214→  const [historyDateFrom, setHistoryDateFrom] = React.useState(() => {
   215→    const today = new Date()
   216→    return today.toISOString().split('T')[0]
   217→  })
   218→  const [historyDateTo, setHistoryDateTo] = React.useState(() => {
   219→    const today = new Date()
   220→    return today.toISOString().split('T')[0]
   221→  })
   222→  const [editingInvoiceId, setEditingInvoiceId] = React.useState<string | null>(null)
   223→  const [editInvoiceValue, setEditInvoiceValue] = React.useState('')
   224→  const [savingInvoiceNum, setSavingInvoiceNum] = React.useState(false)
   225→
   226→  // Fetch history for the CURRENT customer
   227→  const fetchCustomerHistory = React.useCallback(async () => {
   228→    if (!detail || !apiHeaders) return
   229→    try {
   230→      setHistoryLoading(true)
   231→      const params = new URLSearchParams()
   232→      if (historyDateFrom) params.set('dateFrom', historyDateFrom)
   233→      if (historyDateTo) params.set('dateTo', historyDateTo)
   234→      const q = params.toString() ? `?${params.toString()}` : ''
   235→      const res = await fetch(`/api/folio/customers/${detail.id}/history${q}`, { headers: apiHeaders })
   236→      if (res.ok) {
   237→        const data = await res.json() as CustomerHistory
   238→        setCustomerHistory(data)
   239→      }
   240→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to fetch checkout history' }) } finally { setHistoryLoading(false) }
   241→  }, [detail?.id, apiHeaders, historyDateFrom, historyDateTo])
   242→

   244→    if (historyOpen && detail && apiHeaders && fetchCustomerHistory) {
   245→      fetchCustomerHistory()

   247→      setCustomerHistory(null)
   248→      setExpandedFolioId(null)
   249→    }

   265→      fullName: co.customerName || base.fullName,

   274→      companyName: co.customerCompanyName || base.companyName || '',

   283→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})

   290→    w.document.write(`<!DOCTYPE html><html><head><title>Folio Bill - ${invNo}</title><style>${INVOICE_CSS}</style><style>${tplCss}</style></head><body>${buildHotelStyleInvoice(shopSettings, printDetail, co.transactions, invNo, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'FOLIO BILL', undefined, co.checkoutUser)}</body></html>`)
   291→    w.document.close(); w.print()
   292→  }
   293→

   295→    if (!co.transactions.length) return
   296→    let shopSettings: Record<string, string> = {}
   297→    try {
   298→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   299→    } catch { /* use defaults */ }
   300→    const tplCss = getTplCSS(shopSettings)
   301→    const printDetail = buildFrozenDetail(co)
   302→    const invNo = co.customInvoiceNumber || co.invoiceNumber || co.checkoutNumber
   303→    const w = window.open('', '_blank')
   304→    if (!w) return
   305→    w.document.write(`<!DOCTYPE html><html><head><title>Tax Invoice - ${invNo}</title><style>${INVOICE_CSS}</style><style>${tplCss}</style></head><body>${buildHotelStyleInvoice(shopSettings, printDetail, co.transactions, invNo, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'Tax Invoice', { openingBalance: co.openingBalance, closingBalance: co.closingBalance }, co.checkoutUser)}</body></html>`)
   306→    w.document.close(); w.print()
   307→  }
   308→
   309→  const saveCustomInvoiceNumber = async (co: CheckoutRecord) => {

   311→    setSavingInvoiceNum(true)
   312→    try {
   313→      const res = await fetch(`/api/folio/customers/${detail.id}/history`, {

   323→        notify({ type: 'error', title: 'Error', message: 'Failed to save' })
   324→      }
   325→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to save' }) }
   326→    finally { setSavingInvoiceNum(false) }
   327→  }
   328→

   354→        setItems((list || []).map((it: any) => ({ id: it.id, name: it.name, price: it.price || 0 })))

   357→  }, [chargeOpen, apiHeaders])

   400→  }, [invoiceOpen, invoiceData, detail?.id, selectedFolioId])
   401→
   402→  // ─── Entry creation with TXN tracking ───
   403→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   404→    if (!detail || !selectedFolioId || !apiHeaders) {
   405→      return null
   406→    }
   407→    try {
   408→      setLocalSubmitting(true)
   409→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   410→        method: 'POST',
   411→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   412→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   413→      })
   414→      if (res.ok) {
   415→        const entry = await res.json()
   416→        const docNo = entry?.docNo || ''
   417→        setEntrySuccess({ show: true, docNo, type })
   418→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   419→        closeDialogs()
   420→        onRefresh?.()
   421→        setTimeout(() => setEntrySuccess(null), 4000)
   422→        return entry
