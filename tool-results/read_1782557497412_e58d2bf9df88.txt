    50→// Build entry rows HTML (shared across print templates)
    51→function buildEntryRows(entries: { date: string; description?: string; type?: string; debit: number; credit: number; balance: number; tax?: number }[]): string {
    52→  return entries.map((e, i) => {
    53→    const desc = stripInternalDescriptions(e.description || e.type || '—')
    54→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td>${desc}</td><td style="text-align:right;color:#dc2626;font-weight:600">${e.debit > 0 ? fmtINR(e.debit) : '—'}</td><td style="text-align:right;color:#16a34a;font-weight:600">${e.credit > 0 ? fmtINR(e.credit) : '—'}</td><td style="font-weight:700;color:${e.balance >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(e.balance)}</td></tr>`
    55→  }).join('')
    56→}

    66→async function fetchSettingsForPrint(hdrRef: RefObject<Record<string, string>>): Promise<Record<string, string>> {

    86→// ─── Print Receipt ───
    87→export async function printReceipt(
    88→  detail: CustomerDetail,
    89→  entries: LedgerEntry[],
    90→  hdrRef: RefObject<Record<string, string>>
    91→) {
    92→  if (!detail) return
    93→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
    94→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
    95→  const settings = await fetchSettingsForPrint(hdrRef)
    96→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
    97→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
    98→  const totalTax = activeEntries.reduce((s, e) => s + (e.tax || 0), 0)
    99→  const totalDiscounts = activeEntries.reduce((s, e) => s + (e.discount || 0), 0)
   100→  const finalBal = activeEntries[activeEntries.length - 1].balance
   101→  const now = new Date()
   102→  const rows = buildEntryRows(activeEntries)
   103→
   104→  const w = openPrintWindow(`Receipt - ${detail.fullName}`)
   105→  if (!w) return
   106→
   107→  w.document.write(
   108→    `<!DOCTYPE html><html><head><title>Receipt</title><style>${PRINT_CSS}</style></head><body>` +
   109→    buildHeader(settings, 'Receipt') +
   110→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +
   111→    `<div class="datetime">${detail.status === 'CheckedOut' ? 'Status: Checked Out' : `Status: ${detail.status}`} &middot; ${fmtDateTime(now.toISOString())}</div>` +
   112→    `<hr class="divider">` +
   113→    `<table><tr><th>#</th><th>Date</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   114→    `<tr class="totals"><td colspan="3" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(finalBal)}</td></tr></table>` +
   115→    `<div class="tax-summary"><div class="tax-item"><span class="tax-label">Total Tax: </span><span class="tax-val">${fmtINR(totalTax)}</span></div><div class="tax-item"><span class="tax-label">Discounts: </span><span class="tax-val">${fmtINR(totalDiscounts)}</span></div><div class="tax-item"><span class="tax-label">Entries: </span><span class="tax-val">${activeEntries.length}</span></div></div>` +
   116→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Final Balance</div><div class="s-value" style="color:${finalBal >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(finalBal)}</div></div></div>` +
   117→    buildFooter() +
   118→    `</body></html>`
   119→  )
   120→  w.document.close()
   121→  w.print()
   122→}

   124→// ─── Print Invoice (Professional) ───
   125→export async function printInvoice(
   126→  detail: CustomerDetail,
   127→  invoiceData: { entries?: any[]; closingBalance?: number; openingBalance?: number },
   128→  invoiceFrom: string,
   129→  invoiceTo: string
   130→) {
   131→  if (!detail || !invoiceData) return
   132→  const entries = invoiceData.entries || []
   133→  if (entries.length === 0) return
   134→  const totalDebits = entries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   135→  const totalCredits = entries.reduce((s: number, e: any) => s + (e.credit || 0), 0)
   136→  const totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   137→  const totalDiscount = entries.reduce((s: number, e: any) => s + (e.discount || 0), 0)
   138→  const closingBal = invoiceData.closingBalance ?? 0
   139→  const openingBal = invoiceData.openingBalance ?? 0
   140→  const now = new Date()
   141→
   142→  // Fetch shop settings for professional header
   143→  let shopSettings: Record<string, string> = {}
   144→  try {
   145→    const res = await fetch('/api/settings')
   146→    if (res.ok) shopSettings = await res.json()
   147→  } catch { /* ignore */ }
   148→
   149→  // Generate invoice number if not available
   150→  let invNumber = ''
   151→  try {
   152→    const invRes = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, { method: 'POST' })
   153→    if (invRes.ok) { const invData = await invRes.json(); invNumber = invData.invoiceNumber || '' }
   154→  } catch { /* ignore */ }
   155→
   156→  const bizName = shopSettings.shopName || 'Eservices'
   157→  const bizAddress = shopSettings.shopAddress || ''
   158→  const bizPhone = shopSettings.shopPhone || ''
   159→  const bizEmail = shopSettings.shopEmail || ''
   160→  const gstNumber = shopSettings.gstNumber || shopSettings.gstin || ''
   161→  const authorizedSignatory = shopSettings.authorizedSignatory || ''
   162→
   163→  const rows = entries.map((e: any, i: number) => {
   164→    const desc = stripInternalDescriptions(e.description || e.type || '—')
   165→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td style="font-size:9px;color:#666;font-family:monospace">${e.docNo || ''}</td><td>${e.type || ''}</td><td>${desc}</td><td style="text-align:right;color:#dc2626;font-weight:600">${e.debit > 0 ? fmtINR(e.debit) : '—'}</td><td style="text-align:right;color:#16a34a;font-weight:600">${e.credit > 0 ? fmtINR(e.credit) : '—'}</td><td style="font-weight:700;color:${e.balance > 0 ? '#dc2626' : e.balance < 0 ? '#16a34a' : '#666'}">${fmtINR(e.balance)}</td></tr>`
   166→  }).join('')
   167→
   168→  const w = openPrintWindow(`Tax Invoice - ${detail.fullName}`)
   169→  if (!w) return
   170→
   171→  w.document.write(
   172→    `<!DOCTYPE html><html><head><title>Tax Invoice</title><style>${PRINT_CSS}</style></head><body>` +
   173→    // Shop Header
   174→    `<div class="bill-header"><h1>TAX INVOICE</h1>${bizAddress ? `<div class="biz-address">${bizAddress}</div>` : ''}${bizPhone ? `<div class="biz-address">${bizPhone}${bizEmail ? ` | ${bizEmail}` : ''}</div>` : ''}${gstNumber ? `<div class="gst-line">GSTIN: ${gstNumber}</div>` : ''}</div>` +
   175→    // Invoice Meta
   176→    `<div class="invoice-meta">
   177→      <div><div class="inv-label">Invoice No.</div><div class="inv-value">${invNumber || 'N/A'}</div></div>
   178→      <div style="text-align:center"><div class="inv-label">Date</div><div class="inv-value">${fmtDate(now.toISOString())}</div></div>
   179→      <div style="text-align:right"><div class="inv-label">Period</div><div class="inv-value" style="font-size:11px">${invoiceFrom || 'Beginning'} to ${invoiceTo || 'Today'}</div></div>
   180→    </div>` +
   181→    // Customer Info
   182→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">Customer ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +
   183→    (detail.gstNumber ? `<div style="margin:-8px 0 8px 0;font-size:11px;color:#666">Customer GSTIN: ${detail.gstNumber}</div>` : '') +
   184→    `<div class="datetime">Generated: ${fmtDateTime(now.toISOString())}</div>` +
   185→    `<hr class="divider">` +
   186→    // Table
   187→    `<table><tr><th>#</th><th>Date</th><th>TXN ID</th><th>Type</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   188→    `<tr class="totals"><td colspan="5" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(closingBal)}</td></tr></table>` +
   189→    // Tax Summary
   190→    `<div class="tax-summary"><div class="tax-item"><span class="tax-label">Total Tax: </span><span class="tax-val">${fmtINR(totalTax)}</span></div><div class="tax-item"><span class="tax-label">Discount: </span><span class="tax-val">${fmtINR(totalDiscount)}</span></div><div class="tax-item"><span class="tax-label">Net Charges: </span><span class="tax-val">${fmtINR(totalDebits - totalCredits)}</span></div></div>` +
   191→    // Summary Box
   192→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Opening Balance</div><div class="s-value">${fmtINR(openingBal)}</div></div><div class="summary-item"><div class="s-label">Total Charges</div><div class="s-value" style="color:#dc2626">${fmtINR(totalDebits)}</div></div><div class="summary-item"><div class="s-label">Total Payments</div><div class="s-value" style="color:#16a34a">${fmtINR(totalCredits)}</div></div><div class="summary-item"><div class="s-label">Closing Balance</div><div class="s-value" style="color:${closingBal >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(closingBal)}</div></div></div>` +
   193→    // Signature Section
   194→    `<div class="signature-section">
   195→      <div class="signature-box"><div class="signature-line"><div class="sig-label">Customer Signature</div><div class="sig-sub">Authorized Representative</div></div></div>
   196→      <div class="signature-box"><div class="signature-line"><div class="sig-label">${authorizedSignatory || 'Authorized Signatory'}</div><div class="sig-sub">${authorizedSignatory ? 'Authorized Signatory' : 'For ' + bizName}</div></div></div>
   197→    </div>` +
   198→    buildFooter() +
   199→    `</body></html>`
   200→  )
   201→  w.document.close()
   202→  w.print()
   203→}

   239→// ─── Print Voucher / Payment History ───

   257→
   258→  const w = openPrintWindow(`Voucher History - ${detail.fullName}`)
   259→  if (!w) return
   260→
   261→  w.document.write(
   262→    `<!DOCTYPE html><html><head><title>Voucher / Payment History</title><style>${PRINT_CSS}</style></head><body>` +

   269→    buildFooter() +
   270→    `</body></html>`
   271→  )
   272→  w.document.close()
   273→  w.print()
   274→}

   547→export async function printHistoryInvoice(
   548→  detail: CustomerDetail,
   549→  entries: any[],
   550→  openingBalance: number,
   551→  closingBalance: number,
   552→  invoiceFrom: string,
   553→  invoiceTo: string,
   554→) {
   555→  if (!detail || !entries?.length) return
   556→
   557→  const totalDebits = entries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   558→  const totalCredits = entries.reduce((s: number, e: any) => s + (e.credit || 0), 0)
   559→  const totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   560→  const totalDiscount = entries.reduce((s: number, e: any) => s + (e.discount || 0), 0)
   561→  const now = new Date()
   562→
   563→  let shopSettings: Record<string, string> = {}
   564→  try {
   565→    const res = await fetch('/api/settings')
   566→    if (res.ok) shopSettings = await res.json()
   567→  } catch { /* ignore */ }
   568→
   569→  let invNumber = ''
   570→  try {
   571→    const invRes = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, { method: 'POST' })
   572→    if (invRes.ok) { const invData = await invRes.json(); invNumber = invData.invoiceNumber || '' }
   573→  } catch { /* ignore */ }
   574→
   575→  const bizName = shopSettings.shopName || 'Eservices'
   576→  const bizAddress = shopSettings.shopAddress || ''
   577→  const bizPhone = shopSettings.shopPhone || ''
   578→  const bizEmail = shopSettings.shopEmail || ''
   579→  const gstNumber = shopSettings.gstNumber || shopSettings.gstin || ''
   580→  const authorizedSignatory = shopSettings.authorizedSignatory || ''
   581→
   582→  const rows = entries.map((e: any, i: number) => {
   583→    const desc = stripInternalDescriptions(e.description || e.type || '—')
   584→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td style="font-size:9px;color:#666;font-family:monospace">${e.docNo || ''}</td><td>${e.type || ''}</td><td>${desc}</td><td style="text-align:right;color:#dc2626;font-weight:600">${e.debit > 0 ? fmtINR(e.debit) : '—'}</td><td style="text-align:right;color:#16a34a;font-weight:600">${e.credit > 0 ? fmtINR(e.credit) : '—'}</td><td style="font-weight:700;color:${e.balance > 0 ? '#dc2626' : e.balance < 0 ? '#16a34a' : '#666'}">${fmtINR(e.balance)}</td></tr>`
   585→  }).join('')
   586→
   587→  const w = openPrintWindow(`Invoice - ${detail.fullName}`)
   588→  if (!w) return
   589→
   590→  w.document.write(
   591→    `<!DOCTYPE html><html><head><title>Tax Invoice</title><style>${PRINT_CSS}</style></head><body>` +
   592→    `<div class="bill-header"><h1>TAX INVOICE</h1>${bizAddress ? `<div class="biz-address">${bizAddress}</div>` : ''}${bizPhone ? `<div class="biz-address">${bizPhone}${bizEmail ? ` | ${bizEmail}` : ''}</div>` : ''}${gstNumber ? `<div class="gst-line">GSTIN: ${gstNumber}</div>` : ''}</div>` +
   593→    `<div class="invoice-meta">
   594→      <div><div class="inv-label">Invoice No.</div><div class="inv-value">${invNumber || 'N/A'}</div></div>
   595→      <div style="text-align:center"><div class="inv-label">Date</div><div class="inv-value">${fmtDate(now.toISOString())}</div></div>
   596→      <div style="text-align:right"><div class="inv-label">Period</div><div class="inv-value" style="font-size:11px">${invoiceFrom || 'Beginning'} to ${invoiceTo || 'Today'}</div></div>
   597→    </div>` +
   598→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">Customer ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +
   599→    (detail.gstNumber ? `<div style="margin:-8px 0 8px 0;font-size:11px;color:#666">Customer GSTIN: ${detail.gstNumber}</div>` : '') +
   600→    `<div class="datetime">Generated: ${fmtDateTime(now.toISOString())}</div>` +
   601→    `<hr class="divider">` +
   602→    `<table><tr><th>#</th><th>Date</th><th>TXN ID</th><th>Type</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   603→    `<tr class="totals"><td colspan="5" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(closingBalance)}</td></tr></table>` +
