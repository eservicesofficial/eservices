    10→// ─── Helper: Convert number to Indian Rupee words ───
    11→function numberToIndianWords(num: number): string {
    12→  if (num === 0) return 'ZERO'
    13→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    14→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    15→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    16→  let words = ''
    17→  function convertBelowThousand(n: number): string {
    18→    let w = ''
    19→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    20→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    21→    else if (n > 0) { w += ones[n] }
    22→    return w
    23→  }
    24→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    25→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    26→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    27→  if (num > 0) { words += convertBelowThousand(num) }
    28→  return words.trim().toUpperCase() + ' ONLY'
    29→}

   188→// ─── Helper: Compute the most-used payment mode from credit entries ───
   189→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   190→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   191→  if (creditEntries.length === 0) return 'N/A'
   192→  const modeCounts: Record<string, number> = {}
   193→  for (const e of creditEntries) {
   194→    const mode = e.paymentMode || 'Cash'
   195→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   196→  }
   197→  let bestMode = 'Cash'
   198→  let bestAmount = 0
   199→  for (const [mode, amount] of Object.entries(modeCounts)) {
   200→    if (amount > bestAmount) {
   201→      bestAmount = amount
   202→      bestMode = mode
   203→    }
   204→  }
   205→  return bestMode
   206→}

   208→// ─── Helper: Build payment summary HTML ───
   209→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   210→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   211→  if (creditEntries.length === 0) {
   212→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   213→  }
   214→
   215→  const modeTotals: Record<string, number> = {}
   216→  for (const e of creditEntries) {
   217→    const mode = e.paymentMode || 'Cash'
   218→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   219→  }
   220→
   221→  const modes = Object.entries(modeTotals)
   222→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   223→
   224→  if (modes.length === 1) {
   225→    const [mode, amount] = modes[0]
   226→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   227→  }
   228→
   229→  const rows = modes.map(([mode, amount]) => {
   230→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   231→  }).join('')
   232→
   233→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   234→}

   236→// ─── Helper: Get stable document number from entries (never regenerate) ───
   237→// Searches ALL entries for the first non-empty docNo to avoid N/A
   238→function getStableDocId(entries: { docNo?: string; invoiceNo?: string; id?: string; createdAt?: string }[]): string {
   239→  if (entries.length === 0) return `DOC-${Date.now()}`
   240→  for (const e of entries) {
   241→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   242→  }
   243→  for (const e of entries) {
   244→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   245→  }
   246→  // Fallback: use entry ID truncated or timestamp — never show N/A
   247→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   248→  return fallback
   249→}

   402→  const paymentEntries = entries.filter((e: any) => (e.credit || 0) > 0)
   403→  const refundEntries = entries.filter((e: any) => e.type === 'Refund')
   404→  const totalDebits = chargeEntries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   405→  const totalCredits = paymentEntries.reduce((s: number, e: any) => s + (e.credit || 0), 0)

   654→// ─── Print Folio Invoice (formerly "Receipt") ───
   655→// Uses stable doc IDs from existing entries — NEVER regenerates numbers on reprint
   656→export async function printReceipt(
   657→  detail: CustomerDetail,
   658→  entries: LedgerEntry[],
   659→  hdrRef: RefObject<Record<string, string>>
   660→) {
   661→  if (!detail) return
   662→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   663→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   664→  const shopSettings = await fetchSettingsForPrint(hdrRef)
   665→  const docId = getStableDocId(activeEntries)
   666→  const now = new Date()
   667→
   668→  const w = openPrintWindow(`Folio Invoice - ${detail.fullName}`)
   669→  if (!w) return
   670→
   671→  w.document.write(
   672→    `<!DOCTYPE html><html><head><title>Folio Invoice</title><style>${INVOICE_CSS}</style></head><body>` +
   673→    buildHotelStyleInvoice(shopSettings, detail, activeEntries as any[], docId, fmtDate(now.toISOString()), 'FOLIO INVOICE') +
   674→    `</body></html>`
   675→  )
   676→  w.document.close()
   677→  w.print()
   678→}

   701→      openingBalance: invoiceData.openingBalance ?? 0,
   702→      closingBalance: invoiceData.closingBalance ?? 0,

   796→interface ShopSettings {

   804→async function fetchShopSettings(hdrRef?: RefObject<Record<string, string>>): Promise<ShopSettings> {
   805→  try {
   806→    const headers = hdrRef?.current || undefined
   807→    const res = await fetch('/api/settings', headers ? { headers } : undefined)
   808→    if (res.ok) {
   809→      const db = (await res.json()) as Record<string, string>
   810→      return {
   811→        shopName: db.shopName || 'Eservices',
   812→        shopAddress: db.shopAddress || '',
   813→        shopPhone: db.shopPhone || '',
   814→        shopGst: db.gstNumber || db.gstin || '',
   815→        shopGstState: db.gstState || '',
   816→      }
   817→    }
   818→  } catch { /* ignore */ }
   819→  return { shopName: 'Eservices', shopAddress: '', shopPhone: '', shopGst: '', shopGstState: '' }
   820→}

   822→function openThermalPrintWindow(thermalHtml: string, title: string): Window | null {
   823→  const w = window.open('', '_blank', 'width=350,height=700')
   824→  if (!w) {
   825→    notify({ type: 'error', title: 'Error', message: 'Please allow pop-ups to print thermal receipts' })
   826→    return null
   827→  }
   828→  w.document.write(thermalHtml)
   829→  w.document.close()
   830→  setTimeout(() => {
   831→    try { if (!w.closed) w.print() } catch { /* window closed */ }
   832→  }, 600)
   833→  return w
   834→}

   836→// ─── Thermal Print Receipt ───
   837→export async function thermalPrintReceipt(
   838→  detail: CustomerDetail,
   839→  entries: LedgerEntry[],
   840→  hdrRef: RefObject<Record<string, string>>
   841→) {
   842→  if (!detail) return
   843→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   844→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   845→
   846→  const [tConfig, shop] = await Promise.all([
   847→    fetchThermalSettings(),
   848→    fetchShopSettings(hdrRef),
   849→  ])
   850→
   851→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   852→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   853→  const finalBal = activeEntries[activeEntries.length - 1].balance
   854→
   855→  const items: ThermalReceiptData['items'] = activeEntries.map(e => ({
   856→    name: stripInternalDescriptions(e.description || e.type || '—'),
   857→    qty: 1,
   858→    rate: e.debit > 0 ? e.debit : e.credit,
   859→    total: e.debit > 0 ? e.debit : e.credit,
   860→  }))
   861→
   862→  const actualPaymentMethod = computeMostUsedPaymentMode(activeEntries)
   863→
   864→  const receiptData: ThermalReceiptData = {
   865→    shopName: shop.shopName,
   866→    shopAddress: shop.shopAddress,
   867→    shopPhone: shop.shopPhone,
   868→    shopGst: shop.shopGst,
   869→    shopGstState: shop.shopGstState,
   870→    showGst: false,
   871→    invoiceNumber: detail.customerId,
   872→    date: fmtDate(new Date().toISOString()),
   873→    orderType: 'PAYMENT RECEIPT',
   874→    customerName: detail.fullName,
   875→    customerPhone: detail.mobile,
   876→    customerAddress: (detail as any).address || '',
   877→    items,
   878→    subtotal: totalDebits,
   879→    cgst: 0,
   880→    sgst: 0,
   881→    gstRate: 0,
   882→    deliveryCharge: 0,
   883→    discountAmount: 0,
   884→    grandTotal: totalDebits - totalCredits,
   885→    paidAmount: totalCredits,
   886→    balanceAmount: finalBal,
   887→    paymentMethod: actualPaymentMethod,
   888→    billStatus: finalBal <= 0 ? 'Paid' : 'Balance',
   889→  }
   890→
   891→  const thermalHtml = buildRichThermalHtml(receiptData, tConfig, `Receipt - ${detail.fullName}`)
   892→  openThermalPrintWindow(thermalHtml, `Receipt - ${detail.fullName}`)
   893→}
