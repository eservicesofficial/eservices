    12→    12→function numberToIndianWords(num: number): string {
    13→    13→  if (num === 0) return 'ZERO'
    14→    14→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    15→    15→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    16→    16→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    17→    17→  let words = ''
    18→    18→  function convertBelowThousand(n: number): string {
    19→    19→    let w = ''
    20→    20→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    21→    21→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    22→    22→    else if (n > 0) { w += ones[n] }
    23→    23→    return w
    24→    24→  }
    25→    25→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    26→    26→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    27→    27→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    28→    28→  if (num > 0) { words += convertBelowThousand(num) }
    29→    29→  return words.trim().toUpperCase() + ' ONLY'
    30→    30→}

   204→   204→// ─── Helper: Compute the most-used payment mode from credit entries ───
   205→   205→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   206→   206→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   207→   207→  if (creditEntries.length === 0) return 'N/A'
   208→   208→  const modeCounts: Record<string, number> = {}
   209→   209→  for (const e of creditEntries) {
   210→   210→    const mode = e.paymentMode || 'Cash'
   211→   211→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   212→   212→  }
   213→   213→  let bestMode = 'Cash'
   214→   214→  let bestAmount = 0
   215→   215→  for (const [mode, amount] of Object.entries(modeCounts)) {
   216→   216→    if (amount > bestAmount) {
   217→   217→      bestAmount = amount
   218→   218→      bestMode = mode
   219→   219→    }
   220→   220→  }
   221→   221→  return bestMode
   222→   222→}

   224→   224→// ─── Helper: Build payment summary HTML ───
   225→   225→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   226→   226→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   227→   227→  if (creditEntries.length === 0) {
   228→   228→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   229→   229→  }
   230→   230→
   231→   231→  const modeTotals: Record<string, number> = {}
   232→   232→  for (const e of creditEntries) {
   233→   233→    const mode = e.paymentMode || 'Cash'
   234→   234→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   235→   235→  }
   236→   236→
   237→   237→  const modes = Object.entries(modeTotals)
   238→   238→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   239→   239→
   240→   240→  if (modes.length === 1) {
   241→   241→    const [mode, amount] = modes[0]
   242→   242→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   243→   243→  }
   244→   244→
   245→   245→  const rows = modes.map(([mode, amount]) => {
   246→   246→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   247→   247→  }).join('')
   248→   248→
