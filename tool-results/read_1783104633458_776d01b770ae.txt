   157→      if (debounced) params.set('search', debounced)
   158→      if (sort) params.set('sort', sort === 'name' ? 'name-asc' : sort)

   167→    } catch { /* ignore */ } finally { setLoading(false) }
   168→  }, [debounced, filter, sort])

   209→        const ledgerEntries: LedgerEntry[] = activeFolio?.ledgerEntries || []
   210→        const balance = lifetimeStats.currentBalance ?? (ledgerEntries.length > 0 ? ledgerEntries[ledgerEntries.length - 1].balance : 0)
   211→        setDetail({
   212→          ...customer,
   213→          openingBalance: customer.openingBalance ?? 0,
   214→          creditLimit: customer.creditLimit ?? 0,
   215→          balance: balance ?? 0,
   216→          totalVisits: d.visits?.length ?? lifetimeStats.totalVisits ?? 0,
   217→          lastTxnDate: lifetimeStats.lastTransactionDate || '',
   218→          tags: tagAssignments, folios: d.folios || [], ledgerEntries,
   219→          documents: d.documents || [], notes: d.notes || [], visits: d.visits || [],
   220→          settlements: d.settlements || [], auditLogs: [],
   221→          lifetimeRevenue: lifetimeStats.totalCharges ?? 0,
   222→          lifetimePayments: lifetimeStats.totalPayments ?? 0,
   223→          lifetimeRefunds: lifetimeStats.totalRefunds ?? 0,

   227→        })

   229→        setAllFolios(d.folios || [])
   230→        if (activeFolio) { setSelectedFolioId(activeFolio.id); setFolioEntries(ledgerEntries) }
   231→        else if (d.folios?.length > 0) { setSelectedFolioId(d.folios[0].id); fetchFolioEntries(id, d.folios[0].id) }
   232→        else { setSelectedFolioId('master'); fetchMasterFolioEntries(id) }
   233→        setSelectedEntries(new Set())
   234→      }
   235→    } catch { /* ignore */ } finally { setDetailLoading(false) }

   304→  const handleCreate = async () => {
   305→    if (!form.fullName.trim() || !form.mobile.trim()) { notify({ type: 'error', title: 'Error', message: 'Name and Mobile required' }); return }

   307→      const res = await fetch('/api/folio/customers', { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(form) })

   309→      else { const e = await safeJson(res); if (e.duplicate) { setDuplicate(e.duplicate); notify({ type: 'warning', title: 'Duplicate' }) } else notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' }) }
   310→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed' }) } finally { setSubmitting(false) }
   311→  }

   318→      else { const e = await safeJson(res); notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' }) }
   319→    } catch { /* ignore */ } finally { setSubmitting(false) }
   320→  }

   328→    } catch { /* ignore */ } finally { setSubmitting(false) }

   330→

   333→  const handleRefund = () => { if (!refundForm.amount || !refundForm.description) { notify({ type: 'error', title: 'Error', message: 'Amount and reason required' }); return }; addLedgerEntry('Refund', { description: `${refundForm.reason}: ${refundForm.description}`, amount: parseFloat(refundForm.amount), invoiceNo: refundForm.invoiceNo }) }
   334→  const handleDiscount = () => { if (!discountForm.amount) { notify({ type: 'error', title: 'Error', message: 'Amount required' }); return }; addLedgerEntry('Discount', { description: discountForm.description || 'Discount applied', amount: parseFloat(discountForm.amount), discountType: discountForm.discountType }) }
   335→
   336→  const handleSettle = async () => {
   337→    if (!detail || !paymentForm.amount) return
   338→    try { setSubmitting(true)
   339→      const settleAmount = parseFloat(paymentForm.amount)
   340→      // When master folio is selected, use autoAllocate to distribute across all folios
   341→      const isMaster = selectedFolioId === 'master'
   342→      const payload = isMaster
   343→        ? { amount: settleAmount, mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   344→        : { amount: settleAmount, mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   345→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(payload) })
   346→      if (res.ok) {
   347→        notify({ type: 'success', title: 'Settlement Complete' })
   348→        closeDialogs()
   349→        if (!isMaster && selectedFolioId) fetchFolioEntries(detail.id, selectedFolioId)
   350→        fetchMasterFolioEntries(detail.id)
   351→        fetchDetail(detail.id, true)
   352→        fetchCustomers(true)
   353→        onRefreshData()
