    10→// ─── Helper: Convert number to Indian Rupee words ───
    11→function numberToIndianWords(num: number): string {
    12→  if (num === 0) return 'ZERO'
    13→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    14→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    15→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    16→  let words = ''
    17→  function convertBelowThousand(n: number): string {
    18→    let w = ''
    19→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    20→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    21→    else if (n > 0) { w += ones[n] }
    22→    return w
    23→  }
    24→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    25→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    26→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    27→  if (num > 0) { words += convertBelowThousand(num) }
    28→  return words.trim().toUpperCase() + ' ONLY'
    29→}

   193→// ─── Helper: Compute the most-used payment mode from credit entries ───
   194→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   195→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   196→  if (creditEntries.length === 0) return 'N/A'
   197→  const modeCounts: Record<string, number> = {}
   198→  for (const e of creditEntries) {
   199→    const mode = e.paymentMode || 'Cash'
   200→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   201→  }
   202→  let bestMode = 'Cash'
   203→  let bestAmount = 0
   204→  for (const [mode, amount] of Object.entries(modeCounts)) {
   205→    if (amount > bestAmount) {
   206→      bestAmount = amount
   207→      bestMode = mode
   208→    }
   209→  }
   210→  return bestMode
   211→}

   213→// ─── Helper: Build payment summary HTML ───
   214→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   215→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   216→  if (creditEntries.length === 0) {
   217→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   218→  }
   219→
   220→  const modeTotals: Record<string, number> = {}
   221→  for (const e of creditEntries) {
   222→    const mode = e.paymentMode || 'Cash'
   223→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   224→  }
   225→
   226→  const modes = Object.entries(modeTotals)
   227→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   228→
   229→  if (modes.length === 1) {
   230→    const [mode, amount] = modes[0]
   231→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   232→  }
   233→
   234→  const rows = modes.map(([mode, amount]) => {
   235→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   236→  }).join('')
   237→
   238→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   239→}

   241→// ─── Helper: Get stable document number from entries (never regenerate) ───
   242→// Searches ALL entries for the first non-empty docNo to avoid N/A
   243→function getStableDocId(entries: { docNo?: string; invoiceNo?: string; id?: string; createdAt?: string }[]): string {
   244→  if (entries.length === 0) return `DOC-${Date.now()}`
   245→  for (const e of entries) {
   246→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   247→  }
   248→  for (const e of entries) {
   249→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   250→  }
   251→  // Fallback: use entry ID truncated or timestamp — never show N/A
   252→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   253→  return fallback
   254→}

   660→// ─── Print Folio Invoice (formerly "Receipt") ───
   661→// Uses stable doc IDs from existing entries — NEVER regenerates numbers on reprint
   662→export async function printReceipt(
   663→  detail: CustomerDetail,
   664→  entries: LedgerEntry[],
   665→  hdrRef: RefObject<Record<string, string>>,
   666→  printedBy?: string
   667→) {
   668→  if (!detail) return
   669→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   670→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   671→  const shopSettings = await fetchSettingsForPrint(hdrRef)
   672→  const docId = getStableDocId(activeEntries)
   673→  const now = new Date()
   674→
   675→  const w = openPrintWindow(`Folio Invoice - ${detail.fullName}`)
   676→  if (!w) return
   677→
   678→  w.document.write(
   679→    `<!DOCTYPE html><html><head><title>Folio Invoice</title><style>${INVOICE_CSS}</style></head><body>` +
   680→    buildHotelStyleInvoice(shopSettings, detail, activeEntries as any[], docId, fmtDate(now.toISOString()), 'FOLIO INVOICE', undefined, printedBy) +
   681→    `</body></html>`

   804→interface ShopSettings {
   805→  shopName: string
   806→  shopAddress: string
   807→  shopPhone: string

   812→async function fetchShopSettings(hdrRef?: RefObject<Record<string, string>>): Promise<ShopSettings> {
   813→  try {
   814→    const headers = hdrRef?.current || undefined
   815→    const res = await fetch('/api/settings', headers ? { headers } : undefined)
   816→    if (res.ok) {
   817→      const db = (await res.json()) as Record<string, string>
   818→      return {
   819→        shopName: db.shopName || 'Eservices',
   820→        shopAddress: db.shopAddress || '',
   821→        shopPhone: db.shopPhone || '',
   822→        shopGst: db.gstNumber || db.gstin || '',
   823→        shopGstState: db.gstState || '',
   824→      }
   825→    }
   826→  } catch { /* ignore */ }
   827→  return { shopName: 'Eservices', shopAddress: '', shopPhone: '', shopGst: '', shopGstState: '' }
   828→}

   830→function openThermalPrintWindow(thermalHtml: string, title: string): Window | null {
   831→  const w = window.open('', '_blank', 'width=350,height=700')
   832→  if (!w) {
   833→    notify({ type: 'error', title: 'Error', message: 'Please allow pop-ups to print thermal receipts' })
   834→    return null
   835→  }
   836→  w.document.write(thermalHtml)
   837→  w.document.close()
   838→  setTimeout(() => {
   839→    try { if (!w.closed) w.print() } catch { /* window closed */ }
   840→  }, 600)
   841→  return w
   842→}

   846→  detail: CustomerDetail,
   847→  entries: LedgerEntry[],
   848→  hdrRef: RefObject<Record<string, string>>
   849→) {
   850→  if (!detail) return
   851→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   852→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }

   854→  const [tConfig, shop] = await Promise.all([

   858→

   860→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   861→  const finalBal = activeEntries[activeEntries.length - 1].balance
   862→
   863→  const items: ThermalReceiptData['items'] = activeEntries.map(e => ({
   864→    name: stripInternalDescriptions(e.description || e.type || '—'),
   865→    qty: 1,
   866→    rate: e.debit > 0 ? e.debit : e.credit,
   867→    total: e.debit > 0 ? e.debit : e.credit,
   868→  }))
   869→
   870→  const actualPaymentMethod = computeMostUsedPaymentMode(activeEntries)
   871→
   872→  const receiptData: ThermalReceiptData = {
   873→    shopName: shop.shopName,
   874→    shopAddress: shop.shopAddress,
   875→    shopPhone: shop.shopPhone,
   876→    shopGst: shop.shopGst,
   877→    shopGstState: shop.shopGstState,
   878→    showGst: false,
   879→    invoiceNumber: detail.customerId,
   880→    date: fmtDate(new Date().toISOString()),
   881→    orderType: 'PAYMENT RECEIPT',
   882→    customerName: detail.fullName,
   883→    customerPhone: detail.mobile,
   884→    customerAddress: (detail as any).address || '',
   885→    items,
   886→    subtotal: totalDebits,
   887→    cgst: 0,
   888→    sgst: 0,
   889→    gstRate: 0,
   890→    deliveryCharge: 0,
   891→    discountAmount: 0,
   892→    grandTotal: totalDebits - totalCredits,
   893→    paidAmount: totalCredits,
   894→    balanceAmount: finalBal,
   895→    paymentMethod: actualPaymentMethod,
   896→    billStatus: finalBal <= 0 ? 'Paid' : 'Balance',
   897→  }
   898→
   899→  const thermalHtml = buildRichThermalHtml(receiptData, tConfig, `Receipt - ${detail.fullName}`)
   900→  openThermalPrintWindow(thermalHtml, `Receipt - ${detail.fullName}`)
   901→}
