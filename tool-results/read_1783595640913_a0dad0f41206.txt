   174→  splitAmountOpen, shiftOpen, removeOpen, voucherOpen, invoiceOpen,

   176→  form, chargeForm, paymentForm, refundForm, transferForm, discountForm,

   183→  setForm, setChargeForm, setPaymentForm, setRefundForm, setTransferForm,

   190→  fetchInvoiceData, fetchFolioStatement, printInvoice, printCompleteInvoice, printSingleFolio, printVoucherHistory,
   191→  closeDialogs, searchTransferTargets,
   192→  deleteFolioOpen, folioToDelete,
   193→  setDeleteFolioOpen, handleDeleteFolio,
   194→  // Enhanced props
   195→  apiHeaders, onRefresh, canDelete = false, isAdmin = false, onPrintReceipt, onThermalPrintReceipt,
   196→  onThermalPrintInvoice, onThermalPrintVoucher,
   197→  lastCreatedDocNo, lastCreatedType,
   198→  // History
   199→  historyOpen, setHistoryOpen, onPrintHistoryFolio,
   200→}: DialogsProps) {
   201→
   202→  // ─── Internal State ───
   203→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   204→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   205→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   206→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   207→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   208→  // ─── History state ───
   209→  const [historyLoading, setHistoryLoading] = React.useState(false)
   210→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   211→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   212→  const [selectedVoucherIds, setSelectedVoucherIds] = React.useState<Set<string>>(new Set())
   213→  const [historyDate, setHistoryDate] = React.useState(() => {
   214→    const today = new Date()
   215→    return today.toISOString().split('T')[0] // "2025-01-15"
   216→  })
   217→
   218→  // Fetch history for the CURRENT customer
   219→  const fetchCustomerHistory = React.useCallback(async (date?: string) => {
   220→    if (!detail || !apiHeaders) return
   221→    try {
   222→      setHistoryLoading(true)
   223→      const q = date ? `?date=${date}` : ''
   224→      const res = await fetch(`/api/folio/customers/${detail.id}/history${q}`, { headers: apiHeaders })
   225→      if (res.ok) {
   226→        const data = await res.json() as CustomerHistory
   227→        setCustomerHistory(data)
   228→      }
   229→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to fetch checkout history' }) } finally { setHistoryLoading(false) }
   230→  }, [detail?.id, apiHeaders])
   231→
   232→  React.useEffect(() => {
   233→    if (historyOpen && detail && apiHeaders && fetchCustomerHistory) {
   234→      fetchCustomerHistory(historyDate)
   235→    } else if (!historyOpen) {
   236→      setCustomerHistory(null)
   237→      setExpandedFolioId(null)
   238→    }
   239→  }, [historyOpen, detail?.id, fetchCustomerHistory, apiHeaders, historyDate])
   240→
   241→  // ─── Print helpers for checkout records ───
   242→  const printCheckoutFolioBill = async (co: CheckoutRecord) => {
   243→    if (!co.transactions.length) return
   244→    let shopSettings: Record<string, string> = {}
   245→    try {
   246→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   247→    } catch { /* use defaults */ }
   248→    // Build a synthetic CustomerDetail-like object from CheckoutRecord
   249→    const fakeDetail = {
   250→      id: co.customerIdRef,
   251→      fullName: co.customerName,
   252→      customerId: co.customerIdRef,
   253→      mobile: co.customerMobile || '',
   254→      email: co.customerEmail || '',
   255→      address: co.customerAddress || '',
   256→      customerType: co.customerType || '',
   257→      companyName: '',
   258→      gstNumber: '',
   259→      panNumber: '',
   260→      folios: [],
   261→      activeFolio: null,
   262→      totalVisits: co.visitNumber || 0,
   263→      ledgerEntries: [],
   264→    } as unknown as CustomerDetail
   265→    const w = window.open('', '_blank')
   266→    if (!w) return
   267→    w.document.write(`<!DOCTYPE html><html><head><title>Folio Bill - ${co.checkoutNumber}</title><style>${INVOICE_CSS}</style></head><body>${buildHotelStyleInvoice(shopSettings, fakeDetail, co.transactions, co.invoiceNumber, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'FOLIO BILL', undefined, co.checkoutUser)}</body></html>`)
   268→    w.document.close(); w.print()
   269→  }
   270→
   271→  const printCheckoutInvoice = async (co: CheckoutRecord) => {
   272→    if (!co.transactions.length) return
   273→    let shopSettings: Record<string, string> = {}
   274→    try {
   275→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   276→    } catch { /* use defaults */ }
   277→    // Build a synthetic CustomerDetail-like object from CheckoutRecord
   278→    const fakeDetail = {
   279→      id: co.customerIdRef,
   280→      fullName: co.customerName,
   281→      customerId: co.customerIdRef,
   282→      mobile: co.customerMobile || '',
   283→      email: co.customerEmail || '',
   284→      address: co.customerAddress || '',
   285→      customerType: co.customerType || '',
   286→      companyName: '',
   287→      gstNumber: '',
   288→      panNumber: '',
   289→      folios: [],
   290→      activeFolio: null,
   291→      totalVisits: co.visitNumber || 0,
   292→      ledgerEntries: [],
   293→    } as unknown as CustomerDetail
   294→    const w = window.open('', '_blank')
   295→    if (!w) return
   296→    w.document.write(`<!DOCTYPE html><html><head><title>Final Checkout Invoice - ${co.invoiceNumber}</title><style>${INVOICE_CSS}</style></head><body>${buildHotelStyleInvoice(shopSettings, fakeDetail, co.transactions, co.invoiceNumber, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'FINAL CHECKOUT INVOICE', { openingBalance: co.openingBalance, closingBalance: co.closingBalance }, co.checkoutUser)}</body></html>`)
   297→    w.document.close(); w.print()
   298→  }
   299→
   300→  // ─── Track checkout transition ───
   301→  const wasCheckedOut = React.useRef(isCheckedOut)
   302→  React.useEffect(() => {
   303→    if (isCheckedOut && !wasCheckedOut.current) {
   304→      setPostCheckoutOpen(true)
   305→    }
   306→    wasCheckedOut.current = isCheckedOut
   307→  }, [isCheckedOut])
   308→
   309→  // ─── Show TXN number when lastCreatedDocNo changes ───
   310→  React.useEffect(() => {
   311→    if (lastCreatedDocNo && lastCreatedType) {
   312→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   313→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   314→      return () => clearTimeout(timer)
   315→    }
   316→  }, [lastCreatedDocNo, lastCreatedType])
   317→
   318→  // ─── Items for charge selector (Req 5) ───
   319→  const [items, setItems] = React.useState<{ id: string; name: string; price: number }[]>([])
   320→  const [itemsLoading, setItemsLoading] = React.useState(false)
   321→  React.useEffect(() => {
   322→    if (chargeOpen && apiHeaders) {
   323→      setItemsLoading(true)
   324→      fetch('/api/items', { headers: apiHeaders }).then(r => r.ok ? r.json() : []).then((list: any[]) => {
   325→        setItems((list || []).map((it: any) => ({ id: it.id, name: it.name, price: it.price || 0 })))
   326→      }).catch(() => {}).finally(() => setItemsLoading(false))
   327→    }
   328→  }, [chargeOpen, apiHeaders])
   329→
   330→  // Handle item selection
   331→  const handleItemSelect = React.useCallback((itemId: string) => {
   332→    const item = items.find(it => it.id === itemId)
   333→    if (item) {
   334→      setChargeForm({
   335→        ...chargeForm,
   336→        selectedItem: itemId,
   337→        selectedItemPrice: String(item.price),
   338→        description: item.name,
   339→        amount: item.price > 0 ? String(item.price) : chargeForm.amount,
   340→      })
   341→    }
   342→  }, [items, chargeForm])
   343→
   344→  // ─── Fetch invoice number ───
   345→  const fetchInvoiceNumber = React.useCallback(async () => {
   346→    if (!detail || !apiHeaders) return
   347→    try {
   348→      setInvNumLoading(true)
   349→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   350→        method: 'POST',
   351→        headers: apiHeaders,
   352→      })
   353→      if (res.ok) {
   354→        const data = await res.json()
   355→        setInvoiceNumber(data.invoiceNumber)
   356→      }
   357→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   358→  }, [detail?.id, apiHeaders])
   359→
   360→  // ─── Fetch invoice number when invoice dialog opens ───
   361→  React.useEffect(() => {
   362→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber && fetchInvoiceNumber) {
   363→      fetchInvoiceNumber()
   364→    }
   365→  }, [invoiceOpen, detail?.id, apiHeaders, invoiceNumber, fetchInvoiceNumber])
   366→
   367→  React.useEffect(() => {
   368→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   369→      fetchInvoiceData()
   370→    }
   371→  }, [invoiceOpen, invoiceData, detail?.id, selectedFolioId])
   372→
   373→  // ─── Entry creation with TXN tracking ───
   374→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   375→    if (!detail || !selectedFolioId || !apiHeaders) {
   376→      return null
   377→    }
   378→    try {
   379→      setLocalSubmitting(true)
   380→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   381→        method: 'POST',
   382→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   383→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   384→      })
   385→      if (res.ok) {
   386→        const entry = await res.json()
   387→        const docNo = entry?.docNo || ''
   388→        setEntrySuccess({ show: true, docNo, type })
   389→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   390→        closeDialogs()
   391→        onRefresh?.()
   392→        setTimeout(() => setEntrySuccess(null), 4000)
   393→        return entry
   394→      } else {
   395→        const e = await safeJson(res)
   396→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   397→        return null
   398→      }
   399→    } catch {
   400→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   401→      return null
   402→    } finally {
   403→      setLocalSubmitting(false)
   404→    }
   405→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   406→
   407→  // ─── Settlement with TXN tracking ───
   408→  const settleWithTxn = React.useCallback(async () => {
   409→    if (!detail || !paymentForm.amount || !apiHeaders) {
   410→      handleSettle()
   411→      return
   412→    }
   413→    try {
   414→      setLocalSubmitting(true)
   415→      // When master folio is selected, use autoAllocate to distribute across all folios
   416→      const isMaster = selectedFolioId === 'master'
   417→      const payload = isMaster
   418→        ? { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   419→        : { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   420→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   421→        method: 'POST',
   422→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   423→        body: JSON.stringify(payload),
   424→      })
   425→      if (res.ok) {
   426→        const data = await res.json()
   427→        // Auto-allocate may return settledFolios array instead of single ledgerEntry
   428→        const docNo = data?.ledgerEntry?.docNo || (data?.settledFolios?.[0] ? 'Multi-Folio' : '')
   429→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })
   430→        notify({ type: 'success', title: `Settlement ${docNo}`, message: data.message || `Settlement ${docNo} processed successfully` })
