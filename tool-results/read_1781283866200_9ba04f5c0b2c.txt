   166→   166→
   167→   167→  // ─── Internal State ───
   168→   168→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   169→   169→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   170→   170→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   171→   171→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   172→   172→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   173→   173→  const [splitMode, setSplitMode] = React.useState(false)
   174→   174→  const [localSplitItems, setLocalSplitItems] = React.useState<{ method: string; amount: string; reference: string }[]>([

   176→   176→  ])
   177→   177→
   178→   178→  // ─── History state ───
   179→   179→  const [historyLoading, setHistoryLoading] = React.useState(false)
   180→   180→  const [checkedOutCustomers, setCheckedOutCustomers] = React.useState<CustomerHistory[]>([])
   181→   181→  const [expandedFolioCustomer, setExpandedFolioCustomer] = React.useState<string | null>(null)
   182→   182→  const [expandedFolioEntries, setExpandedFolioEntries] = React.useState<Record<string, any[]>>({})
   183→   183→
   184→   184→  // Fetch checked-out customers for history
   185→   185→  React.useEffect(() => {
   186→   186→    if (historyOpen) {
   187→   187→      fetchCheckedOutHistory()
   188→   188→    }
   189→   189→  }, [historyOpen])
   190→   190→
   191→   191→  const fetchCheckedOutHistory = React.useCallback(async () => {
   192→   192→    try {
   193→   193→      setHistoryLoading(true)
   194→   194→      const res = await fetch('/api/folio/customers?status=CheckedOut&limit=100')
   195→   195→      if (res.ok) {
   196→   196→        const data = await res.json()
   197→   197→        const list = Array.isArray(data) ? data : (data.customers || [])
   198→   198→        // Fetch history for each customer
   199→   199→        const histories = await Promise.all(
   200→   200→          list.slice(0, 20).map(async (c: any) => {
   201→   201→            try {
   202→   202→              const hRes = await fetch(`/api/folio/customers/${c.id}/history`)
   203→   203→              if (hRes.ok) return await hRes.json() as CustomerHistory
   204→   204→            } catch { /* ignore */ }
   205→   205→            return null
   206→   206→          })
   207→   207→        )
   208→   208→        setCheckedOutCustomers(histories.filter((h): h is CustomerHistory => h !== null))
   209→   209→      }
   210→   210→    } catch { /* ignore */ } finally { setHistoryLoading(false) }
   211→   211→  }, [])
   212→   212→
   213→   213→  const toggleFolioEntries = (customerId: string, folioId: string) => {
   214→   214→    const key = `${customerId}-${folioId}`
   215→   215→    if (expandedFolioEntries[key]) {
   216→   216→      setExpandedFolioEntries(prev => { const n = { ...prev }; delete n[key]; return n })
   217→   217→    } else {
   218→   218→      const customer = checkedOutCustomers.find(c => c.customer.id === customerId)
   219→   219→      const folio = customer?.historicalFolios.find(f => f.folioId === folioId)
   220→   220→      if (folio) {
   221→   221→        setExpandedFolioEntries(prev => ({ ...prev, [key]: (folio.entries || folio.ledgerEntries || []) }))
   222→   222→      }
   223→   223→    }
   224→   224→  }
   225→   225→
   226→   226→  // ─── Track checkout transition ───
   227→   227→  const wasCheckedOut = React.useRef(isCheckedOut)
   228→   228→  React.useEffect(() => {
   229→   229→    if (isCheckedOut && !wasCheckedOut.current) {
   230→   230→      setPostCheckoutOpen(true)
   231→   231→    }
   232→   232→    wasCheckedOut.current = isCheckedOut
   233→   233→  }, [isCheckedOut])
   234→   234→
   235→   235→  // ─── Show TXN number when lastCreatedDocNo changes ───
   236→   236→  React.useEffect(() => {
   237→   237→    if (lastCreatedDocNo && lastCreatedType) {
   238→   238→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   239→   239→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   240→   240→      return () => clearTimeout(timer)
   241→   241→    }
   242→   242→  }, [lastCreatedDocNo, lastCreatedType])
   243→   243→
   244→   244→  // ─── Fetch invoice number ───
   245→   245→  const fetchInvoiceNumber = React.useCallback(async () => {
   246→   246→    if (!detail || !apiHeaders) return
   247→   247→    try {
   248→   248→      setInvNumLoading(true)
   249→   249→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   250→   250→        method: 'POST',
   251→   251→        headers: apiHeaders,
   252→   252→      })
   253→   253→      if (res.ok) {
   254→   254→        const data = await res.json()
   255→   255→        setInvoiceNumber(data.invoiceNumber)
   256→   256→      }
   257→   257→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   258→   258→  }, [detail?.id, apiHeaders])
   259→   259→
   260→   260→  // ─── Fetch invoice number when invoice dialog opens ───
   261→   261→  React.useEffect(() => {
   262→   262→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber) {
   263→   263→      fetchInvoiceNumber()
   264→   264→    }
   265→   265→  }, [invoiceOpen, detail?.id, apiHeaders])
   266→   266→
   267→   267→  React.useEffect(() => {
   268→   268→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   269→   269→      fetchInvoiceData()
   270→   270→    }
   271→   271→  }, [invoiceOpen])
   272→   272→
   273→   273→  // ─── Entry creation with TXN tracking ───
   274→   274→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   275→   275→    if (!detail || !selectedFolioId || !apiHeaders) {
   276→   276→      return null
   277→   277→    }
   278→   278→    try {
   279→   279→      setLocalSubmitting(true)
   280→   280→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   281→   281→        method: 'POST',
   282→   282→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },

   298→   298→      }
   299→   299→    } catch {
   300→   300→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   301→   301→      return null
   302→   302→    } finally {
   303→   303→      setLocalSubmitting(false)

   319→   319→          amount: parseFloat(paymentForm.amount),
   320→   320→          mode: paymentForm.mode,
   321→   321→          reference: paymentForm.reference,
   322→   322→          description: paymentForm.description,

   331→   331→        closeDialogs()
   332→   332→        onRefresh?.()
   333→   333→        setTimeout(() => setEntrySuccess(null), 4000)
   334→   334→      } else {
   335→   335→        const e = await res.json()
   336→   336→        notify({ type: 'error', title: 'Error', message: e.error || 'Failed' })
   337→   337→      }
   338→   338→    } catch {
   339→   339→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   340→   340→    } finally {
   341→   341→      setLocalSubmitting(false)
   342→   342→    }
   343→   343→  }, [detail?.id, selectedFolioId, apiHeaders, paymentForm, handleSettle, closeDialogs, onRefresh])
   344→   344→
   345→   345→  // ─── Transfer with TXN tracking ───

   348→   348→      handleTransfer()
   349→   349→      return
   350→   350→    }

   353→   353→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   354→   354→        method: 'POST',
   355→   355→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   356→   356→        body: JSON.stringify({
   357→   357→          targetCustomerId: transferForm.targetCustomerId,
   358→   358→          amount: parseFloat(transferForm.amount),
   359→   359→          description: transferForm.description,
   360→   360→        }),
   361→   361→      })
   362→   362→      if (res.ok) {
   363→   363→        const data = await res.json()
   364→   364→        const docNo = data?.source?.entry?.docNo || ''
   365→   365→        setEntrySuccess({ show: true, docNo, type: 'Transfer' })
   366→   366→        notify({ type: 'success', title: `Transfer ${docNo}`, message: `Transfer ${docNo} processed successfully` })
   367→   367→        closeDialogs()
   368→   368→        onRefresh?.()
   369→   369→        setTimeout(() => setEntrySuccess(null), 4000)
   370→   370→      } else {
   371→   371→        const e = await res.json()
   372→   372→        notify({ type: 'error', title: 'Error', message: e.error || 'Failed' })
   373→   373→      }
   374→   374→    } catch {
   375→   375→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   376→   376→    } finally {
   377→   377→      setLocalSubmitting(false)
   378→   378→    }
   379→   379→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   380→   380→
   381→   381→  // ─── Split payment helpers for checkout ───
   382→   382→  const addSplitItem = () => {
   383→   383→    setLocalSplitItems(prev => [...prev, { method: 'Cash', amount: '', reference: '' }])
   384→   384→  }
   385→   385→  const removeSplitItem = (idx: number) => {
   386→   386→    setLocalSplitItems(prev => prev.filter((_, i) => i !== idx))
   387→   387→  }
   388→   388→  const updateSplitItem = (idx: number, field: string, value: string) => {
   389→   389→    setLocalSplitItems(prev => prev.map((item, i) => i === idx ? { ...item, [field]: value } : item))
   390→   390→  }
   391→   391→  const totalSplitAmount = localSplitItems.reduce((sum, item) => sum + (parseFloat(item.amount) || 0), 0)
   392→   392→  const splitRemaining = checkoutSettlementData ? Math.max(0, checkoutSettlementData.totalOutstanding - totalSplitAmount) : 0
   393→   393→
   394→   394→  // ─── Computed GST data from invoiceData ───
   395→   395→  const gstData = React.useMemo(() => {
   396→   396→    if (!invoiceData?.entries) return null
   397→   397→    const entries = invoiceData.entries as any[]
   398→   398→    const taxableEntries = entries.filter((e: any) => e.debit > 0)
