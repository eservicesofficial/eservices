   187→   187→  // ─── History state ───
   188→   188→  const [historyLoading, setHistoryLoading] = React.useState(false)
   189→   189→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   190→   190→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)

   193→   193→
   194→   194→  // Fetch history for the CURRENT customer (not all checked-out customers)
   195→   195→  React.useEffect(() => {
   196→   196→    if (historyOpen && detail && apiHeaders) {
   197→   197→      fetchCustomerHistory()
   198→   198→    } else if (!historyOpen) {
   199→   199→      setCustomerHistory(null)
   200→   200→      setExpandedFolioId(null)
   201→   201→      setExpandedFolioEntries({})
   202→   202→    }
   203→   203→  }, [historyOpen, detail?.id])
   204→   204→
   205→   205→  const fetchCustomerHistory = React.useCallback(async () => {
   206→   206→    if (!detail || !apiHeaders) return
   207→   207→    try {
   208→   208→      setHistoryLoading(true)
   209→   209→      const res = await fetch(`/api/folio/customers/${detail.id}/history`, { headers: apiHeaders })
   210→   210→      if (res.ok) {
   211→   211→        const data = await res.json() as CustomerHistory
   212→   212→        setCustomerHistory(data)
   213→   213→      }
   214→   214→    } catch { /* ignore */ } finally { setHistoryLoading(false) }
   215→   215→  }, [detail?.id, apiHeaders])
   216→   216→
   217→   217→  const toggleFolioEntries = (folioId: string) => {
   218→   218→    if (expandedFolioId === folioId) {
   219→   219→      setExpandedFolioId(null)

   221→   221→      const folio = customerHistory?.historicalFolios.find(f => f.folioId === folioId)
   222→   222→      if (folio) {
   223→   223→        setExpandedFolioId(folioId)
   224→   224→        setExpandedFolioEntries(prev => ({
   225→   225→          ...prev,
   226→   226→          [folioId]: (folio.entries || []),
   227→   227→        }))
   228→   228→      }
   229→   229→    }
   230→   230→  }
   231→   231→
   232→   232→  // ─── Track checkout transition ───

   234→   234→  React.useEffect(() => {
   235→   235→    if (isCheckedOut && !wasCheckedOut.current) {
   236→   236→      setPostCheckoutOpen(true)
   237→   237→    }
   238→   238→    wasCheckedOut.current = isCheckedOut
   239→   239→  }, [isCheckedOut])
   240→   240→
   241→   241→  // ─── Show TXN number when lastCreatedDocNo changes ───

   244→   244→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   245→   245→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   246→   246→      return () => clearTimeout(timer)
   247→   247→    }
   248→   248→  }, [lastCreatedDocNo, lastCreatedType])
   249→   249→
   250→   250→  // ─── Fetch invoice number ───
   251→   251→  const fetchInvoiceNumber = React.useCallback(async () => {
   252→   252→    if (!detail || !apiHeaders) return
   253→   253→    try {
   254→   254→      setInvNumLoading(true)
   255→   255→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   256→   256→        method: 'POST',
   257→   257→        headers: apiHeaders,
   258→   258→      })
   259→   259→      if (res.ok) {
   260→   260→        const data = await res.json()
   261→   261→        setInvoiceNumber(data.invoiceNumber)
   262→   262→      }
   263→   263→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   264→   264→  }, [detail?.id, apiHeaders])

   282→   282→      return null

   303→   303→        return null
   304→   304→      }

   309→   309→      setLocalSubmitting(false)

   328→   328→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },

   330→   330→      })

   333→   333→        // Auto-allocate may return settledFolios array instead of single ledgerEntry

   335→   335→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })

   337→   337→        closeDialogs()
   338→   338→        onRefresh?.()
   339→   339→        setTimeout(() => setEntrySuccess(null), 4000)
   340→   340→      } else {
   341→   341→        const e = await safeJson(res)
   342→   342→        notify({ type: 'error', title: 'Settlement Failed', message: (e.error as string) || 'Failed to process settlement' })
   343→   343→      }
   344→   344→    } catch {
   345→   345→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   346→   346→    } finally {
   347→   347→      setLocalSubmitting(false)
   348→   348→    }
   349→   349→  }, [detail?.id, selectedFolioId, apiHeaders, paymentForm, handleSettle, closeDialogs, onRefresh])
   350→   350→
   351→   351→  // ─── Transfer with TXN tracking ───
   352→   352→  const transferWithTxn = React.useCallback(async () => {
   353→   353→    if (!detail || !transferForm.targetCustomerId || !apiHeaders) {
   354→   354→      handleTransfer()
   355→   355→      return
   356→   356→    }
   357→   357→    try {
   358→   358→      setLocalSubmitting(true)
   359→   359→      // If entries are selected, transfer only those; otherwise pure balance transfer
   360→   360→      const selectedEntryIds = selectedEntries.size > 0 ? Array.from(selectedEntries) : undefined
   361→   361→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   362→   362→        method: 'POST',
   363→   363→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   364→   364→        body: JSON.stringify({
   365→   365→          targetCustomerId: transferForm.targetCustomerId,
   366→   366→          amount: parseFloat(transferForm.amount) || 0,
   367→   367→          description: transferForm.description,
   368→   368→          entryIds: selectedEntryIds,
   369→   369→        }),
   370→   370→      })
   371→   371→      if (res.ok) {
   372→   372→        const data = await res.json()
   373→   373→        const docNo = data?.source?.entry?.docNo || ''
   374→   374→        setEntrySuccess({ show: true, docNo, type: 'Transfer' })
   375→   375→        notify({ type: 'success', title: `Transfer ${docNo}`, message: data.message || `Transfer ${docNo} processed successfully` })
   376→   376→        closeDialogs()
   377→   377→        onRefresh?.()
   378→   378→        setTimeout(() => setEntrySuccess(null), 4000)
   379→   379→      } else {
   380→   380→        const e = await safeJson(res)
   381→   381→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   382→   382→      }
   383→   383→    } catch {
   384→   384→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   385→   385→    } finally {
   386→   386→      setLocalSubmitting(false)
   387→   387→    }
   388→   388→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   389→   389→
   390→   390→  // ─── Split payment helpers for checkout ───
   391→   391→  const addSplitItem = () => {
   392→   392→    setLocalSplitItems(prev => [...prev, { method: 'Cash', amount: '', reference: '' }])
   393→   393→  }
