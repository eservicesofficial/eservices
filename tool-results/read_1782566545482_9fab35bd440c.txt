    68→    68→// ─── Helper: Compute the most-used payment mode from credit entries ───
    69→    69→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
    70→    70→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    71→    71→  if (creditEntries.length === 0) return 'N/A'
    72→    72→  const modeCounts: Record<string, number> = {}
    73→    73→  for (const e of creditEntries) {
    74→    74→    const mode = e.paymentMode || 'Cash'
    75→    75→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
    76→    76→  }
    77→    77→  let bestMode = 'Cash'
    78→    78→  let bestAmount = 0
    79→    79→  for (const [mode, amount] of Object.entries(modeCounts)) {
    80→    80→    if (amount > bestAmount) {
    81→    81→      bestAmount = amount
    82→    82→      bestMode = mode
    83→    83→    }
    84→    84→  }
    85→    85→  return bestMode
    86→    86→}

    89→    89→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
    90→    90→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    91→    91→  if (creditEntries.length === 0) {
    92→    92→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
    93→    93→  }
    94→    94→
    95→    95→  const modeTotals: Record<string, number> = {}
    96→    96→  for (const e of creditEntries) {
    97→    97→    const mode = e.paymentMode || 'Cash'
    98→    98→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
    99→    99→  }
   100→   100→
   101→   101→  const modes = Object.entries(modeTotals)
   102→   102→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   103→   103→
   104→   104→  if (modes.length === 1) {
   105→   105→    const [mode, amount] = modes[0]
   106→   106→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   107→   107→  }
   108→   108→
   109→   109→  const rows = modes.map(([mode, amount]) => {
   110→   110→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   111→   111→  }).join('')
   112→   112→
   113→   113→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   114→   114→}

   191→   191→// ─── Print Receipt ───
   192→   192→export async function printReceipt(
   193→   193→  detail: CustomerDetail,
   194→   194→  entries: LedgerEntry[],
   195→   195→  hdrRef: RefObject<Record<string, string>>
   196→   196→) {
   197→   197→  if (!detail) return
   198→   198→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   199→   199→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   200→   200→  const settings = await fetchSettingsForPrint(hdrRef)
   201→   201→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   202→   202→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   203→   203→  const totalTax = activeEntries.reduce((s, e) => s + (e.tax || 0), 0)
   204→   204→  const totalDiscounts = activeEntries.reduce((s, e) => s + (e.discount || 0), 0)
   205→   205→  const finalBal = activeEntries[activeEntries.length - 1].balance
   206→   206→  const now = new Date()
   207→   207→  const rows = buildEntryRows(activeEntries)
   208→   208→
   209→   209→  const txnId = await fetchTxnId(detail.id)
   210→   210→
   211→   211→  const w = openPrintWindow(`Receipt - ${detail.fullName}`)
   212→   212→  if (!w) return
   213→   213→
   214→   214→  w.document.write(
   215→   215→    `<!DOCTYPE html><html><head><title>Receipt</title><style>${PRINT_CSS}</style></head><body>` +
   216→   216→    buildUnifiedHeader(settings, 'PAYMENT RECEIPT', txnId) +
   217→   217→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +
   218→   218→    `<div class="datetime">${detail.status === 'CheckedOut' ? 'Status: Checked Out' : `Status: ${detail.status}`}</div>` +
   219→   219→    `<hr class="divider">` +
   220→   220→    `<table><tr><th>#</th><th>Date</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   221→   221→    `<tr class="totals"><td colspan="3" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(finalBal)}</td></tr></table>` +
   222→   222→    buildPaymentSummary(activeEntries) +
   223→   223→    `<div class="tax-summary"><div class="tax-item"><span class="tax-label">Total Tax: </span><span class="tax-val">${fmtINR(totalTax)}</span></div><div class="tax-item"><span class="tax-label">Discounts: </span><span class="tax-val">${fmtINR(totalDiscounts)}</span></div><div class="tax-item"><span class="tax-label">Entries: </span><span class="tax-val">${activeEntries.length}</span></div></div>` +
   224→   224→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Final Balance</div><div class="s-value" style="color:${finalBal >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(finalBal)}</div></div></div>` +
   225→   225→    buildFooter() +
   226→   226→    `</body></html>`
   227→   227→  )
   228→   228→  w.document.close()
   229→   229→  w.print()
   230→   230→}
