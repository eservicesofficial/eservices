   100→   100→
   101→   101→  // Note form
   102→   102→  const [noteType, setNoteType] = useState('Internal')
   103→   103→  const [noteContent, setNoteContent] = useState('')
   104→   104→  const [notePrivate, setNotePrivate] = useState(false)
   105→   105→  const [noteImportant, setNoteImportant] = useState(false)
   106→   106→
   107→   107→  // Doc form
   108→   108→  const [docType, setDocType] = useState('Custom')
   109→   109→  const [docName, setDocName] = useState('')
   110→   110→  const [docUrl, setDocUrl] = useState('')
   111→   111→  const [docNotes, setDocNotes] = useState('')

   118→   118→  }

   124→   124→    let notes = (detail?.notes || [])
   125→   125→    if (!isAdmin) notes = notes.filter(n => !n.isPrivate)
   126→   126→    if (noteTypeFilter !== 'All') notes = notes.filter(n => n.type === noteTypeFilter)
   127→   127→    if (noteSearch) { const q = noteSearch.toLowerCase(); notes = notes.filter(n => n.content.toLowerCase().includes(q) || n.type.toLowerCase().includes(q)) }

   129→   129→  }, [detail?.notes, isAdmin, noteTypeFilter, noteSearch])

   132→   132→    let docs = (detail?.documents || [])
   133→   133→    if (docTypeFilter !== 'All') docs = docs.filter(d => d.type === docTypeFilter)
   134→   134→    if (docSearch) { const q = docSearch.toLowerCase(); docs = docs.filter(d => d.name.toLowerCase().includes(q) || d.type.toLowerCase().includes(q) || (d.notes || '').toLowerCase().includes(q)) }
   135→   135→    return docs
   136→   136→  }, [detail?.documents, docTypeFilter, docSearch])

   141→   141→    const byType: Record<string, number> = {}
   142→   142→    for (const n of all) { if (!n.isPrivate || isAdmin) byType[n.type] = (byType[n.type] || 0) + 1 }
   143→   143→    return byType
   144→   144→  }, [detail?.notes, isAdmin])
   145→   145→
   146→   146→  const docStats = useMemo(() => {
   147→   147→    const all = detail?.documents || []
   148→   148→    const byType: Record<string, number> = {}
   149→   149→    for (const d of all) byType[d.type] = (byType[d.type] || 0) + 1
   150→   150→    return byType
   151→   151→  }, [detail?.documents])

   159→   159→      setAddLoading(true)
   160→   160→      const res = await fetch(`/api/folio/customers/${detail.id}/notes`, {
   161→   161→        method: 'POST', headers: { ...authHdrs, 'Content-Type': 'application/json' },
   162→   162→        body: JSON.stringify({ type: noteType, content: noteContent.trim(), isPrivate: isAdmin && notePrivate, isImportant: noteImportant })
   163→   163→      })
   164→   164→      if (res.ok) { notify({ type: 'success', title: 'Note added' }); resetNoteForm(); onRefresh() }
   165→   165→      else { const e = await res.json().catch(() => ({})); notify({ type: 'error', title: 'Error', message: (e as any).error || 'Failed to add note' }) }
   166→   166→    } catch { notify({ type: 'error', title: 'Error', message: 'Network error' }) } finally { setAddLoading(false) }
   167→   167→  }
   168→   168→
   169→   169→  const updateNote = async (noteId: string) => {
   170→   170→    if (!detail?.id || !editContent.trim()) return
   171→   171→    try {
   172→   172→      setAddLoading(true)
   173→   173→      const res = await fetch(`/api/folio/customers/${detail.id}/notes/${noteId}`, {
   174→   174→        method: 'PUT', headers: { ...authHdrs, 'Content-Type': 'application/json' },
   175→   175→        body: JSON.stringify({ content: editContent.trim() })
   176→   176→      })
   177→   177→      if (res.ok) { notify({ type: 'success', title: 'Note updated' }); setEditingNoteId(null); setEditContent(''); onRefresh() }
   178→   178→      else notify({ type: 'error', title: 'Error', message: 'Failed to update note' })
   179→   179→    } catch { notify({ type: 'error', title: 'Error', message: 'Network error' }) } finally { setAddLoading(false) }
   180→   180→  }
   181→   181→
   182→   182→  const addDoc = async () => {
   183→   183→    if (!detail?.id || !docName.trim()) return
   184→   184→    try {
   185→   185→      setAddLoading(true)
   186→   186→      const res = await fetch(`/api/folio/customers/${detail.id}/documents`, {
   187→   187→        method: 'POST', headers: { ...authHdrs, 'Content-Type': 'application/json' },
   188→   188→        body: JSON.stringify({ type: docType, name: docName.trim(), url: docUrl.trim(), notes: docNotes.trim(), fileSize: parseInt(docFileSize) || 0, mimeType: '' })
   189→   189→      })
   190→   190→      if (res.ok) { notify({ type: 'success', title: 'Document added' }); resetDocForm(); onRefresh() }
   191→   191→      else notify({ type: 'error', title: 'Error', message: 'Failed to add document' })
   192→   192→    } catch { notify({ type: 'error', title: 'Error', message: 'Network error' }) } finally { setAddLoading(false) }
   193→   193→  }
   194→   194→
   195→   195→  const deleteNote = async (id: string) => {
   196→   196→    if (!detail) return
   197→   197→    try {
   198→   198→      const res = await fetch(`/api/folio/customers/${detail.id}/notes/${id}`, { method: 'DELETE', headers: authHdrs })
   199→   199→      if (res.ok) { notify({ type: 'success', title: 'Deleted' }); onRefresh() }
   200→   200→      else notify({ type: 'error', title: 'Error', message: 'Failed to delete note' })
   201→   201→    } catch { notify({ type: 'error', title: 'Error', message: 'Network error' }) }
   202→   202→  }
   203→   203→
   204→   204→  const deleteDoc = async (id: string) => {
   205→   205→    if (!detail) return
   206→   206→    try {
   207→   207→      const res = await fetch(`/api/folio/customers/${detail.id}/documents/${id}`, { method: 'DELETE', headers: authHdrs })
   208→   208→      if (res.ok) { notify({ type: 'success', title: 'Deleted' }); onRefresh() }
   209→   209→      else notify({ type: 'error', title: 'Error', message: 'Failed to delete document' })
   210→   210→    } catch { notify({ type: 'error', title: 'Error', message: 'Network error' }) }
   211→   211→  }
   212→   212→
   213→   213→  const fmtSize = (bytes: number) => {
   214→   214→    if (bytes < 1024) return `${bytes} B`
   215→   215→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   216→   216→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   217→   217→  }
   218→   218→
   219→   219→  const totalNotes = detail?.notes ? (isAdmin ? detail.notes.length : detail.notes.filter(n => !n.isPrivate).length) : 0
   220→   220→  const totalDocs = detail?.documents?.length || 0
   221→   221→
   222→   222→  return (
   223→   223→    <div className="flex flex-col h-full">
   224→   224→      {/* ── Header Tabs ── */}
   225→   225→      <div className="flex items-center gap-1 p-3 pb-0">
   226→   226→        <button
   227→   227→          onClick={() => setTab('notes')}
   228→   228→          className={`flex items-center gap-2 px-4 py-2 rounded-t-lg text-xs font-semibold transition-all border-b-2 ${
   229→   229→            tab === 'notes'
   230→   230→              ? 'bg-emerald-50 dark:bg-emerald-950/20 text-emerald-700 dark:text-emerald-300 border-emerald-500'
   231→   231→              : 'text-muted-foreground border-transparent hover:text-foreground hover:bg-muted/50'

   253→   253→        </button>

   255→   255→

   262→   262→              {Object.entries(noteStats).map(([type, count]) => (

   267→   267→                    noteTypeFilter === type
   268→   268→                      ? `${NOTE_COLORS[type] || 'bg-muted text-muted-foreground'} border-current/20 ring-1 ring-current/20`
   269→   269→                      : 'bg-muted/50 text-muted-foreground border-transparent hover:bg-muted'
   270→   270→                  }`}
   271→   271→                >
   272→   272→                  {type} ({count})
   273→   273→                </button>
   274→   274→              ))}
   275→   275→              {noteTypeFilter !== 'All' && (
   276→   276→                <button onClick={() => setNoteTypeFilter('All')} className="text-[10px] px-1.5 py-0.5 rounded-full text-muted-foreground hover:text-foreground hover:bg-muted transition-colors">

   289→   289→            {(isAdmin || canManageDocs) && (
   290→   290→              <Button size="sm" className="h-8 gap-1.5 text-xs bg-emerald-600 hover:bg-emerald-700 shrink-0" onClick={() => setShowNoteForm(!showNoteForm)}>

   292→   292→                {showNoteForm ? 'Cancel' : 'New Note'}
   293→   293→              </Button>
   294→   294→            )}
   295→   295→          </div>
   296→   296→
   297→   297→          {/* Add Note Form */}
   298→   298→          {showNoteForm && (
   299→   299→            <Card className="border-emerald-200 dark:border-emerald-800 bg-emerald-50/50 dark:bg-emerald-950/20 mb-3 shadow-sm">
   300→   300→              <CardContent className="p-3 space-y-2.5">
   301→   301→                <div className="flex items-center gap-2 flex-wrap">
   302→   302→                  <div className="min-w-[120px]">
   303→   303→                    <Label className="text-[10px] text-muted-foreground font-medium">Note Type</Label>

   305→   305→                      <SelectTrigger className="h-8 text-xs mt-0.5"><SelectValue /></SelectTrigger>

   308→   308→                  </div>

   313→   313→                        <Lock className="size-3 text-red-500" /> Private (Admin only)
   314→   314→                      </Label>
   315→   315→                    </div>
   316→   316→                  )}
   317→   317→                  <div className="flex items-center gap-1.5 pt-5">
   318→   318→                    <input type="checkbox" checked={noteImportant} onChange={e => setNoteImportant(e.target.checked)} className="rounded border-slate-300 accent-amber-500" id="note-important-adv" />
   319→   319→                    <Label htmlFor="note-important-adv" className="text-[11px] flex items-center gap-1 cursor-pointer">
   320→   320→                      <Star className="size-3 text-amber-500" /> Mark as Important
   321→   321→                    </Label>
   322→   322→                  </div>
   323→   323→                </div>
   324→   324→                <div>

   326→   326→                  <Textarea value={noteContent} onChange={e => setNoteContent(e.target.value)} rows={3} placeholder="Write your note..." className="text-xs mt-0.5 resize-none" maxLength={2000} />
   327→   327→                  <p className="text-[9px] text-muted-foreground text-right mt-0.5">{noteContent.length}/2000</p>
   328→   328→                </div>
