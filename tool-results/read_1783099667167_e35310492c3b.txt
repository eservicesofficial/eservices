   157→function formatCurrency(amount: number): string {
   158→  return `₹${amount.toLocaleString('en-IN', {
   159→    minimumFractionDigits: 2,
   160→    maximumFractionDigits: 2,
   161→  })}`;
   162→}
   163→
   164→function formatDate(dateStr: string): string {
   165→  try {
   166→    return format(parseISO(dateStr), 'dd MMM yyyy');
   167→  } catch {
   168→    return dateStr;
   169→  }
   170→}
   171→
   172→function formatDateFull(dateStr: string): string {
   173→  try {
   174→    return format(parseISO(dateStr), 'dd/MM/yyyy');
   175→  } catch {
   176→    return dateStr;
   177→  }
   178→}

   262→function getBalanceColorClass(invoice: Invoice): string {
   263→  const s = (invoice.status || '').toLowerCase();
   264→  if (s === 'paid' || s === 'completed') return 'text-emerald-600 dark:text-emerald-400';

   269→  return 'text-amber-600 dark:text-amber-400';

   316→export default function BillHistorySection({
   317→  onNavigate,

   322→  const { token, hasPermission } = useAuth();
   323→  const authHeaders: Record<string, string> = token ? { 'Authorization': `Bearer ${token}` } : {};
   324→  const canAccess = hasPermission('billing', 'access');
   325→  const canEdit = hasPermission('billing', 'edit');
   326→  const canDelete = hasPermission('billing', 'delete');
   327→  const isReadOnly = !canAccess || (!canEdit && !canDelete);
   328→  const noAccess = !canAccess;
   329→
   330→  // Data state
   331→  const [invoices, setInvoices] = useState<Invoice[]>([]);
   332→  const [settings, setSettings] = useState<ShopSettings | null>(null);
   333→  const [loading, setLoading] = useState(true);
   334→  const [searchQuery, setSearchQuery] = useState('');
   335→  const [statusFilter, setStatusFilter] = useState('all');
   336→  const [todayFilter, setTodayFilter] = useState(false);
   337→
   338→  // View dialog state
   339→  const [viewInvoice, setViewInvoice] = useState<Invoice | null>(null);
   340→  const [viewLoading, setViewLoading] = useState(false);
   341→  const invoicePreviewRef = useRef<HTMLDivElement>(null);
   342→
   343→  // Delete dialog state
   344→  const [deleteTarget, setDeleteTarget] = useState<Invoice | null>(null);
   345→  const [deleting, setDeleting] = useState(false);
   346→
   347→  // Update dialog state (for Pending/Balance bills)
   348→  const [updateTarget, setUpdateTarget] = useState<Invoice | null>(null);
   349→  const [updatePaymentMethod, setUpdatePaymentMethod] = useState('Cash');
   350→  const [updateStatus, setUpdateStatus] = useState<string>('Pending');
   351→  const [updateAmountPaid, setUpdateAmountPaid] = useState<string>('');
   352→  const [updating, setUpdating] = useState(false);
   353→
   354→  // Pay Balance dialog state
   355→  const [payBalanceTarget, setPayBalanceTarget] = useState<Invoice | null>(null);
   356→  const [payBalanceAmount, setPayBalanceAmount] = useState<string>('');
   357→  const [payBalanceMethod, setPayBalanceMethod] = useState('Cash');
   358→  const [payingBalance, setPayingBalance] = useState(false);
   359→
   360→  // Refund dialog state
   361→  const [refundTarget, setRefundTarget] = useState<Invoice | null>(null);
   362→  const [refundSelectedItems, setRefundSelectedItems] = useState<Set<number>>(new Set());
   363→  const [refundReason, setRefundReason] = useState('');
   364→  const [processingRefund, setProcessingRefund] = useState(false);
   365→
   366→  // Replace dialog state
   367→  const [replaceTarget, setReplaceTarget] = useState<Invoice | null>(null);
   368→  const [replaceSelectedItems, setReplaceSelectedItems] = useState<Set<number>>(new Set());
   369→  const [replaceReason, setReplaceReason] = useState('');
   370→  const [processingReplace, setProcessingReplace] = useState(false);
   371→
   372→  // ── Data Fetching ──────────────────────────────────────────
   373→  const fetchInvoices = useCallback(async (silent?: boolean) => {
   374→    try {
   375→      if (!silent) setLoading(true);
   376→      const res = await fetch('/api/invoices', { headers: authHeaders });
   377→      if (!res.ok) throw new Error('Failed to fetch');
   378→      const data = await res.json();
   379→      setInvoices(Array.isArray(data) ? data : []);
   380→    } catch {
   381→      notify({ type: 'error', title: 'Error', message: 'Failed to load invoices' });
   382→    } finally {
   383→      setLoading(false);
   384→    }
   385→  }, []);
   386→
   387→  const fetchSettings = useCallback(async () => {
   388→    try {
   389→      const res = await fetch('/api/settings', { headers: authHeaders });
   390→      if (res.ok) {
   391→        const data = await res.json();
   392→        setSettings(data);
   393→      }
   394→    } catch {
   395→      // silently fail
   396→    }
   397→  }, []);
   398→
   399→  useEffect(() => {
   400→    fetchInvoices();
   401→    fetchSettings();
   402→  }, [fetchInvoices, fetchSettings]);
   403→
   404→  // Re-fetch data when dataVersion changes (sync from other tabs/devices)
   405→  const prevDataVersion = useRef(dataVersion);
   406→  useEffect(() => {
   407→    if (dataVersion !== undefined && dataVersion !== prevDataVersion.current) {
   408→      prevDataVersion.current = dataVersion;
   409→      fetchInvoices(true);  // silent=true for background sync
   410→    }
   411→  }, [dataVersion, fetchInvoices]);
   412→
   413→  // ── Filtering ──────────────────────────────────────────────
   414→  const filteredInvoices = useMemo(() => {
   415→    const todayStr = format(new Date(), 'yyyy-MM-dd');
   416→    return invoices
   417→      .filter((inv) => {
   418→        if (searchQuery.trim()) {
   419→          const q = searchQuery.toLowerCase();
   420→          if (
   421→            !(inv.invoiceNumber || '').toLowerCase().includes(q) &&
   422→            !(inv.customerName || '').toLowerCase().includes(q) &&
   423→            !(inv.customerPhone || '').toLowerCase().includes(q)
   424→          ) {
   425→            return false;
   426→          }
   427→        }
   428→        if (statusFilter !== 'all') {
   429→          const s = (inv.status || '').toLowerCase();
   430→          if (statusFilter === 'paid' && s !== 'paid' && s !== 'completed') return false;
   431→          if (statusFilter === 'pending' && s !== 'pending') return false;
   432→          if (statusFilter === 'balance' && s !== 'balance') return false;
   433→          if (statusFilter === 'refunded' && s !== 'refunded') return false;
   434→          if (statusFilter === 'replaced' && s !== 'replaced') return false;
   435→          if (statusFilter === 'cancelled' && s !== 'cancelled') return false;
   436→        }
   437→        if (todayFilter) {
   438→          const invDateStr = (inv.createdAt || inv.date || '').substring(0, 10);
   439→          if (invDateStr !== todayStr) return false;
   440→        }
   441→        return true;
   442→      })
   443→      .sort((a, b) => {
   444→        const dateA = new Date(a.createdAt || a.date).getTime();
   445→        const dateB = new Date(b.createdAt || b.date).getTime();
   446→        return dateB - dateA;
   447→      });
   448→  }, [invoices, searchQuery, statusFilter, todayFilter]);
   449→
   450→  // ── Delete ──────────────────────────────────────────────────
   451→  const handleDelete = async () => {
   452→    if (!deleteTarget) return;
   453→    try {
   454→      setDeleting(true);
   455→      const res = await fetch(`/api/invoices/${deleteTarget.id}`, {
   456→        method: 'DELETE',
   457→        headers: authHeaders,
   458→      });
   459→      if (!res.ok) throw new Error('Delete failed');
   460→      notify({ type: 'success', title: 'Deleted', message: `Invoice ${deleteTarget.invoiceNumber} deleted successfully` });
   461→      setInvoices((prev) => prev.filter((inv) => inv.id !== deleteTarget.id));
   462→      setDeleteTarget(null);
   463→      onRefreshData();
   464→    } catch {
   465→      notify({ type: 'error', title: 'Error', message: 'Failed to delete invoice' });
   466→    } finally {
   467→      setDeleting(false);
   468→    }
   469→  };
   470→
   471→  // ── Edit (fetch full invoice before passing) ──────────────
   472→  const handleEdit = async (invoice: Invoice) => {
   473→    try {
   474→      notify({ type: 'info', title: 'Editing', message: `Loading ${invoice.invoiceNumber} for editing...` });
   475→      const res = await fetch(`/api/invoices/${invoice.id}`, { headers: authHeaders });
   476→      if (res.ok) {
   477→        const fullInvoice = await res.json();
   478→        onEditInvoice(fullInvoice);
   479→        return;
   480→      }
   481→    } catch {
   482→      // fall through to pass list invoice as fallback
   483→    }
   484→    onEditInvoice(invoice);
   485→  };
   486→
   487→  // ── View (fetch full invoice) ──────────────────────────────
   488→  const handleView = async (invoice: Invoice) => {
   489→    try {
   490→      setViewLoading(true);
   491→      const res = await fetch(`/api/invoices/${invoice.id}`, { headers: authHeaders });
   492→      if (res.ok) {
   493→        const data = await res.json();
   494→        setViewInvoice(data);
   495→      } else {
   496→        setViewInvoice(invoice);
   497→      }
   498→    } catch {
   499→      setViewInvoice(invoice);
   500→    } finally {
   501→      setViewLoading(false);
   502→    }
   503→  };
   504→
   505→  // ── Update Dialog Handlers ─────────────────────────────────
   506→  const openUpdateDialog = (invoice: Invoice) => {
   507→    setUpdateTarget(invoice);
   508→    setUpdatePaymentMethod(invoice.paymentMethod || 'Cash');
   509→    setUpdateStatus(invoice.status || 'Pending');
   510→    setUpdateAmountPaid(String(invoice.paidAmount || 0));
   511→  };
   512→
   513→  const handleUpdate = async () => {
   514→    if (!updateTarget) return;
   515→    try {
   516→      setUpdating(true);
   517→      const payload: Record<string, any> = {
   518→        paymentMethod: updatePaymentMethod,
   519→        status: updateStatus,
   520→      };
   521→
   522→      if (updateStatus.toLowerCase() === 'balance') {
   523→        const amtPaid = parseFloat(updateAmountPaid) || 0;
   524→        const balance = updateTarget.grandTotal - amtPaid;
   525→        payload.paidAmount = amtPaid;
   526→        payload.balanceAmount = Math.max(balance, 0);
   527→      } else if (updateStatus.toLowerCase() === 'paid') {
   528→        payload.paidAmount = updateTarget.grandTotal;
   529→        payload.balanceAmount = 0;
   530→      }
   531→
   532→      const res = await fetch(`/api/invoices/${updateTarget.id}`, {
   533→        method: 'PUT',
   534→        headers: { ...authHeaders, 'Content-Type': 'application/json' },
   535→        body: JSON.stringify(payload),
   536→      });
   537→
   538→      if (!res.ok) throw new Error('Update failed');
   539→      const updated = await res.json();
   540→      notify({ type: 'success', title: 'Updated', message: `Invoice ${updateTarget.invoiceNumber} updated to ${getStatusLabel(updateStatus)}` });
   541→
   542→      setInvoices((prev) =>
   543→        prev.map((inv) => (inv.id === updateTarget.id ? { ...inv, ...updated } : inv))
   544→      );
   545→      setUpdateTarget(null);
   546→      onRefreshData();
   547→    } catch {
   548→      notify({ type: 'error', title: 'Error', message: 'Failed to update invoice' });
   549→    } finally {
   550→      setUpdating(false);
   551→    }
   552→  };
   553→
   554→  // ── Pay Balance Dialog Handlers ────────────────────────────
   555→  const openPayBalanceDialog = (invoice: Invoice) => {
   556→    setPayBalanceTarget(invoice);
   557→    setPayBalanceAmount(String(invoice.balanceAmount || 0));
   558→    setPayBalanceMethod(invoice.paymentMethod || 'Cash');
   559→  };
   560→
   561→  const handlePayBalance = async () => {
   562→    if (!payBalanceTarget) return;
   563→    const amount = parseFloat(payBalanceAmount) || 0;
   564→    const maxBalance = payBalanceTarget.balanceAmount || 0;
   565→
   566→    if (amount <= 0) {
   567→      notify({ type: 'error', title: 'Error', message: 'Payment amount must be greater than 0' });
   568→      return;
   569→    }
   570→    if (amount > maxBalance) {
   571→      notify({ type: 'error', title: 'Error', message: `Payment amount cannot exceed balance of ${formatCurrency(maxBalance)}` });
   572→      return;
   573→    }
   574→
   575→    try {
   576→      setPayingBalance(true);
