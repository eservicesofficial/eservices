   137→export function CustomerFolioDialogs({
   138→  detail, allFolios, selectedEntries, submitting, settlingMode,
   139→  selectedFolioId, bal, selectedEntryObj, isActive, isCheckedOut,
   140→  invoiceData, invoiceLoading, folioStatementData, customerTags,
   141→  transferResults, paymentEntries,
   142→  addOpen, editOpen, chargeOpen, paymentOpen, refundOpen, transferOpen,
   143→  discountOpen, docOpen, noteOpen, tagsOpen, checkoutOpen, deleteOpen,
   144→  splitAmountOpen, shiftOpen, removeOpen, voucherOpen, invoiceOpen,
   145→  detailsSheetOpen,
   146→  form, chargeForm, paymentForm, refundForm, transferForm, discountForm,
   147→  docForm, noteForm, splitAmountForm, invoiceFrom, invoiceTo,
   148→  shiftTargetFolioId, duplicate,
   149→  checkoutSettlementData, checkoutSettlementOpen, splitPayments,
   150→  setCheckoutSettlementOpen, setSplitPayments,
   151→  handleMultiFolioSettlement, handleAutoSettleAll,
   152→  setAddOpen, setEditOpen, setChargeOpen, setPaymentOpen, setRefundOpen,
   153→  setTransferOpen, setDiscountOpen, setDocOpen, setNoteOpen, setTagsOpen,
   154→  setCheckoutOpen, setDeleteOpen, setSplitAmountOpen, setShiftOpen,
   155→  setRemoveOpen, setVoucherOpen, setInvoiceOpen, setDetailsSheetOpen,
   156→  setForm, setChargeForm, setPaymentForm, setRefundForm, setTransferForm,
   157→  setDiscountForm, setDocForm, setNoteForm, setSplitAmountForm,
   158→  setInvoiceFrom, setInvoiceTo, setShiftTargetFolioId, setDuplicate,
   159→  handleCreate, handleUpdate, handleCharge, handlePayment, handleSettle,
   160→  handleRefund, handleDiscount, handleTransfer, handleShift, handleSplitAmount,
   161→  handleRemove, handleCheckout, handleDelete, handleAddDoc, handleAddNote,
   162→  handleToggleTag, handleDeleteDoc, handleDeleteNote,
   163→  fetchInvoiceData, fetchFolioStatement, printInvoice, printCompleteInvoice, printSingleFolio, printVoucherHistory,
   164→  closeDialogs, searchTransferTargets,
   165→  deleteFolioOpen, folioToDelete,
   166→  setDeleteFolioOpen, handleDeleteFolio,
   167→  // Enhanced props
   168→  apiHeaders, onRefresh, onPrintReceipt, onThermalPrintReceipt,
   169→  onThermalPrintInvoice, onThermalPrintVoucher,
   170→  lastCreatedDocNo, lastCreatedType,
   171→  // History
   172→  historyOpen, setHistoryOpen, onPrintHistoryFolio,
   173→}: DialogsProps) {
   174→
   175→  // ─── Internal State ───
   176→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   177→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   178→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   179→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   180→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   181→  const [splitMode, setSplitMode] = React.useState(false)
   182→  const [localSplitItems, setLocalSplitItems] = React.useState<{ method: string; amount: string; reference: string }[]>([
   183→    { method: 'Cash', amount: '', reference: '' },
   184→  ])
   185→
   186→  // ─── History state ───
   187→  const [historyLoading, setHistoryLoading] = React.useState(false)
   188→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   189→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   190→  const [expandedFolioEntries, setExpandedFolioEntries] = React.useState<Record<string, any[]>>({})
   191→  const [selectedVoucherIds, setSelectedVoucherIds] = React.useState<Set<string>>(new Set())
   192→
   193→  // Fetch history for the CURRENT customer (not all checked-out customers)
   194→  React.useEffect(() => {
   195→    if (historyOpen && detail && apiHeaders) {
   196→      fetchCustomerHistory()
   197→    } else if (!historyOpen) {
   198→      setCustomerHistory(null)
   199→      setExpandedFolioId(null)
   200→      setExpandedFolioEntries({})
   201→    }
   202→  }, [historyOpen, detail?.id])
   203→
   204→  const fetchCustomerHistory = React.useCallback(async () => {
   205→    if (!detail || !apiHeaders) return
   206→    try {
   207→      setHistoryLoading(true)
   208→      const res = await fetch(`/api/folio/customers/${detail.id}/history`, { headers: apiHeaders })
   209→      if (res.ok) {
   210→        const data = await res.json() as CustomerHistory
   211→        setCustomerHistory(data)
   212→      }
   213→    } catch { /* ignore */ } finally { setHistoryLoading(false) }
   214→  }, [detail?.id, apiHeaders])
   215→
   216→  const toggleFolioEntries = (folioId: string) => {
   217→    if (expandedFolioId === folioId) {
   218→      setExpandedFolioId(null)
   219→    } else {
   220→      const folio = customerHistory?.historicalFolios.find(f => f.folioId === folioId)
   221→      if (folio) {
   222→        setExpandedFolioId(folioId)
   223→        setExpandedFolioEntries(prev => ({
   224→          ...prev,
   225→          [folioId]: (folio.entries || folio.ledgerEntries || []),
   226→        }))
   227→      }
   228→    }
   229→  }
   230→
   231→  // ─── Track checkout transition ───
   232→  const wasCheckedOut = React.useRef(isCheckedOut)
   233→  React.useEffect(() => {
   234→    if (isCheckedOut && !wasCheckedOut.current) {
   235→      setPostCheckoutOpen(true)
   236→    }
   237→    wasCheckedOut.current = isCheckedOut
   238→  }, [isCheckedOut])
   239→
   240→  // ─── Show TXN number when lastCreatedDocNo changes ───
   241→  React.useEffect(() => {
   242→    if (lastCreatedDocNo && lastCreatedType) {
   243→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   244→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   245→      return () => clearTimeout(timer)
   246→    }
   247→  }, [lastCreatedDocNo, lastCreatedType])
   248→
   249→  // ─── Fetch invoice number ───
   250→  const fetchInvoiceNumber = React.useCallback(async () => {
   251→    if (!detail || !apiHeaders) return
   252→    try {
   253→      setInvNumLoading(true)
   254→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   255→        method: 'POST',
   256→        headers: apiHeaders,
   257→      })
   258→      if (res.ok) {
   259→        const data = await res.json()
   260→        setInvoiceNumber(data.invoiceNumber)
   261→      }
   262→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   263→  }, [detail?.id, apiHeaders])
   264→
   265→  // ─── Fetch invoice number when invoice dialog opens ───
   266→  React.useEffect(() => {
   267→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber) {
   268→      fetchInvoiceNumber()
   269→    }
   270→  }, [invoiceOpen, detail?.id, apiHeaders])
   271→
   272→  React.useEffect(() => {
   273→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   274→      fetchInvoiceData()
   275→    }
   276→  }, [invoiceOpen])
   277→
   278→  // ─── Entry creation with TXN tracking ───
   279→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   280→    if (!detail || !selectedFolioId || !apiHeaders) {
   281→      return null
   282→    }
   283→    try {
   284→      setLocalSubmitting(true)
   285→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   286→        method: 'POST',
   287→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   288→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   289→      })
   290→      if (res.ok) {
   291→        const entry = await res.json()
   292→        const docNo = entry?.docNo || ''
   293→        setEntrySuccess({ show: true, docNo, type })
   294→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   295→        closeDialogs()
   296→        onRefresh?.()
   297→        setTimeout(() => setEntrySuccess(null), 4000)
   298→        return entry
   299→      } else {
   300→        const e = await safeJson(res)
   301→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   302→        return null
   303→      }
   304→    } catch {
   305→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   306→      return null
   307→    } finally {
   308→      setLocalSubmitting(false)
   309→    }
   310→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   311→
   312→  // ─── Settlement with TXN tracking ───
   313→  const settleWithTxn = React.useCallback(async () => {
   314→    if (!detail || !paymentForm.amount || !apiHeaders) {
   315→      handleSettle()
   316→      return
   317→    }
   318→    try {
   319→      setLocalSubmitting(true)
   320→      // When master folio is selected, use autoAllocate to distribute across all folios
   321→      const isMaster = selectedFolioId === 'master'
   322→      const payload = isMaster
   323→        ? { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   324→        : { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   325→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   326→        method: 'POST',
   327→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   328→        body: JSON.stringify(payload),
   329→      })
   330→      if (res.ok) {
   331→        const data = await res.json()
   332→        // Auto-allocate may return settledFolios array instead of single ledgerEntry
   333→        const docNo = data?.ledgerEntry?.docNo || (data?.settledFolios?.[0] ? 'Multi-Folio' : '')
   334→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })
   335→        notify({ type: 'success', title: `Settlement ${docNo}`, message: data.message || `Settlement ${docNo} processed successfully` })
   336→        closeDialogs()
   337→        onRefresh?.()
   338→        setTimeout(() => setEntrySuccess(null), 4000)
   339→      } else {
   340→        const e = await safeJson(res)
   341→        notify({ type: 'error', title: 'Settlement Failed', message: (e.error as string) || 'Failed to process settlement' })
   342→      }
   343→    } catch {
   344→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   345→    } finally {
   346→      setLocalSubmitting(false)
   347→    }
   348→  }, [detail?.id, selectedFolioId, apiHeaders, paymentForm, handleSettle, closeDialogs, onRefresh])
   349→
   350→  // ─── Transfer with TXN tracking ───
   351→  const transferWithTxn = React.useCallback(async () => {
   352→    if (!detail || !transferForm.targetCustomerId || !apiHeaders) {
   353→      handleTransfer()
   354→      return
   355→    }
   356→    try {
   357→      setLocalSubmitting(true)
   358→      // If entries are selected, transfer only those; otherwise pure balance transfer
   359→      const selectedEntryIds = selectedEntries.size > 0 ? Array.from(selectedEntries) : undefined
   360→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   361→        method: 'POST',
   362→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   363→        body: JSON.stringify({
   364→          targetCustomerId: transferForm.targetCustomerId,
   365→          amount: parseFloat(transferForm.amount) || 0,
   366→          description: transferForm.description,
   367→          entryIds: selectedEntryIds,
   368→        }),
   369→      })
   370→      if (res.ok) {
   371→        const data = await res.json()
   372→        const docNo = data?.source?.entry?.docNo || ''
   373→        setEntrySuccess({ show: true, docNo, type: 'Transfer' })
   374→        notify({ type: 'success', title: `Transfer ${docNo}`, message: data.message || `Transfer ${docNo} processed successfully` })
   375→        closeDialogs()
   376→        onRefresh?.()
   377→        setTimeout(() => setEntrySuccess(null), 4000)
   378→      } else {
   379→        const e = await safeJson(res)
   380→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   381→      }
   382→    } catch {
   383→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   384→    } finally {
   385→      setLocalSubmitting(false)
   386→    }
   387→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   388→
   389→  // ─── Split payment helpers for checkout ───
   390→  const addSplitItem = () => {
   391→    setLocalSplitItems(prev => [...prev, { method: 'Cash', amount: '', reference: '' }])
   392→  }
   393→  const removeSplitItem = (idx: number) => {
   394→    setLocalSplitItems(prev => prev.filter((_, i) => i !== idx))
   395→  }
   396→  const updateSplitItem = (idx: number, field: string, value: string) => {
   397→    setLocalSplitItems(prev => prev.map((item, i) => i === idx ? { ...item, [field]: value } : item))
   398→  }
   399→  const totalSplitAmount = localSplitItems.reduce((sum, item) => sum + (parseFloat(item.amount) || 0), 0)
   400→  const splitRemaining = checkoutSettlementData ? Math.max(0, checkoutSettlementData.totalOutstanding - totalSplitAmount) : 0
   401→
   402→  // ─── Computed GST data from invoiceData ───
   403→  const gstData = React.useMemo(() => {
   404→    if (!invoiceData?.entries) return null
   405→    const entries = invoiceData.entries as any[]
   406→    const taxableEntries = entries.filter((e: any) => e.debit > 0)
   407→    const taxableAmount = taxableEntries.reduce((sum: number, e: any) => sum + (e.debit || 0) - (e.tax || 0) - (e.discount || 0), 0)
   408→    const totalTax = entries.reduce((sum: number, e: any) => sum + (e.tax || 0), 0)
   409→    const totalDiscount = entries.reduce((sum: number, e: any) => sum + (e.discount || 0), 0)
   410→    const totalCharges = entries.reduce((sum: number, e: any) => sum + (e.debit || 0), 0)
   411→    const cgst = totalTax / 2
   412→    const sgst = totalTax / 2
   413→    const totalPayments = entries.reduce((sum: number, e: any) => sum + (e.credit || 0), 0)
   414→    const advancePayment = bal < 0 ? Math.abs(bal) : 0
   415→    return { taxableAmount, totalTax, totalDiscount, totalCharges, cgst, sgst, totalPayments, advancePayment, entryCount: taxableEntries.length }
   416→  }, [invoiceData, bal])
   417→
   418→  // ─── Active submitting state (local or prop) ───
   419→  const isSubmitting = localSubmitting || submitting
   420→
   421→  // ─── Entry success banner component ───
   422→  const EntrySuccessBanner = () => {
   423→    if (!entrySuccess?.show) return null
   424→    return (
   425→      <div className="fixed bottom-4 right-4 z-[100] animate-in slide-in-from-bottom-2 fade-in duration-300">
   426→        <div className="bg-emerald-600 text-white rounded-xl shadow-lg px-4 py-3 flex items-center gap-3 min-w-[260px]">
   427→          <div className="w-8 h-8 rounded-full bg-emerald-500 flex items-center justify-center shrink-0">
   428→            <CircleCheck className="size-4 text-white" />
   429→          </div>
   430→          <div className="min-w-0">
   431→            <div className="text-xs font-bold">{entrySuccess.type} Created</div>
   432→            <div className="text-[11px] text-emerald-100 flex items-center gap-1 mt-0.5">
   433→              <Hash className="size-3" />
   434→              <span className="font-mono font-bold">{entrySuccess.docNo}</span>
   435→            </div>
   436→          </div>
   437→        </div>
   438→      </div>
   439→    )
   440→  }
   441→
   442→  // ─── Per-entry-type success wrapper ───
   443→  const SuccessOverlay = ({ docNo, type, label }: { docNo: string; type: string; label: string }) => (
   444→    <div className="flex flex-col items-center justify-center py-6 gap-3">
   445→      <div className="w-14 h-14 rounded-2xl bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   446→        <PartyPopper className="size-7 text-emerald-600 dark:text-emerald-400" />
   447→      </div>
