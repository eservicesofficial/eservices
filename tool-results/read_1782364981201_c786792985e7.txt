    51→    51→function buildEntryRows(entries: { date: string; description?: string; type?: string; debit: number; credit: number; balance: number; tax?: number }[]): string {

    86→    86→// ─── Print Receipt ───
    87→    87→export async function printReceipt(
    88→    88→  detail: CustomerDetail,
    89→    89→  entries: LedgerEntry[],
    90→    90→  hdrRef: RefObject<Record<string, string>>
    91→    91→) {
    92→    92→  if (!detail) return
    93→    93→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
    94→    94→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
    95→    95→  const settings = await fetchSettingsForPrint(hdrRef)
    96→    96→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
    97→    97→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
    98→    98→  const totalTax = activeEntries.reduce((s, e) => s + (e.tax || 0), 0)
    99→    99→  const totalDiscounts = activeEntries.reduce((s, e) => s + (e.discount || 0), 0)
   100→   100→  const finalBal = activeEntries[activeEntries.length - 1].balance
   101→   101→  const now = new Date()
   102→   102→  const rows = buildEntryRows(activeEntries)
   103→   103→
   104→   104→  const w = openPrintWindow(`Receipt - ${detail.fullName}`)
   105→   105→  if (!w) return
   106→   106→
   107→   107→  w.document.write(
   108→   108→    `<!DOCTYPE html><html><head><title>Receipt</title><style>${PRINT_CSS}</style></head><body>` +
   109→   109→    buildHeader(settings, 'Receipt') +
   110→   110→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +
   111→   111→    `<div class="datetime">${detail.status === 'CheckedOut' ? 'Status: Checked Out' : `Status: ${detail.status}`} &middot; ${fmtDateTime(now.toISOString())}</div>` +
   112→   112→    `<hr class="divider">` +
   113→   113→    `<table><tr><th>#</th><th>Date</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   114→   114→    `<tr class="totals"><td colspan="3" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(finalBal)}</td></tr></table>` +
   115→   115→    `<div class="tax-summary"><div class="tax-item"><span class="tax-label">Total Tax: </span><span class="tax-val">${fmtINR(totalTax)}</span></div><div class="tax-item"><span class="tax-label">Discounts: </span><span class="tax-val">${fmtINR(totalDiscounts)}</span></div><div class="tax-item"><span class="tax-label">Entries: </span><span class="tax-val">${activeEntries.length}</span></div></div>` +
   116→   116→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Final Balance</div><div class="s-value" style="color:${finalBal >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(finalBal)}</div></div></div>` +
   117→   117→    buildFooter() +
   118→   118→    `</body></html>`
   119→   119→  )
   120→   120→  w.document.close()
   121→   121→  w.print()
   122→   122→}

   307→   307→  return { shopName: 'Eservices', shopAddress: '', shopPhone: '', shopGst: '', shopGstState: '' }

   310→   310→function openThermalPrintWindow(thermalHtml: string, title: string): Window | null {

   324→   324→// ─── Thermal Print Receipt ───
   325→   325→export async function thermalPrintReceipt(
   326→   326→  detail: CustomerDetail,
   327→   327→  entries: LedgerEntry[],
   328→   328→  hdrRef: RefObject<Record<string, string>>
   329→   329→) {
   330→   330→  if (!detail) return
   331→   331→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   332→   332→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   333→   333→
   334→   334→  const [tConfig, shop] = await Promise.all([
   335→   335→    fetchThermalSettings(),
   336→   336→    fetchShopSettings(hdrRef),
   337→   337→  ])
   338→   338→
   339→   339→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   340→   340→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   341→   341→  const finalBal = activeEntries[activeEntries.length - 1].balance
   342→   342→
   343→   343→  const items: ThermalReceiptData['items'] = activeEntries.map(e => ({
   344→   344→    name: stripShiftText(e.description || e.type || '—'),
   345→   345→    qty: 1,
   346→   346→    rate: e.debit > 0 ? e.debit : e.credit,
   347→   347→    total: e.debit > 0 ? e.debit : e.credit,
   348→   348→  }))
   349→   349→
   350→   350→  const receiptData: ThermalReceiptData = {
   351→   351→    shopName: shop.shopName,
   352→   352→    shopAddress: shop.shopAddress,
   353→   353→    shopPhone: shop.shopPhone,
   354→   354→    shopGst: shop.shopGst,
   355→   355→    shopGstState: shop.shopGstState,
   356→   356→    showGst: false,
   357→   357→    invoiceNumber: detail.customerId,
   358→   358→    date: fmtDate(new Date().toISOString()),
   359→   359→    orderType: 'Folio Receipt',
   360→   360→    customerName: detail.fullName,
   361→   361→    customerPhone: detail.mobile,
   362→   362→    customerAddress: '',
   363→   363→    items,
   364→   364→    subtotal: totalDebits,
   365→   365→    cgst: 0,
   366→   366→    sgst: 0,
   367→   367→    gstRate: 0,
   368→   368→    deliveryCharge: 0,
   369→   369→    discountAmount: 0,
   370→   370→    grandTotal: totalDebits - totalCredits,
   371→   371→    paidAmount: totalCredits,
   372→   372→    balanceAmount: finalBal,
   373→   373→    paymentMethod: 'Folio',
   374→   374→    billStatus: finalBal <= 0 ? 'Paid' : 'Balance',
   375→   375→  }
   376→   376→
   377→   377→  const thermalHtml = buildRichThermalHtml(receiptData, tConfig, `Receipt - ${detail.fullName}`)
   378→   378→  openThermalPrintWindow(thermalHtml, `Receipt - ${detail.fullName}`)
   379→   379→}
