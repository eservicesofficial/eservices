    75→// ─── Helpers ───
    76→const fmtCur = (v: number) => `₹${Math.abs(v || 0).toLocaleString('en-IN', { maximumFractionDigits: 2 })}`
    77→const fmtDate = (s: string) => { if (!s) return '—'; try { const d = new Date(s); return `${String(d.getDate()).padStart(2,'0')}/${String(d.getMonth()+1).padStart(2,'0')}/${d.getFullYear()}` } catch { return s } }
    78→const fmtDateTime = (s: string) => { if (!s) return '—'; try { const d = new Date(s); return `${fmtDate(s)} ${d.toLocaleTimeString('en-IN',{hour:'2-digit',minute:'2-digit'})}` } catch { return s } }
    79→const relTime = (s: string) => {
    80→  if (!s) return ''; const n = new Date(); const d = new Date(s); const ms = n.getTime()-d.getTime(); const m = Math.floor(ms/60000); const h = Math.floor(m/60); const dy = Math.floor(h/24)
    81→  if (m < 1) return 'Just now'; if (m < 60) return `${m}m ago`; if (h < 24) return `${h}h ago`; if (dy < 7) return `${dy}d ago`; return fmtDate(s)
    82→}

    86→  let h = 0; for (let i = 0; i < name.length; i++) h = name.charCodeAt(i) + ((h << 5) - h); return g[Math.abs(h) % g.length]
    87→}
