    11→// ─── Helper: Convert number to Indian Rupee words ───
    12→function numberToIndianWords(num: number): string {
    13→  if (num === 0) return 'ZERO'
    14→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    15→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    16→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    17→  let words = ''
    18→  function convertBelowThousand(n: number): string {
    19→    let w = ''
    20→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    21→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    22→    else if (n > 0) { w += ones[n] }
    23→    return w
    24→  }
    25→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    26→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    27→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    28→  if (num > 0) { words += convertBelowThousand(num) }
    29→  return words.trim().toUpperCase() + ' ONLY'
    30→}

   210→// ─── Helper: Compute the most-used payment mode from credit entries ───
   211→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   212→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   213→  if (creditEntries.length === 0) return 'N/A'
   214→  const modeCounts: Record<string, number> = {}
   215→  for (const e of creditEntries) {
   216→    const mode = e.paymentMode || 'Cash'
   217→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   218→  }
   219→  let bestMode = 'Cash'
   220→  let bestAmount = 0
   221→  for (const [mode, amount] of Object.entries(modeCounts)) {
   222→    if (amount > bestAmount) {
   223→      bestAmount = amount
   224→      bestMode = mode
   225→    }
   226→  }
   227→  return bestMode
   228→}

   230→// ─── Helper: Build payment summary HTML ───
   231→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   232→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   233→  if (creditEntries.length === 0) {
   234→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   235→  }
   236→
   237→  const modeTotals: Record<string, number> = {}
   238→  for (const e of creditEntries) {
   239→    const mode = e.paymentMode || 'Cash'
   240→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   241→  }
   242→
   243→  const modes = Object.entries(modeTotals)
   244→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   245→
   246→  if (modes.length === 1) {
   247→    const [mode, amount] = modes[0]
   248→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   249→  }
   250→
   251→  const rows = modes.map(([mode, amount]) => {
   252→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   253→  }).join('')
   254→
   255→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   256→}

   258→// ─── Helper: Get stable document number from entries (never regenerate) ───
   259→// Searches ALL entries for the first non-empty docNo to avoid N/A
   260→function getStableDocId(entries: { docNo?: string; invoiceNo?: string; id?: string; createdAt?: string }[]): string {
   261→  if (entries.length === 0) return `DOC-${Date.now()}`
   262→  for (const e of entries) {
   263→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   264→  }
   265→  for (const e of entries) {
   266→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   267→  }
   268→  // Fallback: use entry ID truncated or timestamp — never show N/A
   269→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   270→  return fallback
   271→}

   274→async function fetchNewInvoiceNumber(customerId: string): Promise<string> {

   279→      return data.invoiceNumber || 'N/A'
   280→    }
   281→  } catch { /* ignore */ }
   282→  return 'N/A'

   396→// BUILD HOTEL-STYLE TAX INVOICE (Redesigned professional layout)
   397→// Supports: P4 specific folio, P8 multi-page, P10a GST inclusive, P11 amount in words
   398→// ═══════════════════════════════════════════════════════════════════
   399→export function buildHotelStyleInvoice(
   400→  settings: Record<string, string>,
   401→  detail: CustomerDetail,
   402→  entries: any[],
   403→  invoiceNo: string,
   404→  invoiceDate: string,
   405→  docTitle: string,
   406→  extraMeta?: { openingBalance?: number; closingBalance?: number; periodFrom?: string; periodTo?: string; specificFolioNumber?: string | number },
   407→  printedBy?: string
   408→): string {
   409→  const d = detail as any
   410→  const bizName = settings.shopName || 'Eservices'
   411→  const bizAddress = settings.shopAddress || ''
   412→  const bizPhone = settings.shopPhone || ''
   413→  const bizEmail = settings.shopEmail || ''
   414→  const gstNumber = settings.gstNumber || settings.gstin || ''
   415→  const panNumber = settings.panNumber || d.panNumber || ''
   416→  const gstState = settings.gstState || settings.gstStateCode || ''
   417→  const stateCode = gstState ? gstState.split('-')[0] || gstState : ''
   418→  const shopLogo = settings.shopLogo || settings.logo || settings.thermalLogo || ''
   419→
   420→  // P10a: GST settings
   421→  const gstEnabled = settings.enableGst === 'true' || settings.enableGst === true
   422→  const gstRate = parseFloat(settings.gstRate) || 0
   423→
   424→  // Customer info
   425→  const custName = d.fullName || ''
   426→  const custId = d.customerId || ''
   427→  const custMobile = d.mobile || ''
   428→  const custEmail = d.email || ''
   429→  const custAddr = d.address || ''
   430→  const custGst = d.gstNumber || ''
   431→  const custType = d.customerType || ''
   432→  const custPan = d.panNumber || ''
   433→  const custCompany = d.companyName || d.company || ''
   434→
   435→  // Calculate totals
   436→  const chargeEntries = entries.filter((e: any) => (e.debit || 0) > 0 && e.type !== 'Refund')
   437→  const paymentEntries = entries.filter((e: any) => (e.credit || 0) > 0)
   438→  const refundEntries = entries.filter((e: any) => e.type === 'Refund')
   439→  const totalDebits = chargeEntries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   440→  const totalCredits = paymentEntries.reduce((s: number, e: any) => s + (e.credit || 0), 0)
   441→  const totalRefunds = refundEntries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   442→
   443→  // P10a: GST Inclusive calculation
   444→  // If GST is enabled in settings, compute GST from charge amounts even if individual entries have tax=0
   445→  let totalTax = 0
   446→  let computedTaxFromSettings = false
   447→  if (gstEnabled && gstRate > 0) {
   448→    // Check if any charge has explicit tax > 0
   449→    const hasExplicitTax = chargeEntries.some((e: any) => (e.tax || 0) > 0)
   450→    if (hasExplicitTax) {
   451→      // Use the explicit tax values from entries
   452→      totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   453→    } else {
   454→      // P10a: No explicit GST on charges — compute from inclusive amounts using settings GST rate
   455→      // For inclusive GST: GST = Amount × Rate / (100 + Rate), Taxable = Amount - GST
   456→      totalTax = Math.round((totalDebits * gstRate / (100 + gstRate)) * 100) / 100
   457→      computedTaxFromSettings = true
   458→    }
   459→  } else {
   460→    // GST not enabled — use whatever tax is on entries (usually 0)
   461→    totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   462→  }
   463→
   464→  const totalDiscount = entries.reduce((s: number, e: any) => s + (e.discount || 0), 0)
   465→  const closingBal = roundBal(extraMeta?.closingBalance ?? (entries.length > 0 ? entries[entries.length - 1].balance : 0))
   466→  const openingBal = roundBal(extraMeta?.openingBalance ?? 0)
   467→
   468→  // Tax calculation (CGST/SGST split from total tax)
   469→  const cgstAmount = Math.round(totalTax / 2 * 100) / 100
   470→  const sgstAmount = Math.round((totalTax - cgstAmount) * 100) / 100
   471→  const netAmount = totalDebits - totalDiscount
   472→  const effectiveGstRate = netAmount > 0 ? ((totalTax / netAmount) * 100) : 0
   473→  const halfRate = effectiveGstRate > 0 ? +(effectiveGstRate / 2).toFixed(2) : 0
   474→  // Grand total = debits (tax-inclusive) - discount - credits + refunds
   475→  const grandTotal = roundBal(totalDebits - totalDiscount - totalCredits + totalRefunds)
   476→
   477→  // Payment mode info
   478→  const mostUsedMode = computeMostUsedPaymentMode(entries)
   479→  const lastPayment = paymentEntries.length > 0 ? paymentEntries[paymentEntries.length - 1] : null
   480→  const paymentDate = lastPayment ? fmtDate(lastPayment.date) : invoiceDate
   481→
   482→  // Arrival / departure from entries
   483→  const firstEntry = entries.length > 0 ? entries[0] : null
   484→  const lastEntry = entries.length > 0 ? entries[entries.length - 1] : null
   485→  const arrivalDate = firstEntry ? fmtDate(firstEntry.date) : '—'
   486→  const departureDate = lastEntry ? fmtDate(lastEntry.date) : '—'
   487→
   488→  // ── Build HTML ──
   489→  // P4: Use specificFolioNumber if provided (for proforma/single folio printing)
   490→  const specificFolioNo = extraMeta?.specificFolioNumber
   491→  let folioNo: string
   492→  if (specificFolioNo != null) {

   494→  } else {

   496→    folioNo = rawFolioNo != null ? (typeof rawFolioNo === 'number' ? fmtFolioNum(rawFolioNo) : String(rawFolioNo)) : 'Master Folio'
   497→  }
   498→
   499→  const visitNo = d.visitNumber || (d.activeFolio && d.activeFolio.visitNumber) || (d.folios && d.folios[0] && d.folios[0].visitNumber) || d.totalVisits || d.visitNo || '—'
   500→
   501→  // P8: Determine invoice number label based on docTitle
   502→  const invoiceLabel = docTitle.toUpperCase().includes('PROFORMA') ? 'Proforma No' : 'Invoice No'
   503→
   504→  let html = `<div class="inv-wrap">`

   506→  // 1) HEADER — white background, flex layout with logo

   512→  html += `<div class="biz-name">${bizName}</div>`

   514→  if (bizAddress) html += `<div class="biz-address">${bizAddress}</div>`

   518→  html += `</div>`
   519→
   520→  // 2) TITLE BAR
   521→  html += `<div class="inv-title-section">`
   522→  html += `<div class="doc-title">${docTitle}</div>`
   523→  html += `<hr class="title-underline">`
   524→  html += `<div class="meta-row">`

   531→
   532→  // 3) CUSTOMER INFO TABLE
   533→  html += `<div class="inv-cust-section"><table>`

   535→    <td class="lbl">Customer Id</td><td class="val">${custId || '—'}</td>

   824→// ─── Print Voucher / Payment History ───
   825→// P5: Uses invoiceNo from payment entries (VOU-xxxx) as permanent voucher number
   826→export async function printVoucherHistory(
   827→  detail: CustomerDetail,
   828→  paymentEntries: LedgerEntry[],
   829→  hdrRef: RefObject<Record<string, string>>,
   830→  printedBy?: string
   831→) {
   832→  if (!detail) return
   833→  if (paymentEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No payment entries to print' }); return }
   834→  const shopSettings = await fetchSettingsForPrint(hdrRef)
   835→  const now = new Date()
   836→
   837→  const sorted = [...paymentEntries].sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
   838→  const totalAmount = sorted.reduce((s, e) => s + (e.credit || 0), 0)
   839→
   840→  // P5: Use first payment's invoiceNo (VOU-xxxx) as permanent voucher number
   841→  // Fallback to docNo if invoiceNo is empty (for entries created before this fix)
   842→  const voucherNo = sorted[0]?.invoiceNo || sorted[0]?.docNo || getStableDocId(sorted)
   843→
   844→  // Build entries in the format expected by buildHotelStyleInvoice
   845→  const voucherEntries = sorted.map(e => ({
   846→    ...e,
   847→    debit: 0,
   848→    credit: e.credit || 0,
   849→    balance: 0,
   850→    type: 'Payment',
   851→    description: e.description || `Payment via ${e.paymentMode || 'Cash'}`,
   852→  }))
   853→
   854→  // Compute running balance for display
   855→  let runBal = 0
   856→  for (const e of voucherEntries) {
   857→    runBal -= (e.credit || 0)
   858→    e.balance = Math.round(runBal * 100) / 100
   859→  }
   860→
   861→  const w = openPrintWindow(`Payment Voucher - ${detail.fullName}`)
   862→  if (!w) return
   863→
   864→  w.document.write(
   865→    `<!DOCTYPE html><html><head><title>Payment Voucher</title><style>${INVOICE_CSS}</style><style>${buildTemplateCSSFromSettings(shopSettings)}</style></head><body>` +
   866→    buildHotelStyleInvoice(shopSettings, detail, voucherEntries, voucherNo, fmtDate(now.toISOString()), 'PAYMENT VOUCHER', {
   867→      openingBalance: 0,
   868→      closingBalance: -totalAmount,
   869→    }, printedBy) +
   870→    `</body></html>`
   871→  )
   872→  w.document.close()
   873→  w.print()
   874→}
