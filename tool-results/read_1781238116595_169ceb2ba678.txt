    23→}) {

    29→    emerald: 'bg-emerald-50 text-emerald-700 border-emerald-200 hover:bg-emerald-100 hover:text-emerald-800 hover:border-emerald-300 dark:bg-emerald-950/40 dark:text-emerald-400 dark:border-emerald-800 dark:hover:bg-emerald-900/50 dark:hover:text-emerald-300',
    30→  }
    31→
    32→  const baseClass = 'flex items-center gap-1.5 px-3 py-2 rounded-lg text-xs font-medium transition-all whitespace-nowrap border cursor-pointer select-none'

    44→        >
    45→          <Icon className="size-3.5 shrink-0" />

    49→          )}

    51→      </TooltipTrigger>
    52→      <TooltipContent side="top" className="text-[10px]">{label}</TooltipContent>
    53→    </Tooltip>
    54→  )
