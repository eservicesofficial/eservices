   336→    if (!detail || selectedEntries.size !== 1 || !splitAmountForm.targetFolioId || !splitAmountForm.amount) { notify({ type: 'error', title: 'Error', message: 'Select one entry, target folio, and amount' }); return }
   337→    try { setSubmitting(true)
   338→      const entryId = Array.from(selectedEntries)[0]
   339→      const res = await fetch(`/api/folio/customers/${detail.id}/split`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify({ entryId, targetFolioId: splitAmountForm.targetFolioId, splitAmount: parseFloat(splitAmountForm.amount), description: 'Split Amount' }) })
   340→      if (res.ok) { notify({ type: 'success', title: 'Split Complete' }); setSplitAmountOpen(false); setSplitAmountForm({ ...DEFAULT_SPLIT_AMOUNT_FORM }); setSelectedEntries(new Set()); fetchFolioEntries(detail.id, selectedFolioId!); fetchDetail(detail.id, true) }
   341→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   342→    } catch { /* ignore */ } finally { setSubmitting(false) }
   343→  }
   344→
   345→  const handleCheckout = async () => {
   346→    if (!detail) return
   347→    try { setSubmitting(true)
   348→      const res = await fetch(`/api/folio/customers/${detail.id}/checkout`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' } })
   349→      if (res.ok) { notify({ type: 'success', title: 'Checkout Complete' }); setCheckoutOpen(false); refreshAll() }
   350→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   351→    } catch { /* ignore */ } finally { setSubmitting(false) }
   352→  }
   353→
   354→  // ── Handle Checkout Settlement (check balances first) ──
   355→  const handleCheckoutSettlement = async () => {
   356→    if (!detail) return
   357→    try { setSubmitting(true)
   358→      const res = await fetch(`/api/folio/customers/${detail.id}/settlement-overview`, { headers: hdrRef.current })
   359→      if (res.ok) { const data = await res.json(); setCheckoutSettlementData(data); setCheckoutSettlementOpen(true) }
   360→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   361→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to check settlement status' }) } finally { setSubmitting(false) }
   362→  }
   363→
   364→  // ── Auto-settle all outstanding folios ──
   365→  const handleAutoSettleAll = async () => {
   366→    if (!detail || !checkoutSettlementData) return
   367→    try { setSubmitting(true)
   368→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   369→        method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' },
   370→        body: JSON.stringify({ amount: checkoutSettlementData.totalOutstanding, mode: 'Cash', reference: '', description: 'Auto-settlement (all folios)', autoAllocate: true }),
   371→      })
   372→      if (res.ok) { notify({ type: 'success', title: 'All Folios Settled', message: `Settled across all folios` }); setCheckoutSettlementOpen(false); refreshAll(); fetchCheckoutSettlementData() }
   373→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   374→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to settle' }) } finally { setSubmitting(false) }
   375→  }
   376→
   377→  const fetchCheckoutSettlementData = async () => {
   378→    if (!detail) return
   379→    try {
   380→      const res = await fetch(`/api/folio/customers/${detail.id}/settlement-overview`, { headers: hdrRef.current })
   381→      if (res.ok) setCheckoutSettlementData(await res.json())
   382→    } catch { /* ignore */ }
   383→  }
   384→
   385→  // ── Multi-folio settlement (settle specific folio) ──
   386→  const handleMultiFolioSettlement = async (payload: SettlementPayload) => {
   387→    if (!detail) return
   388→    try { setSubmitting(true)
   389→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   390→        method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(payload),
   391→      })
   392→      if (res.ok) { notify({ type: 'success', title: 'Settlement Complete' }); fetchCheckoutSettlementData(); if (selectedFolioId && selectedFolioId !== 'master') fetchFolioEntries(detail.id, selectedFolioId); fetchMasterFolioEntries(detail.id); refreshAll() }
   393→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   394→    } catch { /* ignore */ } finally { setSubmitting(false) }
   395→  }
   396→
   397→  const handleDeleteFolio = async () => {
   398→    if (!detail || !folioToDelete) return
   399→    try { setSubmitting(true)
   400→      const res = await fetch(`/api/folio/customers/${detail.id}/folios/${folioToDelete}`, { method: 'DELETE', headers: hdrRef.current })
   401→      if (res.ok) { notify({ type: 'success', title: 'Folio Deleted' }); setDeleteFolioOpen(false); setFolioToDelete(null); fetchDetail(detail.id) }
   402→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   403→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to delete folio' }) } finally { setSubmitting(false) }
   404→  }
   405→
   406→  const handleReopen = async () => {
   407→    if (!detail) return
   408→    try { setSubmitting(true)
   409→      const res = await fetch(`/api/folio/customers/${detail.id}/reopen`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' } })
   410→      if (res.ok) { notify({ type: 'success', title: 'Reopened' }); refreshAll() }
   411→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   412→    } catch { /* ignore */ } finally { setSubmitting(false) }
   413→  }
   414→
   415→  const handleDelete = async () => {
   416→    if (!detail) return
   417→    try { setSubmitting(true)
   418→      const res = await fetch(`/api/folio/customers/${detail.id}`, { method: 'DELETE', headers: hdrRef.current })
   419→      if (res.ok) { notify({ type: 'success', title: 'Archived' }); setDeleteOpen(false); goBack(); fetchCustomers(); onRefreshData() }
   420→      else notify({ type: 'error', title: 'Error', message: 'Failed' })
   421→    } catch { /* ignore */ } finally { setSubmitting(false) }
   422→  }
   423→
   424→  const handleAddDoc = async () => { if (!detail || !docForm.name) { notify({ type: 'error', title: 'Error', message: 'Name required' }); return } try { setSubmitting(true); const res = await fetch(`/api/folio/customers/${detail.id}/documents`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(docForm) }); if (res.ok) { notify({ type: 'success', title: 'Document Added' }); setDocForm({ ...DEFAULT_DOC_FORM }); await fetchDetail(detail.id); } } catch { /* ignore */ } finally { setSubmitting(false) } }
   425→  const handleDeleteDoc = async (docId: string) => { if (!detail) return; try { await fetch(`/api/folio/customers/${detail.id}/documents/${docId}`, { method: 'DELETE', headers: hdrRef.current }); notify({ type: 'success', title: 'Deleted' }); await fetchDetail(detail.id) } catch { /* ignore */ } }
   426→  const handleAddNote = async () => { if (!detail || !noteForm.content) { notify({ type: 'error', title: 'Error', message: 'Content required' }); return } try { setSubmitting(true); const res = await fetch(`/api/folio/customers/${detail.id}/notes`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(noteForm) }); if (res.ok) { notify({ type: 'success', title: 'Note Added' }); setNoteForm({ ...DEFAULT_NOTE_FORM }); await fetchDetail(detail.id); } } catch { /* ignore */ } finally { setSubmitting(false) } }
   427→  const handleDeleteNote = async (noteId: string) => { if (!detail) return; try { await fetch(`/api/folio/customers/${detail.id}/notes/${noteId}`, { method: 'DELETE', headers: hdrRef.current }); notify({ type: 'success', title: 'Deleted' }); await fetchDetail(detail.id) } catch { /* ignore */ } }
   428→  const handleToggleTag = async (tagName: string) => { if (!detail) return; const has = customerTags.includes(tagName); try { if (has) { await fetch(`/api/folio/customers/${detail.id}/tags?tagName=${encodeURIComponent(tagName)}`, { method: 'DELETE', headers: hdrRef.current }); setCustomerTags(prev => prev.filter(t => t !== tagName)) } else { await fetch(`/api/folio/customers/${detail.id}/tags`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify({ tagName }) }); setCustomerTags(prev => [...prev, tagName]) } fetchDetail(detail.id, true); fetchCustomers(true) } catch { /* ignore */ } }
   429→
   430→  // ── Invoice / statement helpers ──
   431→  const fetchInvoiceData = async () => {
   432→    if (!detail) return
   433→    try { setInvoiceLoading(true); const params = new URLSearchParams(); if (invoiceFrom) params.set('from', invoiceFrom); if (invoiceTo) params.set('to', invoiceTo)
   434→      const res = await fetch(`/api/folio/customers/${detail.id}/statement?${params}`, { headers: hdrRef.current }); if (res.ok) setInvoiceData(await res.json()); else notify({ type: 'error', title: 'Error', message: 'Failed' })
   435→    } catch { /* ignore */ } finally { setInvoiceLoading(false) }
   436→  }
   437→  const fetchFolioStatement = async (folioId: string) => {
   438→    if (!detail) return
   439→    try { setFolioStatementData(null)
   440→      const res = await fetch(`/api/folio/customers/${detail.id}/statement?folioId=${folioId}`, { headers: hdrRef.current }); if (res.ok) setFolioStatementData(await res.json())
   441→    } catch { /* ignore */ }
   442→  }
   443→
   444→  // ── Print wrappers (async, using ref) ──
   445→  const bal = detail?.balance ?? 0
   446→  const isActive = detail?.status === 'Active'
   447→  const isCheckedOut = detail?.status === 'CheckedOut'
   448→  // Include all payment/settlement/refund entries from ALL folios for voucher history
   449→  const paymentEntries = masterFolioEntries.filter(e => e.type === 'Payment' || e.type === 'Settlement' || e.type === 'Refund')
   450→  const selectedEntryObj = selectedEntries.size === 1 ? folioEntries.find(e => e.id === Array.from(selectedEntries)[0]) : null
   451→
   452→  const doPrintReceipt = async () => {
   453→    // After checkout, folioEntries may be empty — use masterFolioEntries as fallback
   454→    const entriesToPrint = folioEntries.length > 0 ? folioEntries : masterFolioEntries
   455→    await _printReceipt(detail!, entriesToPrint, hdrRef as any)
   456→  }
   457→  const doPrintInvoice = () => { _printInvoice(detail!, invoiceData, invoiceFrom, invoiceTo) }
   458→  const doPrintSingleFolio = () => { _printSingleFolio(detail!, folioStatementData) }
   459→  const doPrintVoucherHistory = async () => { await _printVoucherHistory(detail!, paymentEntries, hdrRef as any) }
   460→
   461→  // ── Thermal Print wrappers ──
   462→  const doThermalPrintReceipt = async () => {
   463→    const entriesToPrint = folioEntries.length > 0 ? folioEntries : masterFolioEntries
   464→    await _thermalPrintReceipt(detail!, entriesToPrint, hdrRef as any)
   465→  }
   466→  const doThermalPrintInvoice = () => { _thermalPrintInvoice(detail!, invoiceData, invoiceFrom, invoiceTo) }
   467→  const doThermalPrintFolio = () => { _thermalPrintFolioStatement(detail!, folioStatementData) }
   468→  const doThermalPrintVoucher = async () => { await _thermalPrintVoucherHistory(detail!, paymentEntries, hdrRef as any) }
   469→
   470→  // ── RENDER ──
   471→  return (
   472→    <FolioErrorBoundary>
   473→    <TooltipProvider delayDuration={200}>
   474→    <>
   475→    {!canAccess && <ReadOnlyBanner module="Customer Folio" noAccess />}
   476→
   477→    <div className={isFullscreen ? 'fixed inset-0 z-[60] flex flex-col' : ''}>
   478→      <div className={`flex flex-col ${isFullscreen ? 'h-full' : 'min-h-0'} border border-slate-200 dark:border-slate-700 ${isFullscreen ? 'border-0 rounded-none' : 'rounded-lg'} overflow-hidden bg-white dark:bg-slate-900 shadow-sm`}>
   479→
   480→        {/* ── HEADER BAR ── */}
   481→        <div className="bg-white dark:bg-slate-900 border-b border-slate-200 dark:border-slate-700 px-4 py-2.5 flex items-center justify-between shrink-0">
   482→          <div className="flex items-center gap-3">
   483→            <div className="w-9 h-9 rounded-xl bg-gradient-to-br from-emerald-600 to-emerald-700 flex items-center justify-center shadow-sm shadow-emerald-600/20">
   484→              <span className="text-white text-sm font-bold">CF</span>
   485→            </div>
   486→            <div>
   487→              <h2 className="text-sm font-bold text-slate-800 dark:text-slate-100">Customer Folio</h2>
   488→              <p className="text-[10px] text-slate-400 dark:text-slate-500">Guest profiles &amp; billing management</p>
   489→            </div>
   490→          </div>
   491→          <div className="flex items-center gap-2">
   492→            <button onClick={() => setIsFullscreen(!isFullscreen)} className="text-[10px] text-slate-500 dark:text-slate-400 hover:text-slate-700 dark:hover:text-slate-200 px-2 py-1 rounded border border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-800 transition-colors">
   493→              {isFullscreen ? 'Exit Full' : 'Fullscreen'}
   494→            </button>
   495→          </div>
   496→        </div>
   497→
   498→        {/* ── MAIN BODY ── */}
   499→        <div className="flex flex-col flex-1 min-h-0">
   500→          {/* CUSTOMER PROFILE GRID VIEW (when no customer selected) */}
   501→          {!selectedId && (
   502→            <CustomerFolioProfileGrid
   503→              customers={customers} loading={loading} search={search} setSearch={setSearch}
   504→              filter={filter} setFilter={setFilter} sort={sort} setSort={setSort}
   505→              canCreate={canCreate} onAddCustomer={() => { resetForm(); setAddOpen(true) }}
   506→              onSelectCustomer={openCustomer}
   507→            />
   508→          )}
   509→
   510→          {/* CUSTOMER DETAIL VIEW (when customer selected) — full width */}
   511→          {selectedId && detail && detailLoading && !folioEntries.length ? (
   512→            <div className="flex-1 flex flex-col items-center justify-center">
   513→              <Loader2 className="size-8 animate-spin text-emerald-500 mb-3" />
   514→              <p className="text-sm text-muted-foreground">Loading customer details...</p>
   515→            </div>
   516→          ) : selectedId && detail ? (
   517→            <CustomerFolioManagement
   518→              detail={detail} allFolios={allFolios} folioEntries={folioEntries}
   519→              selectedEntries={selectedEntries} selectedFolioId={selectedFolioId}
   520→              folioEntriesLoading={folioEntriesLoading} submitting={submitting}
   521→              isAdmin={isAdmin} isActive={isActive} isCheckedOut={isCheckedOut}
   522→              folioStatementData={folioStatementData}
   523→              masterFolioEntries={masterFolioEntries}
   524→              onCheckoutSettlement={handleCheckoutSettlement}
   525→              goBack={goBack} setSelectedFolioId={setSelectedFolioId}
   526→              setSelectedEntries={setSelectedEntries} fetchFolioEntries={fetchFolioEntries}
   527→              toggleEntrySelection={toggleEntrySelection} toggleAllEntries={toggleAllEntries}
   528→              handleCreateFolio={handleCreateFolio} handleReopen={handleReopen}
   529→              printReceipt={doPrintReceipt} printSingleFolio={doPrintSingleFolio}
   530→              fetchFolioStatement={fetchFolioStatement}
   531→              onVoucher={() => setVoucherOpen(true)}
   532→              onInvoice={() => { setInvoiceFrom(''); setInvoiceTo(''); setInvoiceData(null); fetchInvoiceData(); setInvoiceOpen(true) }}
   533→              onCharge={() => setChargeOpen(true)}
   534→              onSettlement={() => { setPaymentForm({ ...DEFAULT_PAYMENT_FORM }); setSettlingMode(true); setPaymentOpen(true) }}
   535→              onPayment={() => { setPaymentForm({ ...DEFAULT_PAYMENT_FORM }); setSettlingMode(false); setPaymentOpen(true) }}
   536→              onCheckout={() => setCheckoutOpen(true)}
   537→              onRefund={() => { setRefundForm({ ...DEFAULT_REFUND_FORM }); setRefundOpen(true) }}
   538→              onDiscount={() => { setDiscountForm({ ...DEFAULT_DISCOUNT_FORM }); setDiscountOpen(true) }}
   539→              onTransfer={() => setTransferOpen(true)}
   540→              onShift={() => setShiftOpen(true)}
   541→              onSplit={() => { if (selectedEntries.size === 1) { setSplitAmountForm({ ...DEFAULT_SPLIT_AMOUNT_FORM }); setSplitAmountOpen(true) } else notify({ type: 'info', title: 'Info', message: 'Select 1 entry to split' }) }}
   542→              onRemove={() => setRemoveOpen(true)}
   543→              onEdit={() => { populateEditForm(); setEditOpen(true) }}
   544→              onDetails={() => setDetailsSheetOpen(true)}
   545→              onTags={() => setTagsOpen(true)}
   546→              onNotes={() => setNoteOpen(true)}
   547→              onDocs={() => setDocOpen(true)}
   548→              onDelete={() => setDeleteOpen(true)}
   549→              onReports={() => setReportsOpen(true)}
   550→              onDeleteFolio={(folioId: string) => { setFolioToDelete(folioId); setDeleteFolioOpen(true) }}
   551→              isFullscreen={isFullscreen}
   552→              thermalPrintReceipt={doThermalPrintReceipt}
   553→              thermalPrintInvoice={doThermalPrintInvoice}
   554→              thermalPrintFolio={doThermalPrintFolio}
   555→              thermalPrintVoucher={doThermalPrintVoucher}
   556→            />
   557→          ) : null}
   558→        </div>
   559→      </div>
   560→    </div>
   561→
   562→      {/* All Dialogs */}
   563→        <CustomerFolioDialogs
   564→          detail={detail || null} allFolios={allFolios} selectedEntries={selectedEntries}
   565→          submitting={submitting} settlingMode={settlingMode} selectedFolioId={selectedFolioId}
   566→          bal={bal} selectedEntryObj={selectedEntryObj} isActive={isActive}
   567→          isCheckedOut={isCheckedOut} invoiceData={invoiceData} invoiceLoading={invoiceLoading}
   568→          folioStatementData={folioStatementData} customerTags={customerTags}
   569→          transferResults={transferResults} paymentEntries={paymentEntries}
   570→          addOpen={addOpen} editOpen={editOpen} chargeOpen={chargeOpen}
   571→          paymentOpen={paymentOpen} refundOpen={refundOpen} transferOpen={transferOpen}
   572→          discountOpen={discountOpen} docOpen={docOpen} noteOpen={noteOpen}
   573→          tagsOpen={tagsOpen} checkoutOpen={checkoutOpen} deleteOpen={deleteOpen}
   574→          splitAmountOpen={splitAmountOpen} shiftOpen={shiftOpen} removeOpen={removeOpen}
   575→          voucherOpen={voucherOpen} invoiceOpen={invoiceOpen} detailsSheetOpen={detailsSheetOpen}
   576→          form={form} chargeForm={chargeForm} paymentForm={paymentForm}
   577→          refundForm={refundForm} transferForm={transferForm} discountForm={discountForm}
   578→          docForm={docForm} noteForm={noteForm} splitAmountForm={splitAmountForm}
   579→          invoiceFrom={invoiceFrom} invoiceTo={invoiceTo} shiftTargetFolioId={shiftTargetFolioId}
   580→          duplicate={duplicate}
   581→          setAddOpen={setAddOpen} setEditOpen={setEditOpen} setChargeOpen={setChargeOpen}
   582→          setPaymentOpen={setPaymentOpen} setRefundOpen={setRefundOpen} setTransferOpen={setTransferOpen}
   583→          setDiscountOpen={setDiscountOpen} setDocOpen={setDocOpen} setNoteOpen={setNoteOpen}
   584→          setTagsOpen={setTagsOpen} setCheckoutOpen={setCheckoutOpen} setDeleteOpen={setDeleteOpen}
   585→          setSplitAmountOpen={setSplitAmountOpen} setShiftOpen={setShiftOpen} setRemoveOpen={setRemoveOpen}
   586→          setVoucherOpen={setVoucherOpen} setInvoiceOpen={setInvoiceOpen} setDetailsSheetOpen={setDetailsSheetOpen}
   587→          setForm={setForm} setChargeForm={setChargeForm} setPaymentForm={setPaymentForm}
   588→          setRefundForm={setRefundForm} setTransferForm={setTransferForm} setDiscountForm={setDiscountForm}
   589→          setDocForm={setDocForm} setNoteForm={setNoteForm} setSplitAmountForm={setSplitAmountForm}
   590→          setInvoiceFrom={setInvoiceFrom} setInvoiceTo={setInvoiceTo} setShiftTargetFolioId={setShiftTargetFolioId}
   591→          setDuplicate={setDuplicate}
   592→          handleCreate={handleCreate} handleUpdate={handleUpdate} handleCharge={handleCharge}
   593→          handlePayment={handlePayment} handleSettle={handleSettle} handleRefund={handleRefund}
   594→          handleDiscount={handleDiscount} handleTransfer={handleTransfer} handleShift={handleShift}
   595→          handleSplitAmount={handleSplitAmount} handleRemove={handleRemove} handleCheckout={handleCheckout}
   596→          handleDelete={handleDelete} handleAddDoc={handleAddDoc} handleAddNote={handleAddNote}
   597→          handleToggleTag={handleToggleTag} handleDeleteDoc={handleDeleteDoc} handleDeleteNote={handleDeleteNote}
   598→          fetchInvoiceData={fetchInvoiceData} fetchFolioStatement={fetchFolioStatement}
   599→          printInvoice={doPrintInvoice} printSingleFolio={doPrintSingleFolio} printVoucherHistory={doPrintVoucherHistory}
   600→          closeDialogs={closeDialogs} searchTransferTargets={searchTransferTargets}
   601→          checkoutSettlementData={checkoutSettlementData} checkoutSettlementOpen={checkoutSettlementOpen}
   602→          splitPayments={splitPayments} setCheckoutSettlementOpen={setCheckoutSettlementOpen}
   603→          setSplitPayments={setSplitPayments}
   604→          handleMultiFolioSettlement={handleMultiFolioSettlement} handleAutoSettleAll={handleAutoSettleAll}
   605→          deleteFolioOpen={deleteFolioOpen} folioToDelete={folioToDelete}
   606→          setDeleteFolioOpen={setDeleteFolioOpen} handleDeleteFolio={handleDeleteFolio}
   607→        />
   608→
   609→        {/* Reports Sheet */}
   610→        {reportsOpen && (
   611→          <CustomerFolioReports
   612→            onClose={() => setReportsOpen(false)}
   613→            hdrRef={hdrRef as any}
   614→            canAccess={canAccess}
   615→          />
   616→        )}
   617→    </>
   618→    </TooltipProvider>
   619→    </FolioErrorBoundary>
   620→  )
   621→}
