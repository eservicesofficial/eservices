    74→    74→// ─── Helper: Compute the most-used payment mode from credit entries ───
    75→    75→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
    76→    76→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    77→    77→  if (creditEntries.length === 0) return 'N/A'
    78→    78→  const modeCounts: Record<string, number> = {}
    79→    79→  for (const e of creditEntries) {
    80→    80→    const mode = e.paymentMode || 'Cash'
    81→    81→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
    82→    82→  }
    83→    83→  let bestMode = 'Cash'
    84→    84→  let bestAmount = 0
    85→    85→  for (const [mode, amount] of Object.entries(modeCounts)) {
    86→    86→    if (amount > bestAmount) {
    87→    87→      bestAmount = amount
    88→    88→      bestMode = mode
    89→    89→    }
    90→    90→  }
    91→    91→  return bestMode
    92→    92→}

    95→    95→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
    96→    96→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    97→    97→  if (creditEntries.length === 0) {
    98→    98→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
    99→    99→  }
   100→   100→
   101→   101→  const modeTotals: Record<string, number> = {}
   102→   102→  for (const e of creditEntries) {
   103→   103→    const mode = e.paymentMode || 'Cash'
   104→   104→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   105→   105→  }
   106→   106→
   107→   107→  const modes = Object.entries(modeTotals)
   108→   108→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   109→   109→
   110→   110→  if (modes.length === 1) {
   111→   111→    const [mode, amount] = modes[0]
   112→   112→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   113→   113→  }
   114→   114→
   115→   115→  const rows = modes.map(([mode, amount]) => {
   116→   116→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   117→   117→  }).join('')
   118→   118→
   119→   119→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   120→   120→}

   122→   122→// ─── Helper: Get stable document number from entries (never regenerate) ───
   123→   123→// Searches ALL entries for the first non-empty docNo to avoid N/A
   124→   124→function getStableDocId(entries: { docNo?: string; invoiceNo?: string; id?: string; createdAt?: string }[]): string {
   125→   125→  if (entries.length === 0) return `DOC-${Date.now()}`
   126→   126→  for (const e of entries) {
   127→   127→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   128→   128→  }
   129→   129→  for (const e of entries) {
   130→   130→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   131→   131→  }
   132→   132→  // Fallback: use entry ID truncated or timestamp — never show N/A
   133→   133→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   134→   134→  return fallback
   135→   135→}

   247→   247→// ─── Print Folio Invoice (formerly "Receipt") ───
   248→   248→// Uses stable doc IDs from existing entries — NEVER regenerates numbers on reprint
   249→   249→export async function printReceipt(
   250→   250→  detail: CustomerDetail,
   251→   251→  entries: LedgerEntry[],
   252→   252→  hdrRef: RefObject<Record<string, string>>
   253→   253→) {
   254→   254→  if (!detail) return
   255→   255→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   256→   256→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   257→   257→  const settings = await fetchSettingsForPrint(hdrRef)
   258→   258→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   259→   259→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   260→   260→  const totalTax = activeEntries.reduce((s, e) => s + (e.tax || 0), 0)
   261→   261→  const totalDiscounts = activeEntries.reduce((s, e) => s + (e.discount || 0), 0)
   262→   262→  const finalBal = activeEntries[activeEntries.length - 1].balance
   263→   263→  const now = new Date()
   264→   264→  const rows = activeEntries.map((e, i) => {
   265→   265→    const desc = stripInternalDescriptions(e.description || e.type || '—')
   266→   266→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td style="font-size:9px;color:#666;font-family:monospace">${e.docNo || ''}</td><td>${e.type || ''}</td><td>${desc}</td><td style="text-align:right;color:#dc2626;font-weight:600">${e.debit > 0 ? fmtINR(e.debit) : '—'}</td><td style="text-align:right;color:#16a34a;font-weight:600">${e.credit > 0 ? fmtINR(e.credit) : '—'}</td><td style="font-weight:700;color:${e.balance > 0 ? '#dc2626' : e.balance < 0 ? '#16a34a' : '#666'}">${fmtINR(e.balance)}</td></tr>`
   267→   267→  }).join('')
   268→   268→
   269→   269→  // Use stable doc ID from entries — never regenerate
   270→   270→  const docId = getStableDocId(activeEntries)
   271→   271→  const authorizedSignatory = (settings as Record<string, string>).authorizedSignatory || ''
   272→   272→
   273→   273→  const w = openPrintWindow(`Folio Invoice - ${detail.fullName}`)
   274→   274→  if (!w) return
   275→   275→
   276→   276→  w.document.write(
   277→   277→    `<!DOCTYPE html><html><head><title>Folio Invoice</title><style>${PRINT_CSS}</style></head><body>` +

   280→   280→      <div><div class="inv-label">Doc No.</div><div class="inv-value">${docId}</div></div>
   281→   281→      <div style="text-align:center"><div class="inv-label">Date</div><div class="inv-value">${fmtDate(now.toISOString())}</div></div>
   282→   282→      <div style="text-align:right"><div class="inv-label">Status</div><div class="inv-value">${detail.status === 'CheckedOut' ? 'Checked Out' : detail.status}</div></div>
   283→   283→    </div>` +
   284→   284→    buildCustomerInfoBlock(detail) +
   285→   285→    `<hr class="divider">` +
