   184→// ─── CSV Export ───
   185→function downloadCSV(data: Record<string, any>[], filename: string) {
   186→  if (!data.length) return
   187→  const headers = Object.keys(data[0])
   188→  const csv = [
   189→    headers.join(','),
   190→    ...data.map(row => headers.map(h => `"${String(row[h] ?? '').replace(/"/g, '""')}"`).join(','))
   191→  ].join('\n')
   192→  const blob = new Blob([csv], { type: 'text/csv' })
   193→  const url = URL.createObjectURL(blob)
   194→  const a = document.createElement('a')
   195→  a.href = url; a.download = `${filename}.csv`; a.click()
   196→  URL.revokeObjectURL(url)
   197→}

   199→// ─── Format cell value ───
   200→function formatCellValue(value: unknown, col: ColDef): React.ReactNode {
   201→  if (value === null || value === undefined || value === '') return <span className="text-muted-foreground">—</span>
   202→  const strVal = String(value)
   203→
   204→  switch (col.format) {
   205→    case 'currency':
   206→      return <span className="font-medium tabular-nums">{fmtINR(Number(value))}</span>
   207→    case 'date':
   208→      return <span className="tabular-nums">{fmtDate(strVal)}</span>
   209→    case 'datetime':
   210→      return <span className="tabular-nums text-xs">{fmtDate(strVal)}</span>
   211→    case 'amount':
   212→      return <span className="font-medium tabular-nums">{fmtCur(Number(value))}</span>
   213→    case 'tags': {
   214→      const tags = Array.isArray(value) ? value : [value]
   215→      return (
   216→        <div className="flex flex-wrap gap-1">
   217→          {tags.map((tag: string, i: number) => (
   218→            <Badge key={i} variant="outline" className={`text-[10px] px-1.5 py-0 ${TAG_BADGE_MAP[tag] || ''}`}>
   219→              {tag}
   220→            </Badge>
   221→          ))}
   222→        </div>
   223→      )
   224→    }
   225→    case 'badge':
   226→      return (
   227→        <Badge variant="outline" className={`text-[10px] px-1.5 py-0 ${TYPE_BADGE_MAP[strVal] || 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400'}`}>
   228→          {strVal}
   229→        </Badge>
   230→      )
   231→    default:
   232→      return <span className="text-xs">{strVal}</span>
   233→  }
   234→}

   288→      return
   289→    }
   290→
   291→    setLoading(true)
   292→    setError(null)
   293→    setReportData(null)
   294→
   295→    try {
   296→      const params = new URLSearchParams({ type: reportType })
   297→      if (fromDate) params.set('from', fromDate)
   298→      if (toDate) params.set('to', toDate)
   299→
   300→      const res = await fetch(`/api/folio/reports?${params}`, { headers: hdrRef.current })
   301→      if (!res.ok) {
   302→        const errData = await res.json().catch(() => ({}))
   303→        throw new Error(errData.error || `Request failed (${res.status})`)
   304→      }
   305→
   306→      const data = await res.json()
   307→      setReportData(data)
   308→
   309→      if (!data.data || data.data.length === 0) {
   310→        notify({ type: 'info', title: 'No Data', message: 'No records found for the selected criteria' })
   311→      }
   312→    } catch (err: any) {
   313→      const msg = err.message || 'Failed to generate report'
   314→      setError(msg)
   315→      notify({ type: 'error', title: 'Report Error', message: msg })
   316→    } finally {
   317→      setLoading(false)
   318→    }
   319→  }, [reportType, fromDate, toDate, canAccess, hdrRef])
   320→
   321→  // Download CSV
   322→  const handleDownloadCSV = useCallback(() => {
   323→    if (!reportData?.data?.length) {
   324→      notify({ type: 'warning', title: 'No Data', message: 'Generate a report first' })

   326→    }
   327→    // Flatten data for CSV
   328→    const flatData = reportData.data.map((row: any) => {
   329→      const flat: Record<string, any> = {}
   330→      for (const col of columns) {
   331→        const val = row[col.key]
   332→        if (Array.isArray(val)) {
   333→          flat[col.label] = val.join(', ')
   334→        } else if (col.format === 'currency' || col.format === 'amount') {
   335→          flat[col.label] = Number(val) || 0
   336→        } else {
   337→          flat[col.label] = val ?? ''
   338→        }
   339→      }
   340→      return flat
   341→    })
   342→    downloadCSV(flatData, `${reportType}-report-${new Date().toISOString().split('T')[0]}`)

   344→  }, [reportData, columns, reportType])
   345→

   349→      notify({ type: 'warning', title: 'No Data', message: 'Generate a report first' })
   350→      return
   351→    }
   352→    const opt = REPORT_OPTIONS.find(o => o.value === reportType)!
   353→    const summary = reportData.summary || {}
   354→
   355→    const summaryHTML = Object.entries(summary)
   356→      .filter(([, v]) => typeof v === 'number' || typeof v === 'string')
   357→      .map(([k, v]) => {
   358→        const label = k.replace(/([A-Z])/g, ' $1').replace(/^./, s => s.toUpperCase())

   361→          <div style="font-size:16px;font-weight:700;margin-top:2px">${typeof v === 'number' ? '₹' + v.toLocaleString('en-IN', { maximumFractionDigits: 2 }) : v}</div>
   362→        </div>`
   363→      })
   364→      .join('')
   365→
   366→    const headerCols = columns.map(c => `<th style="padding:8px 10px;font-size:10px;text-transform:uppercase;letter-spacing:0.5px;font-weight:600;text-align:${c.align === 'right' ? 'right' : c.align === 'center' ? 'center' : 'left'};border-bottom:2px solid #16a34a;color:#16a34a;background:#f0fdf4">${c.label}</th>`).join('')
   367→
   368→    const rows = reportData.data.map((row: any, idx: number) => {
   369→      const cells = columns.map(col => {
   370→        let val = row[col.key]
   371→        if (Array.isArray(val)) val = val.join(', ')
   372→        if (col.format === 'currency' || col.format === 'amount') val = '₹' + (Number(val) || 0).toLocaleString('en-IN', { minimumFractionDigits: 2 })
   373→        const align = col.align === 'right' ? 'right' : col.align === 'center' ? 'center' : 'left'
   374→        return `<td style="padding:6px 10px;font-size:11px;border-bottom:1px solid #f0f0f0;text-align:${align}">${val ?? '—'}</td>`
   375→      }).join('')
   376→      return `<tr style="${idx % 2 === 0 ? '' : 'background:#f8faf8'}">${cells}</tr>`
   377→    }).join('')
   378→
   379→    const w = window.open('', '_blank')

   382→      return
   383→    }
   384→
   385→    w.document.write(`<!DOCTYPE html><html><head><title>${opt.label}</title>
   386→      <style>
   387→        body{font-family:system-ui,-apple-system,sans-serif;padding:20px;max-width:210mm;margin:0 auto;color:#1a1a1a}
