  1086→  1086→

  1089→  1089→    month: string
  1090→  1090→    totalAmount: number
  1091→  1091→    count: number
  1092→  1092→    modeBreakdown: Record<string, number>

  1111→  1111→  const bestMonth = rows.length > 0 ? rows.reduce((best, r) => r.totalAmount > best.totalAmount ? r : best, rows[0]) : null

  1121→  1121→    },
  1122→  1122→    data: rows,
  1123→  1123→  })
