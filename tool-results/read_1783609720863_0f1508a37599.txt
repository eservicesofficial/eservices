    11→// ─── Helper: Convert number to Indian Rupee words ───
    12→function numberToIndianWords(num: number): string {
    13→  if (num === 0) return 'ZERO'
    14→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    15→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    16→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    17→  let words = ''
    18→  function convertBelowThousand(n: number): string {
    19→    let w = ''
    20→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    21→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    22→    else if (n > 0) { w += ones[n] }
    23→    return w
    24→  }
    25→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    26→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    27→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    28→  if (num > 0) { words += convertBelowThousand(num) }
    29→  return words.trim().toUpperCase() + ' ONLY'
    30→}

   207→// ─── Helper: Compute the most-used payment mode from credit entries ───
   208→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   209→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   210→  if (creditEntries.length === 0) return 'N/A'
   211→  const modeCounts: Record<string, number> = {}
   212→  for (const e of creditEntries) {
   213→    const mode = e.paymentMode || 'Cash'
   214→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   215→  }
   216→  let bestMode = 'Cash'
   217→  let bestAmount = 0
   218→  for (const [mode, amount] of Object.entries(modeCounts)) {
   219→    if (amount > bestAmount) {
   220→      bestAmount = amount
   221→      bestMode = mode
   222→    }
   223→  }
   224→  return bestMode
   225→}

   227→// ─── Helper: Build payment summary HTML ───
   228→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   229→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   230→  if (creditEntries.length === 0) {
   231→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   232→  }
   233→
   234→  const modeTotals: Record<string, number> = {}
   235→  for (const e of creditEntries) {
   236→    const mode = e.paymentMode || 'Cash'
   237→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   238→  }
   239→
   240→  const modes = Object.entries(modeTotals)
   241→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   242→
   243→  if (modes.length === 1) {
   244→    const [mode, amount] = modes[0]
   245→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   246→  }
   247→
   248→  const rows = modes.map(([mode, amount]) => {
   249→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   250→  }).join('')
   251→
   252→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   253→}

   255→// ─── Helper: Get stable document number from entries (never regenerate) ───
   256→// Searches ALL entries for the first non-empty docNo to avoid N/A
   257→function getStableDocId(entries: { docNo?: string; invoiceNo?: string; id?: string; createdAt?: string }[]): string {
   258→  if (entries.length === 0) return `DOC-${Date.now()}`
   259→  for (const e of entries) {
   260→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   261→  }
   262→  for (const e of entries) {
   263→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   264→  }
   265→  // Fallback: use entry ID truncated or timestamp — never show N/A
   266→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   267→  return fallback
   268→}

   271→async function fetchNewInvoiceNumber(customerId: string): Promise<string> {

   279→  return 'N/A'

   392→// ═══════════════════════════════════════════════════════════════════
   393→// BUILD HOTEL-STYLE TAX INVOICE (Redesigned professional layout)
   394→// ═══════════════════════════════════════════════════════════════════
   395→export function buildHotelStyleInvoice(
   396→  settings: Record<string, string>,
   397→  detail: CustomerDetail,
   398→  entries: any[],
   399→  invoiceNo: string,
   400→  invoiceDate: string,
   401→  docTitle: string,
   402→  extraMeta?: { openingBalance?: number; closingBalance?: number; periodFrom?: string; periodTo?: string },
   403→  printedBy?: string
   404→): string {
   405→  const d = detail as any
   406→  const bizName = settings.shopName || 'Eservices'
   407→  const bizAddress = settings.shopAddress || ''
   408→  const bizPhone = settings.shopPhone || ''
   409→  const bizEmail = settings.shopEmail || ''
   410→  const gstNumber = settings.gstNumber || settings.gstin || ''
   411→  const panNumber = settings.panNumber || d.panNumber || ''
   412→  const gstState = settings.gstState || settings.gstStateCode || ''
   413→  const stateCode = gstState ? gstState.split('-')[0] || gstState : ''
   414→  const shopLogo = settings.shopLogo || settings.logo || settings.thermalLogo || ''
   415→
   416→  // Customer info
   417→  const custName = d.fullName || ''
   418→  const custId = d.customerId || ''
   419→  const custMobile = d.mobile || ''
   420→  const custEmail = d.email || ''
   421→  const custAddr = d.address || ''
   422→  const custGst = d.gstNumber || ''
   423→  const custType = d.customerType || ''
   424→  const custPan = d.panNumber || ''
   425→  const custCompany = d.companyName || d.company || ''
   426→
   427→  // Calculate totals
   428→  const chargeEntries = entries.filter((e: any) => (e.debit || 0) > 0 && e.type !== 'Refund')
   429→  const paymentEntries = entries.filter((e: any) => (e.credit || 0) > 0)
   430→  const refundEntries = entries.filter((e: any) => e.type === 'Refund')
   431→  const totalDebits = chargeEntries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   432→  const totalCredits = paymentEntries.reduce((s: number, e: any) => s + (e.credit || 0), 0)
   433→  const totalRefunds = refundEntries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   434→  const totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   435→  const totalDiscount = entries.reduce((s: number, e: any) => s + (e.discount || 0), 0)
   436→  const closingBal = roundBal(extraMeta?.closingBalance ?? (entries.length > 0 ? entries[entries.length - 1].balance : 0))
   437→  const openingBal = roundBal(extraMeta?.openingBalance ?? 0)
   438→
   439→  // Tax calculation (estimate CGST/SGST split from total tax)
   440→  const cgstAmount = Math.round(totalTax / 2 * 100) / 100
   441→  const sgstAmount = Math.round((totalTax - cgstAmount) * 100) / 100
   442→  const netAmount = totalDebits - totalDiscount
   443→  const gstRate = netAmount > 0 ? ((totalTax / netAmount) * 100) : 0
   444→  const halfRate = gstRate > 0 ? +(gstRate / 2).toFixed(2) : 0
   445→  // FIX: Grand total = debits (tax-inclusive) - discount - credits + refunds
   446→  const grandTotal = roundBal(totalDebits - totalDiscount - totalCredits + totalRefunds)
   447→
   448→  // Payment mode info
   449→  const mostUsedMode = computeMostUsedPaymentMode(entries)
   450→  const lastPayment = paymentEntries.length > 0 ? paymentEntries[paymentEntries.length - 1] : null
   451→  const paymentDate = lastPayment ? fmtDate(lastPayment.date) : invoiceDate
   452→
   453→  // Arrival / departure from entries
   454→  const firstEntry = entries.length > 0 ? entries[0] : null
   455→  const lastEntry = entries.length > 0 ? entries[entries.length - 1] : null
   456→  const arrivalDate = firstEntry ? fmtDate(firstEntry.date) : '—'
   457→  const departureDate = lastEntry ? fmtDate(lastEntry.date) : '—'
   458→
   459→  // ── Build HTML ──
   460→  // FIX 2c: Folio Number — extract from folio object properly
   461→  const rawFolioNo = d.folioNumber || (d.folios && d.folios[0] ? d.folios[0].folioNumber : null) || d.folioNo || null
   462→  const folioNo = rawFolioNo != null ? (typeof rawFolioNo === 'number' ? fmtFolioNum(rawFolioNo) : String(rawFolioNo)) : 'Master Folio'
   463→
   464→  // FIX 2d: Visit No — check multiple possible locations
   465→  const visitNo = d.visitNumber || (d.activeFolio && d.activeFolio.visitNumber) || (d.folios && d.folios[0] && d.folios[0].visitNumber) || d.totalVisits || d.visitNo || '—'
   466→
   467→  let html = `<div class="inv-wrap">`
   468→
   469→  // 1) HEADER — white background, flex layout with logo
   470→  html += `<div class="inv-header">`
   471→  if (shopLogo) {
   472→    html += `<div class="header-logo"><img src="${shopLogo}" alt="Logo" /></div>`
   473→  }
   474→  html += `<div class="header-info">`
   475→  html += `<div class="biz-name">${bizName}</div>`

   483→  // 2) TITLE BAR — "TAX INVOICE" centered bold, gradient underline, meta row below
   484→  html += `<div class="inv-title-section">`
   485→  html += `<div class="doc-title">${docTitle}</div>`
   486→  html += `<hr class="title-underline">`
   487→  html += `<div class="meta-row">`
   488→  html += `<span>Invoice No: <strong>${invoiceNo}</strong></span>`
   489→  html += `<span>Date: <strong>${invoiceDate}</strong></span>`

   504→    <td class="lbl">Visit No</td><td class="val">${visitNo}</td>

   507→  html += `<tr>

   510→  </tr>`
   511→  // Row 4: Mobile | val | Email | val
   512→  html += `<tr>
   513→    <td class="lbl">Mobile</td><td class="val">${custMobile || '—'}</td>
   514→    <td class="lbl">Email</td><td class="val">${custEmail || '—'}</td>

   781→export async function printVoucherHistory(
   782→  detail: CustomerDetail,
   783→  paymentEntries: LedgerEntry[],
   784→  hdrRef: RefObject<Record<string, string>>
   785→) {
   786→  if (!detail) return
   787→  if (paymentEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No payment entries to print' }); return }
   788→  const settings = await fetchSettingsForPrint(hdrRef)
   789→  const now = new Date()
   790→
   791→  const sorted = [...paymentEntries].sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
   792→  const totalAmount = sorted.reduce((s, e) => s + (e.credit || 0), 0)
   793→
   794→  // Point 5: Use first entry's docNo as voucher number (permanent, generated at creation time)

   796→

   799→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td>${desc}</td><td style="text-align:right;font-weight:700;color:#16a34a">${fmtINR(e.credit || 0)}</td><td style="text-align:center">${e.paymentMode || '—'}</td></tr>`
   800→  }).join('')
   801→
   802→  const w = openPrintWindow(`Voucher History - ${detail.fullName}`)

   804→
   805→  w.document.write(
   806→    `<!DOCTYPE html><html><head><title>Payment Voucher</title><style>${PRINT_CSS}</style></head><body>` +
   807→    buildUnifiedHeader(settings, 'PAYMENT VOUCHER', voucherNo) +
   808→    buildCustomerInfoBlock(detail) +
   809→    `<div class="datetime">Printed: ${fmtDateTime(now.toISOString())}</div>` +
   810→    `<table><tr><th>#</th><th>Date</th><th>Description</th><th style="text-align:right">Amount (₹)</th><th>Mode</th></tr>${rows}` +
   811→    `<tr class="totals"><td colspan="3" style="text-align:right;font-size:11px;color:#555">TOTAL</td><td style="color:#16a34a;font-weight:700">${fmtINR(totalAmount)}</td><td>${sorted.length} entries</td></tr></table>` +
   812→    buildPaymentSummary(sorted) +
   813→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Total Payments</div><div class="s-value" style="color:#16a34a">${fmtINR(totalAmount)}</div></div><div class="summary-item"><div class="s-label">Entries</div><div class="s-value">${sorted.length}</div></div></div>` +
   814→    buildFooter() +
   815→    `</body></html>`
   816→  )
   817→  w.document.close()
   818→  w.print()
   819→}

   823→interface ShopSettings {

   832→async function fetchShopSettings(hdrRef?: RefObject<Record<string, string>>): Promise<ShopSettings> {
   833→  try {
   834→    const headers = hdrRef?.current || undefined
   835→    const res = await fetch('/api/settings', headers ? { headers } : undefined)
   836→    if (res.ok) {
   837→      const db = (await res.json()) as Record<string, string>
   838→      return {
   839→        shopName: db.shopName || 'Eservices',
   840→        shopAddress: db.shopAddress || '',
   841→        shopPhone: db.shopPhone || '',
   842→        shopGst: db.gstNumber || db.gstin || '',
   843→        shopGstState: db.gstState || '',
   844→        shopLogo: db.shopLogo || db.logo || '',
   845→      }
   846→    }
   847→  } catch { /* ignore */ }
   848→  return { shopName: 'Eservices', shopAddress: '', shopPhone: '', shopGst: '', shopGstState: '', shopLogo: '' }
   849→}

   851→function openThermalPrintWindow(thermalHtml: string, title: string): Window | null {
   852→  const w = window.open('', '_blank', 'width=350,height=700')
   853→  if (!w) {
   854→    notify({ type: 'error', title: 'Error', message: 'Please allow pop-ups to print thermal receipts' })
   855→    return null
   856→  }
   857→  w.document.write(thermalHtml)
   858→  w.document.close()
   859→  setTimeout(() => {
   860→    try { if (!w.closed) w.print() } catch { /* window closed */ }
   861→  }, 600)
   862→  return w

   866→export async function thermalPrintReceipt(
   867→  detail: CustomerDetail,
   868→  entries: LedgerEntry[],
   869→  hdrRef: RefObject<Record<string, string>>
   870→) {
   871→  if (!detail) return
   872→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   873→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   874→
   875→  const [tConfig, shop] = await Promise.all([
   876→    fetchThermalSettings(),

   882→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   883→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   884→  const finalBal = activeEntries[activeEntries.length - 1].balance
   885→
   886→  const items: ThermalReceiptData['items'] = activeEntries.map(e => ({
   887→    name: stripInternalDescriptions(e.description || e.type || '—'),
   888→    qty: 1,
   889→    rate: e.debit > 0 ? e.debit : e.credit,
   890→    total: e.debit > 0 ? e.debit : e.credit,
   891→  }))
   892→
   893→  const actualPaymentMethod = computeMostUsedPaymentMode(activeEntries)
   894→
   895→  const receiptData: ThermalReceiptData = {
   896→    shopName: shop.shopName,
   897→    shopAddress: shop.shopAddress,
   898→    shopPhone: shop.shopPhone,
   899→    shopGst: shop.shopGst,
   900→    shopGstState: shop.shopGstState,
   901→    showGst: false,
   902→    invoiceNumber: detail.customerId,
   903→    date: fmtDate(new Date().toISOString()),
   904→    orderType: 'PAYMENT RECEIPT',
   905→    customerName: detail.fullName,
   906→    customerPhone: detail.mobile,
   907→    customerAddress: (detail as any).address || '',
   908→    items,
   909→    subtotal: totalDebits,
   910→    cgst: 0,
   911→    sgst: 0,
   912→    gstRate: 0,
   913→    deliveryCharge: 0,
   914→    discountAmount: 0,
   915→    grandTotal: totalDebits - totalCredits,
   916→    paidAmount: totalCredits,
   917→    balanceAmount: finalBal,
   918→    paymentMethod: actualPaymentMethod,
   919→    billStatus: finalBal <= 0 ? 'Paid' : 'Balance',
   920→  }
   921→
   922→  const thermalHtml = buildRichThermalHtml(receiptData, tConfig, `Receipt - ${detail.fullName}`)
   923→  openThermalPrintWindow(thermalHtml, `Receipt - ${detail.fullName}`)
