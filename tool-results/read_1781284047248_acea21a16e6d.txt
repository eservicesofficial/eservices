   342→    try { setSubmitting(true)
   343→      const res = await fetch(`/api/folio/customers/${detail.id}/entries/remove`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify({ entryIds: Array.from(selectedEntries) }) })
   344→      if (res.ok) { notify({ type: 'success', title: 'Removed' }); setRemoveOpen(false); setSelectedEntries(new Set()); fetchFolioEntries(detail.id, selectedFolioId!); fetchDetail(detail.id, true) }
   345→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   346→    } catch { /* ignore */ } finally { setSubmitting(false) }
   347→  }
   348→
   349→  const handleSplitAmount = async () => {
   350→    if (!detail || selectedEntries.size !== 1 || !splitAmountForm.targetFolioId || !splitAmountForm.amount) { notify({ type: 'error', title: 'Error', message: 'Select one entry, target folio, and amount' }); return }
   351→    try { setSubmitting(true)
   352→      const entryId = Array.from(selectedEntries)[0]
   353→      const res = await fetch(`/api/folio/customers/${detail.id}/split`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify({ entryId, targetFolioId: splitAmountForm.targetFolioId, splitAmount: parseFloat(splitAmountForm.amount), description: 'Split Amount' }) })
   354→      if (res.ok) { notify({ type: 'success', title: 'Split Complete' }); setSplitAmountOpen(false); setSplitAmountForm({ ...DEFAULT_SPLIT_AMOUNT_FORM }); setSelectedEntries(new Set()); fetchFolioEntries(detail.id, selectedFolioId!); fetchDetail(detail.id, true) }
   355→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   356→    } catch { /* ignore */ } finally { setSubmitting(false) }
   357→  }
   358→
   359→  const handleCheckout = async () => {
   360→    if (!detail) return
   361→    try { setSubmitting(true)
   362→      const res = await fetch(`/api/folio/customers/${detail.id}/checkout`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' } })
   363→      if (res.ok) { notify({ type: 'success', title: 'Checkout Complete' }); setCheckoutOpen(false); refreshAll() }
   364→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   365→    } catch { /* ignore */ } finally { setSubmitting(false) }
   366→  }
   367→
   368→  // ── Handle Checkout Settlement (check balances first) ──
   369→  const handleCheckoutSettlement = async () => {
   370→    if (!detail) return
   371→    try { setSubmitting(true)
   372→      const res = await fetch(`/api/folio/customers/${detail.id}/settlement-overview`, { headers: hdrRef.current })
   373→      if (res.ok) { const data = await res.json(); setCheckoutSettlementData(data); setCheckoutSettlementOpen(true) }
   374→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   375→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to check settlement status' }) } finally { setSubmitting(false) }
   376→  }
   377→
   378→  // ── Auto-settle all outstanding folios ──
   379→  const handleAutoSettleAll = async () => {
   380→    if (!detail || !checkoutSettlementData) return
   381→    try { setSubmitting(true)
   382→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   383→        method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' },
   384→        body: JSON.stringify({ amount: checkoutSettlementData.totalOutstanding, mode: 'Cash', reference: '', description: 'Auto-settlement (all folios)', autoAllocate: true }),
   385→      })
   386→      if (res.ok) { notify({ type: 'success', title: 'All Folios Settled', message: `Settled across all folios` }); setCheckoutSettlementOpen(false); refreshAll(); fetchCheckoutSettlementData() }
   387→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   388→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to settle' }) } finally { setSubmitting(false) }
   389→  }
   390→
   391→  const fetchCheckoutSettlementData = async () => {
   392→    if (!detail) return
   393→    try {
   394→      const res = await fetch(`/api/folio/customers/${detail.id}/settlement-overview`, { headers: hdrRef.current })
   395→      if (res.ok) setCheckoutSettlementData(await res.json())
   396→    } catch { /* ignore */ }
   397→  }
   398→
   399→  // ── Multi-folio settlement (settle specific folio) ──
   400→  const handleMultiFolioSettlement = async (payload: SettlementPayload) => {
   401→    if (!detail) return
   402→    try { setSubmitting(true)
   403→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   404→        method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(payload),
   405→      })
   406→      if (res.ok) { notify({ type: 'success', title: 'Settlement Complete' }); fetchCheckoutSettlementData(); if (selectedFolioId && selectedFolioId !== 'master') fetchFolioEntries(detail.id, selectedFolioId); fetchMasterFolioEntries(detail.id); refreshAll() }
   407→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   408→    } catch { /* ignore */ } finally { setSubmitting(false) }
   409→  }
   410→
   411→  const handleDeleteFolio = async () => {
   412→    if (!detail || !folioToDelete) return
   413→    try { setSubmitting(true)
   414→      const res = await fetch(`/api/folio/customers/${detail.id}/folios/${folioToDelete}`, { method: 'DELETE', headers: hdrRef.current })
   415→      if (res.ok) { notify({ type: 'success', title: 'Folio Deleted' }); setDeleteFolioOpen(false); setFolioToDelete(null); fetchDetail(detail.id) }
   416→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   417→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to delete folio' }) } finally { setSubmitting(false) }
   418→  }
   419→
   420→  // ── Reopen: after success, re-fetch detail properly ──
   421→  const handleReopen = async () => {
   422→    if (!detail) return
   423→    try { setSubmitting(true)
   424→      const res = await fetch(`/api/folio/customers/${detail.id}/reopen`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify({}) })
   425→      if (res.ok) {
   426→        notify({ type: 'success', title: 'Reopened' })
   427→        // Re-fetch detail to get updated state
   428→        await fetchDetail(detail.id, true)
   429→        await fetchMasterFolioEntries(detail.id)
   430→        fetchCustomers(true)
   431→        onRefreshData()
   432→      }
   433→      else { const e = await res.json(); notify({ type: 'error', title: 'Error', message: e.error || 'Failed' }) }
   434→    } catch { /* ignore */ } finally { setSubmitting(false) }
   435→  }
   436→
   437→  const handleDelete = async () => {
   438→    if (!detail) return
   439→    try { setSubmitting(true)
   440→      const res = await fetch(`/api/folio/customers/${detail.id}`, { method: 'DELETE', headers: hdrRef.current })
   441→      if (res.ok) { notify({ type: 'success', title: 'Archived' }); setDeleteOpen(false); goBack(); fetchCustomers(); onRefreshData() }
   442→      else notify({ type: 'error', title: 'Error', message: 'Failed' })
   443→    } catch { /* ignore */ } finally { setSubmitting(false) }
   444→  }
   445→
   446→  const handleAddDoc = async () => { if (!detail || !docForm.name) { notify({ type: 'error', title: 'Error', message: 'Name required' }); return } try { setSubmitting(true); const res = await fetch(`/api/folio/customers/${detail.id}/documents`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(docForm) }); if (res.ok) { notify({ type: 'success', title: 'Document Added' }); setDocForm({ ...DEFAULT_DOC_FORM }); await fetchDetail(detail.id); } } catch { /* ignore */ } finally { setSubmitting(false) } }
   447→  const handleDeleteDoc = async (docId: string) => { if (!detail) return; try { await fetch(`/api/folio/customers/${detail.id}/documents/${docId}`, { method: 'DELETE', headers: hdrRef.current }); notify({ type: 'success', title: 'Deleted' }); await fetchDetail(detail.id) } catch { /* ignore */ } }
   448→  const handleAddNote = async () => { if (!detail || !noteForm.content) { notify({ type: 'error', title: 'Error', message: 'Content required' }); return } try { setSubmitting(true); const res = await fetch(`/api/folio/customers/${detail.id}/notes`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify(noteForm) }); if (res.ok) { notify({ type: 'success', title: 'Note Added' }); setNoteForm({ ...DEFAULT_NOTE_FORM }); await fetchDetail(detail.id); } } catch { /* ignore */ } finally { setSubmitting(false) } }
   449→  const handleDeleteNote = async (noteId: string) => { if (!detail) return; try { await fetch(`/api/folio/customers/${detail.id}/notes/${noteId}`, { method: 'DELETE', headers: hdrRef.current }); notify({ type: 'success', title: 'Deleted' }); await fetchDetail(detail.id) } catch { /* ignore */ } }
   450→  const handleToggleTag = async (tagName: string) => { if (!detail) return; const has = customerTags.includes(tagName); try { if (has) { await fetch(`/api/folio/customers/${detail.id}/tags?tagName=${encodeURIComponent(tagName)}`, { method: 'DELETE', headers: hdrRef.current }); setCustomerTags(prev => prev.filter(t => t !== tagName)) } else { await fetch(`/api/folio/customers/${detail.id}/tags`, { method: 'POST', headers: { ...hdrRef.current, 'Content-Type': 'application/json' }, body: JSON.stringify({ tagName }) }); setCustomerTags(prev => [...prev, tagName]) } fetchDetail(detail.id, true); fetchCustomers(true) } catch { /* ignore */ } }
   451→
   452→  // ── Invoice / statement helpers ──
   453→  const fetchInvoiceData = async () => {
   454→    if (!detail) return
   455→    try { setInvoiceLoading(true); const params = new URLSearchParams(); if (invoiceFrom) params.set('from', invoiceFrom); if (invoiceTo) params.set('to', invoiceTo)
   456→      const res = await fetch(`/api/folio/customers/${detail.id}/statement?${params}`, { headers: hdrRef.current }); if (res.ok) setInvoiceData(await res.json()); else notify({ type: 'error', title: 'Error', message: 'Failed' })
   457→    } catch { /* ignore */ } finally { setInvoiceLoading(false) }
   458→  }
   459→  const fetchFolioStatement = async (folioId: string) => {
   460→    if (!detail) return
   461→    try { setFolioStatementData(null)
   462→      const res = await fetch(`/api/folio/customers/${detail.id}/statement?folioId=${folioId}`, { headers: hdrRef.current }); if (res.ok) setFolioStatementData(await res.json())
   463→    } catch { /* ignore */ }
   464→  }
   465→
   466→  // ── Print wrappers (async, using ref) ──
   467→  const bal = detail?.balance ?? 0
   468→  const isActive = detail?.status === 'Active'
   469→  const isCheckedOut = detail?.status === 'CheckedOut'
   470→  // Include all payment/settlement/refund entries from ALL folios for voucher history
   471→  const paymentEntries = masterFolioEntries.filter(e => e.type === 'Payment' || e.type === 'Settlement' || e.type === 'Refund')
   472→  const selectedEntryObj = selectedEntries.size === 1 ? folioEntries.find(e => e.id === Array.from(selectedEntries)[0]) : null
   473→
   474→  const doPrintReceipt = async () => {
   475→    // After checkout, folioEntries may be empty — use masterFolioEntries as fallback
   476→    const entriesToPrint = folioEntries.length > 0 ? folioEntries : masterFolioEntries
   477→    await _printReceipt(detail!, entriesToPrint, hdrRef as any)
   478→  }
   479→  const doPrintInvoice = () => { _printInvoice(detail!, invoiceData, invoiceFrom, invoiceTo) }
   480→  const doPrintSingleFolio = () => { _printSingleFolio(detail!, folioStatementData) }
   481→  const doPrintVoucherHistory = async () => { await _printVoucherHistory(detail!, paymentEntries, hdrRef as any) }
   482→
   483→  // ── Thermal Print wrappers ──
   484→  const doThermalPrintReceipt = async () => {
   485→    const entriesToPrint = folioEntries.length > 0 ? folioEntries : masterFolioEntries
   486→    await _thermalPrintReceipt(detail!, entriesToPrint, hdrRef as any)
   487→  }
   488→  const doThermalPrintInvoice = () => { _thermalPrintInvoice(detail!, invoiceData, invoiceFrom, invoiceTo) }
   489→  const doThermalPrintFolio = () => { _thermalPrintFolioStatement(detail!, folioStatementData) }
   490→  const doThermalPrintVoucher = async () => { await _thermalPrintVoucherHistory(detail!, paymentEntries, hdrRef as any) }
   491→
   492→  // ── RENDER ──
   493→  return (
   494→    <FolioErrorBoundary>
   495→    <TooltipProvider delayDuration={200}>
   496→    <>
   497→    {!canAccess && <ReadOnlyBanner module="Customer Folio" noAccess />}
   498→
   499→    <div className={isFullscreen ? 'fixed inset-0 z-[60] flex flex-col' : ''}>
   500→      <div className={`flex flex-col ${isFullscreen ? 'h-full' : 'min-h-0'} border border-slate-200 dark:border-slate-700 ${isFullscreen ? 'border-0 rounded-none' : 'rounded-lg'} overflow-hidden bg-white dark:bg-slate-900 shadow-sm`}>
   501→
   502→        {/* ── HEADER BAR ── */}
   503→        <div className="bg-white dark:bg-slate-900 border-b border-slate-200 dark:border-slate-700 px-4 py-2.5 flex items-center justify-between shrink-0">
   504→          <div className="flex items-center gap-3">
   505→            <div className="w-9 h-9 rounded-xl bg-gradient-to-br from-emerald-600 to-emerald-700 flex items-center justify-center shadow-sm shadow-emerald-600/20">
   506→              <span className="text-white text-sm font-bold">CF</span>
   507→            </div>
   508→            <div>
   509→              <h2 className="text-sm font-bold text-slate-800 dark:text-slate-100">Customer Folio</h2>
   510→              <p className="text-[10px] text-slate-400 dark:text-slate-500">Guest profiles &amp; billing management</p>
   511→            </div>
   512→          </div>
   513→          <div className="flex items-center gap-2">
   514→            <button onClick={() => setIsFullscreen(!isFullscreen)} className="text-[10px] text-slate-500 dark:text-slate-400 hover:text-slate-700 dark:hover:text-slate-200 px-2 py-1 rounded border border-slate-200 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-800 transition-colors">
   515→              {isFullscreen ? 'Exit Full' : 'Fullscreen'}
   516→            </button>
   517→          </div>
   518→        </div>
   519→
   520→        {/* ── MAIN BODY ── */}
   521→        <div className="flex flex-col flex-1 min-h-0">
   522→          {/* CUSTOMER PROFILE GRID VIEW (when no customer selected) */}
   523→          {!selectedId && (
   524→            <CustomerFolioProfileGrid
   525→              customers={customers} loading={loading} search={search} setSearch={setSearch}
   526→              filter={filter} setFilter={setFilter} sort={sort} setSort={setSort}
   527→              canCreate={canCreate} onAddCustomer={() => { resetForm(); setAddOpen(true) }}
   528→              onSelectCustomer={openCustomer}
   529→              onReports={() => setReportsOpen(true)}
   530→            />
   531→          )}
   532→
   533→          {/* CUSTOMER DETAIL VIEW (when customer selected) — full width */}
   534→          {selectedId && detail && detailLoading && !folioEntries.length ? (
   535→            <div className="flex-1 flex flex-col items-center justify-center">
   536→              <Loader2 className="size-8 animate-spin text-emerald-500 mb-3" />
   537→              <p className="text-sm text-muted-foreground">Loading customer details...</p>
   538→            </div>
   539→          ) : selectedId && detail ? (
   540→            <CustomerFolioManagement
   541→              detail={detail} allFolios={allFolios} folioEntries={folioEntries}
   542→              selectedEntries={selectedEntries} selectedFolioId={selectedFolioId}
   543→              folioEntriesLoading={folioEntriesLoading} submitting={submitting}
   544→              isAdmin={isAdmin} isActive={isActive} isCheckedOut={isCheckedOut}
   545→              folioStatementData={folioStatementData}
   546→              masterFolioEntries={masterFolioEntries}
   547→              onCheckoutSettlement={handleCheckoutSettlement}
   548→              goBack={goBack} setSelectedFolioId={setSelectedFolioId}
   549→              setSelectedEntries={setSelectedEntries} fetchFolioEntries={fetchFolioEntries}
   550→              toggleEntrySelection={toggleEntrySelection} toggleAllEntries={toggleAllEntries}
   551→              handleCreateFolio={handleCreateFolio} handleReopen={handleReopen}
   552→              printReceipt={doPrintReceipt} printSingleFolio={doPrintSingleFolio}
   553→              fetchFolioStatement={fetchFolioStatement}
   554→              onVoucher={() => setVoucherOpen(true)}
   555→              onInvoice={() => { setInvoiceFrom(''); setInvoiceTo(''); setInvoiceData(null); fetchInvoiceData(); setInvoiceOpen(true) }}
   556→              onCharge={() => setChargeOpen(true)}
   557→              onSettlement={() => { setPaymentForm({ ...DEFAULT_PAYMENT_FORM }); setSettlingMode(true); setPaymentOpen(true) }}
   558→              onPayment={() => { setPaymentForm({ ...DEFAULT_PAYMENT_FORM }); setSettlingMode(false); setPaymentOpen(true) }}
   559→              onCheckout={() => setCheckoutOpen(true)}
   560→              onRefund={() => { setRefundForm({ ...DEFAULT_REFUND_FORM }); setRefundOpen(true) }}
   561→              onDiscount={() => { setDiscountForm({ ...DEFAULT_DISCOUNT_FORM }); setDiscountOpen(true) }}
   562→              onTransfer={() => setTransferOpen(true)}
   563→              onShift={() => setShiftOpen(true)}
   564→              onSplit={() => { if (selectedEntries.size === 1) { setSplitAmountForm({ ...DEFAULT_SPLIT_AMOUNT_FORM }); setSplitAmountOpen(true) } else notify({ type: 'info', title: 'Info', message: 'Select 1 entry to split' }) }}
   565→              onRemove={() => setRemoveOpen(true)}
   566→              onEdit={() => { populateEditForm(); setEditOpen(true) }}
   567→              onDetails={() => setDetailsSheetOpen(true)}
   568→              onTags={() => setTagsOpen(true)}
   569→              onNotes={() => setNoteOpen(true)}
   570→              onDocs={() => setDocOpen(true)}
   571→              onDelete={() => setDeleteOpen(true)}
   572→              onReports={() => setReportsOpen(true)}
   573→              onHistory={() => setHistoryOpen(true)}
   574→              onDeleteFolio={(folioId: string) => { setFolioToDelete(folioId); setDeleteFolioOpen(true) }}
   575→              isFullscreen={isFullscreen}
   576→              thermalPrintReceipt={doThermalPrintReceipt}
   577→              thermalPrintInvoice={doThermalPrintInvoice}
   578→              thermalPrintFolio={doThermalPrintFolio}
   579→              thermalPrintVoucher={doThermalPrintVoucher}
   580→            />
   581→          ) : null}
   582→        </div>
   583→      </div>
   584→    </div>
   585→
   586→      {/* All Dialogs */}
   587→        <CustomerFolioDialogs
   588→          detail={detail || null} allFolios={allFolios} selectedEntries={selectedEntries}
   589→          submitting={submitting} settlingMode={settlingMode} selectedFolioId={selectedFolioId}
   590→          bal={bal} selectedEntryObj={selectedEntryObj} isActive={isActive}
   591→          isCheckedOut={isCheckedOut} invoiceData={invoiceData} invoiceLoading={invoiceLoading}
   592→          folioStatementData={folioStatementData} customerTags={customerTags}
   593→          transferResults={transferResults} paymentEntries={paymentEntries}
   594→          addOpen={addOpen} editOpen={editOpen} chargeOpen={chargeOpen}
   595→          paymentOpen={paymentOpen} refundOpen={refundOpen} transferOpen={transferOpen}
   596→          discountOpen={discountOpen} docOpen={docOpen} noteOpen={noteOpen}
   597→          tagsOpen={tagsOpen} checkoutOpen={checkoutOpen} deleteOpen={deleteOpen}
   598→          splitAmountOpen={splitAmountOpen} shiftOpen={shiftOpen} removeOpen={removeOpen}
   599→          voucherOpen={voucherOpen} invoiceOpen={invoiceOpen} detailsSheetOpen={detailsSheetOpen}
   600→          form={form} chargeForm={chargeForm} paymentForm={paymentForm}
   601→          refundForm={refundForm} transferForm={transferForm} discountForm={discountForm}
   602→          docForm={docForm} noteForm={noteForm} splitAmountForm={splitAmountForm}
   603→          invoiceFrom={invoiceFrom} invoiceTo={invoiceTo} shiftTargetFolioId={shiftTargetFolioId}
   604→          duplicate={duplicate}
   605→          setAddOpen={setAddOpen} setEditOpen={setEditOpen} setChargeOpen={setChargeOpen}
   606→          setPaymentOpen={setPaymentOpen} setRefundOpen={setRefundOpen} setTransferOpen={setTransferOpen}
   607→          setDiscountOpen={setDiscountOpen} setDocOpen={setDocOpen} setNoteOpen={setNoteOpen}
   608→          setTagsOpen={setTagsOpen} setCheckoutOpen={setCheckoutOpen} setDeleteOpen={setDeleteOpen}
   609→          setSplitAmountOpen={setSplitAmountOpen} setShiftOpen={setShiftOpen} setRemoveOpen={setRemoveOpen}
   610→          setVoucherOpen={setVoucherOpen} setInvoiceOpen={setInvoiceOpen} setDetailsSheetOpen={setDetailsSheetOpen}
   611→          setForm={setForm} setChargeForm={setChargeForm} setPaymentForm={setPaymentForm}
   612→          setRefundForm={setRefundForm} setTransferForm={setTransferForm} setDiscountForm={setDiscountForm}
   613→          setDocForm={setDocForm} setNoteForm={setNoteForm} setSplitAmountForm={setSplitAmountForm}
   614→          setInvoiceFrom={setInvoiceFrom} setInvoiceTo={setInvoiceTo} setShiftTargetFolioId={setShiftTargetFolioId}
   615→          setDuplicate={setDuplicate}
   616→          handleCreate={handleCreate} handleUpdate={handleUpdate} handleCharge={handleCharge}
   617→          handlePayment={handlePayment} handleSettle={handleSettle} handleRefund={handleRefund}
   618→          handleDiscount={handleDiscount} handleTransfer={handleTransfer} handleShift={handleShift}
   619→          handleSplitAmount={handleSplitAmount} handleRemove={handleRemove} handleCheckout={handleCheckout}
   620→          handleDelete={handleDelete} handleAddDoc={handleAddDoc} handleAddNote={handleAddNote}
   621→          handleToggleTag={handleToggleTag} handleDeleteDoc={handleDeleteDoc} handleDeleteNote={handleDeleteNote}
   622→          fetchInvoiceData={fetchInvoiceData} fetchFolioStatement={fetchFolioStatement}
   623→          printInvoice={doPrintInvoice} printSingleFolio={doPrintSingleFolio} printVoucherHistory={doPrintVoucherHistory}
   624→          closeDialogs={closeDialogs} searchTransferTargets={searchTransferTargets}
   625→          checkoutSettlementData={checkoutSettlementData} checkoutSettlementOpen={checkoutSettlementOpen}
   626→          splitPayments={splitPayments} setCheckoutSettlementOpen={setCheckoutSettlementOpen}
   627→          setSplitPayments={setSplitPayments}
   628→          handleMultiFolioSettlement={handleMultiFolioSettlement} handleAutoSettleAll={handleAutoSettleAll}
   629→          deleteFolioOpen={deleteFolioOpen} folioToDelete={folioToDelete}
   630→          setDeleteFolioOpen={setDeleteFolioOpen} handleDeleteFolio={handleDeleteFolio}
   631→          onPrintReceipt={doPrintReceipt}
   632→          onThermalPrintReceipt={doThermalPrintReceipt}
   633→          onThermalPrintInvoice={doThermalPrintInvoice}
   634→          onThermalPrintVoucher={doThermalPrintVoucher}
   635→          historyOpen={historyOpen}
   636→          setHistoryOpen={setHistoryOpen}
   637→        />
   638→
   639→        {/* Reports Sheet */}
   640→        {reportsOpen && (
   641→          <CustomerFolioReports
   642→            onClose={() => setReportsOpen(false)}
   643→            hdrRef={hdrRef as any}
   644→            canAccess={canAccess}
   645→          />
   646→        )}
   647→    </>
   648→    </TooltipProvider>
   649→    </FolioErrorBoundary>
   650→  )
   651→}
