   170→  invoiceData, invoiceLoading, folioStatementData, customerTags,

   172→  addOpen, editOpen, chargeOpen, paymentOpen, refundOpen, transferOpen,
   173→  discountOpen, docOpen, noteOpen, tagsOpen, checkoutOpen, deleteOpen,

   189→  handleToggleTag, handleDeleteDoc, handleDeleteNote,
   190→  fetchInvoiceData, fetchFolioStatement, printInvoice, printCompleteInvoice, printSingleFolio, printVoucherHistory,
   191→  closeDialogs, searchTransferTargets,
   192→  deleteFolioOpen, folioToDelete,
   193→  setDeleteFolioOpen, handleDeleteFolio,
   194→  // Enhanced props
   195→  apiHeaders, onRefresh, canDelete = false, isAdmin = false, onPrintReceipt, onThermalPrintReceipt,
   196→  onThermalPrintInvoice, onThermalPrintVoucher,
   197→  lastCreatedDocNo, lastCreatedType,
   198→  // History
   199→  historyOpen, setHistoryOpen, onPrintHistoryFolio,
   200→}: DialogsProps) {
   201→
   202→  // ─── Internal State ───
   203→  const [invoiceNumber, setInvoiceNumber] = React.useState('')
   204→  const [invNumLoading, setInvNumLoading] = React.useState(false)
   205→  const [localSubmitting, setLocalSubmitting] = React.useState(false)
   206→  const [entrySuccess, setEntrySuccess] = React.useState<{ show: boolean; docNo: string; type: string } | null>(null)
   207→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   208→  // ─── History state ───
   209→  const [historyLoading, setHistoryLoading] = React.useState(false)
   210→  const [customerHistory, setCustomerHistory] = React.useState<CustomerHistory | null>(null)
   211→  const [expandedFolioId, setExpandedFolioId] = React.useState<string | null>(null)
   212→  const [selectedVoucherIds, setSelectedVoucherIds] = React.useState<Set<string>>(new Set())
   213→
   214→  // Fetch history for the CURRENT customer
   215→  const fetchCustomerHistory = React.useCallback(async () => {
   216→    if (!detail || !apiHeaders) return
   217→    try {
   218→      setHistoryLoading(true)
   219→      const res = await fetch(`/api/folio/customers/${detail.id}/history`, { headers: apiHeaders })
   220→      if (res.ok) {
   221→        const data = await res.json() as CustomerHistory
   222→        setCustomerHistory(data)
   223→      }
   224→    } catch { notify({ type: 'error', title: 'Error', message: 'Failed to fetch checkout history' }) } finally { setHistoryLoading(false) }
   225→  }, [detail?.id, apiHeaders])
   226→
   227→  React.useEffect(() => {
   228→    if (historyOpen && detail && apiHeaders && fetchCustomerHistory) {
   229→      fetchCustomerHistory()
   230→    } else if (!historyOpen) {
   231→      setCustomerHistory(null)
   232→      setExpandedFolioId(null)
   233→    }
   234→  }, [historyOpen, detail?.id, fetchCustomerHistory, apiHeaders])
   235→
   236→  // ─── Print helpers for checkout records ───
   237→  const printCheckoutFolioBill = async (co: CheckoutRecord) => {
   238→    if (!co.transactions.length) return
   239→    let shopSettings: Record<string, string> = {}
   240→    try {
   241→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   242→    } catch { /* use defaults */ }
   243→    // Build a synthetic CustomerDetail-like object from CheckoutRecord
   244→    const fakeDetail = {
   245→      id: co.customerIdRef,
   246→      fullName: co.customerName,
   247→      customerId: co.customerIdRef,
   248→      mobile: co.customerMobile || '',
   249→      email: co.customerEmail || '',
   250→      address: co.customerAddress || '',
   251→      customerType: co.customerType || '',
   252→      companyName: '',
   253→      gstNumber: '',
   254→      panNumber: '',
   255→      folios: [],
   256→      activeFolio: null,
   257→      totalVisits: co.visitNumber || 0,
   258→      ledgerEntries: [],
   259→    } as unknown as CustomerDetail
   260→    const w = window.open('', '_blank')
   261→    if (!w) return
   262→    w.document.write(`<!DOCTYPE html><html><head><title>Folio Bill - ${co.checkoutNumber}</title><style>${INVOICE_CSS}</style></head><body>${buildHotelStyleInvoice(shopSettings, fakeDetail, co.transactions, co.invoiceNumber, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'FOLIO BILL', undefined, co.checkoutUser)}</body></html>`)
   263→    w.document.close(); w.print()
   264→  }
   265→
   266→  const printCheckoutInvoice = async (co: CheckoutRecord) => {
   267→    if (!co.transactions.length) return
   268→    let shopSettings: Record<string, string> = {}
   269→    try {
   270→      shopSettings = await fetch('/api/settings', { headers: apiHeaders || undefined }).then(r => r.ok ? r.json() : {})
   271→    } catch { /* use defaults */ }
   272→    // Build a synthetic CustomerDetail-like object from CheckoutRecord
   273→    const fakeDetail = {
   274→      id: co.customerIdRef,
   275→      fullName: co.customerName,
   276→      customerId: co.customerIdRef,
   277→      mobile: co.customerMobile || '',
   278→      email: co.customerEmail || '',
   279→      address: co.customerAddress || '',
   280→      customerType: co.customerType || '',
   281→      companyName: '',
   282→      gstNumber: '',
   283→      panNumber: '',
   284→      folios: [],
   285→      activeFolio: null,
   286→      totalVisits: co.visitNumber || 0,
   287→      ledgerEntries: [],
   288→    } as unknown as CustomerDetail
   289→    const w = window.open('', '_blank')
   290→    if (!w) return
   291→    w.document.write(`<!DOCTYPE html><html><head><title>Final Checkout Invoice - ${co.invoiceNumber}</title><style>${INVOICE_CSS}</style></head><body>${buildHotelStyleInvoice(shopSettings, fakeDetail, co.transactions, co.invoiceNumber, new Date(co.checkoutDate).toLocaleDateString('en-IN'), 'FINAL CHECKOUT INVOICE', { openingBalance: co.openingBalance, closingBalance: co.closingBalance }, co.checkoutUser)}</body></html>`)
   292→    w.document.close(); w.print()
   293→  }
   294→
   295→  // ─── Track checkout transition ───
   296→  const wasCheckedOut = React.useRef(isCheckedOut)
   297→  React.useEffect(() => {
   298→    if (isCheckedOut && !wasCheckedOut.current) {
   299→      setPostCheckoutOpen(true)
   300→    }
   301→    wasCheckedOut.current = isCheckedOut
   302→  }, [isCheckedOut])
   303→
   304→  // ─── Show TXN number when lastCreatedDocNo changes ───
   305→  React.useEffect(() => {
   306→    if (lastCreatedDocNo && lastCreatedType) {
   307→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   308→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   309→      return () => clearTimeout(timer)
   310→    }
   311→  }, [lastCreatedDocNo, lastCreatedType])
   312→
   313→  // ─── Items for charge selector (Req 5) ───
   314→  const [items, setItems] = React.useState<{ id: string; name: string; price: number }[]>([])
   315→  const [itemsLoading, setItemsLoading] = React.useState(false)
   316→  React.useEffect(() => {
   317→    if (chargeOpen && apiHeaders) {
   318→      setItemsLoading(true)
   319→      fetch('/api/items', { headers: apiHeaders }).then(r => r.ok ? r.json() : []).then((list: any[]) => {
   320→        setItems((list || []).map((it: any) => ({ id: it.id, name: it.name, price: it.price || 0 })))
   321→      }).catch(() => {}).finally(() => setItemsLoading(false))
   322→    }
   323→  }, [chargeOpen, apiHeaders])
   324→
   325→  // Handle item selection
   326→  const handleItemSelect = React.useCallback((itemId: string) => {
   327→    const item = items.find(it => it.id === itemId)
   328→    if (item) {
   329→      setChargeForm({
   330→        ...chargeForm,
   331→        selectedItem: itemId,
   332→        selectedItemPrice: String(item.price),
   333→        description: item.name,
   334→        amount: item.price > 0 ? String(item.price) : chargeForm.amount,
   335→      })
   336→    }
   337→  }, [items, chargeForm])
   338→
   339→  // ─── Fetch invoice number ───
   340→  const fetchInvoiceNumber = React.useCallback(async () => {
   341→    if (!detail || !apiHeaders) return
   342→    try {
   343→      setInvNumLoading(true)
   344→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   345→        method: 'POST',
   346→        headers: apiHeaders,
   347→      })
   348→      if (res.ok) {
   349→        const data = await res.json()
   350→        setInvoiceNumber(data.invoiceNumber)
   351→      }
   352→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   353→  }, [detail?.id, apiHeaders])
   354→
   355→  // ─── Fetch invoice number when invoice dialog opens ───
   356→  React.useEffect(() => {
   357→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber && fetchInvoiceNumber) {
   358→      fetchInvoiceNumber()
   359→    }
   360→  }, [invoiceOpen, detail?.id, apiHeaders, invoiceNumber, fetchInvoiceNumber])
   361→
   362→  React.useEffect(() => {
   363→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   364→      fetchInvoiceData()
   365→    }
   366→  }, [invoiceOpen, invoiceData, detail?.id, selectedFolioId])
   367→
   368→  // ─── Entry creation with TXN tracking ───
   369→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   370→    if (!detail || !selectedFolioId || !apiHeaders) {
   371→      return null
   372→    }
   373→    try {
   374→      setLocalSubmitting(true)
   375→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   376→        method: 'POST',
   377→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   378→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   379→      })
   380→      if (res.ok) {
   381→        const entry = await res.json()
   382→        const docNo = entry?.docNo || ''
   383→        setEntrySuccess({ show: true, docNo, type })
   384→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   385→        closeDialogs()
   386→        onRefresh?.()
   387→        setTimeout(() => setEntrySuccess(null), 4000)
   388→        return entry
   389→      } else {
   390→        const e = await safeJson(res)
   391→        notify({ type: 'error', title: 'Error', message: (e.error as string) || 'Failed' })
   392→        return null
   393→      }
   394→    } catch {
   395→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   396→      return null
   397→    } finally {
   398→      setLocalSubmitting(false)
   399→    }
   400→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   401→
   402→  // ─── Settlement with TXN tracking ───
   403→  const settleWithTxn = React.useCallback(async () => {
   404→    if (!detail || !paymentForm.amount || !apiHeaders) {
   405→      handleSettle()
   406→      return
   407→    }
   408→    try {
   409→      setLocalSubmitting(true)
   410→      // When master folio is selected, use autoAllocate to distribute across all folios
   411→      const isMaster = selectedFolioId === 'master'
   412→      const payload = isMaster
   413→        ? { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, autoAllocate: true }
   414→        : { amount: parseFloat(paymentForm.amount), mode: paymentForm.mode, reference: paymentForm.reference, description: paymentForm.description || `Settlement via ${paymentForm.mode}`, folioId: selectedFolioId }
   415→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   416→        method: 'POST',
   417→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   418→        body: JSON.stringify(payload),
   419→      })
   420→      if (res.ok) {
   421→        const data = await res.json()
   422→        // Auto-allocate may return settledFolios array instead of single ledgerEntry
   423→        const docNo = data?.ledgerEntry?.docNo || (data?.settledFolios?.[0] ? 'Multi-Folio' : '')
   424→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })
   425→        notify({ type: 'success', title: `Settlement ${docNo}`, message: data.message || `Settlement ${docNo} processed successfully` })
   426→        closeDialogs()
   427→        onRefresh?.()
   428→        setTimeout(() => setEntrySuccess(null), 4000)
   429→      } else {
   430→        const e = await safeJson(res)
   431→        notify({ type: 'error', title: 'Settlement Failed', message: (e.error as string) || 'Failed to process settlement' })
