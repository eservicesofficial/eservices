   354→  const handleSettle = async () => {

   359→      const activeFolio = allFolios.find(f => f.status === 'Active')
   360→      if (!activeFolio) {
   361→        notify({ type: 'error', title: 'Error', message: 'No active folio. Create a folio first.' })
