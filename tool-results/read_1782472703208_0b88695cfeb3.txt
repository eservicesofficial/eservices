   186→// ─── CSV Export with proper headers ───
   187→function downloadCSV(data: Record<string, any>[], columns: ColDef[], filename: string) {
   188→  if (!data.length) return
   189→  const headers = columns.map(c => c.label)
   190→  const csvRows = data.map(row =>
   191→    columns.map(col => {
   192→      const val = row[col.key]
   193→      if (Array.isArray(val)) return `"${val.join(', ').replace(/"/g, '""')}"`
   194→      if (col.format === 'currency' || col.format === 'amount') return Number(val) || 0
   195→      return `"${String(val ?? '').replace(/"/g, '""')}"`
   196→    }).join(',')
   197→  )
   198→  const csv = [headers.join(','), ...csvRows].join('\n')
   199→  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' })
   200→  const url = URL.createObjectURL(blob)
   201→  const a = document.createElement('a')
   202→  a.href = url
   203→  a.download = `${filename}.csv`
   204→  document.body.appendChild(a)
   205→  a.click()
   206→  document.body.removeChild(a)
   207→  URL.revokeObjectURL(url)
   208→}

   210→// ─── Format cell value ───
   211→function formatCellValue(value: unknown, col: ColDef): React.ReactNode {
   212→  if (value === null || value === undefined || value === '') return <span className="text-muted-foreground">—</span>
   213→  const strVal = String(value)
   214→
   215→  switch (col.format) {
   216→    case 'currency':
   217→      return <span className="font-medium tabular-nums">{fmtINR(Number(value))}</span>
   218→    case 'date':
   219→      return <span className="tabular-nums">{fmtDate(strVal)}</span>
   220→    case 'datetime':
   221→      return <span className="tabular-nums text-xs">{fmtDateTime(strVal)}</span>
   222→    case 'amount':
   223→      return <span className="font-medium tabular-nums">{fmtCur(Number(value))}</span>
   224→    case 'tags': {
   225→      const tags = Array.isArray(value) ? value : [value]
   226→      return (
   227→        <div className="flex flex-wrap gap-1">
   228→          {tags.map((tag: string, i: number) => (
   229→            <Badge key={i} variant="outline" className={`text-[10px] px-1.5 py-0 ${TAG_BADGE_MAP[tag] || ''}`}>
   230→              {tag}
   231→            </Badge>
   232→          ))}
   233→        </div>
   234→      )
   235→    }
   236→    case 'badge':
   237→      return (
   238→        <Badge variant="outline" className={`text-[10px] px-1.5 py-0 ${TYPE_BADGE_MAP[strVal] || 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400'}`}>
   239→          {strVal}
   240→        </Badge>
   241→      )
   242→    default:
   243→      return <span className="text-xs">{strVal}</span>
   244→  }
   245→}

   302→// ─── Build PDF HTML ───
   303→function buildReportPdfHtml(
   304→  reportTitle: string,
   305→  reportType: ReportType,
   306→  fromDate: string,
   307→  toDate: string,
   308→  summary: Record<string, any>,
   309→  data: Record<string, any>[],
   310→  columns: ColDef[],
   311→  settings: Record<string, string>,
   312→) {
   313→  const bizName = settings.shopName || 'Eservices'
   314→  const bizAddress = settings.shopAddress || ''
   315→  const gstNumber = settings.gstNumber || settings.gstin || ''
   316→
   317→  // Summary items
   318→  const summaryItems = Object.entries(summary)
   319→    .filter(([, v]) => typeof v === 'number' || typeof v === 'string')
   320→    .map(([k, v]) => {
   321→      const label = k.replace(/([A-Z])/g, ' $1').replace(/^./, s => s.toUpperCase())
   322→      const display = typeof v === 'number' ? '₹' + v.toLocaleString('en-IN', { maximumFractionDigits: 2 }) : v
   323→      return `<div class="summary-item"><div class="s-label">${label}</div><div class="s-value">${display}</div></div>`
   324→    })
   325→    .join('')
   326→
   327→  // Table header
   328→  const ths = columns.map(c => {
   329→    const align = c.align === 'right' ? 'align-right' : c.align === 'center' ? 'align-center' : ''
   330→    return `<th class="${align}">${c.label}</th>`
   331→  }).join('')
   332→
   333→  // Table rows
   334→  const trs = data.map((row, idx) => {
   335→    const cells = columns.map(col => {
   336→      let val = row[col.key]
   337→      if (Array.isArray(val)) val = val.join(', ')
   338→      if (col.format === 'currency' || col.format === 'amount') {
   339→        val = '₹' + (Number(val) || 0).toLocaleString('en-IN', { minimumFractionDigits: 2 })
   340→      } else if (col.format === 'date' && val) {
   341→        val = fmtDate(String(val))
   342→      } else if (col.format === 'datetime' && val) {
   343→        val = fmtDateTime(String(val))
   344→      }
   345→      const align = col.align === 'right' ? 'align-right' : col.align === 'center' ? 'align-center' : ''
   346→      return `<td class="${align}">${val ?? '—'}</td>`
   347→    }).join('')
   348→    return `<tr>${cells}</tr>`
   349→  }).join('')
   350→
   351→  const dateRange = (fromDate || toDate)
   352→    ? `${fromDate || 'All time'} to ${toDate || 'Today'}`
   353→    : 'All Time'
   354→
   355→  return `<!DOCTYPE html>
   356→<html><head><title>${reportTitle}</title>
   357→<style>${REPORT_PRINT_CSS}</style>
   358→</head><body>
   359→  <div class="report-header">
   360→    <h1>${bizName}</h1>
   361→    ${bizAddress ? `<div class="biz-address">${bizAddress}</div>` : ''}
   362→    ${gstNumber ? `<div class="gst-line">GSTIN: ${gstNumber}</div>` : ''}
   363→  </div>
   364→
   365→  <h2 style="margin:0 0 4px 0;font-size:16px;font-weight:700;color:#333">${reportTitle}</h2>
   366→  <div class="report-meta">
   367→    <span class="date-range">Period: ${dateRange}</span>
   368→    <span>Generated: ${new Date().toLocaleString('en-IN')}</span>
   369→  </div>
   370→
   371→  ${summaryItems ? `<div class="summary-box">${summaryItems}</div>` : ''}
   372→
   373→  <table>
   374→    <thead><tr>${ths}</tr></thead>
