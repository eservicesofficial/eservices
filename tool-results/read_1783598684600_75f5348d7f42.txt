    11→// ─── Helper: Convert number to Indian Rupee words ───
    12→function numberToIndianWords(num: number): string {
    13→  if (num === 0) return 'ZERO'
    14→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    15→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    16→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    17→  let words = ''
    18→  function convertBelowThousand(n: number): string {
    19→    let w = ''
    20→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    21→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    22→    else if (n > 0) { w += ones[n] }
    23→    return w
    24→  }
    25→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    26→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    27→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    28→  if (num > 0) { words += convertBelowThousand(num) }
    29→  return words.trim().toUpperCase() + ' ONLY'
    30→}

   200→// ─── Helper: Compute the most-used payment mode from credit entries ───
   201→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   202→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   203→  if (creditEntries.length === 0) return 'N/A'
   204→  const modeCounts: Record<string, number> = {}
   205→  for (const e of creditEntries) {
   206→    const mode = e.paymentMode || 'Cash'
   207→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   208→  }
   209→  let bestMode = 'Cash'
   210→  let bestAmount = 0
   211→  for (const [mode, amount] of Object.entries(modeCounts)) {
   212→    if (amount > bestAmount) {
   213→      bestAmount = amount
   214→      bestMode = mode
   215→    }
   216→  }
   217→  return bestMode
   218→}

   220→// ─── Helper: Build payment summary HTML ───
   221→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   222→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   223→  if (creditEntries.length === 0) {
   224→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   225→  }
   226→
   227→  const modeTotals: Record<string, number> = {}
   228→  for (const e of creditEntries) {
   229→    const mode = e.paymentMode || 'Cash'
   230→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   231→  }
   232→
   233→  const modes = Object.entries(modeTotals)
   234→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   235→
   236→  if (modes.length === 1) {
   237→    const [mode, amount] = modes[0]
   238→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   239→  }
   240→
   241→  const rows = modes.map(([mode, amount]) => {
   242→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   243→  }).join('')
   244→
   245→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   246→}

   248→// ─── Helper: Get stable document number from entries (never regenerate) ───
   249→// Searches ALL entries for the first non-empty docNo to avoid N/A
   250→function getStableDocId(entries: { docNo?: string; invoiceNo?: string; id?: string; createdAt?: string }[]): string {
   251→  if (entries.length === 0) return `DOC-${Date.now()}`
   252→  for (const e of entries) {
   253→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   254→  }
   255→  for (const e of entries) {
   256→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   257→  }
   258→  // Fallback: use entry ID truncated or timestamp — never show N/A
   259→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   260→  return fallback
   261→}

   769→// ─── Print Voucher / Payment History ───
   770→export async function printVoucherHistory(
   771→  detail: CustomerDetail,
   772→  paymentEntries: LedgerEntry[],
   773→  hdrRef: RefObject<Record<string, string>>
   774→) {
   775→  if (!detail) return
   776→  if (paymentEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No payment entries to print' }); return }
   777→  const settings = await fetchSettingsForPrint(hdrRef)
   778→  const now = new Date()
   779→
   780→  const sorted = [...paymentEntries].sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
   781→  const totalAmount = sorted.reduce((s, e) => s + (e.credit || 0), 0)
   782→
   783→  const docId = getStableDocId(sorted)
   784→
   785→  const rows = sorted.map((e, i) => {
   786→    const desc = stripInternalDescriptions(e.description || '—')
   787→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td>${desc}</td><td style="text-align:right;font-weight:700;color:#16a34a">${fmtINR(e.credit || 0)}</td><td style="text-align:center">${e.paymentMode || '—'}</td></tr>`
   788→  }).join('')
   789→
   790→  const w = openPrintWindow(`Voucher History - ${detail.fullName}`)
   791→  if (!w) return
   792→
   793→  w.document.write(
   794→    `<!DOCTYPE html><html><head><title>Payment Voucher</title><style>${PRINT_CSS}</style></head><body>` +
   795→    buildUnifiedHeader(settings, 'PAYMENT VOUCHER', docId) +
   796→    buildCustomerInfoBlock(detail) +
   797→    `<div class="datetime">Printed: ${fmtDateTime(now.toISOString())}</div>` +
   798→    `<table><tr><th>#</th><th>Date</th><th>Description</th><th style="text-align:right">Amount (₹)</th><th>Mode</th></tr>${rows}` +
   799→    `<tr class="totals"><td colspan="3" style="text-align:right;font-size:11px;color:#555">TOTAL</td><td style="color:#16a34a;font-weight:700">${fmtINR(totalAmount)}</td><td>${sorted.length} entries</td></tr></table>` +
   800→    buildPaymentSummary(sorted) +
   801→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Total Payments</div><div class="s-value" style="color:#16a34a">${fmtINR(totalAmount)}</div></div><div class="summary-item"><div class="s-label">Entries</div><div class="s-value">${sorted.length}</div></div></div>` +
   802→    buildFooter() +
   803→    `</body></html>`
   804→  )
   805→  w.document.close()
   806→  w.print()
   807→}

   811→interface ShopSettings {
   812→  shopName: string
   813→  shopAddress: string
   814→  shopPhone: string
   815→  shopGst: string
   816→  shopGstState: string
   817→}
   818→
   819→async function fetchShopSettings(hdrRef?: RefObject<Record<string, string>>): Promise<ShopSettings> {
   820→  try {
   821→    const headers = hdrRef?.current || undefined
   822→    const res = await fetch('/api/settings', headers ? { headers } : undefined)
   823→    if (res.ok) {
   824→      const db = (await res.json()) as Record<string, string>
   825→      return {
   826→        shopName: db.shopName || 'Eservices',
   827→        shopAddress: db.shopAddress || '',
   828→        shopPhone: db.shopPhone || '',
   829→        shopGst: db.gstNumber || db.gstin || '',
   830→        shopGstState: db.gstState || '',
   831→      }
   832→    }
   833→  } catch { /* ignore */ }
   834→  return { shopName: 'Eservices', shopAddress: '', shopPhone: '', shopGst: '', shopGstState: '' }
   835→}
   836→
   837→function openThermalPrintWindow(thermalHtml: string, title: string): Window | null {
   838→  const w = window.open('', '_blank', 'width=350,height=700')
   839→  if (!w) {
   840→    notify({ type: 'error', title: 'Error', message: 'Please allow pop-ups to print thermal receipts' })
   841→    return null
   842→  }
   843→  w.document.write(thermalHtml)
   844→  w.document.close()
   845→  setTimeout(() => {
   846→    try { if (!w.closed) w.print() } catch { /* window closed */ }
   847→  }, 600)
   848→  return w
   849→}

   851→// ─── Thermal Print Receipt ───
   852→export async function thermalPrintReceipt(
   853→  detail: CustomerDetail,
   854→  entries: LedgerEntry[],
   855→  hdrRef: RefObject<Record<string, string>>
   856→) {
   857→  if (!detail) return
   858→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   859→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   860→
   861→  const [tConfig, shop] = await Promise.all([
   862→    fetchThermalSettings(),
   863→    fetchShopSettings(hdrRef),
   864→  ])
   865→
   866→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   867→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   868→  const finalBal = activeEntries[activeEntries.length - 1].balance
   869→
   870→  const items: ThermalReceiptData['items'] = activeEntries.map(e => ({
   871→    name: stripInternalDescriptions(e.description || e.type || '—'),
   872→    qty: 1,
   873→    rate: e.debit > 0 ? e.debit : e.credit,
   874→    total: e.debit > 0 ? e.debit : e.credit,
   875→  }))
   876→
   877→  const actualPaymentMethod = computeMostUsedPaymentMode(activeEntries)
   878→
   879→  const receiptData: ThermalReceiptData = {
   880→    shopName: shop.shopName,
   881→    shopAddress: shop.shopAddress,
   882→    shopPhone: shop.shopPhone,
   883→    shopGst: shop.shopGst,
   884→    shopGstState: shop.shopGstState,
   885→    showGst: false,
   886→    invoiceNumber: detail.customerId,
   887→    date: fmtDate(new Date().toISOString()),
   888→    orderType: 'PAYMENT RECEIPT',
   889→    customerName: detail.fullName,
   890→    customerPhone: detail.mobile,
   891→    customerAddress: (detail as any).address || '',
   892→    items,
   893→    subtotal: totalDebits,
   894→    cgst: 0,
   895→    sgst: 0,
   896→    gstRate: 0,
   897→    deliveryCharge: 0,
   898→    discountAmount: 0,
   899→    grandTotal: totalDebits - totalCredits,
   900→    paidAmount: totalCredits,
   901→    balanceAmount: finalBal,
   902→    paymentMethod: actualPaymentMethod,
   903→    billStatus: finalBal <= 0 ? 'Paid' : 'Balance',
   904→  }
   905→
   906→  const thermalHtml = buildRichThermalHtml(receiptData, tConfig, `Receipt - ${detail.fullName}`)
   907→  openThermalPrintWindow(thermalHtml, `Receipt - ${detail.fullName}`)
   908→}
