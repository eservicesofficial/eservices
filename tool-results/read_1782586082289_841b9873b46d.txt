    68→// ─── Helper: Compute the most-used payment mode from credit entries ───
    69→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
    70→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    71→  if (creditEntries.length === 0) return 'N/A'
    72→  const modeCounts: Record<string, number> = {}
    73→  for (const e of creditEntries) {
    74→    const mode = e.paymentMode || 'Cash'
    75→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
    76→  }
    77→  let bestMode = 'Cash'
    78→  let bestAmount = 0
    79→  for (const [mode, amount] of Object.entries(modeCounts)) {
    80→    if (amount > bestAmount) {
    81→      bestAmount = amount
    82→      bestMode = mode
    83→    }
    84→  }
    85→  return bestMode
    86→}

    88→// ─── Helper: Build payment summary HTML ───
    89→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
    90→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
    91→  if (creditEntries.length === 0) {
    92→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
    93→  }
    94→
    95→  const modeTotals: Record<string, number> = {}
    96→  for (const e of creditEntries) {
    97→    const mode = e.paymentMode || 'Cash'
    98→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
    99→  }
   100→
   101→  const modes = Object.entries(modeTotals)
   102→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   103→
   104→  if (modes.length === 1) {
   105→    const [mode, amount] = modes[0]
   106→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   107→  }
   108→
   109→  const rows = modes.map(([mode, amount]) => {
   110→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   111→  }).join('')
   112→
   113→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   114→}

   116→// ─── Helper: Get stable document number from entries (never regenerate) ───
   117→function getStableDocId(entries: { docNo?: string; invoiceNo?: string }[]): string {
   118→  if (entries.length === 0) return 'N/A'

   198→// ─── Print Folio Invoice (formerly "Receipt") ───
   199→// Uses stable doc IDs from existing entries — NEVER regenerates numbers on reprint
   200→export async function printReceipt(
   201→  detail: CustomerDetail,
   202→  entries: LedgerEntry[],
   203→  hdrRef: RefObject<Record<string, string>>
   204→) {
   205→  if (!detail) return
   206→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   207→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   208→  const settings = await fetchSettingsForPrint(hdrRef)
   209→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   210→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   211→  const totalTax = activeEntries.reduce((s, e) => s + (e.tax || 0), 0)
   212→  const totalDiscounts = activeEntries.reduce((s, e) => s + (e.discount || 0), 0)
   213→  const finalBal = activeEntries[activeEntries.length - 1].balance
   214→  const now = new Date()
   215→  const rows = activeEntries.map((e, i) => {
   216→    const desc = stripInternalDescriptions(e.description || e.type || '—')
   217→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td style="font-size:9px;color:#666;font-family:monospace">${e.docNo || ''}</td><td>${e.type || ''}</td><td>${desc}</td><td style="text-align:right;color:#dc2626;font-weight:600">${e.debit > 0 ? fmtINR(e.debit) : '—'}</td><td style="text-align:right;color:#16a34a;font-weight:600">${e.credit > 0 ? fmtINR(e.credit) : '—'}</td><td style="font-weight:700;color:${e.balance > 0 ? '#dc2626' : e.balance < 0 ? '#16a34a' : '#666'}">${fmtINR(e.balance)}</td></tr>`
   218→  }).join('')
   219→
   220→  // Use stable doc ID from entries — never regenerate
   221→  const docId = getStableDocId(activeEntries)
   222→  const authorizedSignatory = (settings as Record<string, string>).authorizedSignatory || ''
   223→  const gstAddr = (detail as any).gstAddress || ''
   224→
   225→  const w = openPrintWindow(`Folio Invoice - ${detail.fullName}`)
   226→  if (!w) return
   227→
   228→  w.document.write(
   229→    `<!DOCTYPE html><html><head><title>Folio Invoice</title><style>${PRINT_CSS}</style></head><body>` +
   230→    buildUnifiedHeader(settings, 'FOLIO INVOICE', docId) +
   231→    `<div class="invoice-meta">
   232→      <div><div class="inv-label">Doc No.</div><div class="inv-value">${docId}</div></div>
   233→      <div style="text-align:center"><div class="inv-label">Date</div><div class="inv-value">${fmtDate(now.toISOString())}</div></div>
   234→      <div style="text-align:right"><div class="inv-label">Status</div><div class="inv-value">${detail.status === 'CheckedOut' ? 'Checked Out' : detail.status}</div></div>
   235→    </div>` +
   236→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">Customer ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +
   237→    (detail.gstNumber ? `<div style="margin:-8px 0 4px 0;font-size:11px;color:#666">Customer GSTIN: ${detail.gstNumber}</div>` : '') +
   238→    (gstAddr ? `<div style="margin:-4px 0 8px 0;font-size:11px;color:#666">GST Address: ${gstAddr}</div>` : '') +
   239→    `<hr class="divider">` +
   240→    `<table><tr><th>#</th><th>Date</th><th>TXN ID</th><th>Type</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   241→    `<tr class="totals"><td colspan="5" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(finalBal)}</td></tr></table>` +
   242→    buildPaymentSummary(activeEntries) +
   243→    `<div class="tax-summary"><div class="tax-item"><span class="tax-label">Total Tax: </span><span class="tax-val">${fmtINR(totalTax)}</span></div><div class="tax-item"><span class="tax-label">Discount: </span><span class="tax-val">${fmtINR(totalDiscounts)}</span></div><div class="tax-item"><span class="tax-label">Net Charges: </span><span class="tax-val">${fmtINR(totalDebits - totalCredits)}</span></div></div>` +
   244→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Total Charges</div><div class="s-value" style="color:#dc2626">${fmtINR(totalDebits)}</div></div><div class="summary-item"><div class="s-label">Total Payments</div><div class="s-value" style="color:#16a34a">${fmtINR(totalCredits)}</div></div><div class="summary-item"><div class="s-label">Final Balance</div><div class="s-value" style="color:${finalBal > 0 ? '#dc2626' : finalBal < 0 ? '#16a34a' : '#666'}">${fmtINR(finalBal)}</div></div></div>` +
   245→    `<div class="signature-section">
   246→      <div class="signature-box"><div class="signature-line"><div class="sig-label">Customer Signature</div><div class="sig-sub">Authorized Representative</div></div></div>
   247→      <div class="signature-box"><div class="signature-line"><div class="sig-label">${authorizedSignatory || 'Authorized Signatory'}</div><div class="sig-sub">${authorizedSignatory ? 'Authorized Signatory' : 'For ' + (settings.shopName || 'Eservices')}</div></div></div>
   248→    </div>` +
   249→    buildFooter() +
   250→    `</body></html>`
   251→  )
   252→  w.document.close()
   253→  w.print()
   254→}
