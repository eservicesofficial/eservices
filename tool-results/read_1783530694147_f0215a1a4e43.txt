    38→// ─── Section-level Error Boundary ───
    39→// Catches rendering errors in ANY section so they don't crash the entire page
    40→class SectionErrorBoundary extends React.Component<{ children: React.ReactNode; sectionName: string }, { hasError: boolean; error: Error | null }> {
    41→  state = { hasError: false, error: null as Error | null }
    42→  static getDerivedStateFromError(error: Error) {
    43→    return { hasError: true, error }
    44→  }
    45→  componentDidCatch(error: Error, info: React.ErrorInfo) {
    46→    console.error(`[SectionErrorBoundary:${this.props.sectionName}]`, error?.message, error?.stack, info?.componentStack)
    47→  }
    48→  handleRetry = () => {
    49→    this.setState({ hasError: false, error: null })
    50→  }
    51→  render() {
    52→    if (this.state.hasError) {
    53→      const errMsg = this.state.error?.message || 'Unknown error'
    54→      const errName = this.state.error?.name || 'Error'
    55→      const stackLines = (this.state.error?.stack || '').split('\n').filter((l: string) => l.includes('src/')).slice(0, 2).map((l: string) => l.trim())
    56→      return (
    57→        <div className="flex flex-col items-center justify-center py-20 px-6">
    58→          <div className="flex size-14 items-center justify-center rounded-2xl bg-red-100 dark:bg-red-900/30 mb-4">
    59→            <AlertCircle className="size-7 text-red-500" />
    60→          </div>
    61→          <h3 className="text-base font-bold text-slate-800 dark:text-slate-200 mb-1">Section Error: {this.props.sectionName}</h3>
    62→          <p className="text-xs text-red-500 dark:text-red-400 mb-1 font-mono">{errName}: {errMsg}</p>
    63→          {stackLines.length > 0 && (
    64→            <pre className="text-[10px] text-slate-400 dark:text-slate-500 max-w-lg text-center overflow-x-auto mb-4 whitespace-pre-wrap mt-1">{stackLines.join('\n')}</pre>
    65→          )}
    66→          <Button variant="outline" size="sm" onClick={this.handleRetry} className="gap-1.5 mt-2">
    67→            <RefreshCcw className="size-3" /> Try Again
    68→          </Button>
    69→        </div>
    70→      )
    71→    }
    72→    return this.props.children
    73→  }
    74→}
