    12→    12→function numberToIndianWords(num: number): string {
    13→    13→  if (num === 0) return 'ZERO'
    14→    14→  const ones = ['', 'One', 'Two', 'Three', 'Four', 'Five', 'Six', 'Seven', 'Eight', 'Nine', 'Ten', 'Eleven', 'Twelve', 'Thirteen', 'Fourteen', 'Fifteen', 'Sixteen', 'Seventeen', 'Eighteen', 'Nineteen']
    15→    15→  const tens = ['', '', 'Twenty', 'Thirty', 'Forty', 'Fifty', 'Sixty', 'Seventy', 'Eighty', 'Ninety']
    16→    16→  if (num < 0) return 'MINUS ' + numberToIndianWords(-num)
    17→    17→  let words = ''
    18→    18→  function convertBelowThousand(n: number): string {
    19→    19→    let w = ''
    20→    20→    if (n >= 100) { w += ones[Math.floor(n / 100)] + ' Hundred'; n %= 100; if (n > 0) w += ' and ' }
    21→    21→    if (n >= 20) { w += tens[Math.floor(n / 10)]; n %= 10; if (n > 0) w += ' ' + ones[n] }
    22→    22→    else if (n > 0) { w += ones[n] }
    23→    23→    return w
    24→    24→  }
    25→    25→  if (num >= 10000000) { words += convertBelowThousand(Math.floor(num / 10000000)) + ' Crore '; num %= 10000000 }
    26→    26→  if (num >= 100000) { words += convertBelowThousand(Math.floor(num / 100000)) + ' Lakh '; num %= 100000 }
    27→    27→  if (num >= 1000) { words += convertBelowThousand(Math.floor(num / 1000)) + ' Thousand '; num %= 1000 }
    28→    28→  if (num > 0) { words += convertBelowThousand(num) }
    29→    29→  return words.trim().toUpperCase() + ' ONLY'
    30→    30→}

   200→   200→// ─── Helper: Compute the most-used payment mode from credit entries ───
   201→   201→function computeMostUsedPaymentMode(entries: { credit?: number; paymentMode?: string }[]): string {
   202→   202→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   203→   203→  if (creditEntries.length === 0) return 'N/A'
   204→   204→  const modeCounts: Record<string, number> = {}
   205→   205→  for (const e of creditEntries) {
   206→   206→    const mode = e.paymentMode || 'Cash'
   207→   207→    modeCounts[mode] = (modeCounts[mode] || 0) + (e.credit || 0)
   208→   208→  }
   209→   209→  let bestMode = 'Cash'
   210→   210→  let bestAmount = 0
   211→   211→  for (const [mode, amount] of Object.entries(modeCounts)) {
   212→   212→    if (amount > bestAmount) {
   213→   213→      bestAmount = amount
   214→   214→      bestMode = mode
   215→   215→    }
   216→   216→  }
   217→   217→  return bestMode
   218→   218→}

   220→   220→// ─── Helper: Build payment summary HTML ───
   221→   221→function buildPaymentSummary(entries: { credit?: number; paymentMode?: string }[]): string {
   222→   222→  const creditEntries = entries.filter(e => (e.credit || 0) > 0)
   223→   223→  if (creditEntries.length === 0) {
   224→   224→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:11px;color:#888">No payments recorded</div></div>`
   225→   225→  }
   226→   226→
   227→   227→  const modeTotals: Record<string, number> = {}
   228→   228→  for (const e of creditEntries) {
   229→   229→    const mode = e.paymentMode || 'Cash'
   230→   230→    modeTotals[mode] = (modeTotals[mode] || 0) + (e.credit || 0)
   231→   231→  }
   232→   232→
   233→   233→  const modes = Object.entries(modeTotals)
   234→   234→  const grandTotal = modes.reduce((s, [, amt]) => s + amt, 0)
   235→   235→
   236→   236→  if (modes.length === 1) {
   237→   237→    const [mode, amount] = modes[0]
   238→   238→    return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><div style="padding:10px 12px;font-size:12px;font-weight:600;color:#374151">Payment Mode: <span style="color:#16a34a">${mode}</span> — ${fmtINR(amount)}</div></div>`
   239→   239→  }
   240→   240→
   241→   241→  const rows = modes.map(([mode, amount]) => {
   242→   242→    return `<tr><td style="text-align:left">${mode}</td><td style="text-align:right;font-weight:600;color:#374151">${fmtINR(amount)}</td></tr>`
   243→   243→  }).join('')
   244→   244→
   245→   245→  return `<div class="payment-summary"><div class="ps-title">Payment Summary</div><table><tr><th style="text-align:left">Mode</th><th style="text-align:right">Amount</th></tr>${rows}<tr class="ps-total"><td style="text-align:left">Total</td><td style="text-align:right">${fmtINR(grandTotal)}</td></tr></table></div>`
   246→   246→}

   252→   252→  for (const e of entries) {
   253→   253→    if (e.docNo && e.docNo.trim() !== '' && e.docNo !== 'N/A') return e.docNo
   254→   254→  }
   255→   255→  for (const e of entries) {
   256→   256→    if (e.invoiceNo && e.invoiceNo.trim() !== '' && e.invoiceNo !== 'N/A') return e.invoiceNo
   257→   257→  }
   258→   258→  // Fallback: use entry ID truncated or timestamp — never show N/A
   259→   259→  const fallback = entries[0].id ? `DOC-${entries[0].id.slice(0, 8).toUpperCase()}` : `DOC-${Date.now()}`
   260→   260→  return fallback
