   129→   129→  form, chargeForm, paymentForm, refundForm, transferForm, discountForm,

   154→   154→

   158→   158→  const [localSubmitting, setLocalSubmitting] = React.useState(false)

   160→   160→  const [postCheckoutOpen, setPostCheckoutOpen] = React.useState(false)
   161→   161→  const [splitMode, setSplitMode] = React.useState(false)
   162→   162→  const [localSplitItems, setLocalSplitItems] = React.useState<{ method: string; amount: string; reference: string }[]>([
   163→   163→    { method: 'Cash', amount: '', reference: '' },
   164→   164→  ])
   165→   165→
   166→   166→  // ─── Track checkout transition ───
   167→   167→  const wasCheckedOut = React.useRef(isCheckedOut)
   168→   168→  React.useEffect(() => {
   169→   169→    if (isCheckedOut && !wasCheckedOut.current) {
   170→   170→      setPostCheckoutOpen(true)
   171→   171→    }

   174→   174→
   175→   175→  // ─── Show TXN number when lastCreatedDocNo changes ───

   177→   177→    if (lastCreatedDocNo && lastCreatedType) {
   178→   178→      setEntrySuccess({ show: true, docNo: lastCreatedDocNo, type: lastCreatedType })
   179→   179→      const timer = setTimeout(() => setEntrySuccess(null), 4000)
   180→   180→      return () => clearTimeout(timer)
   181→   181→    }
   182→   182→  }, [lastCreatedDocNo, lastCreatedType])
   183→   183→
   184→   184→  // ─── Fetch invoice number ───
   185→   185→  const fetchInvoiceNumber = React.useCallback(async () => {
   186→   186→    if (!detail || !apiHeaders) return

   188→   188→      setInvNumLoading(true)
   189→   189→      const res = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, {
   190→   190→        method: 'POST',
   191→   191→        headers: apiHeaders,
   192→   192→      })
   193→   193→      if (res.ok) {
   194→   194→        const data = await res.json()
   195→   195→        setInvoiceNumber(data.invoiceNumber)
   196→   196→      }
   197→   197→    } catch { /* ignore */ } finally { setInvNumLoading(false) }
   198→   198→  }, [detail?.id, apiHeaders])
   199→   199→
   200→   200→  // ─── Fetch invoice number when invoice dialog opens ───
   201→   201→  React.useEffect(() => {
   202→   202→    if (invoiceOpen && detail && apiHeaders && !invoiceNumber) {
   203→   203→      fetchInvoiceNumber()
   204→   204→    }
   205→   205→  }, [invoiceOpen, detail?.id, apiHeaders])
   206→   206→
   207→   207→  React.useEffect(() => {
   208→   208→    if (invoiceOpen && !invoiceData && detail && selectedFolioId) {
   209→   209→      fetchInvoiceData()
   210→   210→    }
   211→   211→  }, [invoiceOpen])
   212→   212→
   213→   213→  // ─── Entry creation with TXN tracking ───
   214→   214→  const createEntryWithTxn = React.useCallback(async (type: string, data: any) => {
   215→   215→    if (!detail || !selectedFolioId || !apiHeaders) {
   216→   216→      // Fallback to existing handler
   217→   217→      return null
   218→   218→    }
   219→   219→    try {
   220→   220→      setLocalSubmitting(true)
   221→   221→      const res = await fetch(`/api/folio/customers/${detail.id}/ledger`, {
   222→   222→        method: 'POST',
   223→   223→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   224→   224→        body: JSON.stringify({ type, folioId: selectedFolioId, ...data }),
   225→   225→      })
   226→   226→      if (res.ok) {
   227→   227→        const entry = await res.json()
   228→   228→        const docNo = entry?.docNo || ''
   229→   229→        setEntrySuccess({ show: true, docNo, type })
   230→   230→        notify({ type: 'success', title: `${type} ${docNo}`, message: `${type} entry ${docNo} created successfully` })
   231→   231→        closeDialogs()
   232→   232→        onRefresh?.()
   233→   233→        setTimeout(() => setEntrySuccess(null), 4000)
   234→   234→        return entry
   235→   235→      } else {
   236→   236→        const e = await res.json()
   237→   237→        notify({ type: 'error', title: 'Error', message: e.error || 'Failed' })
   238→   238→        return null
   239→   239→      }
   240→   240→    } catch {
   241→   241→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   242→   242→      return null
   243→   243→    } finally {
   244→   244→      setLocalSubmitting(false)
   245→   245→    }
   246→   246→  }, [detail?.id, selectedFolioId, apiHeaders, closeDialogs, onRefresh])
   247→   247→
   248→   248→  // ─── Settlement with TXN tracking ───
   249→   249→  const settleWithTxn = React.useCallback(async () => {
   250→   250→    if (!detail || !paymentForm.amount || !selectedFolioId || !apiHeaders) {
   251→   251→      handleSettle()
   252→   252→      return
   253→   253→    }
   254→   254→    try {
   255→   255→      setLocalSubmitting(true)
   256→   256→      const res = await fetch(`/api/folio/customers/${detail.id}/settle`, {
   257→   257→        method: 'POST',
   258→   258→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   259→   259→        body: JSON.stringify({
   260→   260→          amount: parseFloat(paymentForm.amount),
   261→   261→          mode: paymentForm.mode,
   262→   262→          reference: paymentForm.reference,
   263→   263→          description: paymentForm.description,
   264→   264→          folioId: selectedFolioId,
   265→   265→        }),
   266→   266→      })
   267→   267→      if (res.ok) {
   268→   268→        const data = await res.json()
   269→   269→        const docNo = data?.ledgerEntry?.docNo || ''
   270→   270→        setEntrySuccess({ show: true, docNo, type: 'Settlement' })
   271→   271→        notify({ type: 'success', title: `Settlement ${docNo}`, message: `Settlement ${docNo} processed successfully` })
   272→   272→        closeDialogs()
   273→   273→        onRefresh?.()
   274→   274→        setTimeout(() => setEntrySuccess(null), 4000)
   275→   275→      } else {
   276→   276→        const e = await res.json()
   277→   277→        notify({ type: 'error', title: 'Error', message: e.error || 'Failed' })
   278→   278→      }
   279→   279→    } catch {
   280→   280→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   281→   281→    } finally {
   282→   282→      setLocalSubmitting(false)
   283→   283→    }
   284→   284→  }, [detail?.id, selectedFolioId, apiHeaders, paymentForm, handleSettle, closeDialogs, onRefresh])
   285→   285→
   286→   286→  // ─── Transfer with TXN tracking ───
   287→   287→  const transferWithTxn = React.useCallback(async () => {
   288→   288→    if (!detail || !transferForm.targetCustomerId || !transferForm.amount || !apiHeaders) {
   289→   289→      handleTransfer()
   290→   290→      return
   291→   291→    }
   292→   292→    try {
   293→   293→      setLocalSubmitting(true)
   294→   294→      const res = await fetch(`/api/folio/customers/${detail.id}/transfer`, {
   295→   295→        method: 'POST',
   296→   296→        headers: { ...apiHeaders, 'Content-Type': 'application/json' },
   297→   297→        body: JSON.stringify({
   298→   298→          targetCustomerId: transferForm.targetCustomerId,
   299→   299→          amount: parseFloat(transferForm.amount),
   300→   300→          description: transferForm.description,
   301→   301→        }),
   302→   302→      })
   303→   303→      if (res.ok) {
   304→   304→        const data = await res.json()
   305→   305→        const docNo = data?.source?.entry?.docNo || ''
   306→   306→        setEntrySuccess({ show: true, docNo, type: 'Transfer' })
   307→   307→        notify({ type: 'success', title: `Transfer ${docNo}`, message: `Transfer ${docNo} processed successfully` })
   308→   308→        closeDialogs()
   309→   309→        onRefresh?.()
   310→   310→        setTimeout(() => setEntrySuccess(null), 4000)
   311→   311→      } else {
   312→   312→        const e = await res.json()
   313→   313→        notify({ type: 'error', title: 'Error', message: e.error || 'Failed' })
   314→   314→      }
   315→   315→    } catch {
   316→   316→      notify({ type: 'error', title: 'Error', message: 'Network error' })
   317→   317→    } finally {
   318→   318→      setLocalSubmitting(false)
   319→   319→    }
   320→   320→  }, [detail?.id, apiHeaders, transferForm, handleTransfer, closeDialogs, onRefresh])
   321→   321→
   322→   322→  // ─── Split payment helpers for checkout ───
   323→   323→  const addSplitItem = () => {
   324→   324→    setLocalSplitItems(prev => [...prev, { method: 'Cash', amount: '', reference: '' }])
   325→   325→  }
   326→   326→  const removeSplitItem = (idx: number) => {
   327→   327→    setLocalSplitItems(prev => prev.filter((_, i) => i !== idx))
   328→   328→  }
   329→   329→  const updateSplitItem = (idx: number, field: string, value: string) => {
   330→   330→    setLocalSplitItems(prev => prev.map((item, i) => i === idx ? { ...item, [field]: value } : item))
   331→   331→  }
   332→   332→  const totalSplitAmount = localSplitItems.reduce((sum, item) => sum + (parseFloat(item.amount) || 0), 0)
   333→   333→  const splitRemaining = checkoutSettlementData ? Math.max(0, checkoutSettlementData.totalOutstanding - totalSplitAmount) : 0
   334→   334→
   335→   335→  // ─── Computed GST data from invoiceData ───

   337→   337→    if (!invoiceData?.entries) return null
   338→   338→    const entries = invoiceData.entries as any[]
   339→   339→    const taxableEntries = entries.filter((e: any) => e.debit > 0)
   340→   340→    const taxableAmount = taxableEntries.reduce((sum: number, e: any) => sum + (e.debit || 0) - (e.tax || 0) - (e.discount || 0), 0)
   341→   341→    const totalTax = entries.reduce((sum: number, e: any) => sum + (e.tax || 0), 0)
   342→   342→    const totalDiscount = entries.reduce((sum: number, e: any) => sum + (e.discount || 0), 0)
   343→   343→    const totalCharges = entries.reduce((sum: number, e: any) => sum + (e.debit || 0), 0)
   344→   344→    const cgst = totalTax / 2
   345→   345→    const sgst = totalTax / 2
   346→   346→    const totalPayments = entries.reduce((sum: number, e: any) => sum + (e.credit || 0), 0)
   347→   347→    const advancePayment = bal < 0 ? Math.abs(bal) : 0
   348→   348→    return { taxableAmount, totalTax, totalDiscount, totalCharges, cgst, sgst, totalPayments, advancePayment, entryCount: taxableEntries.length }
   349→   349→  }, [invoiceData, bal])
   350→   350→
   351→   351→  // ─── Active submitting state (local or prop) ───
   352→   352→  const isSubmitting = localSubmitting || submitting
   353→   353→
   354→   354→  // ─── Entry success banner component ───
   355→   355→  const EntrySuccessBanner = () => {
   356→   356→    if (!entrySuccess?.show) return null
   357→   357→    return (
   358→   358→      <div className="fixed bottom-4 right-4 z-[100] animate-in slide-in-from-bottom-2 fade-in duration-300">
   359→   359→        <div className="bg-emerald-600 text-white rounded-xl shadow-lg px-4 py-3 flex items-center gap-3 min-w-[260px]">
   360→   360→          <div className="w-8 h-8 rounded-full bg-emerald-500 flex items-center justify-center shrink-0">
   361→   361→            <CircleCheck className="size-4 text-white" />
   362→   362→          </div>
   363→   363→          <div className="min-w-0">
   364→   364→            <div className="text-xs font-bold">{entrySuccess.type} Created</div>
   365→   365→            <div className="text-[11px] text-emerald-100 flex items-center gap-1 mt-0.5">
   366→   366→              <Hash className="size-3" />
   367→   367→              <span className="font-mono font-bold">{entrySuccess.docNo}</span>
   368→   368→            </div>
   369→   369→          </div>
   370→   370→        </div>
   371→   371→      </div>
   372→   372→    )
   373→   373→  }
   374→   374→
   375→   375→  // ─── Per-entry-type success wrapper ───
   376→   376→  const SuccessOverlay = ({ docNo, type, label }: { docNo: string; type: string; label: string }) => (
   377→   377→    <div className="flex flex-col items-center justify-center py-6 gap-3">
   378→   378→      <div className="w-14 h-14 rounded-2xl bg-emerald-100 dark:bg-emerald-900/40 flex items-center justify-center">
   379→   379→        <PartyPopper className="size-7 text-emerald-600 dark:text-emerald-400" />
   380→   380→      </div>
   381→   381→      <div className="text-center">
   382→   382→        <div className="text-sm font-bold text-slate-800 dark:text-slate-100">{label}</div>
   383→   383→        <div className="mt-2 inline-flex items-center gap-2 bg-emerald-50 dark:bg-emerald-950/40 border border-emerald-200 dark:border-emerald-800 rounded-lg px-4 py-2.5">
   384→   384→          <Hash className="size-4 text-emerald-600 dark:text-emerald-400" />
