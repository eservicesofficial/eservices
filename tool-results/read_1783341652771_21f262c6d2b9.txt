    83→interface Props {
    84→  detail: CustomerDetail | null; submitting: boolean; isAdmin: boolean
    85→  canManageDocs: boolean; canDeleteDocs: boolean; token: string | null; onRefresh: () => void

    88→
    89→export function CustomerFolioNotesDocs({ detail, submitting, isAdmin, canManageDocs, canDeleteDocs, token, onRefresh, initialTab = 'notes' }: Props) {
    90→  const [tab, setTab] = useState<'notes' | 'docs'>(initialTab)
    91→  const [noteSearch, setNoteSearch] = useState('')
    92→  const [noteTypeFilter, setNoteTypeFilter] = useState('All')
    93→  const [docSearch, setDocSearch] = useState('')
    94→  const [docTypeFilter, setDocTypeFilter] = useState('All')
    95→  const [showNoteForm, setShowNoteForm] = useState(false)
    96→  const [showDocForm, setShowDocForm] = useState(false)
    97→  const [addLoading, setAddLoading] = useState(false)
    98→  const [editingNoteId, setEditingNoteId] = useState<string | null>(null)
    99→  const [editContent, setEditContent] = useState('')
   100→
   101→  // Note form
   102→  const [noteType, setNoteType] = useState('Internal')
   103→  const [noteContent, setNoteContent] = useState('')
   104→  const [notePrivate, setNotePrivate] = useState(false)
   105→
   106→  // Doc form
   107→  const [docType, setDocType] = useState('Custom')
   108→  const [docName, setDocName] = useState('')
   109→  const [docUrl, setDocUrl] = useState('')
   110→  const [docNotes, setDocNotes] = useState('')
   111→  const [docFileSize, setDocFileSize] = useState('')
   112→
   113→  // Expanded note for long content
   114→  const [expandedNotes, setExpandedNotes] = useState<Set<string>>(new Set())
   115→  const toggleNoteExpand = (id: string) => {
   116→    setExpandedNotes(prev => { const n = new Set(prev); if (n.has(id)) n.delete(id); else n.add(id); return n })
   117→  }
   118→
   119→  const authHdrs = token ? { Authorization: `Bearer ${token}` } : {}
   120→
   121→  // Filter notes: hide private from non-admin, apply search/type filter
   122→  const visibleNotes = useMemo(() => {
   123→    let notes = (detail?.notes || [])
   124→    if (!isAdmin) notes = notes.filter(n => !n.isPrivate)
   125→    if (noteTypeFilter !== 'All') notes = notes.filter(n => n.type === noteTypeFilter)
   126→    if (noteSearch) { const q = noteSearch.toLowerCase(); notes = notes.filter(n => n.content.toLowerCase().includes(q) || n.type.toLowerCase().includes(q)) }
   127→    return notes
   128→  }, [detail?.notes, isAdmin, noteTypeFilter, noteSearch])
   129→
   130→  const visibleDocs = useMemo(() => {
   131→    let docs = (detail?.documents || [])
   132→    if (docTypeFilter !== 'All') docs = docs.filter(d => d.type === docTypeFilter)
   133→    if (docSearch) { const q = docSearch.toLowerCase(); docs = docs.filter(d => d.name.toLowerCase().includes(q) || d.type.toLowerCase().includes(q) || (d.notes || '').toLowerCase().includes(q)) }
   134→    return docs
   135→  }, [detail?.documents, docTypeFilter, docSearch])
   136→
   137→  // Stats
   138→  const noteStats = useMemo(() => {
   139→    const all = detail?.notes || []
   140→    const byType: Record<string, number> = {}
   141→    for (const n of all) { if (!n.isPrivate || isAdmin) byType[n.type] = (byType[n.type] || 0) + 1 }
   142→    return byType
   143→  }, [detail?.notes, isAdmin])
   144→
   145→  const docStats = useMemo(() => {
   146→    const all = detail?.documents || []
   147→    const byType: Record<string, number> = {}
   148→    for (const d of all) byType[d.type] = (byType[d.type] || 0) + 1
   149→    return byType
   150→  }, [detail?.documents])
   151→
   152→  const resetNoteForm = () => { setNoteType('Internal'); setNoteContent(''); setNotePrivate(false); setShowNoteForm(false) }
   153→  const resetDocForm = () => { setDocType('Custom'); setDocName(''); setDocUrl(''); setDocNotes(''); setDocFileSize(''); setShowDocForm(false) }
   154→
   155→  const addNote = async () => {
   156→    if (!detail?.id || !noteContent.trim()) return
   157→    try {
   158→      setAddLoading(true)
   159→      const res = await fetch(`/api/folio/customers/${detail.id}/notes`, {
   160→        method: 'POST', headers: { ...authHdrs, 'Content-Type': 'application/json' },
   161→        body: JSON.stringify({ type: noteType, content: noteContent.trim(), isPrivate: isAdmin && notePrivate })
   162→      })
   163→      if (res.ok) { notify({ type: 'success', title: 'Note added' }); resetNoteForm(); onRefresh() }
   164→      else { const e = await res.json().catch(() => ({})); notify({ type: 'error', title: 'Error', message: (e as any).error || 'Failed to add note' }) }
   165→    } catch { notify({ type: 'error', title: 'Error', message: 'Network error' }) } finally { setAddLoading(false) }
   166→  }
   167→
   168→  const updateNote = async (noteId: string) => {
   169→    if (!detail?.id || !editContent.trim()) return
   170→    try {
   171→      setAddLoading(true)
   172→      const res = await fetch(`/api/folio/customers/${detail.id}/notes/${noteId}`, {
   173→        method: 'PUT', headers: { ...authHdrs, 'Content-Type': 'application/json' },
   174→        body: JSON.stringify({ content: editContent.trim() })
   175→      })
   176→      if (res.ok) { notify({ type: 'success', title: 'Note updated' }); setEditingNoteId(null); setEditContent(''); onRefresh() }
   177→      else notify({ type: 'error', title: 'Error', message: 'Failed to update note' })
   178→    } catch { notify({ type: 'error', title: 'Error', message: 'Network error' }) } finally { setAddLoading(false) }
   179→  }
   180→
   181→  const addDoc = async () => {
   182→    if (!detail?.id || !docName.trim()) return
   183→    try {
   184→      setAddLoading(true)
   185→      const res = await fetch(`/api/folio/customers/${detail.id}/documents`, {
   186→        method: 'POST', headers: { ...authHdrs, 'Content-Type': 'application/json' },
   187→        body: JSON.stringify({ type: docType, name: docName.trim(), url: docUrl.trim(), notes: docNotes.trim(), fileSize: parseInt(docFileSize) || 0, mimeType: '' })
   188→      })
   189→      if (res.ok) { notify({ type: 'success', title: 'Document added' }); resetDocForm(); onRefresh() }
   190→      else notify({ type: 'error', title: 'Error', message: 'Failed to add document' })
   191→    } catch { notify({ type: 'error', title: 'Error', message: 'Network error' }) } finally { setAddLoading(false) }
   192→  }
   193→
   194→  const deleteNote = async (id: string) => {
   195→    if (!detail) return
   196→    try {
   197→      const res = await fetch(`/api/folio/customers/${detail.id}/notes/${id}`, { method: 'DELETE', headers: authHdrs })
   198→      if (res.ok) { notify({ type: 'success', title: 'Deleted' }); onRefresh() }
   199→      else notify({ type: 'error', title: 'Error', message: 'Failed to delete note' })
   200→    } catch { notify({ type: 'error', title: 'Error', message: 'Network error' }) }
   201→  }
   202→
   203→  const deleteDoc = async (id: string) => {
   204→    if (!detail) return
   205→    try {
   206→      const res = await fetch(`/api/folio/customers/${detail.id}/documents/${id}`, { method: 'DELETE', headers: authHdrs })
   207→      if (res.ok) { notify({ type: 'success', title: 'Deleted' }); onRefresh() }
   208→      else notify({ type: 'error', title: 'Error', message: 'Failed to delete document' })
   209→    } catch { notify({ type: 'error', title: 'Error', message: 'Network error' }) }
   210→  }
   211→
   212→  const fmtSize = (bytes: number) => {
   213→    if (bytes < 1024) return `${bytes} B`
   214→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   215→    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   216→  }
   217→
   218→  const totalNotes = detail?.notes ? (isAdmin ? detail.notes.length : detail.notes.filter(n => !n.isPrivate).length) : 0
   219→  const totalDocs = detail?.documents?.length || 0
   220→
   221→  return (
   222→    <div className="flex flex-col h-full">
   223→      {/* ── Header Tabs ── */}
   224→      <div className="flex items-center gap-1 p-3 pb-0">
   225→        <button
   226→          onClick={() => setTab('notes')}
   227→          className={`flex items-center gap-2 px-4 py-2 rounded-t-lg text-xs font-semibold transition-all border-b-2 ${
   228→            tab === 'notes'
   229→              ? 'bg-emerald-50 dark:bg-emerald-950/20 text-emerald-700 dark:text-emerald-300 border-emerald-500'
   230→              : 'text-muted-foreground border-transparent hover:text-foreground hover:bg-muted/50'
   231→          }`}
   232→        >
   233→          <StickyNote className="size-4" />
   234→          Notes
   235→          <span className={`text-[10px] px-1.5 py-0.5 rounded-full font-bold ${tab === 'notes' ? 'bg-emerald-200 dark:bg-emerald-800 text-emerald-800 dark:text-emerald-200' : 'bg-muted text-muted-foreground'}`}>
   236→            {totalNotes}
   237→          </span>
   238→        </button>
   239→        <button
   240→          onClick={() => setTab('docs')}
   241→          className={`flex items-center gap-2 px-4 py-2 rounded-t-lg text-xs font-semibold transition-all border-b-2 ${
   242→            tab === 'docs'
   243→              ? 'bg-blue-50 dark:bg-blue-950/20 text-blue-700 dark:text-blue-300 border-blue-500'
   244→              : 'text-muted-foreground border-transparent hover:text-foreground hover:bg-muted/50'
   245→          }`}
   246→        >
   247→          <FileText className="size-4" />
   248→          Documents
   249→          <span className={`text-[10px] px-1.5 py-0.5 rounded-full font-bold ${tab === 'docs' ? 'bg-blue-200 dark:bg-blue-800 text-blue-800 dark:text-blue-200' : 'bg-muted text-muted-foreground'}`}>
   250→            {totalDocs}
   251→          </span>
   252→        </button>
   253→      </div>
   254→
   255→      {/* ── NOTES TAB ── */}
   256→      {tab === 'notes' && (
   257→        <div className="flex-1 min-h-0 flex flex-col p-3 pt-2">
   258→          {/* Type Stats Chips */}
   259→          {Object.keys(noteStats).length > 0 && (
   260→            <div className="flex items-center gap-1.5 flex-wrap mb-2">
   261→              {Object.entries(noteStats).map(([type, count]) => (
   262→                <button
   263→                  key={type}
   264→                  onClick={() => setNoteTypeFilter(noteTypeFilter === type ? 'All' : type)}
   265→                  className={`text-[10px] px-2 py-0.5 rounded-full font-medium transition-all border ${
   266→                    noteTypeFilter === type
   267→                      ? `${NOTE_COLORS[type] || 'bg-muted text-muted-foreground'} border-current/20 ring-1 ring-current/20`
   268→                      : 'bg-muted/50 text-muted-foreground border-transparent hover:bg-muted'
   269→                  }`}
   270→                >
   271→                  {type} ({count})
   272→                </button>
   273→              ))}
   274→              {noteTypeFilter !== 'All' && (
   275→                <button onClick={() => setNoteTypeFilter('All')} className="text-[10px] px-1.5 py-0.5 rounded-full text-muted-foreground hover:text-foreground hover:bg-muted transition-colors">
   276→                  <X className="size-3 inline" /> Clear
   277→                </button>
   278→              )}
   279→            </div>
   280→          )}
   281→
   282→          {/* Search + Add */}
   283→          <div className="flex items-center gap-2 mb-3">
   284→            <div className="relative flex-1 min-w-[160px]">
   285→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground" />
   286→              <Input placeholder="Search notes..." className="h-8 pl-8 text-xs" value={noteSearch} onChange={e => setNoteSearch(e.target.value)} />
   287→            </div>
   288→            {(isAdmin || canManageDocs) && (
   289→              <Button size="sm" className="h-8 gap-1.5 text-xs bg-emerald-600 hover:bg-emerald-700 shrink-0" onClick={() => setShowNoteForm(!showNoteForm)}>
   290→                {showNoteForm ? <X className="size-3.5" /> : <Plus className="size-3.5" />}
   291→                {showNoteForm ? 'Cancel' : 'New Note'}
   292→              </Button>
   293→            )}
   294→          </div>
   295→
   296→          {/* Add Note Form */}
   297→          {showNoteForm && (
   298→            <Card className="border-emerald-200 dark:border-emerald-800 bg-emerald-50/50 dark:bg-emerald-950/20 mb-3 shadow-sm">
   299→              <CardContent className="p-3 space-y-2.5">
   300→                <div className="flex items-center gap-2 flex-wrap">
   301→                  <div className="min-w-[120px]">
   302→                    <Label className="text-[10px] text-muted-foreground font-medium">Note Type</Label>
   303→                    <Select value={noteType} onValueChange={setNoteType}>
   304→                      <SelectTrigger className="h-8 text-xs mt-0.5"><SelectValue /></SelectTrigger>
   305→                      <SelectContent>{NOTE_TYPES.map(t => <SelectItem key={t} value={t}><span>{t}</span></SelectItem>)}</SelectContent>
   306→                    </Select>
   307→                  </div>
   308→                  {isAdmin && (
   309→                    <div className="flex items-center gap-1.5 pt-5">
   310→                      <input type="checkbox" checked={notePrivate} onChange={e => setNotePrivate(e.target.checked)} className="rounded border-slate-300" id="note-private-adv" />
   311→                      <Label htmlFor="note-private-adv" className="text-[11px] flex items-center gap-1 cursor-pointer">
   312→                        <Lock className="size-3 text-red-500" /> Private (Admin only)
   313→                      </Label>
   314→                    </div>
   315→                  )}
   316→                </div>
   317→                <div>
   318→                  <Label className="text-[10px] text-muted-foreground font-medium">Content *</Label>
   319→                  <Textarea value={noteContent} onChange={e => setNoteContent(e.target.value)} rows={3} placeholder="Write your note..." className="text-xs mt-0.5 resize-none" maxLength={2000} />
   320→                  <p className="text-[9px] text-muted-foreground text-right mt-0.5">{noteContent.length}/2000</p>
   321→                </div>
   322→                <div className="flex justify-end gap-2">
   323→                  <Button variant="outline" size="sm" className="h-7 text-xs" onClick={resetNoteForm}>Discard</Button>
   324→                  <Button size="sm" className="h-7 text-xs bg-emerald-600 hover:bg-emerald-700" disabled={addLoading || !noteContent.trim()} onClick={addNote}>
   325→                    {addLoading ? <Loader2 className="size-3 animate-spin" /> : <Plus className="size-3" />} Add Note
   326→                  </Button>
   327→                </div>
   328→              </CardContent>
   329→            </Card>
   330→          )}
   331→
   332→          {/* Notes List */}
   333→          <div className="flex-1 min-h-0 overflow-y-auto space-y-1.5 pr-1">
   334→            {visibleNotes.length === 0 ? (
   335→              <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   336→                <div className="w-16 h-16 rounded-2xl bg-muted/50 flex items-center justify-center mb-3">
   337→                  <StickyNote className="size-7 opacity-30" />
   338→                </div>
   339→                <p className="text-sm font-medium">No notes found</p>
   340→                <p className="text-xs mt-1">{noteSearch || noteTypeFilter !== 'All' ? 'Try adjusting your filters' : 'Click "New Note" to add one'}</p>
   341→              </div>
   342→            ) : visibleNotes.map((note, idx) => {
   343→              const isEditing = editingNoteId === note.id
   344→              const isExpanded = expandedNotes.has(note.id)
   345→              const isLong = note.content.length > 200
   346→              const NoteIcon = NOTE_ICONS[note.type] || MessageSquare
   347→              return (
   348→                <div
   349→                  key={note.id}
   350→                  className={`border-l-[3px] ${NOTE_BORDER[note.type] || 'border-l-slate-300'} rounded-r-xl p-3 ${NOTE_BG[note.type] || ''} hover:shadow-sm transition-all group`}
   351→                >
   352→                  <div className="flex items-start gap-2.5">
   353→                    {/* Icon */}
   354→                    <div className={`w-7 h-7 rounded-lg flex items-center justify-center shrink-0 mt-0.5 ${NOTE_COLORS[note.type] || 'bg-muted text-muted-foreground'}`}>
   355→                      <NoteIcon className="size-3.5" />
   356→                    </div>
   357→
   358→                    {/* Content */}
   359→                    <div className="min-w-0 flex-1">
   360→                      <div className="flex items-center gap-2 mb-1 flex-wrap">
   361→                        <span className={`text-[10px] px-1.5 py-0.5 rounded-md font-semibold ${NOTE_COLORS[note.type] || ''}`}>{note.type}</span>
   362→                        {note.isPrivate && <Badge variant="outline" className="text-[9px] h-4 px-1.5 text-red-500 border-red-200 dark:border-red-800 gap-0.5"><Lock className="size-2.5" />Private</Badge>}
   363→                      </div>
   364→
   365→                      {isEditing ? (
   366→                        <div className="space-y-2">
   367→                          <Textarea value={editContent} onChange={e => setEditContent(e.target.value)} rows={3} className="text-xs resize-none" maxLength={2000} />
   368→                          <div className="flex gap-1.5">
   369→                            <Button size="sm" className="h-6 text-[10px] gap-1 bg-emerald-600 hover:bg-emerald-700" disabled={addLoading || !editContent.trim()} onClick={() => updateNote(note.id)}>
   370→                              <Check className="size-3" />Save
   371→                            </Button>
   372→                            <Button variant="outline" size="sm" className="h-6 text-[10px]" onClick={() => { setEditingNoteId(null); setEditContent('') }}>Cancel</Button>
   373→                          </div>
   374→                        </div>
   375→                      ) : (
   376→                        <>

   417→          </div>

   420→
   421→      {/* ── DOCUMENTS TAB ── */}
   422→      {tab === 'docs' && (

   498→            </Card>

   501→          {/* Documents Grid */}
   502→          <div className="flex-1 min-h-0 overflow-y-auto pr-1">

   534→                          {/* Meta */}
