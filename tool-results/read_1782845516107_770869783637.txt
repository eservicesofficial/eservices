    24→// ── Auto-dismiss Toast Popup for Login Page ──────────────────────────────────
    25→// The login page doesn't have the NotificationCenter (it's in the TopBar).
    26→// This component subscribes to the notification store and displays each new
    27→// notification as a popup toast that auto-dismisses after 4 seconds.
    28→function LoginToastContainer() {
    29→  const notifications = useNotificationStore((s) => s.notifications);
    30→  const [visibleToasts, setVisibleToasts] = useState<Array<{ id: string; type: string; title: string; message: string; fading: boolean }>>([]);
    31→  // Track processed IDs with a Set so we never replay or duplicate a notification,
    32→  // even when items are removed from the store (dismissNotification / clearAll).
    33→  const processedIds = useRef<Set<string>>(new Set(notifications.map((n) => n.id)));
    34→
    35→  useEffect(() => {
    36→    // Find notifications we haven't shown yet
    37→    const newNotifs = notifications.filter((n) => !processedIds.current.has(n.id));
    38→    if (newNotifs.length === 0) return;
    39→
    40→    // Mark them as processed
    41→    for (const n of newNotifs) {
    42→      processedIds.current.add(n.id);
    43→    }
    44→
    45→    const newToasts = newNotifs.map((n) => ({ id: n.id, type: n.type, title: n.title, message: n.message, fading: false }));
    46→    setVisibleToasts((prev) => [...prev, ...newToasts]);
    47→
    48→    // Auto-dismiss each toast after 4 seconds
    49→    for (const n of newNotifs) {
    50→      const id = n.id;
    51→      setTimeout(() => {
    52→        setVisibleToasts((prev) => prev.map((t) => t.id === id ? { ...t, fading: true } : t));
    53→        setTimeout(() => {
    54→          setVisibleToasts((prev) => prev.filter((t) => t.id !== id));
    55→          dismissNotification(id);
    56→        }, 300);
    57→      }, 4000);
    58→    }
    59→  }, [notifications]);
    60→
    61→  const iconMap: Record<string, React.ReactNode> = {
    62→    success: <CheckCircle2 className="size-5 text-emerald-500 shrink-0" />,
    63→    error: <AlertCircle className="size-5 text-red-500 shrink-0" />,
    64→    warning: <AlertTriangle className="size-5 text-amber-500 shrink-0" />,
    65→    info: <AlertCircle className="size-5 text-sky-500 shrink-0" />,
    66→  };
    67→  const borderMap: Record<string, string> = {
    68→    success: 'border-l-emerald-500',
    69→    error: 'border-l-red-500',
    70→    warning: 'border-l-amber-500',
    71→    info: 'border-l-sky-500',
    72→  };
    73→  const bgMap: Record<string, string> = {
    74→    success: 'bg-emerald-50 dark:bg-emerald-950/30',
    75→    error: 'bg-red-50 dark:bg-red-950/30',
    76→    warning: 'bg-amber-50 dark:bg-amber-950/30',
    77→    info: 'bg-sky-50 dark:bg-sky-950/30',
    78→  };
    79→
    80→  if (visibleToasts.length === 0) return null;
    81→
    82→  return (
    83→    <div className="fixed top-4 right-4 z-[9999] flex flex-col gap-2 max-w-sm w-full pointer-events-none">
    84→      {visibleToasts.map((t) => (
    85→        <div
    86→          key={t.id}
    87→          className={`pointer-events-auto flex items-start gap-3 p-4 rounded-xl border-l-4 ${borderMap[t.type] || borderMap.info} ${bgMap[t.type] || bgMap.info} shadow-lg shadow-black/10 backdrop-blur-sm transition-all duration-300 ${t.fading ? 'opacity-0 translate-x-4' : 'opacity-100 translate-x-0'}`}
    88→        >
    89→          {iconMap[t.type] || iconMap.info}
    90→          <div className="flex-1 min-w-0">
    91→            <p className="text-sm font-semibold text-foreground">{t.title}</p>
    92→            <p className="text-xs text-muted-foreground mt-0.5">{t.message}</p>
    93→          </div>
    94→          <button onClick={() => { dismissNotification(t.id); setVisibleToasts((prev) => prev.filter((v) => v.id !== t.id)); }} className="text-muted-foreground hover:text-foreground shrink-0">
    95→            <X className="size-4" />
    96→          </button>
    97→        </div>
    98→      ))}
    99→    </div>
   100→  );
   101→}
