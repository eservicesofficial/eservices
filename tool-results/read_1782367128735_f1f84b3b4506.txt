    50→// Build entry rows HTML (shared across print templates)
    51→function buildEntryRows(entries: { date: string; description?: string; type?: string; debit: number; credit: number; balance: number; tax?: number }[]): string {
    52→  return entries.map((e, i) => {
    53→    const desc = stripShiftText(e.description || e.type || '—')
    54→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td>${desc}</td><td style="text-align:right;color:#dc2626;font-weight:600">${e.debit > 0 ? fmtINR(e.debit) : '—'}</td><td style="text-align:right;color:#16a34a;font-weight:600">${e.credit > 0 ? fmtINR(e.credit) : '—'}</td><td style="font-weight:700;color:${e.balance >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(e.balance)}</td></tr>`
    55→  }).join('')
    56→}

    66→async function fetchSettingsForPrint(hdrRef: RefObject<Record<string, string>>): Promise<Record<string, string>> {

    86→// ─── Print Receipt ───
    87→export async function printReceipt(
    88→  detail: CustomerDetail,
    89→  entries: LedgerEntry[],
    90→  hdrRef: RefObject<Record<string, string>>
    91→) {
    92→  if (!detail) return
    93→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
    94→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
    95→  const settings = await fetchSettingsForPrint(hdrRef)
    96→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
    97→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
    98→  const totalTax = activeEntries.reduce((s, e) => s + (e.tax || 0), 0)
    99→  const totalDiscounts = activeEntries.reduce((s, e) => s + (e.discount || 0), 0)
   100→  const finalBal = activeEntries[activeEntries.length - 1].balance
   101→  const now = new Date()
   102→  const rows = buildEntryRows(activeEntries)
   103→
   104→  const w = openPrintWindow(`Receipt - ${detail.fullName}`)
   105→  if (!w) return
   106→
   107→  w.document.write(
   108→    `<!DOCTYPE html><html><head><title>Receipt</title><style>${PRINT_CSS}</style></head><body>` +
   109→    buildHeader(settings, 'Receipt') +
   110→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +
   111→    `<div class="datetime">${detail.status === 'CheckedOut' ? 'Status: Checked Out' : `Status: ${detail.status}`} &middot; ${fmtDateTime(now.toISOString())}</div>` +
   112→    `<hr class="divider">` +
   113→    `<table><tr><th>#</th><th>Date</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   114→    `<tr class="totals"><td colspan="3" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(finalBal)}</td></tr></table>` +
   115→    `<div class="tax-summary"><div class="tax-item"><span class="tax-label">Total Tax: </span><span class="tax-val">${fmtINR(totalTax)}</span></div><div class="tax-item"><span class="tax-label">Discounts: </span><span class="tax-val">${fmtINR(totalDiscounts)}</span></div><div class="tax-item"><span class="tax-label">Entries: </span><span class="tax-val">${activeEntries.length}</span></div></div>` +
   116→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Final Balance</div><div class="s-value" style="color:${finalBal >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(finalBal)}</div></div></div>` +
   117→    buildFooter() +
   118→    `</body></html>`
   119→  )
   120→  w.document.close()
   121→  w.print()
   122→}

   124→// ─── Print Invoice (Professional) ───
   125→export async function printInvoice(
   126→  detail: CustomerDetail,
   127→  invoiceData: { entries?: any[]; closingBalance?: number; openingBalance?: number },
   128→  invoiceFrom: string,
   129→  invoiceTo: string
   130→) {
   131→  if (!detail || !invoiceData) return
   132→  const entries = invoiceData.entries || []
   133→  if (entries.length === 0) return
   134→  const totalDebits = entries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   135→  const totalCredits = entries.reduce((s: number, e: any) => s + (e.credit || 0), 0)
   136→  const totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   137→  const totalDiscount = entries.reduce((s: number, e: any) => s + (e.discount || 0), 0)
   138→  const closingBal = invoiceData.closingBalance ?? 0
   139→  const openingBal = invoiceData.openingBalance ?? 0
   140→  const now = new Date()
   141→
   142→  // Fetch shop settings for professional header
   143→  let shopSettings: Record<string, string> = {}
   144→  try {
   145→    const res = await fetch('/api/settings')
   146→    if (res.ok) shopSettings = await res.json()
   147→  } catch { /* ignore */ }
   148→
   149→  // Generate invoice number if not available
   150→  let invNumber = ''
   151→  try {
   152→    const invRes = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, { method: 'POST' })
   153→    if (invRes.ok) { const invData = await invRes.json(); invNumber = invData.invoiceNumber || '' }
   154→  } catch { /* ignore */ }
   155→
   156→  const bizName = shopSettings.shopName || 'Eservices'
   157→  const bizAddress = shopSettings.shopAddress || ''
   158→  const bizPhone = shopSettings.shopPhone || ''
   159→  const bizEmail = shopSettings.shopEmail || ''
   160→  const gstNumber = shopSettings.gstNumber || shopSettings.gstin || ''
   161→  const authorizedSignatory = shopSettings.authorizedSignatory || ''
   162→
   163→  // Collect TXN IDs from entries
   164→  const txnIds = entries.map((e: any) => e.docNo || e.id).filter(Boolean).slice(0, 10)
   165→
   166→  const rows = entries.map((e: any, i: number) => {
   167→    const desc = stripShiftText(e.description || e.type || '—')
   168→    const txnId = e.docNo || e.id || ''
   169→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td style="font-size:9px;color:#666;font-family:monospace">${txnId}</td><td>${e.type || ''}</td><td>${desc}</td><td style="text-align:right;color:#dc2626;font-weight:600">${e.debit > 0 ? fmtINR(e.debit) : '—'}</td><td style="text-align:right;color:#16a34a;font-weight:600">${e.credit > 0 ? fmtINR(e.credit) : '—'}</td><td style="font-weight:700;color:${e.balance >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(e.balance)}</td></tr>`
   170→  }).join('')
   171→
   172→  const w = openPrintWindow(`Tax Invoice - ${detail.fullName}`)
   173→  if (!w) return
   174→
   175→  w.document.write(
   176→    `<!DOCTYPE html><html><head><title>Tax Invoice</title><style>${PRINT_CSS}</style></head><body>` +
   177→    // Shop Header
   178→    `<div class="bill-header"><h1>TAX INVOICE</h1>${bizAddress ? `<div class="biz-address">${bizAddress}</div>` : ''}${bizPhone ? `<div class="biz-address">${bizPhone}${bizEmail ? ` | ${bizEmail}` : ''}</div>` : ''}${gstNumber ? `<div class="gst-line">GSTIN: ${gstNumber}</div>` : ''}</div>` +
   179→    // Invoice Meta
   180→    `<div class="invoice-meta">
   181→      <div><div class="inv-label">Invoice No.</div><div class="inv-value">${invNumber || 'N/A'}</div></div>
   182→      <div style="text-align:center"><div class="inv-label">Date</div><div class="inv-value">${fmtDate(now.toISOString())}</div></div>
   183→      <div style="text-align:right"><div class="inv-label">Period</div><div class="inv-value" style="font-size:11px">${invoiceFrom || 'Beginning'} to ${invoiceTo || 'Today'}</div></div>
   184→    </div>` +
   185→    // Customer Info
   186→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">Customer ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +
   187→    (detail.gstNumber ? `<div style="margin:-8px 0 8px 0;font-size:11px;color:#666">Customer GSTIN: ${detail.gstNumber}</div>` : '') +
   188→    // TXN IDs
   189→    (txnIds.length > 0 ? `<div style="font-size:9px;color:#999;margin-bottom:8px">Folio TXN IDs: ${txnIds.join(', ')}</div>` : '') +
   190→    `<div class="datetime">Generated: ${fmtDateTime(now.toISOString())}</div>` +
   191→    `<hr class="divider">` +
   192→    // Table
   193→    `<table><tr><th>#</th><th>Date</th><th>TXN ID</th><th>Type</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   194→    `<tr class="totals"><td colspan="5" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(closingBal)}</td></tr></table>` +
   195→    // Tax Summary
   196→    `<div class="tax-summary"><div class="tax-item"><span class="tax-label">Total Tax: </span><span class="tax-val">${fmtINR(totalTax)}</span></div><div class="tax-item"><span class="tax-label">Discount: </span><span class="tax-val">${fmtINR(totalDiscount)}</span></div><div class="tax-item"><span class="tax-label">Net Charges: </span><span class="tax-val">${fmtINR(totalDebits - totalCredits)}</span></div></div>` +
   197→    // Summary Box
   198→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Opening Balance</div><div class="s-value">${fmtINR(openingBal)}</div></div><div class="summary-item"><div class="s-label">Total Charges</div><div class="s-value" style="color:#dc2626">${fmtINR(totalDebits)}</div></div><div class="summary-item"><div class="s-label">Total Payments</div><div class="s-value" style="color:#16a34a">${fmtINR(totalCredits)}</div></div><div class="summary-item"><div class="s-label">Closing Balance</div><div class="s-value" style="color:${closingBal >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(closingBal)}</div></div></div>` +
   199→    // Signature Section
   200→    `<div class="signature-section">
   201→      <div class="signature-box"><div class="signature-line"><div class="sig-label">Customer Signature</div><div class="sig-sub">Authorized Representative</div></div></div>
   202→      <div class="signature-box"><div class="signature-line"><div class="sig-label">${authorizedSignatory || 'Authorized Signatory'}</div><div class="sig-sub">${authorizedSignatory ? 'Authorized Signatory' : 'For ' + bizName}</div></div></div>
   203→    </div>` +
   204→    buildFooter() +
   205→    `</body></html>`
   206→  )
   207→  w.document.close()
   208→  w.print()
   209→}

   245→// ─── Print Voucher / Payment History ───
   246→export async function printVoucherHistory(
   247→  detail: CustomerDetail,
   248→  paymentEntries: LedgerEntry[],
   249→  hdrRef: RefObject<Record<string, string>>
   250→) {

   254→  const now = new Date()

   256→  const sorted = [...paymentEntries].sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())

   259→  const rows = sorted.map((e, i) => {

   262→  }).join('')
   263→
   264→  const w = openPrintWindow(`Voucher History - ${detail.fullName}`)
   265→  if (!w) return
   266→
   267→  w.document.write(
   268→    `<!DOCTYPE html><html><head><title>Voucher / Payment History</title><style>${PRINT_CSS}</style></head><body>` +
   269→    buildHeader(settings, 'Voucher / Payment History') +
   270→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +

   273→    `<tr class="totals"><td colspan="3" style="text-align:right;font-size:11px;color:#555">TOTAL</td><td style="color:#16a34a;font-weight:700">${fmtINR(totalAmount)}</td><td>${sorted.length} entries</td></tr></table>` +
   274→    `<div class="summary-box"><div class="summary-item"><div class="s-label">Total Payments</div><div class="s-value" style="color:#16a34a">${fmtINR(totalAmount)}</div></div><div class="summary-item"><div class="s-label">Entries</div><div class="s-value">${sorted.length}</div></div></div>` +
   275→    buildFooter() +
   276→    `</body></html>`
   277→  )
   278→  w.document.close()
   279→  w.print()
   280→}

   292→async function fetchShopSettings(hdrRef?: RefObject<Record<string, string>>): Promise<ShopSettings> {
   293→  try {
   294→    const headers = hdrRef?.current || undefined
   295→    const res = await fetch('/api/settings', headers ? { headers } : undefined)
   296→    if (res.ok) {
   297→      const db = (await res.json()) as Record<string, string>
   298→      return {

   325→export async function thermalPrintReceipt(
   326→  detail: CustomerDetail,
   327→  entries: LedgerEntry[],
   328→  hdrRef: RefObject<Record<string, string>>
   329→) {
   330→  if (!detail) return
   331→  const activeEntries = entries.length > 0 ? entries : detail.ledgerEntries
   332→  if (activeEntries.length === 0) { notify({ type: 'warning', title: 'No Entries', message: 'No entries to print' }); return }
   333→
   334→  const [tConfig, shop] = await Promise.all([
   335→    fetchThermalSettings(),
   336→    fetchShopSettings(hdrRef),
   337→  ])
   338→
   339→  const totalDebits = activeEntries.reduce((s, e) => s + (e.debit || 0), 0)
   340→  const totalCredits = activeEntries.reduce((s, e) => s + (e.credit || 0), 0)
   341→  const finalBal = activeEntries[activeEntries.length - 1].balance
   342→
   343→  const items: ThermalReceiptData['items'] = activeEntries.map(e => ({
   344→    name: stripShiftText(e.description || e.type || '—'),
   345→    qty: 1,
   346→    rate: e.debit > 0 ? e.debit : e.credit,
   347→    total: e.debit > 0 ? e.debit : e.credit,
   348→  }))
   349→
   350→  const receiptData: ThermalReceiptData = {
   351→    shopName: shop.shopName,
   352→    shopAddress: shop.shopAddress,
   353→    shopPhone: shop.shopPhone,
   354→    shopGst: shop.shopGst,
   355→    shopGstState: shop.shopGstState,
   356→    showGst: false,
   357→    invoiceNumber: detail.customerId,
   358→    date: fmtDate(new Date().toISOString()),
   359→    orderType: 'Folio Receipt',
   360→    customerName: detail.fullName,
   361→    customerPhone: detail.mobile,
   362→    customerAddress: '',
   363→    items,
   364→    subtotal: totalDebits,
   365→    cgst: 0,
   366→    sgst: 0,
   367→    gstRate: 0,
   368→    deliveryCharge: 0,
   369→    discountAmount: 0,
   370→    grandTotal: totalDebits - totalCredits,
   371→    paidAmount: totalCredits,
   372→    balanceAmount: finalBal,
   373→    paymentMethod: 'Folio',
   374→    billStatus: finalBal <= 0 ? 'Paid' : 'Balance',
   375→  }
   376→
   377→  const thermalHtml = buildRichThermalHtml(receiptData, tConfig, `Receipt - ${detail.fullName}`)
   378→  openThermalPrintWindow(thermalHtml, `Receipt - ${detail.fullName}`)
   379→}

   552→// ─── Print History Invoice (Professional invoice for historical folios) ───
   553→export async function printHistoryInvoice(
   554→  detail: CustomerDetail,
   555→  entries: any[],
   556→  openingBalance: number,
   557→  closingBalance: number,
   558→  invoiceFrom: string,
   559→  invoiceTo: string,
   560→) {
   561→  if (!detail || !entries?.length) return
   562→
   563→  const totalDebits = entries.reduce((s: number, e: any) => s + (e.debit || 0), 0)
   564→  const totalCredits = entries.reduce((s: number, e: any) => s + (e.credit || 0), 0)
   565→  const totalTax = entries.reduce((s: number, e: any) => s + (e.tax || 0), 0)
   566→  const totalDiscount = entries.reduce((s: number, e: any) => s + (e.discount || 0), 0)
   567→  const now = new Date()
   568→
   569→  let shopSettings: Record<string, string> = {}
   570→  try {
   571→    const res = await fetch('/api/settings')
   572→    if (res.ok) shopSettings = await res.json()
   573→  } catch { /* ignore */ }
   574→
   575→  let invNumber = ''
   576→  try {
   577→    const invRes = await fetch(`/api/folio/customers/${detail.id}/invoice-number`, { method: 'POST' })
   578→    if (invRes.ok) { const invData = await invRes.json(); invNumber = invData.invoiceNumber || '' }
   579→  } catch { /* ignore */ }
   580→
   581→  const bizName = shopSettings.shopName || 'Eservices'
   582→  const bizAddress = shopSettings.shopAddress || ''
   583→  const bizPhone = shopSettings.shopPhone || ''
   584→  const bizEmail = shopSettings.shopEmail || ''
   585→  const gstNumber = shopSettings.gstNumber || shopSettings.gstin || ''
   586→  const authorizedSignatory = shopSettings.authorizedSignatory || ''
   587→
   588→  const txnIds = entries.map((e: any) => e.docNo || e.id).filter(Boolean).slice(0, 20)
   589→
   590→  const rows = entries.map((e: any, i: number) => {
   591→    const desc = stripShiftText(e.description || e.type || '—')
   592→    const txnId = e.docNo || e.id || ''
   593→    return `<tr><td>${i + 1}</td><td>${fmtDate(e.date)}</td><td style="font-size:9px;color:#666;font-family:monospace">${txnId}</td><td>${e.type || ''}</td><td>${desc}</td><td style="text-align:right;color:#dc2626;font-weight:600">${e.debit > 0 ? fmtINR(e.debit) : '—'}</td><td style="text-align:right;color:#16a34a;font-weight:600">${e.credit > 0 ? fmtINR(e.credit) : '—'}</td><td style="font-weight:700;color:${e.balance >= 0 ? '#16a34a' : '#dc2626'}">${fmtINR(e.balance)}</td></tr>`
   594→  }).join('')
   595→
   596→  const w = openPrintWindow(`Invoice - ${detail.fullName}`)
   597→  if (!w) return
   598→
   599→  w.document.write(
   600→    `<!DOCTYPE html><html><head><title>Tax Invoice</title><style>${PRINT_CSS}</style></head><body>` +
   601→    `<div class="bill-header"><h1>TAX INVOICE</h1>${bizAddress ? `<div class="biz-address">${bizAddress}</div>` : ''}${bizPhone ? `<div class="biz-address">${bizPhone}${bizEmail ? ` | ${bizEmail}` : ''}</div>` : ''}${gstNumber ? `<div class="gst-line">GSTIN: ${gstNumber}</div>` : ''}</div>` +
   602→    `<div class="invoice-meta">
   603→      <div><div class="inv-label">Invoice No.</div><div class="inv-value">${invNumber || 'N/A'}</div></div>
   604→      <div style="text-align:center"><div class="inv-label">Date</div><div class="inv-value">${fmtDate(now.toISOString())}</div></div>
   605→      <div style="text-align:right"><div class="inv-label">Period</div><div class="inv-value" style="font-size:11px">${invoiceFrom || 'Beginning'} to ${invoiceTo || 'Today'}</div></div>
   606→    </div>` +
   607→    `<div class="customer-info"><div><div class="label">Customer</div><div class="value">${detail.fullName}</div></div><div><div class="label">Customer ID</div><div class="value">${detail.customerId}</div></div><div><div class="label">Mobile</div><div class="value">${detail.mobile}</div></div></div>` +
   608→    (detail.gstNumber ? `<div style="margin:-8px 0 8px 0;font-size:11px;color:#666">Customer GSTIN: ${detail.gstNumber}</div>` : '') +
   609→    (txnIds.length > 0 ? `<div style="font-size:9px;color:#999;margin-bottom:8px">Folio TXN IDs: ${txnIds.join(', ')}</div>` : '') +
   610→    `<div class="datetime">Generated: ${fmtDateTime(now.toISOString())}</div>` +
   611→    `<hr class="divider">` +
   612→    `<table><tr><th>#</th><th>Date</th><th>TXN ID</th><th>Type</th><th>Description</th><th>Debit (₹)</th><th>Credit (₹)</th><th>Balance (₹)</th></tr>${rows}` +
   613→    `<tr class="totals"><td colspan="5" style="text-align:right;font-size:11px;color:#555">TOTALS</td><td style="color:#dc2626">${fmtINR(totalDebits)}</td><td style="color:#16a34a">${fmtINR(totalCredits)}</td><td>${fmtINR(closingBalance)}</td></tr></table>` +
